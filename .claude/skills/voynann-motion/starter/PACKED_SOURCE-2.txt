==== 1955 scripts/compare-stills.mjs
// Regression check for refactors: compare two folders of stills (same file names).
// WebGL renders are not bit-exact run to run: re-rendering the same frame with the same code
// measured 52.0–75.6 dB PSNR depending on the frame (GPU/AO noise, a few pixels off by up to
// ~10 levels). A frame passes when it is pixel-identical or its PSNR is at least the threshold
// (default 50 dB). This guards against visible regressions only; to prove a refactor changes
// nothing, compare film state numerically with scripts/snapshot-state.mjs.
//   node scripts/compare-stills.mjs <dirA> <dirB> [minPsnrDb=50]

import fs from "node:fs";
import path from "node:path";
import { spawnSync } from "node:child_process";

const [a, b, minArg = "50"] = process.argv.slice(2);
const min = Number(minArg);
if (!a || !b) {
  console.error("usage: node scripts/compare-stills.mjs <dirA> <dirB>");
  process.exit(1);
}
const md5 = (f) => {
  const r = spawnSync("ffmpeg", ["-v", "error", "-i", f, "-f", "framemd5", "-"], { encoding: "utf8" });
  return r.stdout.trim().split("\n").pop().split(",").pop().trim();
};
const psnr = (x, y) => {
  const r = spawnSync("ffmpeg", ["-v", "info", "-i", x, "-i", y, "-filter_complex", "psnr", "-f", "null", "-"], { encoding: "utf8" });
  return /average:(\S+)/.exec(r.stderr)?.[1] ?? "?";
};
const files = fs.readdirSync(a).filter((f) => f.endsWith(".png")).sort();
let diff = 0;
for (const f of files) {
  const fb = path.join(b, f);
  if (!fs.existsSync(fb)) {
    console.log(`${f}: missing in ${b}`);
    diff++;
    continue;
  }
  if (md5(path.join(a, f)) === md5(fb)) {
    console.log(`${f}: identical`);
    continue;
  }
  const db = psnr(path.join(a, f), fb);
  const ok = db === "inf" || Number(db) >= min;
  if (!ok) diff++;
  console.log(`${f}: ${ok ? "within render noise" : "CHANGED"} (PSNR ${db} dB)`);
}
console.log(`${files.length - diff}/${files.length} pass (threshold ${min} dB)`);
process.exit(diff ? 2 : 0);

==== 2500 scripts/deliver.mjs
// Delivery master: measure integrated loudness, apply gain + a true limiter so the file
// lands at the target without clipping, copy the video stream untouched, then verify.
//   node scripts/deliver.mjs <in.mp4> <out.mp4> [targetLUFS=-14] [ceilingDb=-2]
// (loudnorm alone can't stop AAC inter-sample overshoot; a limiter before the encoder can.)

import { spawnSync } from "node:child_process";

const [input, output, targetArg = "-14", ceilArg = "-2"] = process.argv.slice(2);
if (!input || !output) {
  console.error("usage: node scripts/deliver.mjs <in.mp4> <out.mp4> [targetLUFS] [ceilingDb]");
  process.exit(1);
}
const target = Number(targetArg);
const ceiling = 10 ** (Number(ceilArg) / 20);

const ffmpeg = (args) => {
  const r = spawnSync("ffmpeg", args, { encoding: "utf8" });
  if (r.status !== 0) throw new Error(r.stderr);
  return r.stderr;
};

/** ebur128 prints its summary to stderr. */
const loudness = (file) => {
  const log = ffmpeg(["-nostats", "-i", file, "-af", "ebur128=peak=true", "-f", "null", "-"]);
  const s = log.slice(log.lastIndexOf("Summary"));
  return {
    I: Number(/I:\s+(-?[\d.]+) LUFS/.exec(s)?.[1]),
    peak: Number(/Peak:\s+(-?[\d.]+) dBFS/.exec(s)?.[1]),
  };
};

const before = loudness(input);
let gain = target - before.I + 0.4; // the limiter shaves a little; aim slightly high
let after;
// Peaky voices make the limiter eat more than expected, so measure and correct (max 5 passes).
for (let pass = 1; pass <= 5; pass++) {
  ffmpeg([
    "-v", "error", "-y", "-i", input, "-c:v", "copy",
    "-af", `volume=${gain.toFixed(2)}dB,alimiter=limit=${ceiling.toFixed(3)}:level=false:attack=2:release=80,aresample=48000`,
    "-c:a", "aac", "-b:a", "256k", "-movflags", "+faststart", output,
  ]);
  after = loudness(output);
  if (Math.abs(after.I - target) <= 0.5) break;
  gain += 1.6 * (target - after.I); // the limiter absorbs part of every boost
}
const probe = spawnSync(
  "ffprobe",
  ["-v", "error", "-select_streams", "v:0", "-show_entries", "stream=codec_name,width,height,r_frame_rate,nb_frames", "-of", "csv=p=0", output],
  { encoding: "utf8" },
).stdout.trim();
console.log(`in : ${before.I} LUFS, true peak ${before.peak} dBFS`);
console.log(`out: ${after.I} LUFS, true peak ${after.peak} dBFS (gain ${gain.toFixed(2)} dB)`);
console.log(`video: ${probe}`);
if (Math.abs(after.I - target) > 0.5) console.warn(`WARNING: integrated loudness ${after.I} LUFS is more than 0.5 LU from the ${target} LUFS target; report this honestly.`);

==== 1469 scripts/gl.mjs
// WebGL renderer choice for headless Chrome, shared by every render script.
// Order: GL env var > the project's .gl-mode file > "angle" (GPU). When the GPU path cannot create a WebGL
// context (no GPU, some VMs/drivers), the scripts switch to "swangle" (software, much slower) automatically
// and remember it in .gl-mode, so later renders and `npx remotion render` (remotion.config.ts) use it too.
import fs from "node:fs";

const FILE = ".gl-mode";

// When the GPU path crashes, Remotion may reject an internal "Target closed" promise that nothing awaits.
// That rejection is noise from the dead tab, not a render failure; real errors still reach the caller.
export const tolerateClosedTargets = () =>
  process.on("unhandledRejection", (e) => {
    if (/Target closed/i.test(String(e?.message ?? e))) return;
    console.error(e);
    process.exit(1);
  });

export const glMode = () =>
  process.env.GL || (fs.existsSync(FILE) ? fs.readFileSync(FILE, "utf8").trim() : "") || "angle";

export const isWebGLError = (e) =>
  /webgl|WebGLRenderer|GL context|BindToCurrentSequence|GL_VENDOR = Disabled/i.test(String(e?.stack ?? e?.message ?? e));

export const switchToSoftware = () => {
  fs.writeFileSync(FILE, "swangle\n");
  console.log(
    "\n! GPU 3D rendering is not available on this computer: switched to software rendering (GL=swangle, much slower)." +
      "\n  Saved in .gl-mode; delete that file to try the GPU again.\n",
  );
  return "swangle";
};

==== 3817 scripts/qa-gate.mjs
// The VOYNANN QA gate in two steps, used by every make / render script (and runnable alone):
//   1. before the final render: scripts/qa-scene.mjs on the PREVIEW composition — tembus, one-frame jumps and blinks
//      of tagged solids, camera snaps, and a video scan of the 3D alone (captions left out) for kedip / loncat / flicker
//   2. after the final render: scripts/qa-video.mjs --hard-only on the delivered MP4 — black / white runs, full-frame
//      blinks, cuts that are not in the plan
//   node scripts/qa-gate.mjs --comp=<PreviewComp> [--cuts=s,s,…] [--out=dir] [--props=file] [--final=film.mp4]
// A FAIL stops the build. A CHECK stops it too until every strip has a verdict in <out>/qa-video.review.json
// ("ok: <why it is intended>"); --force-qa goes on anyway and writes that into the review (never for a delivery
// the buyer has not seen).
import { spawnSync } from "node:child_process";
import path from "node:path";
import { fileURLToPath } from "node:url";

const here = path.dirname(fileURLToPath(import.meta.url));
const node = (script, args) => spawnSync(process.execPath, [path.join(here, script), ...args], { stdio: "inherit" }).status;
const WORD = { 0: "PASS", 1: "FAIL", 2: "INCOMPLETE", 3: "CHECK" };

/** Step 1. `allow`: seconds of designed instant changes (a panel reveal, a page snap) the video scan should accept.
 *  Returns "PASS" | "CHECK" | "FAIL" | "INCOMPLETE" (and exits the process on FAIL / CHECK unless force). */
export const sceneGate = ({ comp, cuts, out, props, allow, force = false, extra = [] }) => {
  console.log(`\n▶ QA gate (scene + clean video scan) on ${comp}`);
  const code = node("qa-scene.mjs", [`--comp=${comp}`, ...(cuts ? [`--cuts=${cuts.map((t) => (+t).toFixed(3)).join(",")}`] : []), ...(allow?.length ? [`--allow=${allow.map((t) => (+t).toFixed(3)).join(",")}`] : []), `--out=${out}`, ...(props ? [`--props=${props}`] : []), ...extra]);
  const word = WORD[code] ?? "INCOMPLETE";
  if ((word === "FAIL" || word === "CHECK") && !force) {
    console.error(
      word === "FAIL"
        ? `\n✗ QA gate FAIL — open the images in ${out}, fix the cause (choreography, geometry, a z-fight), then run again. The final is not rendered.`
        : `\n✗ QA gate CHECK — open each qa-video_*.jpg in ${out}; for a change that is designed, write "ok: <why>" for its id in ${path.join(out, "qa-video.review.json")}; fix the rest; then run again.`,
    );
    process.exit(1);
  }
  return word;
};

/** Step 2. Returns "PASS" | "FAIL" | "INCOMPLETE". */
export const deliveryScan = ({ file, cuts, out }) => {
  console.log(`\n▶ QA scan of the delivered film ${path.basename(file)}`);
  const code = node("qa-video.mjs", [file, "--hard-only", `--out=${out}`, ...(cuts ? [`--cuts=${cuts.map((t) => (+t).toFixed(3)).join(",")}`] : [])]);
  return WORD[code] ?? "INCOMPLETE";
};

if (process.argv[1] && path.resolve(process.argv[1]) === fileURLToPath(import.meta.url)) {
  const args = process.argv.slice(2);
  const opt = (k) => args.find((x) => x.startsWith(`--${k}=`))?.slice(k.length + 3);
  const comp = opt("comp");
  if (!comp && !opt("final")) {
    console.error("usage: node scripts/qa-gate.mjs --comp=<PreviewComp> [--cuts=s,s,…] [--out=dir] [--props=file] [--final=film.mp4]");
    process.exit(2);
  }
  const cuts = opt("cuts") !== undefined ? opt("cuts").split(",").filter(Boolean).map(Number) : null;
  const out = opt("out") ?? path.join("out", "qa", comp ?? "final");
  const a = comp ? sceneGate({ comp, cuts, out, props: opt("props"), force: args.includes("--force-qa") }) : "PASS";
  const b = opt("final") ? deliveryScan({ file: opt("final"), cuts, out: path.join(out, "delivered") }) : "PASS";
  console.log(`\nQA gate: scene ${a}${opt("final") ? `, delivered film ${b}` : ""}`);
  process.exit(b === "FAIL" ? 1 : 0);
}

// VOYNANN video QA gate: finds, in any rendered MP4 of any style, the glitches a viewer reads as cheap:
//   KEDIP  (pop)    part of the picture changes for 1–3 frames and comes back: a blink, a one-frame pose,
//                   z-fighting, a flash, a camera shake
//   LONCAT (jump)   part of the picture steps far more in one frame than in the frames around it and stays:
//                   a limb / prop / camera snapping, something appearing from nowhere, an unplanned cut
//   FLICKER         a region oscillating up-down-up for a sustained time: shimmer, z-fighting, shadow acne
//   BLACK / WHITE   near-black or burnt-white runs in the middle of the film
// Planned cuts are skipped (±2 frames). Fades, smooth moves and eased pop-ins pass.
//   node scripts/qa-video.mjs <video.mp4> [--cuts=s,s,…] [--allow=s,s,…] [--ignore=x0,y0,x1,y1;…] [--out=dir]
//     --cuts    planned hard cuts in seconds (from the plan). Without it, full-frame jumps count as cuts and are
//               listed for the visual review; with it, a full-frame jump that is not in the list fails.
//     --allow   seconds where an instant change is intended (a title slam, a page that is meant to snap);
//               every allowed second must be justified in the review — never allow a glitch away
//     --ignore  regions (0…1 picture coordinates) where instant changes are by design, e.g. a word-caption band
//               that switches words without a fade: --ignore=0,0.70,1,0.86
//     --hard-only  only the FAIL tier (for the delivered film with its captions; the local scan runs on the clean 3D
//               frames from scripts/qa-scene.mjs)
// Writes <out>/qa-video.json and <out>/qa-video_<n>.jpg (5 frames around each finding, red box on the place).
// Exit code 1 when anything is found.

const blk = new Int16Array(PX);
for (let y = 0; y < H; y++) for (let x = 0; x < W; x++) blk[y * W + x] = ((y / B) | 0) * BX + ((x / B) | 0);
const ignored = new Uint8Array(NB);
for (let k = 0; k < NB; k++) {
  const cx = ((k % BX) + 0.5) / BX, cy = (((k / BX) | 0) + 0.5) / BY;
  if (ignore.some(([x0, y0, x1, y1]) => cx >= x0 && cx <= x1 && cy >= y0 && cy <= y1)) ignored[k] = 1;
}

// per-block mean absolute step between consecutive frames, frame mean luma
const D1 = [new Float32Array(NB)];
const G = new Float32Array(N);
const Y = new Float32Array(N);
const M = []; // block mean luma per frame
for (let f = 0; f < N; f++) {
  const a = f * PX;
  let s = 0;
  const m = new Float32Array(NB);
  for (let p = 0; p < PX; p++) {
    s += raw[a + p];
    m[blk[p]] += raw[a + p];
  }
  for (let k = 0; k < NB; k++) m[k] /= BP;
  M.push(m);
  Y[f] = s / PX;
  if (!f) continue;
  const d = new Float32Array(NB), b = a - PX;
  for (let p = 0; p < PX; p++) d[blk[p]] += Math.abs(raw[a + p] - raw[b + p]);
  let g = 0;
  for (let k = 0; k < NB; k++) g += (d[k] /= BP);
  D1.push(d);
  G[f] = g / NB;
}
/** mean |frame i − frame j| inside block k */
const bd = (i, j, k) => {
  const x0 = (k % BX) * B, y0 = ((k / BX) | 0) * B, a = i * PX, b = j * PX;
  let s = 0;
  for (let y = 0; y < B; y++) {
    const r = (y0 + y) * W + x0;
    for (let x = 0; x < B; x++) s += Math.abs(raw[a + r + x] - raw[b + r + x]);
  }
  return s / BP;
};
/** Can block k of frame j be built from frame i moved by up to R_MOVE pixels? Returns the best mean difference. */
const moved = (i, j, k) => {
  const x0 = (k % BX) * B, y0 = ((k / BX) | 0) * B, a = i * PX, b = j * PX;
  const sad = (dx, dy) => {
    let s = 0;
    for (let y = 0; y < B; y++) {
      const yy = Math.min(H - 1, Math.max(0, y0 + y + dy)), r = (y0 + y) * W + x0, rs = yy * W;
      for (let x = 0; x < B; x++) s += Math.abs(raw[b + r + x] - raw[a + rs + Math.min(W - 1, Math.max(0, x0 + x + dx))]);
    }
    return s / BP;
  };
  let best = sad(0, 0), bx = 0, by = 0;
  for (let dy = -R_MOVE; dy <= R_MOVE; dy += 2)
    for (let dx = -R_MOVE; dx <= R_MOVE; dx += 2) {
      const v = sad(dx, dy);
      if (v < best) (best = v), (bx = dx), (by = dy);
    }
  for (let dy = by - 1; dy <= by + 1; dy++) for (let dx = bx - 1; dx <= bx + 1; dx++) best = Math.min(best, sad(dx, dy));
  return best;
};
/** Share of block k's pixels that changed clearly between frames i and j (a thin line or a raindrop: low). */
const cover = (i, j, k) => {
  const x0 = (k % BX) * B, y0 = ((k / BX) | 0) * B, a = i * PX, b = j * PX;
  let n = 0;
  for (let y = 0; y < B; y++) {
    const r = (y0 + y) * W + x0;
    for (let x = 0; x < B; x++) if (Math.abs(raw[a + r + x] - raw[b + r + x]) > 12) n++;
  }
  return n / BP;
};
const COVER = 0.35;
/** Largest step of block k's 3×3 neighbourhood into frame f (0 outside the film or across a cut). */
const around = (f, k) => {
  if (f < 1 || f >= N) return 0;
  const bx = k % BX, by = (k / BX) | 0;
  let m = 0;
  for (let y = Math.max(0, by - 1); y <= Math.min(BY - 1, by + 1); y++)
    for (let x = Math.max(0, bx - 1); x <= Math.min(BX - 1, bx + 1); x++) m = Math.max(m, D1[f][y * BX + x]);
  return m;
};
/** A glitch happens in a calm place: the neighbourhood barely changes in the two frames before and after it.
 *  Fast moves, page turns, bubbles that grow and settle, counters and pans all keep their neighbourhood busy. */
const calm = (f0, f1, k, step) => {
  const lim = Math.max(4, 0.3 * step);
  for (const g of [f0 - 2, f0 - 1, f1 + 1, f1 + 2]) if (!nearCut(g) && around(g, k) > lim) return false;
  return true;
};
// both ways: on a flat surface a darkened patch "matches" plain surface a few pixels away, but not the other way round
const isMotion = (i, j, k, step) => {
  const lim = Math.max(3, 0.35 * step);
  return moved(i, j, k) < lim && moved(j, i, k) < lim;
};
const fd = (i, j) => {
  let s = 0;
  for (let k = 0; k < NB; k++) s += bd(i, j, k);
  return s / NB;
};

// ---- cuts: a full-frame step that does not come back and is far bigger than its neighbours ----
const detected = [];
for (let f = 1; f < N; f++) {
  const g = G[f], nb = Math.max(G[f - 1] ?? 0, G[f + 1] ?? 0);
  if (g < T_CUT || g < 1.6 * nb) continue;
  let many = 0;
  for (let k = 0; k < NB; k++) if (D1[f][k] > 6) many++;
  if (many < 0.5 * NB) continue;
  if (f + 1 < N && fd(f - 1, f + 1) < 0.6 * g) continue; // comes back: a blink, handled below
  detected.push(f);
}
const plannedF = (planned ?? []).map((t) => Math.round(t * fps));
const isPlanned = (f) => plannedF.some((c) => Math.abs(c - f) <= 2);
const unplanned = planned ? detected.filter((f) => !isPlanned(f)) : [];
const cutF = [...plannedF, ...detected];
const nearCut = (f) => cutF.some((c) => f >= c - 2 && f <= c + 2);

// ---- hits per block ----
const hits = [];
for (const f of unplanned) hits.push({ type: "LONCAT", f, k: -1, mag: G[f], note: "full-frame jump that is not a planned cut (camera snap or unplanned cut)" });
for (let f = 1; f < N - 1; f++) {
  if (nearCut(f)) continue;
  const d = D1[f];
  for (let k = 0; k < NB; k++) {
    if (ignored[k]) continue;
    const step = d[k];
    // KEDIP: in at f, out at f+len, frame f−1 ≈ frame f+len, the frames between stay away from f−1
    if (step > T_POP && cover(f - 1, f, k) >= COVER) {
      let found = false;
      for (let len = 1; len <= 3 && f + len < N; len++) {
        if (nearCut(f + len)) break;
        const out = D1[f + len][k];
        if (out < 0.7 * T_POP) continue;
        const back = bd(f - 1, f + len, k);
        if (back >= 0.35 * Math.min(step, out)) continue;
        let away = true;
        for (let j = 1; j < len; j++) if (bd(f - 1, f + j, k) < 0.5 * step) away = false;
        if (!away || !calm(f, f + len, k, Math.min(step, out))) continue;
        hits.push({ type: "KEDIP", f, len, k, mag: Math.min(step, out) });
        found = true;
        break;
      }
      if (found) continue;
    }
    // LONCAT: one step far bigger than the steps around it, and it stays
    if (step > T_JUMP && f >= 2 && f + 2 < N) {
      const ctx = Math.max(D1[f - 2][k], D1[f - 1][k], D1[f + 1][k], D1[f + 2][k]);
      if (step > JUMP_RATIO * Math.max(ctx, 1.5) && bd(f - 1, f + 1, k) >= 0.6 * step && cover(f - 1, f, k) >= COVER && calm(f, f, k, step)) hits.push({ type: "LONCAT", f, k, mag: step, ctx });
    }
  }
}

// ---- FLICKER: the block's mean brightness going up-down-up, sustained over a 1 s window. Block means ignore film
// grain and thin rain; z-fighting, shadow acne and shimmering patches move them. ----
{
  const win = Math.max(2, Math.round(fps));
  const sign = new Int8Array(NB);
  const ring = Array.from({ length: win }, () => new Uint8Array(NB));
  const sum = new Float32Array(NB);
  const flagged = new Int32Array(NB).fill(-1e9);
  const flickRate = Math.min(F_FLICK, 0.6 * fps); // 18 at a 30 fps preview: sparkle and thin-edge shimmer stay below, z-fights alternate faster
  for (let f = 1; f < N; f++) {
    const slot = ring[f % win];
    for (let k = 0; k < NB; k++) sum[k] -= slot[k];
    slot.fill(0);
    if (nearCut(f)) {
      sign.fill(0);
      for (const r of ring) r.fill(0);
      sum.fill(0);
      continue;
    }
    for (let k = 0; k < NB; k++) {
      const dv = M[f][k] - M[f - 1][k];
      if (dv >= T_FLICK || dv <= -T_FLICK) {
        const s = dv > 0 ? 1 : -1;
        if (sign[k] && sign[k] !== s) slot[k] = 1;
        sign[k] = s;
      }
      sum[k] += slot[k];
      if (ignored[k]) continue;
      const rate = sum[k] * (fps / win); // flips per second
      if (rate >= flickRate && f - flagged[k] > win) {
        // recount without the steps that are only motion (a moving camera, a turning page, a waving hand)
        let n = 0, sg = 0;
        for (let g = Math.max(1, f - win + 1); g <= f; g++) {
          const dv = M[g][k] - M[g - 1][k];
          if (Math.abs(dv) < T_FLICK || isMotion(g - 1, g, k, D1[g][k])) continue;
          const s = dv > 0 ? 1 : -1;
          if (sg && sg !== s) n++;
          sg = s;
        }
        const real = n * (fps / win);
        if (real >= flickRate) {
          hits.push({ type: "FLICKER", f: f - win + 1, f1: f, k, mag: real });
          flagged[k] = f;
        }
      }
    }
  }
}

// ---- BLACK / WHITE runs (the first 0.5 s and the last 3 s may fade on purpose) ----
const runs = (test) => {
  const out = [];
  let s = -1;
  for (let f = 0; f <= N; f++) {
    const on = f < N && test(Y[f]);
    if (on && s < 0) s = f;
    if (!on && s >= 0) {
      if (f - s > 0.7 * fps && s > 0.5 * fps && f < N - 3 * fps) out.push([s, f - 1]);
      s = -1;
    }
  }
  return out;
};
for (const [a, b] of runs((y) => y < 20)) hits.push({ type: "BLACK", f: a, f1: b, k: -1, mag: (b - a + 1) / fps });
for (const [a, b] of runs((y) => y > 232)) hits.push({ type: "WHITE", f: a, f1: b, k: -1, mag: (b - a + 1) / fps });

// ---- group hits into findings: same type, touching blocks, within 1 frame (flicker: overlapping windows) ----
const ev = [];
hits.sort((a, b) => a.f - b.f);
for (const h of hits) {
  const bx = h.k < 0 ? -1 : h.k % BX, by = h.k < 0 ? -1 : (h.k / BX) | 0;
  const f1 = h.f1 ?? h.f + (h.len ?? 1) - 1;
  const gap = h.type === "FLICKER" ? Math.round(fps / 2) : 1;
  const e = ev.find((q) => q.type === h.type && h.f <= q.f1 + gap && (h.k < 0 ? q.x0 < 0 : q.x0 >= 0 && bx >= q.x0 - 1 && bx <= q.x1 + 1 && by >= q.y0 - 1 && by <= q.y1 + 1));
  if (e) {
    e.f1 = Math.max(e.f1, f1);
    if (bx >= 0) Object.assign(e, { x0: Math.min(e.x0, bx), x1: Math.max(e.x1, bx), y0: Math.min(e.y0, by), y1: Math.max(e.y1, by) });
    e.blocks++;
    e.per.set(h.f, (e.per.get(h.f) ?? 0) + 1);
    e.mag = Math.max(e.mag, h.mag);
    e.sum += h.mag;
  } else ev.push({ type: h.type, f0: h.f, f1, x0: bx, x1: bx, y0: by, y1: by, blocks: 1, per: new Map([[h.f, 1]]), mag: h.mag, sum: h.mag, note: h.note });
}
// A KEDIP / LONCAT chain longer than 4 frames is either something fast moving through the picture (its box keeps
// growing: motion, fine) or the same place blinking again and again (a flicker).
const info = { motion: 0, particles: 0, faint: 0, edge: 0 };
for (const e of ev) {
  if (e.type === "FLICKER" || e.x0 < 0 || e.f1 - e.f0 < 4) continue;
  const area = (e.x1 - e.x0 + 1) * (e.y1 - e.y0 + 1), perMax = Math.max(...e.per.values());
  if (area > 2.5 * perMax) (e.drop = "motion"), info.motion++;
  else e.type = "FLICKER";
}
// Many small short events spread over the picture within half a second = a particle field (rain, snow, sparks,
// dust), not a glitch. Big ones (a quarter of the frame) always count. A flicker inside such a shot can't be told
// from the particles by numbers: it is left to the visual review of that shot.
const shotOf = (f) => cutF.filter((c) => c <= f).length;
const small = ev.filter((e) => e.type !== "FLICKER" && e.x0 >= 0 && e.blocks < NB / 4);
const busy = (f) => small.filter((q) => shotOf(q.f0) === shotOf(f) && Math.abs(q.f0 - f) <= fps / 2).length >= 7;
const edge = (e) => e.x0 === 0 || e.y0 === 0 || e.x1 === BX - 1 || e.y1 === BY - 1;
for (const e of ev) {
  if (e.drop || e.x0 < 0) continue;
  // something small entering or leaving at the frame edge while the camera moves
  if (e.type !== "FLICKER" && edge(e) && e.blocks < 6) {
    e.drop = "edge";
    info.edge++;
    continue;
  }
  if (e.type === "FLICKER" ? busy(e.f0) || busy(e.f1) : e.blocks < NB / 4 && busy(e.f0)) (e.drop = "particles"), info.particles++;
  else if (e.type !== "FLICKER" && e.sum < S_MIN) (e.drop = "faint"), info.faint++;
}
const live = ev.filter((e) => !e.drop);
const near = (list) => (e) => list.some((t) => t >= tOf(e.f0) - 0.15 && t <= tOf(e.f1) + 0.15);
const isAllowed = near(allow), isSkipped = near(skip);
const skippedEv = live.filter((e) => isSkipped(e) && !isAllowed(e));
const findings0 = live.filter((e) => !isAllowed(e) && !isSkipped(e));
const allowedEv = live.filter(isAllowed);
// Tiers. FAIL: black / white runs, an unplanned full-frame jump, a blink over a quarter of the frame — never by design.
// CHECK: every other KEDIP / LONCAT / FLICKER. Numbers can't tell a glitch from an instant change that is designed
// (a cartoon eye blink, typewriter letters, a counter, a part placed without animation), so a person looks at the strip
// and writes a verdict in <out>/qa-video.review.json: "ok: <why it is intended>" or anything else (= still to fix).
const hardOnly = args.includes("--hard-only"); // on a delivered file with captions: only what is never by design
const isHard = (e) => e.type === "BLACK" || e.type === "WHITE" || e.x0 < 0 || (e.type === "KEDIP" && e.blocks >= NB / 4);
// one review item per happening: CHECK findings of one type in one shot that overlap in time (± ½ s) are merged
// (a lighthouse beam or sea glitter shows as many neighbouring blocks — one look, one verdict)
const merged = [];
for (const e of [...findings0].sort((a, b) => a.f0 - b.f0)) {
  const m = !isHard(e) && merged.find((q) => !isHard(q) && q.type === e.type && shotOf(q.f0) === shotOf(e.f0) && e.f0 <= q.f1 + fps / 2);
  if (m) {
    Object.assign(m, { f1: Math.max(m.f1, e.f1), x0: Math.min(m.x0, e.x0), x1: Math.max(m.x1, e.x1), y0: Math.min(m.y0, e.y0), y1: Math.max(m.y1, e.y1), mag: Math.max(m.mag, e.mag), sum: m.sum + e.sum, blocks: m.blocks + e.blocks });
    m.parts = (m.parts ?? 1) + 1;
  } else merged.push({ ...e });
}
const findings = hardOnly ? merged.filter(isHard) : merged;
const severity = (e) => (isHard(e) ? 1e6 : 0) + e.mag * Math.sqrt(e.blocks);
findings.sort((a, b) => severity(b) - severity(a));
const idOf = (e) => `${e.type}@${tOf(e.f0).toFixed(2)}`;
const reviewFile = path.join(outDir, "qa-video.review.json");
let verdicts = {};
try {
  verdicts = JSON.parse(fs.readFileSync(reviewFile, "utf8"));
} catch {}
const accepted = (e) => !isHard(e) && /^ok\b.{6,}/i.test(String(verdicts[idOf(e)] ?? "").trim());

// ---- strips: 5 frames around each finding, red box on the place ----
const SW = portrait ? 216 : 384, SH = Math.round((SW * st.height) / st.width / 2) * 2;
findings.slice(0, 30).forEach((e, i) => {
  const F0 = Math.max(0, Math.min(N - 5, e.f0 - 2));
  const name = `qa-video_${String(i + 1).padStart(2, "0")}.jpg`;
  const box = e.x0 < 0 ? `drawbox=x=0:y=0:w=iw:h=ih:color=red@0.9:t=4` : `drawbox=x=${Math.round(((e.x0 * B) / W) * SW) - 2}:y=${Math.round(((e.y0 * B) / H) * SH) - 2}:w=${Math.round((((e.x1 - e.x0 + 1) * B) / W) * SW) + 4}:h=${Math.round((((e.y1 - e.y0 + 1) * B) / H) * SH) + 4}:color=red@0.9:t=2`;
  try {
    execFileSync("ffmpeg", ["-v", "error", "-y", "-ss", (F0 / fps).toFixed(4), "-i", file, "-an", "-vf", `scale=${SW}:${SH},${box},tile=5x1:padding=3:color=black`, "-frames:v", "1", path.join(outDir, name)]);
    e.strip = name;
  } catch {}
});

const pct = (v, n) => Math.round((v / n) * 100);
const rows = findings.map((e) => ({
  id: idOf(e),
  tier: isHard(e) ? "FAIL" : accepted(e) ? "OK" : "CHECK",
  type: e.type,
  from: +tOf(e.f0).toFixed(3),
  to: +tOf(e.f1).toFixed(3),
  frames: [e.f0, e.f1],
  where: e.x0 < 0 ? "whole frame" : `x ${pct(e.x0, BX)}–${pct(e.x1 + 1, BX)} %, y ${pct(e.y0, BY)}–${pct(e.y1 + 1, BY)} %`,
  size: e.blocks,
  strength: +e.mag.toFixed(1),
  ...(e.note ? { note: e.note } : {}),
  ...(e.strip ? { strip: e.strip } : {}),
  ...(verdicts[idOf(e)] ? { verdict: verdicts[idOf(e)] } : {}),
}));
const hard = rows.filter((r) => r.tier === "FAIL"), open = rows.filter((r) => r.tier === "CHECK");
// keep the verdict file in step with this render: every finding has a line, old ids are dropped
const nextVerdicts = Object.fromEntries(rows.filter((r) => r.tier !== "FAIL").map((r) => [r.id, verdicts[r.id] ?? ""]));
if (Object.keys(nextVerdicts).length) fs.writeFileSync(reviewFile, JSON.stringify(nextVerdicts, null, 2) + "\n");
else fs.rmSync(reviewFile, { force: true });
const report = {
  file: path.resolve(file),
  size: `${st.width}x${st.height}`,
  fps: +fps.toFixed(3),
  frames: N,
  cuts: { planned: planned ?? "not given", detected: detected.map((f) => +tOf(f).toFixed(3)), unplanned: unplanned.map((f) => +tOf(f).toFixed(3)) },
  ignore,
  ignoredAsMotion: info.motion,
  ignoredAsParticles: info.particles,
  ignoredAsFaint: info.faint,
  ignoredAtEdge: info.edge,
  allowed: allowedEv.map((e) => ({ type: e.type, from: +tOf(e.f0).toFixed(3), to: +tOf(e.f1).toFixed(3) })),
  ...(args.includes("--debug") ? { dropped: ev.filter((e) => e.drop).map((e) => ({ type: e.type, why: e.drop, from: +tOf(e.f0).toFixed(3), to: +tOf(e.f1).toFixed(3), x: [e.x0, e.x1], y: [e.y0, e.y1], blocks: e.blocks, mag: +e.mag.toFixed(1) })) } : {}),
  findings: rows,
  result: hard.length ? "FAIL" : open.length ? "CHECK" : "PASS",
};
fs.writeFileSync(path.join(outDir, "qa-video.json"), JSON.stringify(report, null, 2) + "\n");

console.log(`qa-video ${path.basename(file)}: ${N} frames @ ${fps.toFixed(2)} fps, ${planned ? `${plannedF.length} planned cuts` : `${detected.length} cuts detected (no plan given — check each is intended)`}`);
for (const r of rows.slice(0, 30)) console.log(`  ${r.tier.padEnd(5)} ${r.id.padEnd(15)} ${r.from.toFixed(2)}–${r.to.toFixed(2)} s, ${r.where}, strength ${r.strength}${r.strip ? ` → ${path.join(outDir, r.strip)}` : ""}${r.verdict ? `  [${r.verdict}]` : ""}`);
if (rows.length > 30) console.log(`  … ${rows.length - 30} more in ${path.join(outDir, "qa-video.json")}`);
if (info.particles) console.log(`  note: ${info.particles} changes inside a particle field (rain, sparks…) were left to the eye — look at those shots in the review`);
if (allowedEv.length) console.log(`  allowed by --allow: ${allowedEv.length} (justify each in the review)`);
if (skippedEv.length) console.log(`  ${skippedEv.length} change${skippedEv.length > 1 ? "s" : ""} during fast camera moves not scanned (the scene checks cover them)`);
if (hard.length) console.log(`qa-video: FAIL (${hard.length}) — a real glitch: open the strips, fix the cause, render again`);
else if (open.length) console.log(`qa-video: CHECK (${open.length}) — open each strip; in ${reviewFile} write "ok: <why it is intended>" for designed changes, fix the rest, then run qa-video again`);
else console.log("qa-video: PASS");
process.exit(hard.length ? 1 : open.length ? 3 : 0);

// Final delivery, cross-platform and restartable: bundle once, render the composition muted in
// segments (finished segments are kept and skipped when re-run — delete out/seg after a source
// edit), join them, lay the voice-over in at its offset, then master the audio to -14 LUFS.

// join, the delivered file is scanned for hard glitches. Pass the film's planned cuts with --cuts.

//                                 [--cuts=s,s,…] [--qa-comp=<PreviewId>] [--no-qa] [--force-qa]

// Camera director: a spherical rig (target, azimuth, elevation, distance, fov) sampled from
// C1 keyframe tracks, so moves hand over without kinks and nothing depends on wall time.

import { sample, type Key } from "../motion/track.ts";
import type { Shot } from "../render/Post.tsx";

const D = Math.PI / 180;

/** Track channels: [targetX, targetY, targetZ, azimuthDeg, elevationDeg, distance, fovDeg]. */
export type RigKeys = Key[];

export const shotAt = (rig: RigKeys, frame: number, lens: { aperture: number; maxBlur: number; exposure: number }): Shot => {
  const [tx, ty, tz, az, el, dist, fov] = sample(rig, frame);
  const pos: [number, number, number] = [
    tx + dist * Math.cos(el * D) * Math.sin(az * D),
    ty + dist * Math.sin(el * D),
    tz + dist * Math.cos(el * D) * Math.cos(az * D),
  ];
  return { pos, target: [tx, ty, tz], fov, focus: dist, ...lens };
};

// Parametric rounded slab with a FIXED topology: every parameter change only moves
// vertices, so a slab can morph continuously from a 1990s brick to a 2026 glass slab
// (or a sensor orb) with no remeshing, popping or flicker.
//
// Surface = outline (front view, rounded rectangle with per-corner radius) × side profile
// (front cap → front fillet → side wall with optional seam groove → back fillet → back cap).
// The outline is offset inward by the profile inset; corners use the exact inward offset of
// a rounded corner (radius rc − inset, mitred once the inset exceeds rc), so the same
// generator makes sharp-cornered modules, rounded phones and — with rc = hw = hh and
// fillets = d/2 — a sphere. Normals are analytic (sharp edges stay sharp).

import { BufferAttribute, BufferGeometry, Sphere, Vector3 } from "three";

export type SlabSpec = {
  /** Half width (x), half height (y), full depth (z). */
  hw: number;
  hh: number;
  d: number;
  /** Outline corner radius, or per corner [TR, TL, BL, BR]. */
  rc: number | readonly [number, number, number, number];
  /** Front / back edge fillet radii. */
  reF: number;
  reB: number;
  /** Groove around the side wall (z of its centre, depth, width). */
  seamZ?: number;
  seamDepth?: number;
  seamWidth?: number;
  /** Dome height of the front / back caps. */
  bulgeF?: number;
  bulgeB?: number;
  /** Offset added to the shader-space position attribute `tfPos` (e.g. a module's place in its parent). */
  origin?: readonly [number, number];
  /** Planar UV: u = (x + origin.x) / uvSize.x + 0.5. */
  uvSize?: readonly [number, number];
};

const N_CORNER = 10; // segments per corner arc
const CAP_RHO = [0, 0.38, 0.68, 0.88]; // cap rings (edge ring comes from the fillet)
const N_FILLET = 8;
const GROOVE_U = [0.85, 0.5, 0.18, 0, -0.18, -0.5, -0.85];

const P = 4 * (N_CORNER + 1);
type Row = { kind: "capF" | "filletF" | "wall" | "filletB" | "capB"; a: number };
const ROWS: Row[] = [
  ...CAP_RHO.map((a) => ({ kind: "capF" as const, a })),
  ...Array.from({ length: N_FILLET + 1 }, (_, k) => ({ kind: "filletF" as const, a: (k / N_FILLET) * (Math.PI / 2) })),
  ...GROOVE_U.map((a) => ({ kind: "wall" as const, a })),
  ...Array.from({ length: N_FILLET + 1 }, (_, k) => ({ kind: "filletB" as const, a: ((N_FILLET - k) / N_FILLET) * (Math.PI / 2) })),
  ...[...CAP_RHO].reverse().map((a) => ({ kind: "capB" as const, a })),
];
const V = ROWS.length;
/** Band index ranges per material group: front cap, side (fillets + wall), back cap. */
const FRONT_END = CAP_RHO.length; // bands [0, FRONT_END)
const BACK_START = V - 1 - CAP_RHO.length; // bands [BACK_START, V-1)

const SIGN: [number, number][] = [
  [1, 1],
  [-1, 1],
  [-1, -1],
  [1, -1],
];

/** Outline point at inset i for sample c (0..P-1); writes [x, y, nx, ny]. */
const outline = (hw: number, hh: number, rc: readonly number[], i: number, c: number, out: number[]) => {
  const k = Math.floor(c / (N_CORNER + 1));
  const j = c % (N_CORNER + 1);
  const th = (k + j / N_CORNER) * (Math.PI / 2);
  const m = Math.max(rc[k], i);
  const r = Math.max(rc[k] - i, 0);
  const [sx, sy] = SIGN[k];
  const cx = sx * Math.max(hw - m, 0);
  const cy = sy * Math.max(hh - m, 0);
  out[0] = cx + r * Math.cos(th);
  out[1] = cy + r * Math.sin(th);
  out[2] = Math.cos(th);
  out[3] = Math.sin(th);
};

export class Slab {
  readonly geometry: BufferGeometry;
  private pos: Float32Array;
  private nor: Float32Array;
  private uv: Float32Array;
  private tf: Float32Array;

  constructor() {
    const n = P * V;
    this.pos = new Float32Array(n * 3);
    this.nor = new Float32Array(n * 3);
    this.uv = new Float32Array(n * 2);
    this.tf = new Float32Array(n * 3);
    const idx: number[] = [];
    for (let r = 0; r < V - 1; r++)
      for (let c = 0; c < P; c++) {
        const c1 = (c + 1) % P;
        const a = r * P + c, b = (r + 1) * P + c, d = (r + 1) * P + c1, e = r * P + c1;
        idx.push(a, b, d, a, d, e);
      }
    const g = new BufferGeometry();
    g.setIndex(idx);
    g.setAttribute("position", new BufferAttribute(this.pos, 3));
    g.setAttribute("normal", new BufferAttribute(this.nor, 3));
    g.setAttribute("uv", new BufferAttribute(this.uv, 2));
    g.setAttribute("tfPos", new BufferAttribute(this.tf, 3));
    const per = P * 6;
    g.addGroup(0, FRONT_END * per, 0);
    g.addGroup(FRONT_END * per, (BACK_START - FRONT_END) * per, 1);
    g.addGroup(BACK_START * per, (V - 1 - BACK_START) * per, 2);
    g.boundingSphere = new Sphere(new Vector3(), 1);
    this.geometry = g;
  }

  update(s: SlabSpec) {
    const rc = typeof s.rc === "number" ? [s.rc, s.rc, s.rc, s.rc] : [...s.rc];
    const half = s.d / 2;
    // fillets can never exceed the depth or the outline
    let reF = Math.max(0, Math.min(s.reF, s.d, s.hw, s.hh));
    let reB = Math.max(0, Math.min(s.reB, s.d, s.hw, s.hh));
    if (reF + reB > s.d) {
      const k = s.d / (reF + reB);
      reF *= k;
      reB *= k;
    }
    for (let k = 0; k < 4; k++) rc[k] = Math.max(0, Math.min(rc[k], s.hw, s.hh));
    const zTop = half - reF, zBot = -half + reB;
    const wallLen = zTop - zBot;
    const w = Math.min(s.seamWidth ?? 0.1, wallLen * 0.9);
    const seamZ = Math.min(zTop - w / 2, Math.max(zBot + w / 2, s.seamZ ?? (zTop + zBot) / 2));
    const g = Math.min(s.seamDepth ?? 0, Math.min(s.hw, s.hh) * 0.5);
    const bF = s.bulgeF ?? 0, bB = s.bulgeB ?? 0;
    const [ox, oy] = s.origin ?? [0, 0];
    const [us, vs] = s.uvSize ?? [2 * s.hw, 2 * s.hh];
    const o = [0, 0, 0, 0];
    const edge = [0, 0, 0, 0];

    for (let r = 0; r < V; r++) {
      const row = ROWS[r];
      for (let c = 0; c < P; c++) {
        let x: number, y: number, z: number, nx: number, ny: number, nz: number;
        if (row.kind === "capF" || row.kind === "capB") {
          const front = row.kind === "capF";
          outline(s.hw, s.hh, rc, front ? reF : reB, c, edge);
          const rho = row.a;
          x = edge[0] * rho;
          y = edge[1] * rho;
          const R2 = Math.max(edge[0] * edge[0] + edge[1] * edge[1], 1e-9);
          const b = front ? bF : bB;
          z = front ? half + b * (1 - rho * rho) : -half - b * (1 - rho * rho);
          // dome slope: n ∝ (2b·r/R² · radial, ±1)
          const k = (2 * b) / R2;
          nx = k * x;
          ny = k * y;
          nz = front ? 1 : -1;
        } else {
          let inset: number, nOut: number, nZ: number;
          if (row.kind === "filletF") {
            inset = reF * (1 - Math.sin(row.a));
            z = zTop + reF * Math.cos(row.a);
            nOut = Math.sin(row.a);
            nZ = Math.cos(row.a);
          } else if (row.kind === "filletB") {
            inset = reB * (1 - Math.sin(row.a));
            z = zBot - reB * Math.cos(row.a);
            nOut = Math.sin(row.a);
            nZ = -Math.cos(row.a);
          } else {
            const u = row.a;
            z = seamZ + (w / 2) * u;
            inset = (g * (1 + Math.cos(Math.PI * u))) / 2;
            const slope = w > 1e-6 ? (-g * Math.PI * Math.sin(Math.PI * u)) / w : 0;
            const l = Math.hypot(1, slope);
            nOut = 1 / l;
            nZ = slope / l;
          }
          outline(s.hw, s.hh, rc, inset, c, o);
          x = o[0];
          y = o[1];
          nx = nOut * o[2];
          ny = nOut * o[3];
          nz = nZ;
        }
        const l = Math.hypot(nx, ny, nz) || 1;
        const v = r * P + c;
        this.pos[v * 3] = x;
        this.pos[v * 3 + 1] = y;
        this.pos[v * 3 + 2] = z;
        this.nor[v * 3] = nx / l;
        this.nor[v * 3 + 1] = ny / l;
        this.nor[v * 3 + 2] = nz / l;
        this.uv[v * 2] = (x + ox) / us + 0.5;
        this.uv[v * 2 + 1] = (y + oy) / vs + 0.5;
        this.tf[v * 3] = x + ox;
        this.tf[v * 3 + 1] = y + oy;
        this.tf[v * 3 + 2] = z;
      }
    }
    const g0 = this.geometry;
    g0.attributes.position.needsUpdate = true;
    g0.attributes.normal.needsUpdate = true;
    g0.attributes.uv.needsUpdate = true;
    g0.attributes.tfPos.needsUpdate = true;
    g0.boundingSphere!.radius = Math.hypot(s.hw, s.hh, half + Math.max(bF, bB)) + 0.01;
  }

  dispose() {
    this.geometry.dispose();
  }
}

// Material transitions. A material's look is a plain number tree (MatState), so it can live
// inside era definitions. Between eras the change does not fade globally: a front travels
// along the object (top → bottom in object space, attribute `tfPos`) and each fragment flips
// from state A to state B as it passes, with a thin light band riding the front — the
// TIMEFORGE "time scan". Clearcoat, sheen and iridescence blend with the same progress.

import { Color, MeshPhysicalMaterial, SRGBColorSpace, type Side, type Texture } from "three";
import { patchSoftboxOnly } from "./softboxOnly.ts";

export type Rgb = [number, number, number];
export type MatState = {
  color: Rgb;
  rough: number;
  metal: number;
  cc: number;
  ccRough: number;
  sheen: number;
  irid: number;
  aniso: number;
  emissive: Rgb;
  emissiveI: number;
};

export const hex = (h: string): Rgb => [
  parseInt(h.slice(1, 3), 16) / 255,
  parseInt(h.slice(3, 5), 16) / 255,
  parseInt(h.slice(5, 7), 16) / 255,
];

export const mat = (color: string, o: Partial<Omit<MatState, "color">> = {}): MatState => ({
  color: hex(color),
  rough: 0.5,
  metal: 0,
  cc: 0,
  ccRough: 0.1,
  sheen: 0,
  irid: 0,
  aniso: 0,
  emissive: [0, 0, 0],
  emissiveI: 0,
  ...o,
});

/** Where the scan front is: 0 = not started (all A), 1 = done (all B). */
export type Sweep = { progress: number; top: number; bottom: number; band: number };

type TfUniforms = {
  tfColor2: { value: Color };
  tfRough2: { value: number };
  tfMetal2: { value: number };
  tfFront: { value: number };
  tfSoft: { value: number };
  tfBand: { value: number };
  tfBandW: { value: number };
  tfBandColor: { value: Color };
};

const BAND_COLOR = new Color("#bfeff2"); // light cyan, never neon

type MatOpts = { map?: Texture; emissiveMap?: Texture; side?: Side; transparent?: boolean };
const defined = (o: MatOpts) => Object.fromEntries(Object.entries(o).filter(([, v]) => v !== undefined));

export const sweepMaterial = (o: MatOpts = {}) => {
  const m = new MeshPhysicalMaterial(defined(o));
  const u: TfUniforms = {
    tfColor2: { value: new Color() },
    tfRough2: { value: 0.5 },
    tfMetal2: { value: 0 },
    tfFront: { value: 1e4 },
    tfSoft: { value: 0.35 },
    tfBand: { value: 0 },
    tfBandW: { value: 0.09 },
    tfBandColor: { value: BAND_COLOR.clone() },
  };
  m.userData.tf = u;
  m.onBeforeCompile = (shader) => {
    patchSoftboxOnly(shader);
    Object.assign(shader.uniforms, u);
    shader.vertexShader = shader.vertexShader
      .replace("#include <common>", "#include <common>\nattribute vec3 tfPos;\nvarying vec3 vTf;")
      .replace("#include <begin_vertex>", "#include <begin_vertex>\nvTf = tfPos;");
    shader.fragmentShader = shader.fragmentShader
      .replace(
        "#include <common>",
        `#include <common>
uniform vec3 tfColor2; uniform float tfRough2; uniform float tfMetal2; uniform float tfFront; uniform float tfSoft;
uniform float tfBand; uniform float tfBandW; uniform vec3 tfBandColor;
varying vec3 vTf;`,
      )
      .replace(
        "#include <color_fragment>",
        "#include <color_fragment>\nfloat tfk = smoothstep(tfFront - tfSoft, tfFront + tfSoft, vTf.y);\ndiffuseColor.rgb = mix(diffuseColor.rgb, tfColor2, tfk);",
      )
      .replace("#include <roughnessmap_fragment>", "#include <roughnessmap_fragment>\nroughnessFactor = mix(roughnessFactor, tfRough2, tfk);")
      .replace("#include <metalnessmap_fragment>", "#include <metalnessmap_fragment>\nmetalnessFactor = mix(metalnessFactor, tfMetal2, tfk);")
      .replace(
        "#include <emissivemap_fragment>",
        "#include <emissivemap_fragment>\nfloat tfd = (vTf.y - tfFront) / tfBandW;\ntotalEmissiveRadiance += tfBandColor * tfBand * exp(-tfd * tfd);",
      );
  };
  m.customProgramCacheKey = () => "tf-sweep";
  return m;
};

/** A plain physical material that follows the same look rules (no sweep). */
export const plainMaterial = (o: MatOpts = {}) => {
  const m = new MeshPhysicalMaterial(defined(o));
  m.onBeforeCompile = (shader) => patchSoftboxOnly(shader);
  m.customProgramCacheKey = () => "tf-plain";
  return m;
};

const mix = (a: number, b: number, t: number) => a + (b - a) * t;

/** Apply A→B with a scan front at `sw.progress`; plain materials just take the blend. */
export const applyMat = (m: MeshPhysicalMaterial, A: MatState, B: MatState, sw: Sweep) => {
  const t = Math.min(1, Math.max(0, sw.progress));
  const u = m.userData.tf as TfUniforms | undefined;
  const lerp3 = (a: Rgb, b: Rgb, k: number): Rgb => [mix(a[0], b[0], k), mix(a[1], b[1], k), mix(a[2], b[2], k)];
  if (u) {
    m.color.setRGB(...A.color, SRGBColorSpace);
    m.roughness = A.rough;
    m.metalness = A.metal;
    u.tfColor2.value.setRGB(...B.color, SRGBColorSpace);
    u.tfRough2.value = B.rough;
    u.tfMetal2.value = B.metal;
    const soft = u.tfSoft.value;
    u.tfFront.value = mix(sw.top + soft, sw.bottom - soft, t);
    // the band lives only while the front is on the object
    u.tfBand.value = sw.band * Math.sin(Math.PI * t) ** 0.6;
  } else {
    m.color.setRGB(...lerp3(A.color, B.color, t), SRGBColorSpace);
    m.roughness = mix(A.rough, B.rough, t);
    m.metalness = mix(A.metal, B.metal, t);
  }
  m.clearcoat = mix(A.cc, B.cc, t);
  m.clearcoatRoughness = mix(A.ccRough, B.ccRough, t);
  m.sheen = mix(A.sheen, B.sheen, t);
  m.sheenRoughness = 0.5;
  m.sheenColor.setRGB(1, 1, 1);
  m.iridescence = mix(A.irid, B.irid, t);
  m.iridescenceIOR = 1.3;
  m.anisotropy = mix(A.aniso, B.aniso, t);
  m.emissive.setRGB(...lerp3(A.emissive, B.emissive, t), SRGBColorSpace);
  m.emissiveIntensity = mix(A.emissiveI, B.emissiveI, t);
};

import { DataTexture, LinearFilter, LinearMipmapLinearFilter, RepeatWrapping, RGBAFormat, SRGBColorSpace, type ColorSpace } from "three";

  const t = new DataTexture(data, w, h, RGBAFormat);
  t.colorSpace = colorSpace;
  t.generateMipmaps = true;
  t.minFilter = LinearMipmapLinearFilter;
  t.magFilter = LinearFilter;
  t.wrapS = t.wrapT = RepeatWrapping;
  t.anisotropy = 8;
  t.needsUpdate = true;
  return t;

type Rgb = [number, number, number];
const hex = (h: string): Rgb => [parseInt(h.slice(1, 3), 16), parseInt(h.slice(3, 5), 16), parseInt(h.slice(5, 7), 16)];

/**
 * Hex-packed micro perforation. Alpha carries the outline mask (stadium / circle) for
 * alpha-to-coverage, so edges stay anti-aliased under MSAA.
 */
export const perforation = ({
  w = 1024, h = 1024, across = 18, hole = 0.36, base = "#3a3a3d", rim = "#55565a", deep = "#0d0d0f",
  mask = "none" as "none" | "stadium" | "circle",
}) => {
  const data = new Uint8Array(w * h * 4);
  const pitch = w / across;
  const rowH = pitch * Math.sqrt(3) / 2;
  const B = hex(base), R = hex(rim), D = hex(deep);
  const r = pitch * hole;
  for (let y = 0; y < h; y++)
    for (let x = 0; x < w; x++) {
      const row = Math.round(y / rowH);
      const cy = row * rowH;
      const off = row % 2 ? pitch / 2 : 0;
      const cx = Math.round((x - off) / pitch) * pitch + off;
      // nearest of the two candidate rows keeps the hex lattice exact
      let d = Math.hypot(x - cx, y - cy);
      const row2 = row + (y > cy ? 1 : -1);
      const off2 = row2 % 2 ? pitch / 2 : 0;
      const cx2 = Math.round((x - off2) / pitch) * pitch + off2;
      d = Math.min(d, Math.hypot(x - cx2, y - row2 * rowH));
      const inHole = Math.min(1, Math.max(0, (r - d) / 1.2));
      const onRim = Math.min(1, Math.max(0, 1 - Math.abs(d - r - 1.2) / 1.6));
      const c = B.map((b, i) => b + (R[i] - b) * onRim * 0.8 + (D[i] - b) * inHole);
      let a = 255;
      if (mask !== "none") {
        const u = (x + 0.5) / w - 0.5, v = (y + 0.5) / h - 0.5;
        let dist: number;
        if (mask === "circle") dist = 0.5 - Math.hypot(u, v);
        else {
          // stadium in a w×h box (aspect-correct)
          const aspect = w / h;
          const hx = 0.5 * aspect - 0.5;
          const px = Math.max(Math.abs(u * aspect) - hx, 0);
          dist = 0.5 - Math.hypot(px, v);
        }
        a = Math.round(Math.min(1, Math.max(0, dist * h * 0.6 + 0.5)) * 255);
      }
      const k = (y * w + x) * 4;
      data[k] = c[0];
      data[k + 1] = c[1];
      data[k + 2] = c[2];
      data[k + 3] = a;
    }
  return toTexture(data, w, h);
};

/** Seeded PRNG (Park–Miller). */
export const prng = (seed: number) => () => (seed = (seed * 16807) % 2147483647) / 2147483647;

/** Circuit board: graphite substrate, routed traces, vias and champagne pads. */
export const circuitBoard = (size = 1024, seed = 11) => {
  const rnd = prng(seed);
  const S = size;
  const data = new Uint8Array(S * S * 4);
  const base = hex("#2c2d30"), trace = hex("#45474b"), pad = hex("#c8b48c"), via = hex("#8d8a84");
  const set = (x: number, y: number, c: Rgb, a = 1) => {
    if (x < 0 || y < 0 || x >= S || y >= S) return;
    const k = (y * S + x) * 4;
    for (let i = 0; i < 3; i++) data[k + i] = data[k + i] + (c[i] - data[k + i]) * a;
  };
  for (let i = 0; i < S * S; i++) {
    const n = (rnd() - 0.5) * 6;
    data[i * 4] = base[0] + n;
    data[i * 4 + 1] = base[1] + n;
    data[i * 4 + 2] = base[2] + n;
    data[i * 4 + 3] = 255;
  }
  const disc = (cx: number, cy: number, r: number, c: Rgb, hollow = 0) => {
    for (let y = Math.floor(cy - r - 1); y <= cy + r + 1; y++)
      for (let x = Math.floor(cx - r - 1); x <= cx + r + 1; x++) {
        const d = Math.hypot(x - cx, y - cy);
        const a = Math.min(1, Math.max(0, r - d + 0.5)) * (hollow ? Math.min(1, Math.max(0, d - hollow + 0.5)) : 1);
        if (a > 0) set(x, y, c, a);
      }
  };
  const line = (x0: number, y0: number, x1: number, y1: number, wdt: number) => {
    const L = Math.hypot(x1 - x0, y1 - y0);
    for (let s = 0; s <= L; s += 0.5) {
      const x = x0 + ((x1 - x0) * s) / L, y = y0 + ((y1 - y0) * s) / L;
      disc(x, y, wdt / 2, trace);
    }
  };
  // Manhattan + 45° routing from random pins
  for (let i = 0; i < 70; i++) {
    let x = rnd() * S, y = rnd() * S;
    const steps = 2 + Math.floor(rnd() * 3);
    const w = 3 + rnd() * 4;
    for (let s = 0; s < steps; s++) {
      const dir = Math.floor(rnd() * 8) * (Math.PI / 4);
      const len = 40 + rnd() * 180;
      const nx = x + Math.cos(dir) * len, ny = y + Math.sin(dir) * len;
      line(x, y, nx, ny, w);
      x = nx;
      y = ny;
    }
    disc(x, y, w * 1.6, via, w * 0.7);
  }
  for (let i = 0; i < 40; i++) disc(rnd() * S, rnd() * S, 5 + rnd() * 5, pad);
  return toTexture(data, S, S);
};

// PRECISION easing vocabulary. Every curve is a pure function of normalised time, so the
// whole film stays deterministic: same frame in, same pose out.

export const clamp = (x: number, lo = 0, hi = 1) => Math.min(hi, Math.max(lo, x));
export const mix = (a: number, b: number, t: number) => a + (b - a) * t;
export const remap = (x: number, a: number, b: number) => clamp((x - a) / (b - a));

export const smoothstep = (t: number) => {
  const x = clamp(t);
  return x * x * (3 - 2 * x);
};

/** C2-continuous: zero velocity and acceleration at both ends. The house curve. */
export const smootherstep = (t: number) => {
  const x = clamp(t);
  return x * x * x * (x * (x * 6 - 15) + 10);
};

/** CSS-style cubic bezier easing (x1,y1,x2,y2), solved by Newton + bisection. */
export const bezier = (x1: number, y1: number, x2: number, y2: number) => {
  const cx = 3 * x1, bx = 3 * (x2 - x1) - cx, ax = 1 - cx - bx;
  const cy = 3 * y1, by = 3 * (y2 - y1) - cy, ay = 1 - cy - by;
  const sx = (t: number) => ((ax * t + bx) * t + cx) * t;
  const sy = (t: number) => ((ay * t + by) * t + cy) * t;
  const dx = (t: number) => (3 * ax * t + 2 * bx) * t + cx;
  return (x: number) => {
    const X = clamp(x);
    let t = X;
    for (let i = 0; i < 8; i++) {
      const e = sx(t) - X;
      const d = dx(t);
      if (Math.abs(e) < 1e-6) return sy(t);
      if (Math.abs(d) < 1e-6) break;
      t -= e / d;
    }
    let lo = 0, hi = 1;
    t = X;
    for (let i = 0; i < 30; i++) {
      const v = sx(t);
      if (Math.abs(v - X) < 1e-6) break;
      if (v < X) lo = t;
      else hi = t;
      t = (lo + hi) / 2;
    }
    return sy(t);
  };
};

// Named curves, tuned for product cinematography (long, confident tails; no bounce).
export const ease = {
  /** Default move: decisive start, very long settle. */
  glide: bezier(0.45, 0, 0.12, 1),
  /** Symmetric, weighty. Camera and large objects. */
  inOut: bezier(0.65, 0, 0.35, 1),
  /** Soft symmetric for subtle drifts. */
  sine: (t: number) => 0.5 - 0.5 * Math.cos(Math.PI * clamp(t)),
  /** Starts with energy (a magnet letting go), long deceleration. */
  release: bezier(0.2, 0.7, 0.1, 1),
  /** Gathers speed and arrives fast (magnetic pull into a dock). */

// Keyframe tracks with C1 continuity. Interior keys pass through with Catmull-Rom tangents
// (so a camera never "kinks" when one move hands over to the next); `hold` keys and the
// ends get zero tangents (a soft ease in/out). Tangents are clamped Fritsch–Carlson style
// so a curve never overshoots its keys — no accidental bounces.

/**
 * `m` overrides the tangent (units per frame), e.g. to arrive at speed for a magnetic snap;
 * `stop` makes the key an impact: it is reached at `m` and left at rest.
 */
export type Key = { f: number; v: number[]; hold?: boolean; m?: number[]; stop?: boolean };

const hermite = (p0: number, p1: number, m0: number, m1: number, t: number, h: number) => {
  const t2 = t * t, t3 = t2 * t;
  return (2 * t3 - 3 * t2 + 1) * p0 + (t3 - 2 * t2 + t) * h * m0 + (-2 * t3 + 3 * t2) * p1 + (t3 - t2) * h * m1;
};

const tangent = (keys: Key[], i: number, c: number) => {
  const k = keys[i];
  if (k.m) return k.m[c];
  if (k.hold || i === 0 || i === keys.length - 1) return 0;
  const a = keys[i - 1], b = keys[i + 1];
  const m = (b.v[c] - a.v[c]) / (b.f - a.f);
  // Monotone clamp: no overshoot past either neighbour.
  const d0 = (k.v[c] - a.v[c]) / (k.f - a.f);
  const d1 = (b.v[c] - k.v[c]) / (b.f - k.f);
  if (d0 * d1 <= 0) return 0;
  const lim = 3 * Math.min(Math.abs(d0), Math.abs(d1));
  return Math.sign(m) * Math.min(Math.abs(m), lim);
};

/** Sample a multi-channel track at `frame`. */
export const sample = (keys: Key[], frame: number): number[] => {
  if (frame <= keys[0].f) return keys[0].v.slice();
  const last = keys[keys.length - 1];
  if (frame >= last.f) return last.v.slice();
  let i = 0;
  while (i < keys.length - 2 && frame >= keys[i + 1].f) i++;
  const a = keys[i], b = keys[i + 1];
  const h = b.f - a.f;
  const t = (frame - a.f) / h;
  return a.v.map((_, c) =>
    hermite(a.v[c], b.v[c], a.stop ? 0 : tangent(keys, i, c), tangent(keys, i + 1, c), t, h),
  );
};

/** Convenience: build keys from [frame, ...values] tuples; append "hold" to pin a key. */
export const keys = (...rows: (number | "hold")[][]): Key[] =>
  rows.map((r) => {
    const hold = r[r.length - 1] === "hold";
    const nums = r.filter((x): x is number => typeof x === "number");
    return { f: nums[0], v: nums.slice(1), hold };
  });

const r4 = (v: number) => Math.round(v * 1e4) / 1e4;

/** Closed (watertight) after welding equal positions: every edge shared by exactly two triangles. */
const closedOf = (g: BufferGeometry): boolean => {
  const pos = g.attributes.position;
  if (!pos || pos.count > 60000) return false;
  if (!g.boundingBox) g.computeBoundingBox();
  const s = g.boundingBox!.getSize(new Vector3()).length() || 1;
  const q = s * 1e-4;
  const ids = new Int32Array(pos.count);
  const weld = new Map<string, number>();
  for (let i = 0; i < pos.count; i++) {
    const k = `${Math.round(pos.getX(i) / q)},${Math.round(pos.getY(i) / q)},${Math.round(pos.getZ(i) / q)}`;
    let v = weld.get(k);
    if (v === undefined) weld.set(k, (v = weld.size));
    ids[i] = v;
  }
  const idx = g.index;
  const tri = idx ? idx.count / 3 : pos.count / 3;
  if (!tri) return false;
  const edges = new Map<number, number>();
  const W = weld.size + 1;
  const add = (a: number, b: number) => {
    if (a === b) return;
    const k = a < b ? a * W + b : b * W + a;
    edges.set(k, (edges.get(k) ?? 0) + 1);
  };
  for (let t = 0; t < tri; t++) {
    const a = ids[idx ? idx.getX(t * 3) : t * 3], b = ids[idx ? idx.getX(t * 3 + 1) : t * 3 + 1], c = ids[idx ? idx.getX(t * 3 + 2) : t * 3 + 2];
    add(a, b);
    add(b, c);
    add(c, a);
  }
  for (const n of edges.values()) if (n !== 2) return false;
  return true;
};

/** Up to `n` points spread over the surface by area (deterministic), in the geometry's own space. */
const surfaceSamples = (g: BufferGeometry, n = 600): Float32Array => {
  const pos = g.attributes.position, idx = g.index;
  const tri = idx ? idx.count / 3 : pos.count / 3;
  const v = (t: number, k: number, out: Vector3) => out.fromBufferAttribute(pos, idx ? idx.getX(t * 3 + k) : t * 3 + k);
  const a = new Vector3(), b = new Vector3(), c = new Vector3(), e1 = new Vector3(), e2 = new Vector3();
  const area = new Float64Array(tri);
  let total = 0;
  for (let t = 0; t < tri; t++) {
    v(t, 0, a), v(t, 1, b), v(t, 2, c);
    total += area[t] = e1.subVectors(b, a).cross(e2.subVectors(c, a)).length() / 2;
  }
  const out: number[] = [];
  // every vertex first (corners and thin limbs), then area-weighted points on the faces
  const stride = Math.max(1, Math.ceil(pos.count / (n / 2)));
  for (let i = 0; i < pos.count; i += stride) out.push(pos.getX(i), pos.getY(i), pos.getZ(i));
  let seed = 12345;
  const rnd = () => ((seed = (seed * 1103515245 + 12345) & 0x7fffffff) / 0x7fffffff);
  const want = Math.max(0, n - out.length / 3);
  if (total > 0)
    for (let t = 0, acc = 0, next = rnd() * (total / Math.max(1, want)); t < tri && out.length / 3 < n; t++) {
      acc += area[t];
      while (next <= acc && out.length / 3 < n) {
        let u = rnd(), w = rnd();
        if (u + w > 1) (u = 1 - u), (w = 1 - w);
        v(t, 0, a), v(t, 1, b), v(t, 2, c);
        out.push(a.x + (b.x - a.x) * u + (c.x - a.x) * w, a.y + (b.y - a.y) * u + (c.y - a.y) * w, a.z + (b.z - a.z) * u + (c.z - a.z) * w);
        next += total / Math.max(1, want);
      }
    }
  return new Float32Array(out);
};

const partsOf = (s: SolidNow): Part[] => {
  const parts: Part[] = [];
  for (const m of s.meshes) {
    const g = m.geometry as BufferGeometry;
    if (!g?.attributes?.position) continue;
    if ((m as SkinnedMesh).isSkinnedMesh || (g.morphAttributes?.position?.length ?? 0) > 0) {
      // bake the posed (skinned / morphed) shape in world space for this frame
      const gen = new StaticGeometryGenerator([m]);
      gen.applyWorldTransforms = true;
      const baked = gen.generate();
      let closed = closedCache.get(g);
      if (closed === undefined) closedCache.set(g, (closed = closedOf(g)));
      baked.computeBoundingBox();
      const info = { bvh: new MeshBVH(baked, { indirect: true }), closed, samples: surfaceSamples(baked) };
      parts.push({ info, toWorld: new Matrix4(), toLocal: new Matrix4(), box: baked.boundingBox!.clone(), mesh: m });
    } else if ((m as InstancedMesh).isInstancedMesh) {
      const im = m as InstancedMesh, info = rigidInfo(g);
      if (!g.boundingBox) g.computeBoundingBox();
      const mi = new Matrix4();
      for (let i = 0; i < Math.min(im.count, 256); i++) {
        im.getMatrixAt(i, mi);
        const w = new Matrix4().multiplyMatrices(m.matrixWorld, mi);
        parts.push({ info, toWorld: w, toLocal: w.clone().invert(), box: g.boundingBox!.clone().applyMatrix4(w), mesh: m });
      }
    } else {
      const info = rigidInfo(g);
      if (!g.boundingBox) g.computeBoundingBox();
      const w = m.matrixWorld.clone();
      parts.push({ info, toWorld: w, toLocal: w.clone().invert(), box: g.boundingBox!.clone().applyMatrix4(w), mesh: m });

const DIR = new Vector3(0.5774, 0.5774, 0.5774);
/** Deepest point of part `a`'s surface inside closed part `b`: depth (scene units), the point and b's surface there. */
const deepest = (a: Part, b: Part, limitBox: Box3) => {
  const s = a.info.samples, p = new Vector3(), pl = new Vector3(), ray = new Ray();
  let best = 0;
  let at: Vector3 | null = null, from: Vector3 | null = null;
  for (let i = 0; i < s.length; i += 3) {
    p.set(s[i], s[i + 1], s[i + 2]).applyMatrix4(a.toWorld);
    if (!limitBox.containsPoint(p) || !b.box.containsPoint(p)) continue;
    pl.copy(p).applyMatrix4(b.toLocal);
    ray.set(pl, DIR);
    const hits = b.info.bvh.raycast(ray, DoubleSide);
    if (hits.length % 2 === 0) continue;
    const c = b.info.bvh.closestPointToPoint(pl);
    if (!c) continue;
    const sw = c.point.clone().applyMatrix4(b.toWorld);
    const d = sw.distanceTo(p);
    if (d > best) (best = d), (at = sw), (from = p.clone());
  }
  return at ? { depth: best, at, from: from! } : null;
};

const backdrop = () =>
  new ShaderMaterial({
    side: BackSide,
    depthWrite: false,
    uniforms: {
      uTop: { value: new Color("#f3f1ec") },
      uMid: { value: new Color("#e7e4de") },
      uLow: { value: new Color("#d6d2cb") },
      uTint: { value: new Color("#ffffff") },
      uTintAmt: { value: 0 },
    },
    vertexShader: /* glsl */ `
      varying vec3 vWorld; varying vec4 vClip;
      void main(){ vec4 w = modelMatrix * vec4(position,1.0); vWorld = w.xyz; vClip = projectionMatrix * viewMatrix * w; gl_Position = vClip; }`,
    fragmentShader: /* glsl */ `
      uniform vec3 uTop; uniform vec3 uMid; uniform vec3 uLow; uniform vec3 uTint; uniform float uTintAmt;
      varying vec3 vWorld; varying vec4 vClip;
      void main(){
        vec2 s = vClip.xy / vClip.w;
        vec3 dir = normalize(vWorld - cameraPosition);
        float t = clamp(0.5 + 0.5 * (0.7 * s.y + 0.3 * dir.y * 1.6), 0.0, 1.0);
        vec3 c = mix(uLow, uMid, smoothstep(0.0, 0.55, t));
        c = mix(c, uTop, smoothstep(0.5, 1.0, t));

      }`,
  });

      plinthMat,
      ring: new TorusGeometry(P.r - 0.45, 0.03, 8, 192),

      blobGeo: new PlaneGeometry(1, 1),
      blobMat,
    };

  }, [look.keyAz, look.keyEl]);

  );

// In VOYNANN Motion the brand lives in overlays (title, kicker, end card, labels); the 3D world keeps the
// style's art direction unless the buyer explicitly asks for brand colours in the scene.
import brandFile from "../../brand.json";

export type BrandKit = {
  brand?: string; handle?: string; cta?: string; logo?: string; font?: string;
  colors?: { primary?: string; accent?: string; background?: string };
};
export const BRAND = brandFile as unknown as BrandKit;

const isHex = (c?: string): c is string => !!c && /^#([0-9a-f]{3}|[0-9a-f]{6})$/i.test(c.trim());
const lum = (c: string) => {
  let h = c.trim().slice(1);
  if (h.length === 3) h = [...h].map((x) => x + x).join("");
  const [r, g, b] = [0, 2, 4].map((i) => parseInt(h.slice(i, i + 2), 16) / 255)
    .map((v) => (v <= 0.03928 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4));
  return 0.2126 * r + 0.7152 * g + 0.0722 * b;
};
/** WCAG contrast ratio of two hex colours (1–21). */
export const contrast = (a: string, b: string) => {
  const [x, y] = [lum(a), lum(b)].sort((p, q) => q - p);
  return (x + 0.05) / (y + 0.05);
};
/** The brand accent when it stays readable on `background` (≥ 3:1), otherwise the style's own `fallback`. */
export const brandAccent = (background: string, fallback: string) => {
  const a = BRAND.colors?.accent;
  return isHex(a) && isHex(background) && contrast(a, background) >= 3 ? a : fallback;
};
/** Brand name for overlays, or the fallback text. */
export const brandName = (fallback = "") => BRAND.brand?.trim() || fallback;

import { continueRender, delayRender, staticFile } from "remotion";

// Default typeface: Manrope (SIL Open Font License, see public/fonts/OFL-manrope.txt).
// Bundled locally in public/fonts so rendering never depends on Google Fonts being reachable
// and every OS renders the same glyph metrics.
const FAMILY = "Manrope";
const WEIGHTS = ["200", "300", "400", "500", "600"] as const;

let ready: Promise<void> | null = null;
const load = (): Promise<void> => {
  if (ready) return ready;
  if (typeof document === "undefined") return (ready = Promise.resolve());
  const handle = delayRender("Loading bundled Manrope font");
  ready = Promise.all(
    WEIGHTS.map(async (w) => {
      const face = new FontFace(FAMILY, `url(${staticFile(`fonts/manrope-latin-${w}-normal.woff2`)}) format("woff2")`, {
        weight: w,
        style: "normal",
      });
      await face.load();
      (document.fonts as unknown as { add(f: FontFace): void }).add(face);
    }),
  ).then(
    () => continueRender(handle),
    (err) => {
      continueRender(handle);
      throw err;
    },
  );
  return ready;
};
load();

import { PerspectiveCamera, Vector3 } from "three";
import type { Shot } from "../render/Post.tsx";

/** Project a world point to pixel coordinates with exactly the film camera for this frame. */
export const projector = (shot: Shot, width: number, height: number) => {
  const cam = new PerspectiveCamera(shot.fov, width / height, 0.05, 1200);
  cam.position.set(...shot.pos);
  cam.up.set(Math.sin(shot.roll ?? 0), Math.cos(shot.roll ?? 0), 0);
  cam.lookAt(...shot.target);
  cam.updateProjectionMatrix();
  cam.updateMatrixWorld();
  return (p: Vector3): [number, number] => {
    const v = p.clone().project(cam);
    return [((v.x + 1) / 2) * width, ((1 - v.y) / 2) * height];
  };
};
