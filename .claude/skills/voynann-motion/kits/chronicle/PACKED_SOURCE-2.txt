// ---------------------------------------------------------------- helpers
/** Indonesian number words, for estimating how long a line takes to say. */
function terbilang(n) {
  const S = ["", "satu", "dua", "tiga", "empat", "lima", "enam", "tujuh", "delapan", "sembilan", "sepuluh", "sebelas"];
  if (n < 12) return S[n];
  if (n < 20) return terbilang(n - 10) + " belas";
  if (n < 100) return terbilang(Math.floor(n / 10)) + " puluh " + terbilang(n % 10);
  if (n < 200) return "seratus " + terbilang(n - 100);
  if (n < 1000) return terbilang(Math.floor(n / 100)) + " ratus " + terbilang(n % 100);
  if (n < 2000) return "seribu " + terbilang(n - 1000);
  if (n < 1e6) return terbilang(Math.floor(n / 1000)) + " ribu " + terbilang(n % 1000);
  if (n < 1e9) return terbilang(Math.floor(n / 1e6)) + " juta " + terbilang(n % 1e6);
  return String(n);
}
function spoken(text) {
  return text.replace(/\*/g, "").replace(/\d[\d.]*/g, (m) => { const n = parseInt(m.replace(/\./g, ""), 10); return Number.isFinite(n) ? terbilang(n) : m; });
}
/** Silences [start, end] in a voice file. */
function silences(file) {
  const txt = spawnSync("ffmpeg", ["-v", "info", "-i", file, "-af", "silencedetect=noise=-45dB:d=0.22", "-f", "null", "-"], { encoding: "utf8" }).stderr ?? "";
  const st = [...txt.matchAll(/silence_start: ([\d.]+)/g)].map((m) => +m[1]);
  const en = [...txt.matchAll(/silence_end: ([\d.]+)/g)].map((m) => +m[1]);
  return st.map((a, i) => [a, en[i] ?? Infinity]);
}

/** Speech runs [start, end] of a voice file (the gaps between silences ≥ 0.15 s: an expressive VO breathes short). */
function speechRuns(file, db = -45, minSil = 0.15) {
  const dur = parseFloat(sh("ffprobe", ["-v", "error", "-show_entries", "format=duration", "-of", "csv=p=0", file]));
  const txt = spawnSync("ffmpeg", ["-v", "info", "-i", file, "-af", `silencedetect=noise=${db}dB:d=${minSil}`, "-f", "null", "-"], { encoding: "utf8" }).stderr ?? "";
  const starts = [...txt.matchAll(/silence_start: ([\d.]+)/g)].map((m) => +m[1]);
  const ends = [...txt.matchAll(/silence_end: ([\d.]+)/g)].map((m) => +m[1]);
  const runs = [];
  let cur = 0;
  starts.forEach((s0, i) => { if (s0 - cur > 0.08) runs.push([cur, s0]); cur = ends[i] ?? dur; });
  if (dur - cur > 0.08) runs.push([cur, dur]);
  return runs;
}

/**
 * Re-seats whisper-timed words on the VO's speech runs (in place). Every run takes a block of consecutive words (or
 * none: a breath); a block's spoken length should fill its run, its first word should be near whisper's time (whisper
 * runs early, so early is cheap), and blocks like to start after punctuation. Inside a run, words are paced by
 * spoken length. Returns how many words moved more than 0.25 s.
 */
function wordsOntoRuns(file, times) {
  const runs = speechRuns(file);
  const lines = times.flat();
  const W = lines.flatMap((T, li) => T.words.map((w, k) => ({ ref: w, li, first: k === 0, t: w.t, c: spoken(w.w).replace(/[^\p{L}\p{N}]/gu, "").length + 2, punct: /[.,;:!?—–-]["”']?$/.test(w.w) || k === T.words.length - 1 })));
  const n = W.length, m = runs.length;
  if (!n || m < 2) return 0;
  const C = [0];
  for (const w of W) C.push(C[C.length - 1] + w.c);
  const speech = runs.reduce((s, [a, b]) => s + b - a, 0);
  const rho = speech / C[n]; // seconds per spoken char
  const INF = 1e18;
  const f = Array.from({ length: m + 1 }, () => new Float64Array(n + 1).fill(INF));
  const from = Array.from({ length: m + 1 }, () => new Int32Array(n + 1).fill(-2));
  f[0][0] = 0;
  for (let r = 1; r <= m; r++) {
    const [s, e] = runs[r - 1], d = e - s;
    for (let j = 0; j <= n; j++) {
      // skip run r (a breath or a noise): cheap when short
      if (f[r - 1][j] < INF) { const c = f[r - 1][j] + 0.4 + 6 * d * d; if (c < f[r][j]) { f[r][j] = c; from[r][j] = -1; } }
    }
    for (let k = 0; k < n; k++) {
      if (f[r - 1][k] >= INF) continue;
      const lead = W[k].t - s; // whisper start of the block's first word vs the run start
      const hint = lead < -1.6 ? 1 + 2 * (-1.6 - lead) : lead < 0 ? 0.12 * -lead : lead < 0.35 ? 0.3 * lead : 0.1 + 2 * (lead - 0.35);
      const brk = k === 0 || W[k - 1].punct ? 0 : 0.35;
      for (let j = k + 1; j <= n; j++) {
        const exp = (C[j] - C[k]) * rho;
        const err = (d - exp) / Math.max(0.6, exp);
        const c = f[r - 1][k] + 3 * err * err + hint + brk;
        if (c < f[r][j]) { f[r][j] = c; from[r][j] = k; }
        if (exp > d * 3 + 1) break;
      }
    }
  }
  if (f[m][n] >= INF) return 0;
  const blocks = [];
  for (let r = m, j = n; r > 0; r--) { const k = from[r][j]; if (k >= 0) { blocks.unshift({ r: r - 1, k, j }); j = k; } }
  let moved = 0;
  for (const { r, k, j } of blocks) {
    const [s, e] = runs[r];
    const tot = C[j] - C[k];
    for (let i = k; i < j; i++) {
      const t = s + ((e - s) * (C[i] - C[k])) / tot;
      if (Math.abs(t - W[i].ref.t) > 0.25) moved++;
      W[i].ref.t = +t.toFixed(3);
    }
    lines[W[j - 1].li].b = e; // a line ends where its last word's run ends (or later, if the line goes on)
  }
  for (const T of lines) { T.a = T.words[0].t; T.b = Math.max(T.b, T.words[T.words.length - 1].t + 0.2); }
  return moved;
}

// ---------------------------------------------------------------- no whisper: lines onto speech runs
function alignBySpeech(file, story) {
  const dur = parseFloat(sh("ffprobe", ["-v", "error", "-show_entries", "format=duration", "-of", "csv=p=0", file]));
  // silencedetect reports on stderr
  const txt = spawnSync("ffmpeg", ["-v", "info", "-i", file, "-af", "silencedetect=noise=-55dB:d=0.22", "-f", "null", "-"], { encoding: "utf8" }).stderr ?? "";
  const starts = [...txt.matchAll(/silence_start: ([\d.]+)/g)].map((m) => +m[1]);
  const ends = [...txt.matchAll(/silence_end: ([\d.]+)/g)].map((m) => +m[1]);
  // speech runs = the gaps between silences
  const runs = [];
  let cur = 0;
  starts.forEach((s0, i) => { if (s0 - cur > 0.08) runs.push([cur, s0]); cur = ends[i] ?? dur; });
  if (dur - cur > 0.08) runs.push([cur, dur]);
  if (!runs.length) runs.push([0, dur]);
  // cumulative speech time at the end of each run; candidate line boundaries = every pause between runs
  const cum = [0];
  for (const [a, b] of runs) cum.push(cum[cum.length - 1] + (b - a));
  const total = cum[cum.length - 1];

  } else {
    // fewer runs than lines: split the speech by text length
    let acc = 0;
    for (let i = 0; i < L; i++) { spans.push([acc, acc + chars[i] * rate]); acc += chars[i] * rate; }
  }
  const toReal = (st) => { let acc = 0; for (const [a, b] of runs) { if (acc + (b - a) >= st - 1e-9) return a + (st - acc); acc += b - a; } return runs[M - 1][1]; };

    return { a: toReal(s0 + 1e-6), b: toReal(s1), words };
  }));

const canvasTex = (w: number, h: number, paint: (g: CanvasRenderingContext2D) => void, rep: [number, number]) => {
  const c = document.createElement("canvas");
  c.width = w; c.height = h;
  paint(c.getContext("2d")!);
  const t = new CanvasTexture(c);
  t.colorSpace = SRGBColorSpace;
  t.wrapS = t.wrapT = RepeatWrapping;
  t.repeat.set(...rep);
  return t;
};
let planks: MeshStandardMaterial | null = null;

const C = (x: string) => new Color(x);
const mixC = (a: string, b: string, t: number) => "#" + C(a).lerp(C(b), t).getHexString();
const m = (a: number, b: number, t: number) => a + (b - a) * t;
const m3 = (a: V3, b: V3, t: number): V3 => [m(a[0], b[0], t), m(a[1], b[1], t), m(a[2], b[2], t)];

/** Blend two looks (light changes inside a shot: a flash, dusk falling, fire flaring). */
export const mixLook = (a: Look, b: Look, t: number): Look => {
  if (t <= 0) return a;
  if (t >= 1) return b;
  return {
    key: { pos: m3(a.key.pos, b.key.pos, t), color: mixC(a.key.color, b.key.color, t), intensity: m(a.key.intensity, b.key.intensity, t), shadow: a.key.shadow },
    hemi: { sky: mixC(a.hemi.sky, b.hemi.sky, t), ground: mixC(a.hemi.ground, b.hemi.ground, t), intensity: m(a.hemi.intensity, b.hemi.intensity, t) },
    fog: { color: mixC(a.fog.color, b.fog.color, t), density: m(a.fog.density, b.fog.density, t) },
    sky: {
      zenith: mixC(a.sky.zenith, b.sky.zenith, t), horizon: mixC(a.sky.horizon, b.sky.horizon, t), ground: mixC(a.sky.ground, b.sky.ground, t),
      sunDir: m3(a.sky.sunDir, b.sky.sunDir, t), sunColor: mixC(a.sky.sunColor, b.sky.sunColor, t), sunSize: m(a.sky.sunSize, b.sky.sunSize, t),
      glow: m(a.sky.glow, b.sky.glow, t), stars: m(a.sky.stars ?? 0, b.sky.stars ?? 0, t),
    },
    exposure: m(a.exposure, b.exposure, t),
    bloom: { strength: m(a.bloom.strength, b.bloom.strength, t), radius: m(a.bloom.radius, b.bloom.radius, t), threshold: m(a.bloom.threshold, b.bloom.threshold, t) },
    grade: {
      shadow: mixC(a.grade.shadow, b.grade.shadow, t), high: mixC(a.grade.high, b.grade.high, t), amount: m(a.grade.amount, b.grade.amount, t),
      contrast: m(a.grade.contrast, b.grade.contrast, t), vignette: m(a.grade.vignette, b.grade.vignette, t), grain: m(a.grade.grain, b.grade.grain, t),
    },
    fill: a.fill && b.fill ? { pos: m3(a.fill.pos, b.fill.pos, t), color: mixC(a.fill.color, b.fill.color, t), intensity: m(a.fill.intensity, b.fill.intensity, t) } : a.fill,
    outline: a.outline,
    near: a.near,
    far: a.far,
  };
};

    }`,
};

export const Stage: React.FC<{ shot: Shot; look: Look; frame: number; children: React.ReactNode }> = ({ shot, look, frame, children }) => {
  const { gl, scene, camera, size } = useThree();

  const rig = useMemo(() => {
    const target = new Object3D();
    const key = new DirectionalLight("#fff", 2);
    key.target = target;
    key.shadow.mapSize.set(4096, 4096);
    key.shadow.radius = 4;
    const hemi = new HemisphereLight("#fff", "#444", 1);
    const fill = new DirectionalLight("#fff", 0);
    fill.target = target;
    const sky = new Mesh(new SphereGeometry(1, 48, 24), new ShaderMaterial({ ...SkyShader, side: BackSide, depthWrite: false, fog: false }));
    sky.frustumCulled = false;
    sky.renderOrder = -10;
    sky.userData.noMask = true;
    return { target, key, hemi, fill, sky };
  }, []);

  const chain = useMemo(() => {
    const w = size.width, h = size.height;
    const rt = new WebGLRenderTarget(w, h, { type: HalfFloatType, samples: 4 });
    const composer = new EffectComposer(gl, rt);
    composer.setPixelRatio(1);
    composer.setSize(w, h);
    composer.addPass(new RenderPass(scene, camera));
    composer.addPass(new ShaderPass(SanitizeShader));
    const outline = new SilhouetteOutlinePass(scene, camera, w, h);
    composer.addPass(outline);
    const bloom = new UnrealBloomPass(new Vector2(w, h), 0.3, 0.5, 0.9);
    composer.addPass(bloom);
    composer.addPass(new OutputPass());
    const grade = new ShaderPass(GradeShader);
    (grade.uniforms.uRes.value as Vector2).set(w, h);
    composer.addPass(grade);
    return { composer, outline, bloom, grade };
  }, [gl, scene, camera, size.width, size.height]);

  useLayoutEffect(() => {
    const cam = camera as PerspectiveCamera;
    cam.position.set(...shot.pos);
    cam.up.set(Math.sin(shot.roll ?? 0), Math.cos(shot.roll ?? 0), 0);
    cam.lookAt(...shot.target);
    cam.fov = shot.fov;
    cam.near = look.near ?? 0.1;
    cam.far = look.far ?? 4000;
    cam.updateProjectionMatrix();
    cam.updateMatrixWorld();
    gl.toneMapping = NeutralToneMapping;
    gl.shadowMap.autoUpdate = false;
  }, [shot, camera, gl, look.near, look.far]);

  useFrame(() => {
    const L = look;
    const k = rig.key;
    const sh = L.key.shadow;
    const at = sh?.at ?? [0, 0, 0];
    rig.target.position.set(...at);
    rig.target.updateMatrixWorld();
    k.position.set(at[0] + L.key.pos[0], at[1] + L.key.pos[1], at[2] + L.key.pos[2]);
    k.color.set(L.key.color);
    k.intensity = L.key.intensity;
    k.castShadow = !!sh;
    if (sh) {
      const b = sh.box;
      const sc = k.shadow.camera;
      sc.left = -b; sc.right = b; sc.top = b; sc.bottom = -b;
      sc.near = 0.5;
      sc.far = sh.far ?? 400;
      sc.updateProjectionMatrix();
      k.shadow.bias = sh.bias ?? -0.0004;
      k.shadow.normalBias = 0.02;
    }
    k.updateMatrixWorld();
    rig.hemi.color.set(L.hemi.sky);
    rig.hemi.groundColor.set(L.hemi.ground);
    rig.hemi.intensity = L.hemi.intensity;
    const F = L.fill;
    rig.fill.intensity = F ? F.intensity : 0;
    if (F) { rig.fill.position.set(at[0] + F.pos[0], at[1] + F.pos[1], at[2] + F.pos[2]); rig.fill.color.set(F.color); }
    rig.fill.updateMatrixWorld();

    const fog = scene.fog as FogExp2 | null;
    if (!fog) scene.fog = new FogExp2(L.fog.color, L.fog.density);
    else { fog.color.set(L.fog.color); fog.density = L.fog.density; }
    scene.background = new Color(L.sky.horizon);

    const cam = camera as PerspectiveCamera;
    rig.sky.position.copy(cam.position);
    rig.sky.scale.setScalar((cam.far ?? 1000) * 0.9);
    rig.sky.updateMatrixWorld();
    const su = (rig.sky.material as ShaderMaterial).uniforms;
    su.uZenith.value.set(L.sky.zenith);
    su.uHorizon.value.set(L.sky.horizon);
    su.uGround.value.set(L.sky.ground);
    su.uSunDir.value.set(...L.sky.sunDir).normalize();
    su.uSun.value.set(L.sky.sunColor);
    su.uSunSize.value = L.sky.sunSize;
    su.uGlow.value = L.sky.glow;
    su.uStars.value = L.sky.stars ?? 0;

    gl.toneMappingExposure = L.exposure;
    chain.bloom.strength = L.bloom.strength;
    chain.bloom.radius = L.bloom.radius;
    chain.bloom.threshold = L.bloom.threshold;
    chain.outline.setRadius(size.width, L.outline?.width ?? 1);
    chain.outline.mat.uniforms.uHalo.value = L.outline?.halo ?? 0.35;
    chain.outline.mat.uniforms.uGain.value = L.outline?.gain ?? 1.5;
    const g = chain.grade.uniforms;
    g.uTime.value = frame % 997;
    g.uShadow.value.set(L.grade.shadow);
    g.uHigh.value.set(L.grade.high);
    g.uAmount.value = L.grade.amount;
    g.uContrast.value = L.grade.contrast;
    g.uVignette.value = L.grade.vignette;
    g.uGrain.value = L.grade.grain;
    gl.shadowMap.needsUpdate = true;
    chain.composer.render();
  }, 1);

  return (
    <>
      <primitive object={rig.target} />
      <primitive object={rig.key} />
      <primitive object={rig.hemi} />
      <primitive object={rig.fill} />
      <primitive object={rig.sky} />
      {children}
    </>
  );
};
