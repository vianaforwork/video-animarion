#!/usr/bin/env node
// CHRONICLE sound department — an ORIGINAL score + SFX synthesised from scratch (no samples), shaped by the plan.
//   node scripts/sound.mjs out/<id>/plan.json public/stories/<id>/mix.wav [vo file]
//
// Music: string-like pads (additive, slow swells), low drone, a gamelan-flavoured metallophone (inharmonic bars,
// pelog-like scale), frame drums / low hits, gong. One "cue" per scene picked from its kind / set / mood, with
// cross-fades, punctuation on cuts, dates, names and counters. The bed ducks under the voice.
// SFX: per-set ambiences (wind, sea, fire crackle + roar, forest birds, room tone, volcanic rumble) and events
// (sound-ring booms, route ticks, counter ticks, whooshes on cuts).
// Mix: VO + music + SFX → loudness-normalised (−14 LUFS, −1.5 dBTP) 48 kHz stereo WAV.

import fs from "node:fs";
import path from "node:path";
import os from "node:os";
import { execFileSync } from "node:child_process";

const [planFile, outFile, voFile] = process.argv.slice(2);
if (!planFile || !outFile) { console.error("usage: node scripts/sound.mjs <plan.json> <out.wav> [vo]"); process.exit(2); }
const plan = JSON.parse(fs.readFileSync(planFile, "utf8"));
const SR = 48000;
const DUR = plan.duration + 0.5;
const N = Math.ceil(DUR * SR);

// ---------------------------------------------------------------- buffers & helpers
const music = [new Float32Array(N), new Float32Array(N)];
const sfx = [new Float32Array(N), new Float32Array(N)];
let seed = 12345;
const rnd = () => { seed = (seed * 1664525 + 1013904223) >>> 0; return seed / 4294967296; };
const hz = (m) => 440 * 2 ** ((m - 69) / 12);
const TAB = 8192;
const SIN = new Float32Array(TAB + 1);
for (let i = 0; i <= TAB; i++) SIN[i] = Math.sin((i / TAB) * Math.PI * 2);
const sinp = (ph) => { const x = (ph - Math.floor(ph)) * TAB; const i = x | 0; return SIN[i] + (SIN[i + 1] - SIN[i]) * (x - i); };
const clamp = (x, a, b) => (x < a ? a : x > b ? b : x);
const smooth = (x) => { const k = clamp(x, 0, 1); return k * k * (3 - 2 * k); };

// biquad (RBJ) for noise shaping
const biquad = (type, f, q = 0.707) => {
  const w = (2 * Math.PI * f) / SR, c = Math.cos(w), s = Math.sin(w), al = s / (2 * q);
  let b0, b1, b2, a0, a1, a2;
  if (type === "lp") { b0 = (1 - c) / 2; b1 = 1 - c; b2 = (1 - c) / 2; a0 = 1 + al; a1 = -2 * c; a2 = 1 - al; }
  else if (type === "hp") { b0 = (1 + c) / 2; b1 = -(1 + c); b2 = (1 + c) / 2; a0 = 1 + al; a1 = -2 * c; a2 = 1 - al; }
  else { b0 = al; b1 = 0; b2 = -al; a0 = 1 + al; a1 = -2 * c; a2 = 1 - al; } // band-pass
  let x1 = 0, x2 = 0, y1 = 0, y2 = 0;
  return (x) => { const y = (b0 * x + b1 * x1 + b2 * x2 - a1 * y1 - a2 * y2) / a0; x2 = x1; x1 = x; y2 = y1; y1 = y; return y; };
};
const add = (buf, i, l, r) => { if (i >= 0 && i < N) { buf[0][i] += l; buf[1][i] += r; } };

// ---------------------------------------------------------------- instruments
/** Pad: additive "strings" chord over [t0, t1] with swells; detuned voices per side. */
const pad = (t0, t1, notes, gain, bright = 0.5, fadeIn = 1.2, fadeOut = 1.4) => {
  const i0 = Math.max(0, Math.floor(t0 * SR)), i1 = Math.min(N, Math.floor((t1 + fadeOut) * SR));
  const partials = 9;
  for (const m of notes) {
    for (const det of [-0.08, 0.07]) {
      const f = hz(m + det * 0.1) * (1 + det * 0.004);
      let ph = rnd();
      const side = det < 0 ? 0 : 1;
      for (let i = i0; i < i1; i++) {
        const t = i / SR;
        const env = smooth((t - t0) / fadeIn) * (t > t1 ? 1 - smooth((t - t1) / fadeOut) : 1) * (0.85 + 0.15 * Math.sin(t * 0.7 + m));
        const vib = 1 + 0.0025 * Math.sin(t * 5.1 + m);
        ph += (f * vib) / SR;
        let v = 0;
        for (let k = 1; k <= partials; k++) v += sinp(ph * k) * (1 / k) * Math.exp(-(k - 1) * (1.15 - bright));
        const s = v * env * gain * 0.06;
        if (side === 0) music[0][i] += s * 0.9, music[1][i] += s * 0.45;
        else music[1][i] += s * 0.9, music[0][i] += s * 0.45;
      }
    }
  }
};
/** Low drone with slow breathing. */
const drone = (t0, t1, m, gain) => {
  const i0 = Math.max(0, Math.floor(t0 * SR)), i1 = Math.min(N, Math.floor((t1 + 1.5) * SR));
  const f = hz(m);
  let p1 = 0, p2 = 0;
  for (let i = i0; i < i1; i++) {
    const t = i / SR;
    const env = smooth((t - t0) / 1.5) * (t > t1 ? 1 - smooth((t - t1) / 1.5) : 1) * (0.75 + 0.25 * Math.sin(t * 0.45));
    p1 += f / SR; p2 += (f * 2.003) / SR;
    const s = (sinp(p1) * 0.8 + sinp(p2) * 0.25 + sinp(p1 * 0.5) * 0.35) * env * gain * 0.12;
    music[0][i] += s; music[1][i] += s;
  }
};
/** Metallophone bar (saron / bonang-like): inharmonic partials, fast attack, ringing decay. */
const bar = (t, m, gain, pan = 0, decay = 1.6) => {
  const i0 = Math.floor(t * SR), len = Math.floor(decay * 3 * SR);
  const f = hz(m);
  const P = [[1, 1], [2.76, 0.42], [5.4, 0.2], [8.93, 0.09], [1.006, 0.5]];
  const ph = P.map(() => rnd());
  for (let j = 0; j < len; j++) {
    const tt = j / SR;
    let v = 0;
    for (let k = 0; k < P.length; k++) v += sinp(ph[k] + f * P[k][0] * tt) * P[k][1] * Math.exp(-tt * (k === 0 || k === 4 ? 1 / decay : 3 / decay * (k + 1)));
    const a = Math.min(1, tt / 0.003);
    const s = v * a * gain * 0.09;
    add(music, i0 + j, s * (1 - pan) * 0.8, s * (1 + pan) * 0.8);
  }
};
/** Gong: very low inharmonic hum with slow beating. */
const gong = (t, gain, m = 38) => {
  const i0 = Math.floor(t * SR), len = Math.floor(7 * SR);
  const f = hz(m);
  const P = [[1, 1], [1.5, 0.35], [2.0, 0.3], [2.98, 0.12], [1.012, 0.6]];
  for (let j = 0; j < len; j++) {
    const tt = j / SR;
    let v = 0;
    for (const [r, a] of P) v += sinp(f * r * tt) * a;
    const env = Math.min(1, tt / 0.02) * Math.exp(-tt / 2.2);
    const s = v * env * gain * 0.13;
    add(music, i0 + j, s, s);
  }
};
/** Frame drum / low tom hit: pitch drop + filtered noise skin. */
const drum = (t, gain, low = 52, pan = 0) => {
  const i0 = Math.floor(t * SR), len = Math.floor(0.9 * SR);
  const bp = biquad("bp", 180, 1.2);
  let ph = 0;
  for (let j = 0; j < len; j++) {
    const tt = j / SR;
    const f = low + 70 * Math.exp(-tt * 28);
    ph += f / SR;
    const body = sinp(ph) * Math.exp(-tt * 6.5);
    const skin = bp(rnd() * 2 - 1) * Math.exp(-tt * 30) * 0.6;
    const s = (body + skin) * gain * 0.32;
    add(music, i0 + j, s * (1 - pan * 0.5), s * (1 + pan * 0.5));
  }
};
/** Big low impact (cuts, booms). */
const impact = (t, gain, buf = music) => {
  const i0 = Math.floor(t * SR), len = Math.floor(3 * SR);
  const lp = biquad("lp", 220, 0.8);
  let ph = 0;
  for (let j = 0; j < len; j++) {
    const tt = j / SR;
    ph += (38 + 60 * Math.exp(-tt * 9)) / SR;
    const s = (sinp(ph) * Math.exp(-tt * 2.2) + lp(rnd() * 2 - 1) * Math.exp(-tt * 3.5) * 0.8) * gain * 0.42;
    add(buf, i0 + j, s, s);
  }
};
/** Noise whoosh / riser into time t. */
const whoosh = (t, gain, len = 0.9, buf = sfx) => {
  const i1 = Math.floor(t * SR), i0 = i1 - Math.floor(len * SR);
  const bpL = biquad("bp", 900, 0.9), bpR = biquad("bp", 1100, 0.9);
  for (let i = i0; i < i1 + Math.floor(0.25 * SR); i++) {
    const k = (i - i0) / (i1 - i0);
    const env = k < 1 ? Math.pow(k, 2.2) : Math.exp(-(i - i1) / SR * 14);
    add(buf, i, bpL(rnd() * 2 - 1) * env * gain * 0.5, bpR(rnd() * 2 - 1) * env * gain * 0.5);
  }
};

// ---------------------------------------------------------------- ambiences
const amb = (t0, t1, kind, gain = 1) => {
  const i0 = Math.max(0, Math.floor((t0 - 0.3) * SR)), i1 = Math.min(N, Math.floor((t1 + 0.4) * SR));
  const env = (i) => { const t = i / SR; return smooth((t - t0 + 0.3) / 0.6) * (1 - smooth((t - t1) / 0.4)); };
  const L1 = biquad("lp", 400), R1 = biquad("lp", 400), L2 = biquad("bp", 1800, 0.6), R2 = biquad("bp", 2100, 0.6);
  const H = biquad("hp", 3000), lowL = biquad("lp", 90), lowR = biquad("lp", 90);
  for (let i = i0; i < i1; i++) {
    const t = i / SR, e = env(i) * gain;
    const nl = rnd() * 2 - 1, nr = rnd() * 2 - 1;
    let l = 0, r = 0;
    if (kind === "wind" || kind === "sea" || kind === "forest" || kind === "makam") {
      const g = 0.5 + 0.5 * Math.sin(t * 0.37) * Math.sin(t * 0.21 + 1);
      l += L1(nl) * 0.5 * (0.4 + g); r += R1(nr) * 0.5 * (0.4 + g * 0.8);
    }
    if (kind === "sea") {
      const wave = Math.pow(0.5 + 0.5 * Math.sin(t * 0.9), 3);
      l += L2(nl) * 0.22 * wave; r += R2(nr) * 0.22 * (0.5 + 0.5 * Math.sin(t * 0.9 + 0.8)) ** 3;
    }
    if (kind === "fire") {
      l += L1(nl) * 0.6 * (0.7 + 0.3 * Math.sin(t * 3.1)); r += R1(nr) * 0.6 * (0.7 + 0.3 * Math.sin(t * 2.7));
      if (rnd() < 0.0009) { const c = (rnd() * 2 - 1) * 0.9; l += c; r += c * 0.6; }
      l += H(nl) * 0.05;
    }
    if (kind === "rumble" || kind === "fire") { const rr = lowL(nl) * 2.2, rl = lowR(nr) * 2.2; l += rr; r += rl; }
    if (kind === "room") { l += L1(nl) * 0.12; r += R1(nr) * 0.12; }
    add(sfx, i, l * e * 0.5, r * e * 0.5);
  }
  if (kind === "forest" || kind === "makam") {
    // birds: short frequency-swept chirps
    for (let t = t0 + 0.5; t < t1; t += 0.7 + rnd() * 1.8) {
      const f0 = 2600 + rnd() * 1800, i = Math.floor(t * SR), len = Math.floor((0.06 + rnd() * 0.08) * SR), pan = rnd() * 2 - 1;
      const n = 1 + Math.floor(rnd() * 3);
      for (let c = 0; c < n; c++) {
        let ph = 0;
        for (let j = 0; j < len; j++) {
          const k = j / len; ph += (f0 * (1 + 0.35 * Math.sin(k * Math.PI)) + c * 120) / SR;
          const s = sinp(ph) * Math.sin(k * Math.PI) * 0.03 * Math.min(gain, 1.6);
          add(sfx, i + j + c * Math.floor(0.11 * SR), s * (1 - pan) * 0.7, s * (1 + pan) * 0.7);
        }
      }
    }
  }
};

// ---------------------------------------------------------------- the score
const ROOT = 50; // D3
const PELOG = [0, 1, 3, 7, 8]; // D Eb F A Bb — a pelog-like five-note colour
const scale = (deg, oct = 0) => ROOT + 12 + PELOG[((deg % 5) + 5) % 5] + 12 * (Math.floor(deg / 5) + oct);
const PROG = {
  reflective: [[50, 57, 62, 65, 69], [46, 53, 58, 62, 65], [43, 50, 55, 58, 62], [45, 52, 57, 61, 64]],
  tense: [[50, 51, 57, 62], [49, 50, 56, 62], [50, 51, 57, 63], [46, 50, 57, 62]],
  curious: [[50, 57, 62, 64, 69], [48, 55, 60, 64, 67], [46, 53, 58, 62, 65], [45, 52, 57, 60, 64]],
  determined: [[50, 57, 62, 65], [48, 55, 60, 63], [46, 53, 58, 62], [48, 55, 60, 64]],
  somber: [[50, 57, 60, 65], [46, 53, 58, 61], [43, 50, 55, 58], [45, 52, 57, 60]],
};
const cueOf = (sc) => {
  if (sc.kind === "end") return { prog: "reflective", pad: 0.9, drone: 0.6, drums: 0, metal: 0.35, bright: 0.45 };
  if (sc.kind === "map") return sc.map?.route ? { prog: "determined", pad: 0.7, drone: 0.7, drums: 0.5, bpm: 92, metal: 0.6, bright: 0.5 }
    : sc.map?.rings || sc.map?.erupt ? { prog: "tense", pad: 0.8, drone: 1.0, drums: 0.35, bpm: 70, metal: 0.25, bright: 0.4 }
    : { prog: "curious", pad: 0.75, drone: 0.6, drums: 0, metal: 0.55, bright: 0.55 };
  const p = sc.place;
  if (p.fire || p.volcano || p.mood === "api") return { prog: "tense", pad: 0.9, drone: 1.1, drums: 0.9, bpm: 104, metal: 0.3, bright: 0.55 };
  if (p.set === "kamar" || p.set === "makam") return { prog: p.set === "kamar" ? "somber" : "reflective", pad: 0.85, drone: 0.5, drums: 0, metal: 0.4, bright: 0.35 };
  if (p.set === "balai") return { prog: "tense", pad: 0.7, drone: 0.8, drums: 0.25, bpm: 64, metal: 0.25, bright: 0.35 };
  if (p.set === "hutan" || p.set === "medan") return { prog: "determined", pad: 0.85, drone: 0.8, drums: 0.6, bpm: 84, metal: 0.4, bright: 0.45 };
  return { prog: "curious", pad: 0.8, drone: 0.6, drums: 0.2, bpm: 80, metal: 0.45, bright: 0.5 };
};
const ambOf = (sc) => {
  if (sc.kind !== "place") return sc.kind === "map" && (sc.map?.rings || sc.map?.erupt) ? ["rumble", 0.9] : ["wind", 0.8];
  const p = sc.place;
  if (p.fire) return ["fire", 1.0];
  if (p.volcano) return ["rumble", 1.0];
  if (p.set === "balai" || p.set === "kamar") return ["room", 2.2];
  if (p.set === "pelabuhan" || p.set === "dek" || p.set === "pantai") return ["sea", 2.4];
  if (p.set === "hutan") return ["forest", 2.4];
  if (p.set === "makam") return ["makam", 2.2];
  return ["wind", 2.2];
};

plan.scenes.forEach((sc, si) => {
  const a = sc.a, b = sc.b, d = b - a;
  const cue = cueOf(sc);
  const prog = PROG[cue.prog];
  const barLen = Math.max(2.4, Math.min(4.2, d / 2));
  // pads: one chord per bar, overlapping tails
  for (let k = 0, t = a; t < b; k++, t += barLen) pad(Math.max(0, t - 0.3), Math.min(b, t + barLen), prog[(k + si) % prog.length], cue.pad, cue.bright, k === 0 ? 0.9 : 1.6, 1.6);
  if (cue.drone) drone(Math.max(0, a - 0.2), b, prog[si % prog.length][0] - 12, cue.drone);
  // drums: a low pulse (frame drum on beats 1 and 3½, soft taps between)
  if (cue.drums) {
    const beat = 60 / cue.bpm;
    let k = 0;
    for (let t = a + 0.05; t < b - 0.1; t += beat / 2, k++) {
      const pos = k % 8;
      if (pos === 0) drum(t, cue.drums, 48);
      else if (pos === 5) drum(t, cue.drums * 0.7, 54, 0.3);
      else if (pos === 3 || pos === 7) drum(t, cue.drums * 0.25, 70, -0.4);
    }
  }
  // metallophone: a sparse pelog motif, denser when the cue is lively
  if (cue.metal) {
    const step = cue.drums ? 60 / (cue.bpm || 80) : 0.75;
    const motif = [0, 2, 3, 1, 4, 3, 2, 0, 1, 3];
    let k = 0;
    for (let t = a + 0.4; t < b - 0.3; t += step, k++) {
      if (rnd() > cue.metal + 0.15) continue;
      const deg = motif[(k + si * 3) % motif.length];
      bar(t, scale(deg, deg > 3 ? 0 : 0), cue.metal * (k % 4 === 0 ? 1 : 0.6), (rnd() - 0.5) * 0.8, cue.drums ? 1.0 : 1.9);
    }
  }
  // ambience + transitions
  const [kind, g] = ambOf(sc);
  amb(a, b, kind, g);
  if (si > 0) whoosh(a, 0.5, 0.7);
  // map events
  if (sc.kind === "map") {
    if (sc.map.rings) [0.6, 1.05, 1.5].forEach((t0, i) => impact(a + t0, 0.9 - i * 0.2, sfx));
    if (sc.map.erupt) impact(a + 0.2, 0.8, sfx);
    if (sc.map.route) {
      const n = Math.max(1, sc.map.points.length - 1);
      for (let i = 1; i <= n; i++) bar(a + 0.15 * d + (0.7 * d * i) / n, scale(4, 1), 0.5, 0.2, 0.8);
    }
  }
});

/** All places named in a text, in order of appearance (no duplicates). */
export const findPlaces = (text: string): Place[] => {
  const t = " " + norm(text).replace(/[^a-z0-9\- ]/g, " ") + " ";
  const hits: { at: number; p: Place }[] = [];
  const taken: [number, number][] = [];
  for (const p of SORTED) {
    const n = " " + norm(p.name) + " ";
    let i = t.indexOf(n);
    while (i >= 0) {
      const a = i, b = i + n.length;
      if (!taken.some(([x, y]) => a < y && b > x)) {
        hits.push({ at: a, p });
        taken.push([a, b]);
      }
      i = t.indexOf(n, i + 1);
    }
  }
  hits.sort((a, b) => a.at - b.at);
  const out: Place[] = [];
  for (const h of hits) if (!out.some((o) => o.name === h.p.name)) out.push(h.p);
  return out;
};

const splitTop = (s: string) => {
  const out: string[] = [];
  let depth = 0, cur = "";
  for (const ch of s) {
    if (ch === "(") depth++;
    if (ch === ")") depth--;
    if ((ch === "," || ch === ";") && depth === 0) { out.push(cur); cur = ""; } else cur += ch;
  }
  if (cur.trim()) out.push(cur);
  return out.map((x) => x.trim()).filter(Boolean);
};

export const parseCast = (tokoh: string, era: number): Group[] => {
  const groups: Group[] = [];
  for (const part0 of splitTop(tokoh)) {
    let part = part0;
    let color: string | null = null;
    const eq = /=\s*([^=]+)$/.exec(part);
    if (eq) {
      const c = COLORS.find(([re]) => re.test(eq[1].toLowerCase()));
      color = c ? c[1] : null;
      part = part.slice(0, eq.index).trim();
    }
    const attrs = [...part.matchAll(/\(([^)]*)\)/g)].map((m) => m[1]).join(", ");
    part = part.replace(/\([^)]*\)/g, " ").replace(/\s+/g, " ").trim();
    const m = /^(\d+|[a-z]+)\s+(.*)$/i.exec(part);
    let count = 1, rest = part;
    if (m) {
      const n = /^\d+$/.test(m[1]) ? parseInt(m[1], 10) : NUM[m[1].toLowerCase()];
      if (n) { count = n; rest = m[2]; }
    }
    const low = (rest + " " + attrs).toLowerCase();
    let action: Action | null = null;
    let actionWord = "";
    for (const [re, a] of ACTIONS) {
      const mm = re.exec(rest.toLowerCase());
      if (mm) { action = a; actionWord = mm[0]; break; }
    }
    const role = ROLES.find(([re]) => re.test(low));
    const costume: CostumeSpec = role ? role[1](era) : { hat: "destar", wear: "sarong" };
    for (const [re, f] of ATTRS) if (re.test(attrs.toLowerCase())) f(costume);
    if (!action) action = role?.[2] ?? "stand";
    let name = rest;
    if (actionWord) name = name.replace(new RegExp(actionWord, "i"), "").trim();
    groups.push({ count: Math.max(1, Math.min(40, count)), name, costume, action, color, lead: false });
  }
  const lead = groups.find((g) => g.count === 1 && g.color) ?? groups.find((g) => g.count === 1) ?? groups[0];
  if (lead) lead.lead = true;
  return groups;
};

const parseCount = (t: string) => {
  const [main, label] = t.split(/,(.+)/).map((x) => x?.trim());
  const m = /([\d.,]+)\s*(?:→|->|ke)\s*([\d.,]+)\s*(.*)/.exec(main) ?? /([\d.,]+)\s*(.*)/.exec(main);

const splitDash = (t: string) => {
  const m = /\s+[—–-]\s+/.exec(t);
  return m ? [t.slice(0, m.index).trim(), t.slice(m.index + m[0].length).trim()] : [t.trim(), ""];

export type LineTime = { a: number; b: number; words?: { w: string; t: number }[] };
const WPS = 2.55;
const paceLines = (story: RawStory): LineTime[][] => {
  let t = 0.5;
  return story.scenes.map((sc, si) => {
    if (si > 0) t += 0.35;
    const isMap = sc.kind === "PETA" || sc.kind === "RUTE";
    const start = t;
    const out = sc.lines.map((l) => {
      const n = l.words.length;
      const d = Math.max(1.05, n / WPS + 0.12);
      const a = t, b = t + d;
      t = b + (/[.?!]$/.test(l.text) ? 0.42 : 0.18);
      return { a, b };
    });
    if (sc.kind === "PENUTUP") t += 0.2;
    // map scenes need time to travel
    const min = sc.kind === "RUTE" ? 4.4 : isMap ? 3.0 : 0;
    if (t - start < min) t = start + min;
    return out;
  });
};

export const planStory = (src: string, opts: { id: string; times?: LineTime[][]; audio?: string } = { id: "story" }): Plan => {
  const story = parseNaskah(src);
  const warnings: string[] = [];

  const allText = story.scenes.map((s) => s.lines.map((l) => l.text).join(" ") + " " + s.tags.map((t) => t.text).join(" ")).join(" ");
  const yearM = /\b(1[0-9]{3}|20[0-9]{2})\b/.exec(allText);

  const era = year ?? 1800;

  // scene boundaries: from just before the first line to just before the next scene's first line
  const firstA = (i: number) => times[i]?.[0]?.a;
  const scenes: PlanScene[] = [];
  const words: Word[] = [];
  const sources: string[] = [];
  let prevB = 0;
  story.scenes.forEach((sc, i) => {
    const lt = times[i];
    const lineStart = lt[0]?.a ?? prevB + 0.2;
    const a = i === 0 ? 0 : prevB;
    let nextA = Infinity;
    for (let j = i + 1; j < story.scenes.length; j++) { const f = firstA(j); if (f !== undefined) { nextA = f; break; } }
    const lastB = lt.length ? lt[lt.length - 1].b : lineStart + 1;
    let b = Number.isFinite(nextA) ? Math.max(lastB + 0.15, nextA - 0.3) : lastB + 0.8;
    if (sc.kind === "PENUTUP") b = Math.max(b, a + 4.2);
    const minDur = sc.kind === "RUTE" ? 4.4 : sc.kind === "PETA" ? 3.0 : 1.6;
    if (b - a < minDur && !opts.times) b = a + minDur;
    prevB = b;

    // captions
    sc.lines.forEach((l, li) => {
      const T = lt[li];
      if (T.words && T.words.length === l.words.length) l.words.forEach((w, k) => words.push({ w: w.w, t: T.words![k].t, key: w.key }));
      else {
        const tot = l.words.reduce((s, w) => s + 0.6 + w.w.length * 0.08, 0);
        let acc = 0;
        l.words.forEach((w) => { words.push({ w: w.w, t: T.a + ((T.b - T.a) * acc) / tot, key: w.key }); acc += 0.6 + w.w.length * 0.08; });
      }
    });

    // overlays
    const overlays: Overlay[] = [];
    const lineA = (k: number) => (lt[Math.min(k, lt.length - 1)]?.a ?? a + 0.3);
    for (const tg of sc.tags) {
      const at = tg.line === 0 ? a + 0.25 : lineA(tg.line) - 0.1;
      if (tg.kind === "TANGGAL") {
        const [top, big] = tg.text.includes("/") ? tg.text.split("/").map((x) => x.trim()) : ["", tg.text.trim()];
        overlays.push({ kind: "date", a: at, b: Math.min(at + 3.4, b - 0.1), top: top.toUpperCase(), big: big.toUpperCase() });
      } else if (tg.kind === "NAMA") {
        const [name, role] = splitDash(tg.text);
        overlays.push({ kind: "name", a: at + 0.15, b: Math.min(at + 3.8, b - 0.1), name, role: role || undefined });
      } else if (tg.kind === "ANGKA") {
        const c = parseCount(tg.text);
        if (c) overlays.push({ kind: "count", a: at, b: b - 0.1, from: c.from, to: c.to, roll: [at + 0.3, at + 1.7], unit: c.unit, label: c.label });
        else warnings.push(`ANGKA tidak terbaca: "${tg.text}"`);
      } else if (tg.kind === "SUMBER") {
        const [text, src] = splitDash(tg.text);
        if (sc.kind === "PENUTUP") sources.push(...tg.text.split(/;\s*/).map((x) => x.trim()).filter(Boolean));
        else {
          overlays.push({ kind: "cite", a: at + 0.2, b: b - 0.15, text: src ? text : "", src: src || text });
          if (src) sources.push(src);
        }
      }
    }

    const text = sc.lines.map((l) => l.text).join(" ");
    if (sc.kind === "PENUTUP") {
      // the closing card lists the sources written under [PENUTUP]; without them, every cited source in the film
      const own = sc.tags.filter((tg) => tg.kind === "SUMBER").flatMap((tg) => tg.text.split(/;\s*/).map((x) => x.trim()).filter(Boolean));
      scenes.push({ a, b, kind: "end", overlays: [...overlays.filter((o) => o.kind !== "cite"), { kind: "end", a: a + 0.2, b, title: sc.spec, sources: own.length ? own : [...new Set(sources)] }] });
      return;
    }
    let kind = sc.kind;
    if (kind === "AUTO") {
      // untagged paragraph: a place named in it → map, otherwise a location from its words
      const pl = findPlaces(text).filter((p) => p.kind !== "sea");
      kind = pl.length && !SET_RULES.some(([re]) => re.test(text.toLowerCase())) ? "PETA" : "TEMPAT";
      if (kind === "TEMPAT") sc.spec = text;
    }
    if (kind === "PETA" || kind === "RUTE") {
      let places: Place[] = findPlaces(sc.spec);
      if (!places.length) places = findPlaces(text);
      if (!places.length) warnings.push(`Peta tanpa nama tempat yang dikenal: "${sc.spec}"`);
      const route = kind === "RUTE" || /→|->/.test(sc.spec);
      const points: MapPoint[] = places.map((p) => ({ name: p.name, lon: p.lon, lat: p.lat, volcano: p.kind === "volcano" }));
      scenes.push({
        a, b, kind: "map", overlays,
        map: { points, route, rings: /cincin|gelombang|dentum/.test(sc.spec.toLowerCase()), erupt: /meletus|letusan/.test(sc.spec.toLowerCase()), mode: route ? "route" : points.length > 1 ? "fit" : "close" },
      });
      return;
    }
    const pl = resolvePlace(sc, text);
    const groups = sc.tokoh ? parseCast(sc.tokoh, era) : [];
    if (!groups.length) warnings.push(`Adegan tanpa TOKOH (dipakai figuran otomatis): "${sc.spec.slice(0, 40)}"`);
    // camera beats: one per line, merged when lines are short
    const solo = groups.length > 0 && groups.every((g) => g.count === 1);
    const cycle: Shot[] = solo ? ["wide", "medium", "close", "low", "medium"] : ["wide", "medium", "group", "close", "wide"];
    const beats: Beat[] = [];
    lt.forEach((T, k) => {
      const ba = k === 0 ? a : T.a - 0.15;
      if (beats.length && ba - beats[beats.length - 1].a < 1.7) return;
      if (beats.length) beats[beats.length - 1].b = ba;
      beats.push({ a: ba, b, shot: cycle[beats.length % cycle.length] });
    });
    if (!beats.length) beats.push({ a, b, shot: "wide" });
    // a counter needs the top of the frame clear: keep that beat wide
    for (const o of overlays) if (o.kind === "count") { const bt = beats.find((x) => o.a >= x.a && o.a < x.b); if (bt) bt.shot = "wide"; }
    scenes.push({ a, b, kind: "place", overlays, place: { ...pl, groups, beats, seed: i + 1 } });
  });

  // relief regions: each map scene gets its own (fine coastlines even when a film's places are far apart), and one
  // region spans every map point for the end card
  const regionFor = (pts: MapPoint[], key: string, minW: number, minH: number, padW: number, padH: number): Region => {
    let lon0 = Math.min(...pts.map((p) => p.lon)), lon1 = Math.max(...pts.map((p) => p.lon));
    let lat0 = Math.min(...pts.map((p) => p.lat)), lat1 = Math.max(...pts.map((p) => p.lat));
    const cx = (lon0 + lon1) / 2, cy = (lat0 + lat1) / 2;
    const w = Math.max(minW, lon1 - lon0 + padW), h = Math.max(minH, lat1 - lat0 + padH);
    lon0 = Math.max(86.5, cx - w / 2); lon1 = Math.min(159.5, cx + w / 2);
    lat0 = Math.max(-47, cy - h / 2); lat1 = Math.min(47, cy + h / 2);
    const area = (lon1 - lon0) * 43.24 * (lat1 - lat0) * 44;
    const cell = Math.max(0.6, Math.sqrt(area / 1.3e6));
    return { lon0, lon1, lat0, lat1, cell: Math.round(cell * 100) / 100, key: `${opts.id}-${key}-${lon0.toFixed(1)}-${lat1.toFixed(1)}` };
  };
  scenes.forEach((s, i) => { if (s.map && s.map.points.length) s.map.region = regionFor(s.map.points, `m${i}`, 18, 12, 12, 9); });
  const pts = scenes.flatMap((s) => s.map?.points ?? []);
  const region: Region | null = pts.length ? regionFor(pts, "all", 24, 16, 16, 12) : null;
  const duration = Math.ceil((scenes[scenes.length - 1]?.b ?? 5) * 10) / 10;
  return { id: opts.id, title: story.title, series: story.series, duration, year, region, scenes, words, audio: opts.audio, warnings };
};

  const xs = pts.map((p) => p[0]), zs = pts.map((p) => p[1]);

// Stager: turns planned groups into actors with a pose / position / facing for every moment, using the set's anchors.
// Also the camera grammar: one shot per beat (wide, group, medium, close, low), framed on the lead or the pack.

import type { Anchors, V3 } from "./sets.tsx";

export type ActorState = { pos: V3; rotY: number; pose: Pose };
export type Actor = { id: number; group: number; lead: boolean; color: string | null; gain?: number; costume: Group["costume"]; seated: boolean; moving: boolean; at: (t: number) => ActorState };

const hsh = (n: number) => {
  const x = Math.sin(n * 91.3458 + 47.123) * 43758.5453;
  return x - Math.floor(x);
};
const face = (from: V3, to: V3) => Math.atan2(to[0] - from[0], to[2] - from[2]);
const armed = (g: Group) => g.costume.itemR === "musket" || g.costume.itemR === "spear" || g.costume.itemL === "musket";

const staticPose = (g: Group, t: number, seed: number, seatKind?: "chair" | "floor"): Pose => {
  switch (g.action) {
    case "akimbo": return idle(POSES.akimbo(), t, seed);
    case "point": return idle(POSES.point("R", 0.1), t, seed, 0.6);
    case "telescope": return idle(POSES.telescope(), t, seed, 0.4);
    case "pray": return idle(POSES.pray(), t, seed, 0.5);
    case "kneel": return idle(POSES.kneel(0), t, seed, 0.5);
    case "bow": return idle(POSES.kneel(0.85), t, seed, 0.3);
    case "sit": return idle(seatKind === "chair" ? POSES.sitChair() : POSES.sitCross(), t, seed, 0.6);
    case "write": return idle(POSES.write(t + seed), t, seed, 0.3);
    case "guard": case "march": return idle(armed(g) ? POSES.shoulderArms() : POSES.stand(), t, seed, 0.5);
    case "raise": return idle(POSES.raiseFist(), t, seed, 0.8);
    case "talk": return idle(POSES.talk(t + seed), t, seed);
    case "look": return idle({ ...POSES.stand(), head: [0.3, 0] }, t, seed);
    case "carry": return idle(POSES.carryHead(), t, seed);
    default: return idle(POSES.stand(), t, seed);
  }

export const stageCast = (groups: Group[], A: Anchors, dur: number, seed: number, dark: boolean): Actor[] => {
  const actors: Actor[] = [];
  let id = 1;
  const leadG = groups.find((g) => g.lead);
  const leadMoving = leadG && (leadG.action === "run" || leadG.action === "walk");
  const second = groups.find((g) => !g.lead && g.count === 1);
  const seatFor = (g: Group) => (g.action === "sit" || g.action === "write") && A.seat ? A.seat : undefined;
  const leadPos: V3 = leadG && seatFor(leadG) ? A.seat!.pos : A.lead;
  const pathDir = [A.path[1][0] - A.path[0][0], A.path[1][2] - A.path[0][2]];
  const pathLen = Math.hypot(pathDir[0], pathDir[1]);
  const ux = pathDir[0] / pathLen, uz = pathDir[1] / pathLen;
  const pathYaw = Math.atan2(ux, uz);
  let guardSlot = 0;
  let moverSlot = 0;
  const totalMovers = groups.filter((g) => g.action === "run" || g.action === "walk").reduce((s, g) => s + g.count, 0);

  groups.forEach((g, gi) => {
    const dim = !g.color && dark ? { color: "#b4a58c", gain: 0.55 } : null;
    const color = g.color ?? dim?.color ?? null;
    const gain = g.color ? 1.5 : dim?.gain;
    for (let k = 0; k < g.count; k++) {
      const sd = seed * 31 + gi * 7 + k;
      const mover = g.action === "run" || g.action === "walk";
      if (mover) {
        // stream along the path so the pack fills the shot for the whole scene; the lead runs mid-pack
        const slot = g.lead ? Math.floor(totalMovers / 2) : moverSlot++ + (leadMoving && moverSlot > totalMovers / 2 ? 1 : 0);
        const run = g.action === "run" ? 0.75 + 0.25 * hsh(sd) : 0.05;
        const speed = Math.min(g.action === "run" ? 3.0 + hsh(sd + 1) * 0.6 : 1.3 + hsh(sd + 1) * 0.3, (pathLen * 0.75) / Math.max(1, dur));
        // everyone crosses the middle of the path during the scene; stagger by slot
        const mid = pathLen * 0.5;
        const t0 = dur * 0.5 - mid / speed + (slot - totalMovers / 2) * (1.1 / speed) + (hsh(sd + 2) - 0.5) * 0.25;
        const off = g.lead ? 0 : ((slot % 3) - 1) * 0.9 + (hsh(sd + 3) - 0.5) * 0.4;
        const sc = g.costume.scale ?? 1;
        actors.push({
          id: id++, group: gi, lead: g.lead, color, gain, costume: g.costume, seated: false, moving: true,
          at: (t) => {

            const x = A.path[0][0] + ux * d - uz * off, z = A.path[0][2] + uz * d + ux * off;
            const stride = (g.action === "run" ? 1.1 : 0.72) * sc;
            let pose = gait((d / stride) * Math.PI, run);
            if (g.costume.load === "head") pose = { ...pose, armL: { fwd: 0.35, out: 1.25, elbow: 2.1, twist: -0.4 } };
            return { pos: [x, 0, z], rotY: pathYaw, pose };
          },
        });
        continue;
      }
      let pos: V3;
      let seated = false;
      const seat = seatFor(g);
      if (g.lead) { pos = leadPos; seated = !!seat; }
      else if (g === second) pos = A.second;
      else if (g.action === "pray" || g.action === "kneel" || g.action === "bow") {
        // a loose arc in front of the target

        const [x0, x1, z0, z1] = A.rows;
        const cols = Math.max(1, Math.min(g.count, Math.round((x1 - x0) / 1.0)));
        const row = Math.floor(k / cols), col = k % cols;
        const rowsN = Math.ceil(g.count / cols);
        pos = [x0 + ((col + 0.5) / cols) * (x1 - x0) + (row % 2) * 0.4 + (hsh(sd) - 0.5) * 0.25, 0, z1 - (row / Math.max(1, rowsN - 1 || 1)) * (z1 - z0) * (rowsN > 1 ? 1 : 0) - (hsh(sd + 1) - 0.5) * 0.3];
      } else {
        const gp = A.guards[guardSlot++ % A.guards.length];
        const extra = Math.floor((guardSlot - 1) / A.guards.length);
        pos = [gp[0] + extra * 0.7, 0, gp[2] - extra * 0.6];
      }
      // who looks where
      let rotY: number;
      if (seat && g.lead) rotY = seat.rotY;
      else if (g.lead) rotY = second ? face(pos, A.second) : g.action === "point" || g.action === "akimbo" ? face(pos, A.target) + 0.5 : face(pos, A.target);

      else if (g.count > 6 || g.action === "march") rotY = leadG ? face(pos, leadPos) : Math.PI;
      else rotY = leadG ? face(pos, leadPos) : face(pos, A.target);
      const p = pos;
      const ry = rotY;
      actors.push({
        id: id++, group: gi, lead: g.lead, color, gain, costume: g.costume, seated: seated || g.action === "sit" || g.action === "kneel" || g.action === "pray" || g.action === "bow", moving: false,
        at: (t) => ({ pos: p, rotY: ry, pose: staticPose(g, t, sd, seat?.kind) }),
      });
    }
  });
  return actors;

const add = (a: V3, b: V3): V3 => [a[0] + b[0], a[1] + b[1], a[2] + b[2]];

export const shotKeys = (shot0: Shot, a: number, b: number, focusAt0: (t: number) => V3, groupAt: (t: number) => V3, A: Anchors, seated: boolean, side: number, moving = false): CamKey[] => {
  const lowY = seated ? -0.5 : 0;
  // a running lead leaves any tight frame: stay wide on the pack
  const shot: Shot = moving && shot0 !== "wide" ? "group" : shot0;
  const focusAt = moving ? groupAt : focusAt0;
  const mk = (t: number, push: number): CamKey => {
    const F = focusAt(t);
    const C = groupAt(t);
    let pos: V3, target: V3, fov: number;
    const sx = side;
    switch (shot) {
      // lateral offsets stay small: a 9:16 frame is only ~22° wide
      case "wide": pos = add(F, [0.9 * sx, 1.7, 10.5 - push * 1.2]); target = add(F, [-0.25 * sx, 1.9, -22]); fov = 46; break;
      case "group": pos = add(C, [-0.9 * sx, 1.7, 9.0 - push]); target = add(C, [0.2 * sx, 1.3, -14]); fov = 46; break;
      case "medium": pos = add(F, [0.5 * sx, 1.35 + lowY, 4.6 - push * 0.5]); target = add(F, [-0.12 * sx, 1.15 + lowY, -10]); fov = 40; break;
      case "close": pos = add(F, [0.3 * sx, 1.45 + lowY, 2.5 - push * 0.3]); target = add(F, [-0.05 * sx, 1.4 + lowY * 1.2, -8]); fov = 38; break;
      default: pos = add(F, [0.45 * sx, 0.35, 3.0 - push * 0.3]); target = add(F, [-0.1 * sx, 2.5 + lowY, -8]); fov = 44; break;
    }
    if (A.camMaxZ !== undefined && pos[2] > A.camMaxZ) {
      // inside a room: stay inside, widen instead
      const over = pos[2] - A.camMaxZ;
      pos = [pos[0] * 0.8, pos[1], A.camMaxZ];
      fov = Math.min(58, fov + over * 2.2);
    }
    pos[1] = Math.max(pos[1], 0.3);
    return { t, pos, target, fov };
  };
  return [mk(a, 0), mk(b, 1)];
};

// Figure — a posable 3D person rendered as a flat black silhouette (optionally with a coloured contour, see
// outline.ts). Built from simple solids on a joint hierarchy; because it is pure black, overlaps inside the body
// never show — only the outer shape reads, so the costume and the pose carry the character:
//   hats  : peci, destar (pointed head-cloth), caping (wide cone), shako, bicorne, turban, bun (hair), none
//   wear  : sarong / kain (cloth that follows the legs), robe (ankle-long), coat (short flare), pants
//   items : musket, spear, telescope, torch (lit flame), keris, flag, bundle (back / head), paddle
// Units are metres; the root is at the feet, facing +z. Everything is a pure function of the props.

const limbCache = new Map<string, CylinderGeometry>();
const limb = (rTop: number, rBot: number) => {
  const k = `${rTop.toFixed(3)}|${rBot.toFixed(3)}`;
  let g = limbCache.get(k);
  if (!g) { g = G.limb(rBot, rTop); limbCache.set(k, g); }

/** Breathing + weight shift + small head moves layered on any pose so held silhouettes never freeze. */
export const idle = (p: Pose, t: number, seed = 0, amt = 1): Pose => {
  const b = Math.sin(t * 1.45 + seed * 2.1);
  const w = Math.sin(t * 0.37 + seed * 1.3);
  const h = Math.sin(t * 0.53 + seed * 0.7);
  const arm = (a: Arm | undefined, k: number): Arm | undefined => (a ? { ...a, fwd: (a.fwd ?? 0) + 0.025 * amt * Math.sin(t * 0.8 + seed + k), out: (a.out ?? 0) + 0.012 * amt * b } : a);
  return {
    ...p,
    spine: (p.spine ?? 0) + 0.018 * amt * b,
    side: (p.side ?? 0) + 0.02 * amt * w,
    yaw: (p.yaw ?? 0) + 0.03 * amt * w,
    head: [(p.head?.[0] ?? 0) + 0.03 * amt * h, (p.head?.[1] ?? 0) + 0.06 * amt * Math.sin(t * 0.29 + seed)],
    armL: arm(p.armL, 0), armR: arm(p.armR, 2),
  };
};

/** Walk / run cycle. `phase` advances by π per step; `run` 0…1 blends walk → run. */
export const gait = (phase: number, run = 0, base: Pose = {}): Pose => {
  const amp = 0.42 + 0.32 * run;
  const leg = (p: number): Leg => {
    const s = Math.sin(p), c = Math.cos(p);
    return { fwd: amp * s + 0.05 * run, out: 0.03, knee: 0.08 + (0.75 + 0.85 * run) * Math.pow(Math.max(0, c), 1.4) + 0.1 * run, ankle: -0.2 * Math.max(0, -s) };
  };
  const arm = (p: number, side: number): Arm => ({ fwd: -(0.35 + 0.5 * run) * Math.sin(p) + 0.1 * run, out: 0.1, elbow: 0.2 + 1.0 * run, twist: side * 0.1 });
  return {
    ...base,
    hipY: 0.93 - 0.025 - 0.05 * run + (0.025 + 0.03 * run) * Math.abs(Math.cos(phase)),
    lean: 0.04 + 0.26 * run + (base.lean ?? 0),
    spine: base.spine ?? 0.02 * run,
    yaw: 0.08 * Math.sin(phase),
    twist: -0.12 * Math.sin(phase),
    legL: leg(phase), legR: leg(phase + Math.PI),
    armL: base.armL ?? arm(phase + Math.PI, 1), armR: base.armR ?? arm(phase, -1),
  };
};

// ------------------------------------------------------------------ kinematics used by the cloth
const legFK = (hx: number, l: Leg) => {
  const f = l.fwd ?? 0, k = l.knee ?? 0;
  const knee = [hx + Math.sin(l.out ?? 0) * 0.42 * Math.sign(hx), -0.02 - 0.42 * Math.cos(f), 0.42 * Math.sin(f)];
  const a = f - k;
  const ankle = [knee[0], knee[1] - 0.41 * Math.cos(a), knee[2] + 0.41 * Math.sin(a)];
  return { knee, ankle };
};

/** Cloth tube from the waist ring to a bottom ring that follows the knees/shins (sarong, kain, robe, coat flare). */
const Skirt: React.FC<{ pose: Pose; len: number; top: number; bottom: number; tag: SilTag; front?: number }> = ({ pose, len, top, bottom, tag, front = 1 }) => {
  const SEG = 22, RINGS = 7;
  const geo = useMemo(() => {
    const g = new BufferGeometry();
    g.setAttribute("position", new BufferAttribute(new Float32Array((SEG + 1) * RINGS * 3), 3));
    const idx: number[] = [];
    for (let r = 0; r < RINGS - 1; r++) for (let s = 0; s < SEG; s++) {
      const a = r * (SEG + 1) + s, b = a + SEG + 1;
      idx.push(a, b, a + 1, a + 1, b, b + 1);
    }
    g.setIndex(idx);
    return g;
  }, []);
  useLayoutEffect(() => () => geo.dispose(), [geo]);
  const L = legFK(0.09, pose.legL ?? {}), R = legFK(-0.09, pose.legR ?? {});
  const pos = geo.attributes.position as BufferAttribute;
  for (let r = 0; r < RINGS; r++) {
    const t = r / (RINGS - 1);
    const y = 0.06 - t * len;
    // where the legs are at this height (interpolate hip → knee → ankle by drop)
    const at = (fk: ReturnType<typeof legFK>, hx: number) => {
      const drop = -y + 0.02;
      if (drop <= 0.42) { const u = drop / 0.42; return [hx + (fk.knee[0] - hx) * u, fk.knee[2] * u]; }
      const u = Math.min(1, (drop - 0.42) / 0.41);
      return [fk.knee[0] + (fk.ankle[0] - fk.knee[0]) * u, fk.knee[2] + (fk.ankle[2] - fk.knee[2]) * u];
    };
    const l = at(L, 0.09), rr = at(R, -0.09);
    const cz = (l[1] + rr[1]) / 2, cx = (l[0] + rr[0]) / 2;
    const spreadZ = Math.abs(l[1] - rr[1]) / 2;
    const rx = top + (bottom - top) * t;
    const rz = rx * 0.78 + spreadZ * Math.min(1, t * 1.6) * 0.95;
    for (let s = 0; s <= SEG; s++) {
      const a = (s / SEG) * Math.PI * 2;
      const sx = Math.cos(a), sz = Math.sin(a);
      const k = sz > 0 ? front : 1;
      pos.setXYZ(r * (SEG + 1) + s, cx * t + sx * rx, y, cz * Math.min(1, t * 1.3) + sz * rz * k);
    }
  }
  pos.needsUpdate = true;
  geo.computeBoundingSphere();
  return <mesh geometry={geo} material={INK} userData={{ sil: tag }} />;
};

const PuffShader = {
  vertexShader: /* glsl */ `
    attribute vec3 aPos; attribute float aSize; attribute float aSeed; attribute float aAlpha; attribute float aHeat; attribute float aRot;
    varying vec2 vUv; varying float vSeed; varying float vAlpha; varying float vHeat;
    #include <fog_pars_vertex>
    void main(){
      vUv = uv; vSeed = aSeed; vAlpha = aAlpha; vHeat = aHeat;
      vec4 mv = modelViewMatrix * vec4(aPos, 1.0);
      float c = cos(aRot), s = sin(aRot);
      vec2 p = mat2(c, -s, s, c) * position.xy;
      mv.xy += p * aSize;
      gl_Position = projectionMatrix * mv;
      vec4 mvPosition = mv;
      #include <fog_vertex>
    }`,
  fragmentShader: /* glsl */ `
    uniform vec3 uDark; uniform vec3 uLit; uniform vec3 uHot; uniform float uFlash; uniform vec3 uFlashCol; uniform float uDensity;
    varying vec2 vUv; varying float vSeed; varying float vAlpha; varying float vHeat;
    #include <fog_pars_fragment>
    float h(vec2 p){ return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453); }
    float n(vec2 p){ vec2 i = floor(p), f = fract(p); f = f*f*(3.0-2.0*f);
      return mix(mix(h(i), h(i+vec2(1,0)), f.x), mix(h(i+vec2(0,1)), h(i+vec2(1,1)), f.x), f.y); }
    float fbm(vec2 p){ float s = 0.0, a = 0.5; for (int i = 0; i < 4; i++){ s += a * n(p); p = p * 2.03 + 7.1; a *= 0.5; } return s; }
    void main(){
      vec2 q = vUv - 0.5;
      float r = length(q) * 2.0;
      float f = fbm(vUv * 3.2 + vSeed * 17.0);
      float m = smoothstep(1.0, 0.35, r + (f - 0.5) * 0.9);
      if (m < 0.01) discard;
      // lit from below: the underside of each puff catches the vent glow
      float under = smoothstep(0.35, -0.45, q.y + (f - 0.5) * 0.4);
      vec3 col = mix(uDark, uLit, under * vHeat);
      col = mix(col, uHot, pow(vHeat, 3.0) * under * 0.85);
      col *= 0.75 + 0.5 * f;
      col += uFlashCol * uFlash * (0.4 + 0.6 * f);
      gl_FragColor = vec4(col, m * vAlpha * uDensity);
      #include <fog_fragment>
    }`,
};

export const AshColumn: React.FC<AshProps> = (p) => {

    g.instanceCount = N;

        fogColor: { value: new Color() }, fogNear: { value: 1 }, fogFar: { value: 1000 }, fogDensity: { value: 0.001 },
      },
      transparent: true, depthWrite: false, fog: true, blending: NormalBlending, side: DoubleSide,
    });
    return { geo: g, mat: m };
  }, [N]);
  useLayoutEffect(() => () => { geo.dispose(); mat.dispose(); }, [geo, mat]);

  const amt = p.amount ?? 1;
  const seed = p.seed ?? 1;
  const aPos = geo.getAttribute("aPos") as InstancedBufferAttribute;
  const aSize = geo.getAttribute("aSize") as InstancedBufferAttribute;
  const aSeed = geo.getAttribute("aSeed") as InstancedBufferAttribute;
  const aAlpha = geo.getAttribute("aAlpha") as InstancedBufferAttribute;
  const aHeat = geo.getAttribute("aHeat") as InstancedBufferAttribute;
  const aRot = geo.getAttribute("aRot") as InstancedBufferAttribute;
  const wind = p.wind ?? [0, 0];
  for (let i = 0; i < N; i++) {
    const r0 = hash(i * 3.1 + seed), r1 = hash(i * 7.7 + seed), r2 = hash(i * 1.3 + seed * 2);
    const age = (((p.t + r0 * L) % L) + L) % L;
    const k = age / L;
    const rise = Math.pow(k, 0.75);
    const spread = p.width * (0.18 + 0.82 * Math.pow(k, 0.9));
    const ang = r1 * Math.PI * 2 + p.t * 0.15 * (r2 - 0.5);
    const rad = Math.sqrt(r2) * spread * 0.5;
    const y = p.base[1] + p.height * rise;

    const z = p.base[2] + Math.sin(ang) * rad * 0.7 + wind[1] * k * p.height;

    aSize.setX(i, p.width * (0.32 + 0.75 * k) * (0.8 + 0.4 * r0));
    aSeed.setX(i, r1 * 10);
    const fadeIn = Math.min(1, age / 0.8), fadeOut = Math.min(1, (L - age) / 2.0);
    // born only after the eruption has started (amount ramps the column up)
    const born = Math.min(1, Math.max(0, (amt * 1.3 - k) * 3));
    aAlpha.setX(i, fadeIn * fadeOut * born * (0.55 + 0.45 * r2));
    aHeat.setX(i, Math.max(0, 1 - k * 1.6));
    aRot.setX(i, r0 * 6.28 + p.t * 0.08 * (r1 - 0.5));
  }
  for (const a of [aPos, aSize, aSeed, aAlpha, aHeat, aRot]) a.needsUpdate = true;

  u.uDark.value.set(p.dark ?? "#3b3433");
  u.uLit.value.set(p.lit ?? "#b4502a");
  u.uHot.value.set(p.hot ?? "#ffb15a");

  u.uFlashCol.value.set(p.flashColor ?? "#b9a0ff");

/** Parse "Kata *kunci* lain" with a start/end time into evenly paced words (until real VO timings exist). */
export const pace = (text: string, t0: number, t1: number): Word[] => {
  const raw = text.split(/\s+/).filter(Boolean);
  const weights = raw.map((w) => 0.6 + w.replace(/[*,.]/g, "").length * 0.08 + (/[,.]$/.test(w) ? 0.35 : 0));

  return raw.map((w, i) => {

    acc += weights[i];
    return { w: w.replace(/\*/g, ""), t, key: /^\*.*\*[,.!?]?$/.test(w) };
  });
};

/** Screen-space polyline (0..1 points) drawn progressively with a bright head. */
export const RouteLine: React.FC<{ pts: { x: number; y: number }[]; progress: number; color?: string; dash?: boolean; width?: number; opacity?: number }> = ({
  pts, progress, color = AMBER, dash = false, width = 6, opacity = 1,

  const C = useCanvas();
  if (pts.length < 2 || progress <= 0) return null;
  const P = pts.map((p) => [p.x * C.W, p.y * C.H]);
  const seg: number[] = [];
  let tot = 0;
  for (let i = 1; i < P.length; i++) { const l = Math.hypot(P[i][0] - P[i - 1][0], P[i][1] - P[i - 1][1]); seg.push(l); tot += l; }
  const want = tot * c01(progress);
  const out: number[][] = [P[0]];
  let acc = 0;
  for (let i = 1; i < P.length; i++) {
    if (acc + seg[i - 1] >= want) {
      const u = (want - acc) / seg[i - 1];
      out.push([P[i - 1][0] + (P[i][0] - P[i - 1][0]) * u, P[i - 1][1] + (P[i][1] - P[i - 1][1]) * u]);
      break;
    }
    out.push(P[i]);
    acc += seg[i - 1];
  }
  const d = out.map((p, i) => `${i ? "L" : "M"}${p[0].toFixed(1)} ${p[1].toFixed(1)}`).join(" ");
  const head = out[out.length - 1];
  return (
    <svg width={C.W} height={C.H} style={{ position: "absolute", left: 0, top: 0, opacity }}>
      <path d={d} fill="none" stroke="rgba(0,0,0,0.35)" strokeWidth={width + 6} strokeLinecap="round" strokeLinejoin="round" />

      {progress < 1 && <circle cx={head[0]} cy={head[1]} r={width * 1.6} fill={IVORY} />}
    </svg>
  );

      const ridge = Math.pow(Math.abs(Math.sin(a * 11 + n * 3)), 3) * 0.04;
      const k = 1 + n * 0.12 - ridge;
      p.setXYZ(i, x * k, y + n * height * 0.04, z * k);
    }
    g.computeVertexNormals();

      c.fillStyle = "#000";

      // lathe uv: u around, v along the profile (0 at the rim side → 1 at the base)
      for (let s = 0; s < 7; s++) {
        let x = 30 + hash(s * 5 + seed) * 452;
        const len = 120 + hash(s * 7 + seed) * 220;
        c.lineCap = "round";

          c.lineTo(x, y + 4);

  }, [radius, height, seed]);

  mat.emissiveIntensity = 2.5 * glow;

// Relief map — a real-coastline 3D terrain of a map region, built in the browser from the baked Natural Earth
// paths (src/data/map-*.json, same format as the PETA kit: Mercator "map px", optional 10 m detail box).
//
// Pipeline (once per region, cached): rasterise land → several blurred copies (coastal shelf, inland distance,
// massif) → height = plateau + ridged noise + listed volcano cones, × vertical exaggeration → one grid mesh with
// vertex colours (lowland olive-grey → rock → pale peaks); plus a painted sea texture (deep → shelf, thin coast).
// World units: 1 unit = 1 map px (≈ 2.6 km at the bundled scale); y up; north = −z.

import { BufferAttribute, BufferGeometry, CanvasTexture, Color, SRGBColorSpace, LinearFilter } from "three";
import MAP from "../../data/map-nusantara.json";

export type GeoMap = {
  w: number; h: number; bbox: number[];
  land: string;
  detail?: { rect: number[]; land: string };
};
export const NUSANTARA = MAP as unknown as GeoMap;

const RAD = Math.PI / 180;
const merc = (lat: number) => Math.log(Math.tan(Math.PI / 4 + (lat * RAD) / 2));
export const projPx = (lon: number, lat: number, map: GeoMap = NUSANTARA): [number, number] => {
  const k = map.w / ((map.bbox[1] - map.bbox[0]) * RAD);
  return [(lon - map.bbox[0]) * RAD * k, (merc(map.bbox[3]) - merc(lat)) * k];
};
/** km per map px at a latitude (Mercator stretch). */
export const kmPerPx = (lat: number, map: GeoMap = NUSANTARA) => ((map.bbox[1] - map.bbox[0]) * 111.32 * Math.cos(lat * RAD)) / map.w;

// ---------------------------------------------------------------- noise (gradient noise, rotated octaves)
const grad = (ix: number, iy: number, x: number, y: number) => {
  const h = Math.sin(ix * 127.1 + iy * 311.7) * 43758.5453;
  const a = (h - Math.floor(h)) * Math.PI * 2;
  return Math.cos(a) * (x - ix) + Math.sin(a) * (y - iy);
};
const q = (t: number) => t * t * t * (t * (t * 6 - 15) + 10);
/** Perlin-style gradient noise, roughly −0.7…0.7 → mapped to 0…1. */
const gnoise = (x: number, y: number) => {
  const xi = Math.floor(x), yi = Math.floor(y);
  const u = q(x - xi), v = q(y - yi);
  const a = grad(xi, yi, x, y), b = grad(xi + 1, yi, x, y), c = grad(xi, yi + 1, x, y), d = grad(xi + 1, yi + 1, x, y);
  return 0.5 + 0.75 * (a + (b - a) * u + (c - a) * v + (a - b - c + d) * u * v);
};
const ROT = [0.8, 0.6];
export const fbm = (x: number, y: number, oct = 5) => {
  let s = 0, a = 0.5, f = 1, px = x, py = y;
  for (let i = 0; i < oct; i++) {
    s += a * gnoise(px * f, py * f);
    const nx = px * ROT[0] - py * ROT[1], ny = px * ROT[1] + py * ROT[0];
    px = nx + 17.3; py = ny - 9.1;
    f *= 2.02; a *= 0.5;
  }
  return s;
};
export const ridged = (x: number, y: number, oct = 5) => {
  let s = 0, a = 0.6, f = 1, px = x, py = y, w = 1;
  for (let i = 0; i < oct; i++) {
    let n = 1 - Math.abs(gnoise(px * f, py * f) * 2 - 1);
    n = n * n * w;
    s += a * n;
    w = Math.min(1, n * 1.6);
    const nx = px * ROT[0] - py * ROT[1], ny = px * ROT[1] + py * ROT[0];
    px = nx + 5.7; py = ny + 3.3;
    f *= 2.15; a *= 0.5;
  }
  return s;
};

const LAND_LOW = new Color("#626c5c"), LAND_MID = new Color("#7b7e70"), LAND_HIGH = new Color("#9c978a"), LAND_PEAK = new Color("#cbc4b5");
const ROCK = new Color("#66645c");
export const reliefColor = (km: number, slope: number, n: number): [number, number, number] => {
  const c = new Color();
  if (km < 0.6) c.copy(LAND_LOW).lerp(LAND_MID, km / 0.6);
  else if (km < 1.8) c.copy(LAND_MID).lerp(LAND_HIGH, (km - 0.6) / 1.2);
  else c.copy(LAND_HIGH).lerp(LAND_PEAK, Math.min(1, (km - 1.8) / 1.8));
  c.lerp(ROCK, Math.min(0.55, Math.max(0, slope - 0.35) * 0.9));

type Raster = { w: number; h: number; d: Float32Array };
const readRed = (g: CanvasRenderingContext2D, w: number, h: number): Raster => {
  const img = g.getImageData(0, 0, w, h).data;
  const d = new Float32Array(w * h);

const bilinear = (r: Raster, x: number, y: number) => {
  const fx = Math.max(0, Math.min(r.w - 1.001, x)), fy = Math.max(0, Math.min(r.h - 1.001, y));
  const xi = Math.floor(fx), yi = Math.floor(fy), u = fx - xi, v = fy - yi;
  const i = yi * r.w + xi;
  return r.d[i] * (1 - u) * (1 - v) + r.d[i + 1] * u * (1 - v) + r.d[i + r.w] * (1 - u) * v + r.d[i + r.w + 1] * u * v;
};

export const buildRelief = (R: Region): Relief => {
  const hit = cache.get(R.key);
  if (hit) return hit;
  const map = R.map ?? NUSANTARA;
  const [x0, y0] = projPx(R.lon0, R.lat1, map);
  const [x1, y1] = projPx(R.lon1, R.lat0, map);
  const W = x1 - x0, H = y1 - y0;
  const cx = x0 + W / 2, cy = y0 + H / 2;
  const S = 4; // raster px per map px
  const rw = Math.ceil(W * S), rh = Math.ceil(H * S);
  const cvs = document.createElement("canvas");
  cvs.width = rw;
  cvs.height = rh;
  const g = cvs.getContext("2d", { willReadFrequently: true })!;
  const drawLand = (ctx: CanvasRenderingContext2D, color: string) => {
    ctx.save();
    ctx.setTransform(S, 0, 0, S, -x0 * S, -y0 * S);
    ctx.fillStyle = color;
    ctx.fill(new Path2D(map.land));
    if (map.detail) {
      const [dx0, dy0, dx1, dy1] = map.detail.rect;
      ctx.save();
      ctx.beginPath();
      ctx.rect(dx0, dy0, dx1 - dx0, dy1 - dy0);
      ctx.clip();
      ctx.fillStyle = "#000";
      ctx.fillRect(dx0, dy0, dx1 - dx0, dy1 - dy0);
      ctx.fillStyle = color;
      ctx.fill(new Path2D(map.detail.land));
      ctx.restore();
    }
    ctx.restore();
  };
  g.fillStyle = "#000";
  g.fillRect(0, 0, rw, rh);
  drawLand(g, "#fff");
  const mask = readRed(g, rw, rh);
  const blurOf = (px: number) => {
    const c2 = document.createElement("canvas");
    c2.width = rw;
    c2.height = rh;
    const g2 = c2.getContext("2d", { willReadFrequently: true })!;
    g2.fillStyle = "#000";
    g2.fillRect(0, 0, rw, rh);
    g2.filter = `blur(${px * S}px)`;
    g2.drawImage(cvs, 0, 0);
    return readRed(g2, rw, rh);
  };
  const b1 = blurOf(1.2), b2 = blurOf(5), b3 = blurOf(14);

  // land sinks into the sea along the region's border, so a cut coastline never shows as a straight edge: the
  // sinking line wobbles (low-frequency noise), so a cut island ends in a ragged coast, not a ruler line
  const mn = Math.min(W, H);
  const edgeE = (px: number, pz: number) => {
    const wob = (fbm((px / mn) * 7 + 3.1, (pz / mn) * 7 + 7.7, 3) - 0.44) * 0.16 * mn;
    const t = (Math.min(px, W - px, pz, H - pz) - 0.02 * mn + wob) / (0.1 * mn);
    return t >= 1 ? 1 : t <= 0 ? 0 : t * t * (3 - 2 * t);
  };

  // ---- sea texture (deep → shelf glow → thin pale coastline)
  const sea = document.createElement("canvas");
  const SW = 2048, SH = Math.round((2048 * H) / W);
  sea.width = SW;
  sea.height = SH;
  const sg = sea.getContext("2d")!;
  sg.fillStyle = "#183c48";
  sg.fillRect(0, 0, SW, SH);
  const k = SW / W;
  const shelf = (blur: number, col: string, alpha: number) => {
    sg.save();
    sg.globalAlpha = alpha;
    sg.filter = `blur(${blur * k}px)`;
    sg.setTransform(k, 0, 0, k, -x0 * k, -y0 * k);
    sg.fillStyle = col;
    sg.fill(new Path2D(map.land));
    if (map.detail) sg.fill(new Path2D(map.detail.land));
    sg.restore();
  };
  shelf(10, "#23525f", 0.8);
  shelf(3, "#2f6570", 0.65);
  sg.save();
  sg.setTransform(k, 0, 0, k, -x0 * k, -y0 * k);
  sg.strokeStyle = "rgba(196, 220, 212, 0.75)";
  sg.lineWidth = 0.32;
  sg.stroke(new Path2D(map.land));
  sg.restore();
  // the shelf glow and coastline of sunk land fade out with it (else they draw the region's border as a box)
  {
    const fw = 256, fh = Math.max(2, Math.round((256 * H) / W));
    const fc = document.createElement("canvas");
    fc.width = fw;
    fc.height = fh;
    const fg = fc.getContext("2d")!;
    const im = fg.createImageData(fw, fh);
    for (let y = 0; y < fh; y++) for (let x = 0; x < fw; x++) {
      const e = edgeE(((x + 0.5) / fw) * W, ((y + 0.5) / fh) * H);
      const al = Math.min(1, Math.max(0, (0.92 - e) / 0.5));
      const o = (y * fw + x) * 4;
      im.data[o] = 0x18; im.data[o + 1] = 0x3c; im.data[o + 2] = 0x48; im.data[o + 3] = Math.round(al * 255);
    }
    fg.putImageData(im, 0, 0);
    sg.imageSmoothingEnabled = true;
    sg.drawImage(fc, 0, 0, SW, SH);
  }
  // faint graticule every 2°
  sg.save();
  sg.strokeStyle = "rgba(170, 205, 205, 0.07)";
  sg.lineWidth = 1;
  for (let lon = Math.ceil(R.lon0 / 2) * 2; lon <= R.lon1; lon += 2) { const [px] = projPx(lon, 0, map); sg.beginPath(); sg.moveTo((px - x0) * k, 0); sg.lineTo((px - x0) * k, SH); sg.stroke(); }
  for (let lat = Math.ceil(R.lat0 / 2) * 2; lat <= R.lat1; lat += 2) { const [, py] = projPx(0, lat, map); sg.beginPath(); sg.moveTo(0, (py - y0) * k); sg.lineTo(SW, (py - y0) * k); sg.stroke(); }
  sg.restore();
  const seaTex = new CanvasTexture(sea);
  seaTex.colorSpace = SRGBColorSpace;
  seaTex.minFilter = LinearFilter;
  seaTex.anisotropy = 8;

  // ---- heights
  const vex = R.vex ?? 5;
  const yPerKm = (vex / kmPerPx((R.lat0 + R.lat1) / 2, map));
  const vols = (R.volcanoes ?? []).map((v) => {
    const [px, py] = projPx(v.lon, v.lat, map);
    return { x: (px - x0) * S, y: (py - y0) * S, km: v.km, r: (v.r ?? 7) * S * 1.8, skip: v.skip };
  });
  const kmAtRaster = (rx: number, ry: number) => {
    const m = bilinear(mask, rx, ry);
    const c1 = bilinear(b1, rx, ry), c2 = bilinear(b2, rx, ry), c3 = bilinear(b3, rx, ry);
    const land = Math.min(1, Math.max(0, (c1 - 0.3) / 0.4));
    const mx = (rx / S + x0) * 0.06, my = (ry / S + y0) * 0.06;
    let km = 0.02 + 0.12 * c2 + 0.45 * Math.pow(c3, 1.6);
    const inland = Math.min(1, Math.max(0, (c2 - 0.5) / 0.4));
    const hills = Math.min(1, Math.max(0, (fbm(mx * 0.22 + 3.1, my * 0.22 - 1.7, 3) - 0.42) / 0.22));
    km += (0.08 + 0.9 * Math.pow(c2, 1.2)) * ridged(mx * 0.8, my * 0.8, 6) * inland * inland * hills;
    km += 0.06 * (fbm(mx * 5, my * 5, 3) - 0.5) * c1;
    for (const v of vols) {
      if (v.skip) continue;
      const dx = rx - v.x, dy = ry - v.y;
      const u = Math.sqrt(dx * dx + dy * dy) / v.r;
      if (u > 3) continue;
      const ang = Math.atan2(dy, dx);
      const gull = 1 - 0.07 * Math.pow(Math.abs(Math.sin(ang * 9 + fbm(u * 3, ang, 2) * 3)), 3) * Math.min(1, u * 2);
      const prof = Math.exp(-Math.pow(u * 1.7, 1.5));
      km = Math.max(km, v.km * prof * gull + km * 0.4);
    }
    const seaDepth = -0.9 * (1 - c1) - 0.1;
    return land > 0 ? km * land + seaDepth * (1 - land) * (m < 0.5 ? 1 : 0) : seaDepth;
  };

  const cell = R.cell ?? 0.65;
  const nx = Math.ceil(W / cell) + 1, nz = Math.ceil(H / cell) + 1;
  const pos = new Float32Array(nx * nz * 3);
  const col = new Float32Array(nx * nz * 3);
  const kmGrid = new Float32Array(nx * nz);
  for (let j = 0; j < nz; j++) {
    for (let i = 0; i < nx; i++) {
      const px = Math.min(W, i * cell), pz = Math.min(H, j * cell);
      const e = edgeE(px, pz);
      const km = kmAtRaster(px * S, pz * S) * e - 0.6 * (1 - e);
      const id = j * nx + i;
      kmGrid[id] = km;
      pos[id * 3] = px - W / 2;
      pos[id * 3 + 1] = km * yPerKm;
      pos[id * 3 + 2] = pz - H / 2;
    }
  }
  for (let j = 0; j < nz; j++) {
    for (let i = 0; i < nx; i++) {
      const id = j * nx + i;
      const kx = kmGrid[j * nx + Math.min(nx - 1, i + 1)] - kmGrid[j * nx + Math.max(0, i - 1)];
      const kz = kmGrid[Math.min(nz - 1, j + 1) * nx + i] - kmGrid[Math.max(0, j - 1) * nx + i];
      const slope = (Math.sqrt(kx * kx + kz * kz) * yPerKm) / (2 * cell);
      const n = fbm(i * 0.21, j * 0.21, 3);
      const [r, gg, b] = kmGrid[id] <= 0 ? [0.06, 0.15, 0.17] : reliefColor(kmGrid[id], slope, n);
      col[id * 3] = r; col[id * 3 + 1] = gg; col[id * 3 + 2] = b;
    }
  }
  const idx = new Uint32Array((nx - 1) * (nz - 1) * 6);
  let p = 0;
  for (let j = 0; j < nz - 1; j++) for (let i = 0; i < nx - 1; i++) {
    const a = j * nx + i, b = a + 1, c = a + nx, d = c + 1;
    idx[p++] = a; idx[p++] = c; idx[p++] = b; idx[p++] = b; idx[p++] = c; idx[p++] = d;
  }
  const geo = new BufferGeometry();
  geo.setAttribute("position", new BufferAttribute(pos, 3));
  geo.setAttribute("color", new BufferAttribute(col, 3));
  geo.setIndex(new BufferAttribute(idx, 1));
  geo.computeVertexNormals();
  geo.computeBoundingSphere();

  const toWorld = (lon: number, lat: number): [number, number] => {
    const [px, py] = projPx(lon, lat, map);
    return [px - cx, py - cy];
  };
  const heightAt = (lon: number, lat: number) => {
    const [px, py] = projPx(lon, lat, map);
    return Math.max(0, kmAtRaster((px - x0) * S, (py - y0) * S)) * yPerKm;
  };
  const heightAtWorld = (x: number, z: number) => kmAtRaster((x + W / 2) * S, (z + H / 2) * S) * yPerKm;
  const out: Relief = { geo, sea: seaTex, size: [W, H], toWorld, heightAt, heightAtWorld, yPerKm, color: reliefColor };
  cache.set(R.key, out);
  return out;
};
