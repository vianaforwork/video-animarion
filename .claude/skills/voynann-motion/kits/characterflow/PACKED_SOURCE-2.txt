`node <skill folder>/new-project.mjs <project folder> --kit=<style>` unpacks them into the project.
Format: per file one line with four '=' signs, the byte length and the path, then exactly that many bytes.

// One command from a story spec to a checked video.
//   node scripts/make.mjs stories/<id>/story.json [--preview] [--final] [--stills=3.2,10,18] [--wide] [--no-qa]
//
//   1. resolve   story.json (+ words.json) → src/stories/data/<id>.json, regenerate src/stories/registry.ts
//   2. direct    compile the story (blocking, acting, shots, sound cues) and print the plan; stops on errors
//   3. sound     scripts/sound-story.ts → public/audio/<id>-mix.wav
//   4. render    --preview: 540×960@30 with audio · --final (default): 1080×1920@60, mastered to −14 LUFS
//                (interrupted? run the same command again — finished segments are reused)
//   5. QA        QA gate (scripts/qa-gate.mjs: tembus / loncat / kedip of the tagged people and props, camera snaps,
//                and a scan of the 3D alone for flicker) — after --preview as a report, before --final as a stop;
//                then scripts/qa.mjs → out/<id>/qa/ (contact sheets, frame scan, sync, duration) + REVIEW.md
//
//   --wide       16:9 instead of 9:16, only when the buyer asks for landscape: the story is directed again for the
//                wide frame (camera, captions, cards) → out/<id>/stills-wide/, <id>_preview_960x540.mp4,
//                <id>_1920x1080_60fps.mp4, out/<id>/qa-wide/. The story file and the sound mix are the same.

==== 13146 scripts/qa.mjs
// Automated QA of a rendered video. Writes out/<id>/qa/{qa.json, contact_1fps_001.jpg …, shots_start_mid_end.jpg}
// (a 16:9 video — "1920x1080" / "960x540" in its file name, from `make.mjs --wide` — is checked as landscape, in qa-wide/)
// (the 1-fps contact sheet covers the whole video in pages of 96 seconds, each tile stamped with its second)
// and prints PASS/FAIL per check. The contact sheets are for the visual review that follows.
//   node scripts/qa.mjs <video.mp4> --vo=public/audio/x.mp3 --offset=0.4 --expect=49.8 [--shots=out/<id>/shots.json] [--final]
//                       [--size=1920x1080] [--from=12]   (--from: the video is a section starting at that film second)

import { spawnSync } from "node:child_process";
import fs from "node:fs";
import path from "node:path";

const args = process.argv.slice(2);
const video = args.find((a) => !a.startsWith("--"));
const opt = (k, d) => (args.find((a) => a.startsWith(`--${k}=`)) ?? `=${d ?? ""}`).split("=").slice(1).join("=");
const final = args.includes("--final");
if (!video || !fs.existsSync(video)) { console.error("usage: node scripts/qa.mjs <video.mp4> --vo=… --offset=… --expect=… [--shots=…] [--final]"); process.exit(1); }
const run = (c, a) => spawnSync(c, a, { encoding: "utf8", maxBuffer: 1 << 30 });
// the delivery size: --size, else from the file name (make.mjs names a 16:9 video …_1920x1080_60fps / …_preview_960x540)
const [sw, shh] = opt("size", /1920x1080|960x540/.test(path.basename(video)) ? "1920x1080" : "1080x1920").split("x").map(Number);
const portraitV = shh > sw;
// each format has its own QA folder, so the portrait review is never overwritten by the landscape one
const qaDir = path.join(path.dirname(video), portraitV ? "qa" : "qa-wide");
fs.mkdirSync(qaDir, { recursive: true });
const checks = [];
const check = (name, ok, detail) => { checks.push({ name, ok, detail }); console.log(`  ${ok ? "PASS" : "FAIL"}  ${name}: ${detail}`); };

// streams
const probe = JSON.parse(run("ffprobe", ["-v", "error", "-show_streams", "-show_format", "-of", "json", video]).stdout);
const v = probe.streams.find((s) => s.codec_type === "video"), a = probe.streams.find((s) => s.codec_type === "audio");
const dur = Number(probe.format.duration), expect = Number(opt("expect", dur));
const [fn, fd] = v.r_frame_rate.split("/").map(Number);
const fps = fn / fd;
// previews are half size; finals must be the exact delivery size at 60 fps
const sizeOk = final ? v.width === sw && v.height === shh && Math.abs(fps - 60) < 0.01 : Math.abs(v.width / v.height - sw / shh) < 0.01;
check("video stream", v.codec_name === "h264" && sizeOk, `${v.codec_name} ${v.width}×${v.height} @ ${fps} fps${final ? ` (final must be ${sw}×${shh} @ 60)` : ` (${portraitV ? "portrait" : "landscape"} aspect)`}`);
check("audio stream", !!a && a.codec_name === "aac", a ? `${a.codec_name} ${a.sample_rate} Hz, ${Number(a.duration ?? dur).toFixed(2)} s` : "missing");
check("duration", Math.abs(dur - expect) < 0.12, `${dur.toFixed(2)} s (expected ${expect.toFixed(2)} s)`);
if (a) check("audio length = video length", Math.abs(Number(a.duration ?? dur) - dur) < 0.15, `${Number(a.duration ?? dur).toFixed(2)} vs ${dur.toFixed(2)} s`);

// frame scan: white / black / sudden flashes
const ylog = path.join(qaDir, "yavg.txt");
run("ffmpeg", ["-v", "error", "-i", video, "-vf", `scale=192:108,signalstats,metadata=mode=print:file=${ylog}`, "-f", "null", "-"]);
const ytxt = fs.readFileSync(ylog, "utf8");
const y = (ytxt.match(/YAVG=[\d.]+/g) ?? []).map((s) => Number(s.slice(5)));
// 90th percentile luma per frame (video range: black is 16) — a frame whose p90 stays ≤ 24 is practically black
const y90 = (ytxt.match(/YHIGH=[\d.]+/g) ?? []).map((s) => Number(s.slice(6)));
// black: the opening fade, scene fades and the end card may be dark on purpose — only long runs fail
const white = y.filter((x) => x > 225).length;
// blown highlights: share of the picture burnt to white (a lamp in close-up, bloom) — the mean luma misses it
const hlog = path.join(qaDir, "hot.txt");
// the middle of the frame (a bright sky at the top is not a lamp in the lens); video range: white is 235
run("ffmpeg", ["-v", "error", "-i", video, "-vf", `crop=iw*0.7:ih*0.6:iw*0.15:ih*0.2,scale=160:120,lutyuv=y='if(gt(val,222),255,0)',signalstats,metadata=mode=print:key=lavfi.signalstats.YAVG:file=${hlog}`, "-f", "null", "-"]);
const hot = (fs.readFileSync(hlog, "utf8").match(/YAVG=[\d.]+/g) ?? []).map((s) => Number(s.slice(5)) / 255);
let hr = 0, hotLongest = 0, hotAt = -1;
hot.forEach((x, i) => { hr = x > 0.12 ? hr + 1 : 0; if (hr > hotLongest) { hotLongest = hr; hotAt = (i - hr + 1) / fps; } });
let run0 = 0, longest = 0;
y.slice(Math.round(fps * 0.9), y.length - Math.round(fps * 1.2)).forEach((x) => { run0 = x < 7 ? run0 + 1 : 0; longest = Math.max(longest, run0); });
const black = longest > fps * 0.7 ? longest : 0;
check("no white frames", white === 0, `${white} frames with mean luma > 225`);
check("no blown-out highlights", hotLongest < fps * 0.5, hotLongest ? `${(hotLongest / fps).toFixed(1)} s with > 12 % of the middle of the picture burnt white (from ${(hotAt + Number(opt("from", 0))).toFixed(1)} s) — a lamp / bloom too close` : "none");
check("no black gaps (outside fades)", black === 0, `longest near-black run ${(longest / fps).toFixed(2)} s (fails above 0.7 s)`);
// frozen picture (a stuck render shows as identical frames)
const fz = run("ffmpeg", ["-v", "info", "-i", video, "-vf", "scale=320:180,freezedetect=n=0.0006:d=2.5", "-an", "-f", "null", "-"]).stderr;
const freezes = [...fz.matchAll(/freeze_start: ([\d.]+)/g)].map((m) => Number(m[1])).filter((t) => t > 1 && t < dur - 6);
check("no frozen picture", freezes.length === 0, freezes.length ? `frozen from ${freezes.map((t) => t.toFixed(1)).join(", ")} s` : "none (≥ 2.5 s)");
// loudness (final delivery is mastered to −14 LUFS, true peak ≤ −1 dBTP)
if (a) {
  const lu = run("ffmpeg", ["-v", "info", "-nostats", "-i", video, "-vn", "-af", "ebur128=peak=true", "-f", "null", "-"]).stderr;
  const I = Number((lu.match(/I:\s+(-?[\d.]+) LUFS/g) ?? []).pop()?.match(/-?[\d.]+/)?.[0]);
  const TP = Number((lu.match(/Peak:\s+(-?[\d.]+) dBFS/g) ?? []).pop()?.match(/-?[\d.]+/)?.[0]);
  check("loudness", !final || (Math.abs(I + 14) < 1.2 && TP <= -0.8), `${I.toFixed(1)} LUFS integrated, true peak ${TP.toFixed(1)} dBTP${final ? " (target −14 LUFS, ≤ −1 dBTP)" : ""}`);
}
const jumps = [];
for (let i = 1; i < y.length; i++) if (Math.abs(y[i] - y[i - 1]) > 45) jumps.push(+(i / fps).toFixed(2));
console.log(`  info  luma ${Math.min(...y).toFixed(0)}–${Math.max(...y).toFixed(0)}, big jumps (cuts) at ${jumps.join(", ") || "none"}`);

// VO sync: cross-correlate the film's audio with the VO around the expected offset
const voFile = opt("vo"), off0 = Number(opt("offset", 0)), section = opt("from") !== "";
// a section that starts after the VO has begun: compare with the VO from that point (expected lag 0)
const voSkip = Math.max(0, -off0), off = Math.max(0, off0);
if (voFile && fs.existsSync(voFile)) {
  const SR = 8000;
  const raw = (f, ss = 0) => { const r = spawnSync("ffmpeg", ["-v", "error", ...(ss ? ["-ss", ss.toFixed(3)] : []), "-i", f, "-f", "f32le", "-ac", "1", "-ar", String(SR), "-"], { maxBuffer: 1 << 30 }); return new Float32Array(r.stdout.buffer, r.stdout.byteOffset, Math.floor(r.stdout.length / 4)); };
  const film = raw(video), vo = raw(voFile, voSkip);
  let best = -1, lag = 0;
  for (let L = Math.round((off - 0.25) * SR); L <= Math.round((off + 0.25) * SR); L++) {
    if (L < 0) continue;
    let s = 0, e1 = 0, e2 = 0;
    for (let i = 0; i < vo.length && i + L < film.length; i += 3) { const p = vo[i], q = film[i + L]; s += p * q; e1 += p * p; e2 += q * q; }
    const c = s / Math.sqrt(e1 * e2 + 1e-12);
    if (c > best) { best = c; lag = L; }
  }
  const err = lag / SR - off;
  check("VO sync", Math.abs(err) < 0.03 && best > 0.6, `VO found at ${(lag / SR).toFixed(3)} s (expected ${off.toFixed(3)}, error ${(err * 1000).toFixed(0)} ms, correlation ${best.toFixed(2)})`);
  const voEnd = off + vo.length / SR;
  if (!section) check("VO not cut", voEnd <= dur + 0.02, `VO ends at ${voEnd.toFixed(2)} s, video ${dur.toFixed(2)} s`);
}

// contact sheets (portrait: tall tiles): one tile per second for the WHOLE video, 96 per page
// (contact_1fps_001.jpg, _002.jpg …), every tile stamped with its film second so nothing can be skipped unseen
const tw = portraitV ? 180 : 320, thh = portraitV ? 320 : 180;
const grid = portraitV ? [12, 8] : [8, 12], perPage = grid[0] * grid[1];
for (const f of fs.readdirSync(qaDir)) if (/^contact_1fps(_\d+)?\.jpg$/.test(f)) fs.rmSync(path.join(qaDir, f));
// a second stamp on each tile when this ffmpeg has drawtext (not every build does); the page ranges are always listed
const hasDrawtext = /drawtext/.test(run("ffmpeg", ["-hide_banner", "-filters"]).stdout ?? "");

export type Subject = { face: V3; fwd: V3; pos: V3 };
export type ShotKind =
  | "establish" | "wide" | "closeUp" | "medium" | "mediumWide" | "twoShot" | "ots" | "track" | "pathDown"
  | "reaction" | "explain" | "think" | "sitStand" | "turn" | "boardWide" | "bedWide" | "bedClose" | "bedThink";
export type ShotSpec = { kind: ShotKind; a: string; b?: string; side?: 1 | -1; t0: number; t1: number; /** wide two-person: camera direction from the pair (chosen by the director to fit the set) */ dir?: V3; /** framed for the 16:9 frame */ wide?: boolean };

const V = (v: V3) => new Vector3(...v);
const A = (v: Vector3): V3 => [v.x, v.y, v.z];
const flat = (v: V3) => { const x = V(v).setY(0); return x.lengthSq() < 1e-6 ? new Vector3(0, 0, 1) : x.normalize(); };
const drift = (t: number, amt: number): V3 => [amt * 0.012 * (Math.sin(t * 0.83) + 0.5 * Math.sin(t * 2.1 + 1)), amt * 0.008 * Math.sin(t * 1.07 + 2), amt * 0.01 * Math.sin(t * 0.61 + 4)];
const add = (a: V3, b: V3): V3 => [a[0] + b[0], a[1] + b[1], a[2] + b[2]];
/** distance that frames `h` metres of height with vertical fov `fov` (deg) */
const distFor = (h: number, fov: number) => h / 2 / Math.tan((fov * Math.PI) / 360);
/** a direction rotated `deg` around +Y */
const rotY = (v: Vector3, deg: number) => v.clone().applyAxisAngle(new Vector3(0, 1, 0), (deg * Math.PI) / 180);
const ease = (x: number) => { const k = Math.min(1, Math.max(0, x)); return k * k * (3 - 2 * k); };

/** in front of a subject at an angle (deg, + = toward his left), framing height h */
const front = (s: Subject, angle: number, h: number, fov: number, up = 0, lookUp = 0): { pos: V3; target: V3 } => {
  const d = distFor(h, fov);
  const dir = rotY(flat(s.fwd), angle);
  const p = V(s.face).addScaledVector(dir, d).add(new Vector3(0, up, 0));
  return { pos: A(p), target: [s.face[0], s.face[1] + lookUp, s.face[2]] };
};

/** Width / height of the picture. */
export const aspectOf = (wide?: boolean) => (wide ? 16 / 9 : 9 / 16);
/** Horizontal half-width of the frame per metre of distance at a vertical fov of `fov`°. */
const halfW = (fov: number, wide?: boolean) => Math.tan((fov * Math.PI) / 360) * aspectOf(wide);
/** Distance that fits two people (positions a, b) side by side when seen from direction dir. */
export const pairDist = (a: V3, b: V3, dir: V3, wide?: boolean) => {
  const right = new Vector3(-dir[2], 0, dir[0]).normalize();
  const w = Math.abs(V(b).sub(V(a)).dot(right)) / 2 + 0.45;
  // 16:9 has the width for both at full-body height: no need to stand back as far
  return Math.max(distFor(wide ? 2.2 : 3.0, 42), w / halfW(42, wide));
};
/**
 * The best direction for a wide shot of two people inside a set: both inside the frame, the camera
 * inside the room, as many faces toward the lens as possible. null when no direction fits.
 */
export const pairDir = (a: Subject, b: Subject, inside: (p: V3) => boolean, wide?: boolean): V3 | null => {
  const mid = V(a.pos).add(V(b.pos)).multiplyScalar(0.5);
  let best: V3 | null = null, bestS = -Infinity;
  for (let deg = 0; deg < 360; deg += 10) {
    const dir: V3 = [Math.sin((deg * Math.PI) / 180), 0, Math.cos((deg * Math.PI) / 180)];
    const d = pairDist(a.pos, b.pos, dir, wide);
    // (16:9: a camera that would stand just outside is pulled in by safeCam — the two still fit the width)
    const k = wide ? 0.85 : 1;
    const p: V3 = [mid.x + dir[0] * d * k, 1.7, mid.z + dir[2] * d * k];
    if (!inside(p)) continue;
    const face = (s: Subject) => Math.max(-0.5, flat(s.fwd).dot(V(dir)));
    // 16:9: side by side across the width beats one person behind the other (two profiles are fine)
    const spread = wide ? 0.9 * Math.abs(V(b.pos).sub(V(a.pos)).setY(0).normalize().dot(new Vector3(-dir[2], 0, dir[0]))) : 0;
    const score = face(a) + face(b) - 0.08 * d + spread;
    if (score > bestS) { bestS = score; best = dir; }
  }
  // two people in profile (facing each other) score about 0; both turned away score below −0.5
  return best && bestS > -0.45 ? best : null;
};

  establish: (S, sp, t) => {
    const s = S[sp.a], k = ease((t - sp.t0) / Math.max(0.5, sp.t1 - sp.t0));

    const f = sp.wide ? front(s, 28 * (sp.side ?? 1), 1.42, 40, 0.1, 0.1) : front(s, 28 * (sp.side ?? 1), 1.75, 40, 0.15, 0.22);

/**
 * 16:9 only: the picture slid sideways so `subject` sits `k` half-frames off centre (k > 0: left of centre,
 * the room on the right) — looking room for a single person instead of an empty frame around a centred one.
 */
export const leadCam = (c: Cam, subject: V3, k: number): Cam => {
  if (!k) return c;
  const view = V(c.target).sub(V(c.pos)).normalize();
  const right = new Vector3().crossVectors(view, new Vector3(0, 1, 0)).normalize();
  const s = k * V(subject).sub(V(c.pos)).dot(view) * halfW(c.fov, true);
  return { ...c, pos: A(V(c.pos).addScaledVector(right, s)), target: A(V(c.target).addScaledVector(right, s)) };
};
/** Where a point lands in a camera's picture: x in half-frames from the centre (±1 = the frame edge), its depth
 *  along the lens axis (m) and the half-width of the picture at that depth (m). */
export const screenX = (c: Cam, p: V3, wide?: boolean): { x: number; depth: number; half: number } => {
  const view = V(c.target).sub(V(c.pos)).normalize();
  const right = new Vector3().crossVectors(view, new Vector3(0, 1, 0)).normalize();
  const rel = V(p).sub(V(c.pos)), depth = rel.dot(view), half = Math.max(1e-6, depth) * halfW(c.fov, wide);
  return { x: rel.dot(right) / half, depth, half };
};

/** The camera moved around `pivot` (a vertical axis) by `deg`; it keeps looking at the same target. */
export const orbitCam = (c: Cam, pivot: V3, deg: number): Cam => {
  if (!deg) return c;
  const off = V(c.pos).sub(V(pivot));
  return { ...c, pos: A(V(pivot).add(rotY(off, deg))) };
};

/** True when someone else's head or chest is within `clear` m of the line from the lens to the subject's face. */
export const peopleBlock = (c: Cam, me: Subject, others: Subject[], clear = 0.3) => {
  const p = V(c.pos), f = V(me.face), d = f.clone().sub(p);
  const len = d.length();
  d.normalize();
  return others.some((o) => [V(o.face), V(o.face).add(new Vector3(0, -0.35, 0))].some((q) => {
    const s = q.clone().sub(p).dot(d);
    if (s < 0 || s > len - 0.25) return false;
    return p.clone().addScaledVector(d, s).distanceTo(q) < clear;
  }));
};

/** True when this camera sits on the set's open side (where cameras belong). */
export const onViewSide = (c: Cam, env: EnvData) => V(c.pos).sub(V(c.target)).setY(0).dot(V(env.viewSide)) >= -0.05;

/**
 * Keep a camera inside the environment, outside its keep-out spheres, and with nothing between the
 * lens and the subject: anything in the way makes the camera slide along the view line past it.
 */
export const safeCam = (c: Cam, env: EnvData): Cam => {
  const p = V(c.pos), tg = V(c.target);
  for (let i = 0; i < 8; i++) {
    let moved = false;
    for (const b of env.blockers) {
      const C = V(b.c), dir = tg.clone().sub(p);
      const len = dir.length();
      dir.normalize();
      const s = Math.min(len, Math.max(0, C.clone().sub(p).dot(dir)));
      const closest = p.clone().addScaledVector(dir, s);
      if (closest.distanceTo(C) < b.r && s < len - 0.6) {
        // past the far side of the sphere along the view line (never closer than 0.6 m to the subject)
        const half = Math.sqrt(Math.max(0, b.r * b.r - closest.distanceToSquared(C)));
        p.addScaledVector(dir, Math.min(len - 0.6, s + half + 0.12));
        moved = true;
      }
    }
    if (!moved) break;
  }
  const { min, max } = env.bounds;
  p.set(Math.min(max[0], Math.max(min[0], p.x)), Math.min(max[1], Math.max(min[1], p.y)), Math.min(max[2], Math.max(min[2], p.z)));
  return { ...c, pos: A(p) };
};

// Performance engine: turns a list of acting beats into a Pose at any time t (pure, no state).
//
// A beat sets any of five independent channels — body, left arm, right arm, face, look — and blends
// into it over `dur` seconds with an acting curve. Channels evaluate live (a moving IK target or a
// run cycle keeps moving during a blend). On top sit the procedural layers that keep a character
// alive between beats: breathing, blinks, eye saccades, idle sway and tremble.
//
// Beats know nothing about a topic: scenes give them times (usually voice-over cues) and presets.

import { bezier, clamp, remap } from "../../../core/motion/easing.ts";
import { hash1, noise1 } from "../lib/rng.ts";
import { BODY, type Anchor, type BodyName } from "./presets/bodies.ts";
import { EXPR, type ExprName } from "./presets/expressions.ts";
import { GESTURE, type Gesture, type GestureName } from "./presets/gestures.ts";
import { blend, clonePose, lerpArm, lerpFace, lerpHand, lerpN, type Face, type Pose } from "./pose.ts";
import { lookAngles, resolveIK } from "./skeleton.ts";
import type { CharacterSpec, V3 } from "./spec.ts";

export const CURVES = {
  /** default acting move: quick start, long settle */
  act: bezier(0.3, 0, 0.15, 1),
  inOut: bezier(0.65, 0, 0.35, 1),
  soft: (t: number) => 0.5 - 0.5 * Math.cos(Math.PI * clamp(t)),
  /** fast with a small overshoot (startle, snap) */
  snap: bezier(0.2, 0.9, 0.3, 1.18),
  /** anticipation: dips back before going (big body moves) */
  antic: bezier(0.55, -0.22, 0.25, 1.06),
  linear: (t: number) => clamp(t),
};
export type CurveName = keyof typeof CURVES;

type Val<T> = T | ((t: number) => T);
export type BodySel = BodyName | ((t: number) => Pose);
export type GestureSel = GestureName | Gesture | ((t: number) => Gesture);
export type FaceSel = ExprName | Face | { base: ExprName; mix?: ExprName; k?: number; add?: Partial<Face> };
export type LookSel = V3 | ((t: number) => V3) | null;

export type Beat = {
  at: number;
  dur?: number;
  curve?: CurveName;
  body?: BodySel;
  /** tweak the body pose after the preset (live) */
  bodyMod?: (p: Pose, t: number) => void;
  /** null hands the arm back to the body (e.g. the walk's arm swing) */
  armL?: GestureSel | null;
  armR?: GestureSel | null;
  arms?: GestureSel | null;
  face?: FaceSel;
  look?: LookSel;
  /** how far the head follows the look target (0…1; eyes always follow) */
  headFollow?: number;
  breath?: { amp?: number; rate?: number };
  tremble?: number;
  /** eye darting 0…1 */
  nervous?: number;
  blink?: boolean;
  /** deep breath (inhale ≈ 1.1 s, exhale ≈ 1.5 s) starting at `at` */
  sigh?: boolean;
  label?: string;
};

export type Performance = { spec: CharacterSpec; anchor: Val<Anchor>; beats: Beat[] };

// ── generic channel evaluation ───────────────────────────────────────────────
type Ch<T> = { at: number; dur: number; curve: CurveName; v: (t: number) => T };
const channel = <T>(list: Ch<T>[], t: number, mix: (a: T, b: T, k: number) => T, i = -1): T => {
  if (i < 0) {
    i = 0;
    while (i < list.length - 1 && list[i + 1].at <= t) i++;
  }
  const c = list[i];
  const cur = c.v(t);
  if (i === 0) return cur;
  const k = CURVES[c.curve](remap(t, c.at, c.at + Math.max(1e-3, c.dur)));
  if (k >= 1 && t >= c.at + c.dur) return cur;
  return mix(channel(list, t, mix, i - 1), cur, k);
};

const val = <T>(v: Val<T>, t: number): T => (typeof v === "function" ? (v as (t: number) => T)(t) : v);

// ── performance → compiled channels ─────────────────────────────────────────
type Compiled = {
  body: Ch<Pose>[]; armL: Ch<Gesture | null>[]; armR: Ch<Gesture | null>[]; face: Ch<Face>[];
  lookH: Ch<{ p: V3; w: number }>[]; lookE: Ch<{ p: V3; w: number }>[]; follow: Ch<number>[];
  scal: Ch<{ amp: number; rate: number; tremble: number; nervous: number }>[];
  blinks: number[]; sighs: number[];
};

const faceOf = (s: FaceSel): Face => {
  if (typeof s === "string") return EXPR[s];
  if ("base" in s) {
    let f = EXPR[s.base];
    if (s.mix) f = lerpFace(f, EXPR[s.mix], s.k ?? 0.5);
    return { ...f, ...(s.add ?? {}) };
  }
  return s;
};
const gestureOf = (s: GestureSel, t: number): Gesture => (typeof s === "string" ? GESTURE[s] : typeof s === "function" ? s(t) : s);

const cache = new WeakMap<Performance, Compiled>();
const compile = (perf: Performance): Compiled => {
  let c = cache.get(perf);
  if (c) return c;
  const beats = [...perf.beats].sort((a, b) => a.at - b.at);
  const spec = perf.spec;
  const anchor = (t: number) => val(perf.anchor, t);
  c = { body: [], armL: [], armR: [], face: [], lookH: [], lookE: [], follow: [], scal: [], blinks: [], sighs: [] };
  let sc = { amp: 1, rate: 0.24, tremble: 0, nervous: 0 };
  for (const b of beats) {
    const dur = b.dur ?? 0.45, curve = b.curve ?? "act";
    if (b.body) {
      const sel = b.body, mod = b.bodyMod;
      c.body.push({ at: b.at, dur, curve, v: (t) => {
        const p = typeof sel === "string" ? clonePose(BODY[sel](anchor(t), spec, t)) : sel(t);
        if (mod) mod(p, t);
        return p;
      } });
    }
    // `null` is a real value (give the arm back to the body), so only `undefined` falls back to `arms`
    const armL = b.armL !== undefined ? b.armL : b.arms, armR = b.armR !== undefined ? b.armR : b.arms;
    if (armL !== undefined) c.armL.push({ at: b.at, dur, curve, v: (t) => (armL === null ? null : gestureOf(armL, t)) });
    if (armR !== undefined) c.armR.push({ at: b.at, dur, curve, v: (t) => (armR === null ? null : gestureOf(armR, t)) });
    if (b.face) {
      const f = faceOf(b.face);
      c.face.push({ at: b.at, dur: b.dur ?? 0.3, curve, v: () => f });
    }
    if (b.look !== undefined) {
      const L = b.look;
      const v = (t: number) => (L === null ? { p: [0, 0, 0] as V3, w: 0 } : { p: val(L, t), w: 1 });
      c.lookH.push({ at: b.at, dur: dur * 1.15, curve: "inOut", v });
      c.lookE.push({ at: b.at, dur: Math.min(0.14, dur), curve: "act", v });
    }
    if (b.headFollow !== undefined) c.follow.push({ at: b.at, dur, curve: "inOut", v: () => b.headFollow! });
    if (b.breath || b.tremble !== undefined || b.nervous !== undefined) {
      sc = { amp: b.breath?.amp ?? sc.amp, rate: b.breath?.rate ?? sc.rate, tremble: b.tremble ?? sc.tremble, nervous: b.nervous ?? sc.nervous };
      const s = sc;
      c.scal.push({ at: b.at, dur: Math.max(0.6, dur), curve: "soft", v: () => s });
    }
    if (b.blink) c.blinks.push(b.at);
    if (b.sigh) c.sighs.push(b.at);
  }
  if (!c.body.length) c.body.push({ at: 0, dur: 0, curve: "act", v: (t) => BODY.stand(anchor(t), spec, t) });
  if (!c.face.length) c.face.push({ at: 0, dur: 0, curve: "act", v: () => EXPR.neutral });
  if (!c.armL.length) c.armL.push({ at: 0, dur: 0, curve: "act", v: () => null });
  if (!c.armR.length) c.armR.push({ at: 0, dur: 0, curve: "act", v: () => null });
  if (!c.lookH.length) { c.lookH.push({ at: 0, dur: 0, curve: "act", v: () => ({ p: [0, 0, 0], w: 0 }) }); c.lookE.push(c.lookH[0]); }
  if (!c.follow.length) c.follow.push({ at: 0, dur: 0, curve: "act", v: () => 0.85 });
  if (!c.scal.length) c.scal.push({ at: 0, dur: 0, curve: "act", v: () => sc });
  cache.set(perf, c);
  return c;
};

// ── procedural life layers ──────────────────────────────────────────────────
const blinkShape = (dt: number) => (dt < 0 ? 0 : dt < 0.065 ? dt / 0.065 : dt < 0.095 ? 1 : dt < 0.22 ? 1 - (dt - 0.095) / 0.125 : 0);
/** Natural blinks every 2–5 s (deterministic), plus forced ones. */
export const blinkAt = (t: number, forced: number[], seed = 1) => {
  let b = 0;
  for (const f of forced) b = Math.max(b, blinkShape(t - f));
  let tb = 0.7 + hash1(0, seed) * 1.5;
  for (let i = 1; tb <= t + 0.3 && i < 400; i++) {
    b = Math.max(b, blinkShape(t - tb));
    tb += 2.1 + 2.9 * hash1(i, seed);
    if (hash1(i, seed + 7) < 0.15) tb -= 1.7; // occasional double blink
  }
  return b;
};

/** Fixation jumps: piecewise-constant gaze offsets with 50 ms transitions. */
const saccade = (t: number, nervous: number, seed: number): [number, number] => {
  const rate = 0.9 - 0.6 * nervous;
  const i = Math.floor(t / rate), f = t / rate - i;
  const pick = (k: number): [number, number] => [(hash1(k, seed) - 0.5) * 2, (hash1(k, seed + 3) - 0.5) * 2];
  const a = pick(i - 1), b = pick(i);
  const k = clamp(f * rate / 0.05);
  const amp = 0.025 + 0.13 * nervous;
  return [lerpN(a[0], b[0], k) * amp, lerpN(a[1], b[1], k) * amp * 0.6];
};

/** Deep breath envelope at time since start (0…1 inhale, back to 0 on exhale). */
const sighShape = (dt: number) => (dt < 0 ? 0 : dt < 1.1 ? Math.sin((dt / 1.1) * Math.PI / 2) : dt < 2.6 ? Math.cos(((dt - 1.1) / 1.5) * Math.PI / 2) : 0);

const mixGesture = (a: Gesture | null, b: Gesture | null, k: number, base: Gesture): Gesture => {
  const A = a ?? base, B = b ?? base;
  return { arm: lerpArm(A.arm, B.arm, k), hand: lerpHand(A.hand, B.hand, k) };
};

export type Evaluated = { pose: Pose; look: V3 | null };

/** The pose of a performance at time t (seconds). */
export const perform = (perf: Performance, t: number, seed = 1): Evaluated => {
  const c = compile(perf);
  const spec = perf.spec;
  const body = channel(c.body, t, blend);
  const p = clonePose(body);

  // arms: resolve IK targets for this body, then blend like-for-like (chest space)
  const resolveG = (g: Gesture | null, side: 1 | -1): Gesture | null => {
    if (!g) return null;
    if (!g.arm.ik) return g;
    return { arm: { ...g.arm, ik: resolveIK(g.arm.ik, body, spec, side) }, hand: g.hand };
  };
  const baseL: Gesture = { arm: body.armL, hand: body.handL }, baseR: Gesture = { arm: body.armR, hand: body.handR };
  const gl = channel(c.armL.map((x) => ({ ...x, v: (tt: number) => resolveG(x.v(tt), 1) })), t, (a, b, k) => mixGesture(a, b, k, baseL));
  const gr = channel(c.armR.map((x) => ({ ...x, v: (tt: number) => resolveG(x.v(tt), -1) })), t, (a, b, k) => mixGesture(a, b, k, baseR));
  if (gl) { p.armL = gl.arm; p.handL = gl.hand; }
  if (gr) { p.armR = gr.arm; p.handR = gr.hand; }

  p.face = { ...channel(c.face, t, lerpFace) };
  const sc = channel(c.scal, t, (a, b, k) => ({ amp: lerpN(a.amp, b.amp, k), rate: lerpN(a.rate, b.rate, k), tremble: lerpN(a.tremble, b.tremble, k), nervous: lerpN(a.nervous, b.nervous, k) }));
  p.breath = sc.amp * body.breath;
  p.breathRate = sc.rate;
  p.tremble = sc.tremble;

  // deep breaths: shoulders rise, chest opens, head lifts
  let sg = 0;
  for (const s0 of c.sighs) sg = Math.max(sg, sighShape(t - s0));
  if (sg > 0) {
    p.shrug += 0.55 * sg;
    p.chest = [p.chest[0] - 0.07 * sg, p.chest[1], p.chest[2]];
    p.head = [p.head[0] - 0.08 * sg, p.head[1], p.head[2]];
  }
  // breathing shoulders + idle sway (more when nervous)
  const bs = Math.sin(t * p.breathRate * Math.PI * 2);
  p.shrug += 0.07 * p.breath * Math.max(0, bs);
  const sway = 0.012 + 0.02 * sc.nervous;
  p.spine = [p.spine[0] + sway * 0.6 * noise1(t * 0.35, seed), p.spine[1] + sway * noise1(t * 0.27, seed + 1), p.spine[2] + sway * 0.5 * noise1(t * 0.31, seed + 2)];
  p.head = [p.head[0] + sway * 1.4 * noise1(t * 0.5 + 9, seed), p.head[1] + sway * 1.6 * noise1(t * 0.43 + 4, seed + 1), p.head[2] + sway * noise1(t * 0.38, seed + 5)];

  // look: head follows the slow channel, eyes the fast one
  const lh = channel(c.lookH, t, (a, b, k) => ({ p: a.w === 0 ? b.p : b.w === 0 ? a.p : ([lerpN(a.p[0], b.p[0], k), lerpN(a.p[1], b.p[1], k), lerpN(a.p[2], b.p[2], k)] as V3), w: lerpN(a.w, b.w, k) }));
  const le = channel(c.lookE, t, (a, b, k) => ({ p: a.w === 0 ? b.p : b.w === 0 ? a.p : ([lerpN(a.p[0], b.p[0], k), lerpN(a.p[1], b.p[1], k), lerpN(a.p[2], b.p[2], k)] as V3), w: lerpN(a.w, b.w, k) }));
  const follow = channel(c.follow, t, lerpN);
  let headYaw = p.neck[1] + p.head[1], headPitch = p.neck[0] + p.head[0];
  if (lh.w > 0.001) {
    const [yaw0, pitch0] = lookAngles(body, spec, lh.p);
    const yaw = clamp(yaw0, -1.25, 1.25) * follow, pitch = clamp(pitch0, -0.8, 0.8) * follow;
    const w = lh.w;
    p.neck = [lerpN(p.neck[0], body.neck[0] * 0.4 + pitch * 0.35, w), lerpN(p.neck[1], yaw * 0.38, w), p.neck[2]];
    p.head = [lerpN(p.head[0], body.head[0] * 0.4 + pitch * 0.55, w), lerpN(p.head[1], yaw * 0.6, w), p.head[2]];
    headYaw = p.neck[1] + p.head[1];
    headPitch = p.neck[0] + p.head[0];
  }
  if (le.w > 0.001) {
    const [ey, ep] = lookAngles(body, spec, le.p);
    p.face.eyeX = lerpN(p.face.eyeX, clamp(ey - headYaw, -0.6, 0.6), le.w);
    p.face.eyeY = lerpN(p.face.eyeY, clamp(-(ep - headPitch), -0.45, 0.45), le.w);
  }
  const [sx, sy] = saccade(t, sc.nervous, seed);
  p.face.eyeX += sx;
  p.face.eyeY += sy;

  // blinks (not while the eyes are already shut)
  const bl = blinkAt(t, c.blinks, seed) * clamp(p.face.lid / 0.4);
  p.face.lid = p.face.lid * (1 - bl);
  p.face.squint = lerpN(p.face.squint, Math.max(p.face.squint, 0.3), bl);
  if (sg > 0) p.face.lid *= 1 - 0.25 * sg;

// Indoor set kit (Node-safe data). A room is a box — back wall at z0, side walls at x0 / x1, the camera
// side (+z) open — with windows and a door cut into the walls and a list of furniture pieces. Room.tsx
// draws a layout; roomEnv() (registry.ts) derives the director's data from it: camera bounds, keep-out
// zones, seats, marks and the light per time of day. A new indoor environment is one more layout here.

import type { V3 } from "../character/spec.ts";
import type { Seat } from "../locomotion/stand.ts";

export type Wall = "back" | "left" | "right";
/** an opening along a wall: back wall → x from a to b, side walls → z from a to b */
export type Opening = { wall: Wall; a: number; b: number; y0: number; y1: number; kind: "window" | "door" };
type XZ = [number, number];
/** yaw = the direction a person using the piece faces (0 = +z, toward the cameras) */
export type Piece =
  | { k: "desk"; at: XZ; yaw?: number; w?: number; d?: number; top?: ("laptop" | "books" | "mug" | "monitor" | "plant")[]; seat?: string; color?: string }
  | { k: "schoolDesk"; at: XZ; yaw?: number; seat?: string }
  | { k: "chair"; at: XZ; yaw?: number; seat?: string; color?: string }
  | { k: "stool"; at: XZ; yaw?: number; seat?: string }
  | { k: "sofa"; at: XZ; yaw?: number; w?: number; seats?: string[]; color?: string }
  | { k: "armchair"; at: XZ; yaw?: number; seat?: string; color?: string }
  | { k: "coffeeTable"; at: XZ; yaw?: number; w?: number; d?: number }
  | { k: "cafeTable"; at: XZ; r?: number; cups?: number }
  | { k: "counter"; at: XZ; yaw?: number; w?: number; d?: number; h?: number; machine?: boolean }
  | { k: "shelf"; at: XZ; yaw?: number; w?: number; h?: number }
  | { k: "board"; at: XZ; yaw?: number; w?: number; h?: number; y?: number; style?: "white" | "chalk" | "menu" }
  | { k: "tv"; at: XZ; yaw?: number; w?: number }
  | { k: "plant"; at: XZ; s?: number }
  | { k: "rug"; at: XZ; w: number; d: number; color: string; yaw?: number }
  | { k: "frame"; at: XZ; yaw?: number; y?: number; w?: number; h?: number; colors?: [string, string, string] }
  | { k: "clock"; at: XZ; yaw?: number; y?: number }
  | { k: "dispenser"; at: XZ; yaw?: number }
  | { k: "cabinet"; at: XZ; yaw?: number; w?: number; h?: number; d?: number; color?: string }
  | { k: "lamp"; at: XZ; y: number; style?: "pendant" | "tube" };

export type RoomLayout = {
  id: string;
  room: { x0: number; x1: number; z0: number; z1: number; h: number };
  openings: Opening[];
  pieces: Piece[];
  marks: Record<string, { pos: V3; yaw: number; what: string }>;
  palette: { wall: string; wallLow?: string; floor: string; floorKind: "wood" | "tile" | "carpet"; trim: string; accent: string; ceiling?: string };
  /** what is seen through the windows by day */
  view: "city" | "garden" | "street";
  /** sound of the room */
  ambience: "room" | "class" | "office" | "cafe";
  /** horizontal direction cameras prefer around a subject (default +z) */

const FW = (yaw: number): XZ => [Math.sin(yaw), Math.cos(yaw)];
const R = (yaw: number, lx: number, lz: number): XZ => [lx * Math.cos(yaw) + lz * Math.sin(yaw), -lx * Math.sin(yaw) + lz * Math.cos(yaw)];

/** Chair offset behind a desk (the sitter faces the desk's yaw). */
export const DESK_CHAIR = 0.62;
export const SEAT_H = { chair: 0.47, stool: 0.62, sofa: 0.45, schoolDesk: 0.45 };

/** Every seat a layout offers (pelvis point on the seat, facing yaw). */
export const roomSeats = (L: RoomLayout): Record<string, Seat> => {
  const out: Record<string, Seat> = {};
  for (const p of L.pieces) {
    const yaw = "yaw" in p && p.yaw !== undefined ? p.yaw : 0;
    const f = FW(yaw);
    if ((p.k === "desk" || p.k === "schoolDesk") && p.seat) {
      const back = p.k === "desk" ? DESK_CHAIR : 0.5;
      out[p.seat] = { pos: [p.at[0] - f[0] * back, 0, p.at[1] - f[1] * back], yaw, height: p.k === "desk" ? SEAT_H.chair : SEAT_H.schoolDesk };
    }
    if ((p.k === "chair" || p.k === "stool") && p.seat) out[p.seat] = { pos: [p.at[0] + f[0] * 0.03, 0, p.at[1] + f[1] * 0.03], yaw, height: p.k === "stool" ? SEAT_H.stool : SEAT_H.chair };
    if (p.k === "armchair" && p.seat) out[p.seat] = { pos: [p.at[0] + f[0] * 0.06, 0, p.at[1] + f[1] * 0.06], yaw, height: SEAT_H.sofa };
    if (p.k === "sofa" && p.seats) {
      const w = p.w ?? 2.0, n = p.seats.length;
      p.seats.forEach((id, i) => {
        const lx = -w / 2 + 0.25 + ((w - 0.5) * (i + 0.5)) / n;
        const o = R(yaw, lx, 0.06);
        out[id] = { pos: [p.at[0] + o[0], 0, p.at[1] + o[1]], yaw, height: SEAT_H.sofa };
      });
    }
  }
  return out;
};

/** Keep-out spheres for cameras: tall furniture and the walls' near zone. */
export const roomBlockers = (L: RoomLayout): { c: V3; r: number }[] => {
  const out: { c: V3; r: number }[] = [];
  for (const p of L.pieces) {
    const [x, z] = p.at;
    switch (p.k) {
      case "shelf": out.push({ c: [x, 1.0, z], r: Math.max(0.55, (p.w ?? 1) * 0.55) }); break;
      case "counter": out.push({ c: [x, 0.6, z], r: 0.6 }); break;
      case "plant": out.push({ c: [x, 0.8, z], r: 0.45 * (p.s ?? 1) }); break;
      case "dispenser": out.push({ c: [x, 0.7, z], r: 0.35 }); break;
      case "cabinet": out.push({ c: [x, (p.h ?? 0.9) / 2, z], r: 0.45 }); break;
      case "desk": out.push({ c: [x, 0.5, z], r: 0.5 }); break;
      case "cafeTable": out.push({ c: [x, 0.5, z], r: 0.4 }); break;
      case "sofa": case "armchair": out.push({ c: [x, 0.45, z], r: 0.55 }); break;
      case "lamp": out.push({ c: [x, p.y, z], r: 0.3 }); break;
      default: break;
    }
  }
  return out;
};

// field's behaviour over time is a FieldMood that the scene computes (drift → swirl, calm → ghost →
// alarm, freeze, dissolve). Large bubbles hold staged mini scenes (vignettes.tsx) or a glyph; the swarm
// is instanced (cheap at 100+). Everything glowing sits on FX_LAYER so depth passes ignore it.
//
// Placement frame: `back` is the direction from the camera side toward the set (bubbles start behind
// and beside the head); when the swirl carries a bubble round to the camera side it is pushed up/down
// and outward so it never covers the face.

import React, { useMemo } from "react";
import {
  AdditiveBlending,
  Color,
  DoubleSide,
  InstancedMesh,
  Matrix4,
  MeshBasicMaterial,
  PlaneGeometry,
  Quaternion,
  SphereGeometry,
  Vector3,
} from "three";
import { FX_LAYER } from "../../../core/render/Post.tsx";
import { setMiniFade } from "../character/materials.ts";
import { BIMA, BOSS, FRIEND, NAYA, RAKA, RAKA_OUT, type V3 } from "../character/spec.ts";

export type ThoughtDef = {
  id: string;
  at: number;
  vignette?: VignetteName;
  icon?: IconName;
  /** offset from the field centre in the field frame: [right, up, back] metres (right/back are
   *  horizontal and follow `back`, the direction away from the main camera) */
  slot: [number, number, number];
  size: number;
  trail?: boolean;
  /** scale with the alarm pulse */
  pulses?: boolean;
  /** this bubble only exists from `at` until `until` */
  until?: number;
  /** who plays in the vignette */
  cast?: Cast;
  /** short text written inside the bubble (any topic, no vignette needed) */
  text?: string;
};

export type FieldMood = {
  /** accumulated orbit angle (rad) — the scene integrates its own speed */
  phase: number;
  /** motion clock (stops at a freeze) */
  tm: number;
  agitation: number;
  ghost: number;
  alarm: number;
  pulse: number;
  dissolve: number;
  tight: number;
  /** overall visibility of the large bubbles */
  show: number;
};

export type SwarmDef = { at: number; spread: number; count: number; rMin: number; rMax: number };

const COOL = { disc: "#f3f9ff", rim: "#8fdcff", glow: "#7fd0ff", ink: "#2f5d8a" };
const HOT = { disc: "#ffd9d2", rim: "#ff4a3d", glow: "#ff3d2e", ink: "#b3201b" };
const GHOST = { disc: "#c9d3e0", rim: "#b8c6d8", glow: "#8aa0c0", ink: "#7a8aa3" };

const colors = (m: FieldMood) => {
  const pick = (k: keyof typeof COOL) => mixHex(mixHex(COOL[k], GHOST[k], m.ghost), HOT[k], m.alarm);
  return { disc: pick("disc"), rim: pick("rim"), glow: pick("glow"), ink: pick("ink") };
};

const popIn = (x: number) => {
  if (x <= 0) return 0;
  if (x >= 1) return 1;
  const c1 = 1.9, c3 = c1 + 1;
  return 1 + c3 * Math.pow(x - 1, 3) + c1 * Math.pow(x - 1, 2);
};

const billboard = (pos: V3, cam: V3) =>
  new Quaternion().setFromRotationMatrix(new Matrix4().lookAt(new Vector3(...cam), new Vector3(...pos), new Vector3(0, 1, 0)));

/** World position of a slot at time t. The (right, back) part orbits round the vertical axis with the
 *  field phase; while it passes the camera side it is pushed up/down and outward, clear of the face. */
export const slotPos = (slot: [number, number, number], center: V3, back: V3, m: FieldMood, seed: number): V3 => {
  const [x0, y0, d0] = slot;
  const b = new Vector3(...back).setY(0).normalize();
  const rt = new Vector3(-b.z, 0, b.x);
  const ang = Math.atan2(x0, d0) + m.phase * (0.8 + 0.4 * hash1(seed, 3)) + 0.1 * Math.sin(m.tm * 0.6 + seed);
  const rad = Math.hypot(x0, d0) * m.tight;
  const front = Math.max(0, -Math.cos(ang));
  const y = y0 + Math.sign(y0 - 0.05 || 1) * 0.45 * front + 0.035 * Math.sin(m.tm * (1.1 + m.agitation * 3) + seed * 1.7) * (1 + 2 * m.agitation);
  const r = rad * (1 + 0.5 * front) + 0.015 * Math.sin(m.tm * 2.3 + seed);
  const d = b.clone().multiplyScalar(Math.cos(ang) * r).addScaledVector(rt, Math.sin(ang) * r);
  return [center[0] + d.x, center[1] + y, center[2] + d.z];
};

/** 0 when a bubble would sit on the lens, 1 from a comfortable distance. */
const nearFade = (p: V3 | Vector3, cam: V3) => {
  const v = p instanceof Vector3 ? p : new Vector3(...p);
  const dd = v.distanceTo(new Vector3(...cam));
  const k = Math.min(1, Math.max(0, (dd - 0.45) / 0.5));
  return k * k * (3 - 2 * k);
};

const glowMat = () => new MeshBasicMaterial({ map: haloTex(), transparent: true, depthWrite: false, blending: AdditiveBlending, toneMapped: false });
const mats = new Map<string, MeshBasicMaterial>();
const M = (key: string, make: () => MeshBasicMaterial) => {
  let m = mats.get(key);
  if (!m) { m = make(); mats.set(key, m); }
  return m;
};
const unit = () => cached("cfunitplane", () => new PlaneGeometry(1, 1));

const Bubble: React.FC<{ d: ThoughtDef; i: number; t: number; pos: V3; cam: V3; head: V3; mood: FieldMood; txt: VignetteText; C: ReturnType<typeof colors> }> = ({ d, i, t, pos, cam, head, mood, txt, C }) => {
  const age = t - d.at;
  const out = d.until !== undefined ? 1 - Math.min(1, Math.max(0, (t - d.until) / 0.3)) : 1;
  let s = d.size * popIn(age / 0.45) * out * mood.show * (1 - 0.85 * mood.dissolve);
  if (d.pulses) s *= 1 + 0.12 * mood.pulse;
  s *= nearFade(pos, cam);
  if (s <= 0.002) return null;
  const q = billboard(pos, cam);
  const flick = mood.ghost > 0 ? 1 - mood.ghost * (0.28 + 0.22 * Math.sin(t * 23 + i * 2.1)) : 1;
  const disc = M(`disc${i}`, () => new MeshBasicMaterial({ map: bubbleTex(), transparent: true, depthWrite: false, side: DoubleSide }));
  disc.color.set(C.disc);
  disc.opacity = 0.97 * flick * (1 - mood.dissolve);
  const halo = M(`halo${i}`, glowMat);
  halo.color.set(C.glow);
  halo.opacity = (0.3 + 0.25 * mood.alarm * mood.pulse) * (1 - mood.dissolve);
  const rim = M(`rim${i}`, () => new MeshBasicMaterial({ toneMapped: false }));
  rim.color.set(C.rim).multiplyScalar(1.3 + 0.8 * mood.alarm * mood.pulse);
  const ink = M(`ink${i}`, () => new MeshBasicMaterial({ transparent: true, depthWrite: false, side: DoubleSide }));
  if (d.icon) ink.map = iconTex(d.icon);
  ink.color.set(C.ink);
  ink.opacity = flick * (1 - mood.dissolve);
  const dashes = 28, dashK = 1 - 0.55 * mood.ghost;
  const trail = d.trail ? [0.3, 0.5, 0.68].map((f) => [head[0] + (pos[0] - head[0]) * f, head[1] + 0.12 + (pos[1] - head[1] - 0.12) * f, head[2] + (pos[2] - head[2]) * f] as V3) : [];
  return (
    <>
      <group position={pos} quaternion={q} scale={s}>
        <mesh geometry={unit()} material={halo} scale={3.2} position={[0, 0, -0.05]} layers={FX_LAYER} />
        <mesh geometry={cached("cfbubbledisc", () => new PlaneGeometry(2.04, 2.04))} material={disc} layers={FX_LAYER} />
        {Array.from({ length: dashes }, (_, k) => (
          <mesh key={k} geometry={torus(1.0, 0.03, 6, 6, ((Math.PI * 2) / dashes) * dashK)} material={rim} rotation={[0, 0, (k / dashes) * Math.PI * 2 + mood.tm * 0.4]} layers={FX_LAYER} />
        ))}
        {d.icon ? <mesh geometry={unit()} material={ink} scale={1.25} position={[0, 0, 0.02]} layers={FX_LAYER} /> : null}
        {d.text ? <mesh geometry={unit()} material={M(`txt${d.text}`, () => new MeshBasicMaterial({ map: textTex(d.text!), transparent: true, depthWrite: false, toneMapped: false }))} scale={1.7} position={[0, 0, 0.025]} layers={FX_LAYER} /> : null}
        {d.vignette ? (
          <group position={[0, -0.8, 0.08]} scale={0.6}>
            <Vignette name={d.vignette} t={t + i * 1.3} txt={txt} cast={d.cast} />
          </group>
        ) : null}
      </group>
      {trail.map((p, k) => (
        <mesh key={`tr${k}`} geometry={sphere(1, 16, 12)} material={disc} position={p} scale={s * (0.08 + 0.07 * k)} layers={FX_LAYER} />
      ))}
    </>
  );
};

// ── swarm (instanced) ───────────────────────────────────────────────────────
type Inst = { dir: Vector3; r: number; at: number; icon: number; size: number; seed: number };
const swarmData = (S: SwarmDef, back: V3): Inst[] => {
  const out: Inst[] = [];
  const b = new Vector3(...back).normalize();
  const golden = Math.PI * (3 - Math.sqrt(5));
  for (let i = 0, k = 0; out.length < S.count && k < S.count * 4; k++) {
    const y = 1 - (k / (S.count * 4 - 1)) * 2;
    const rr = Math.sqrt(1 - y * y);
    const dir = new Vector3(Math.cos(golden * k) * rr * 0.62, y * 1.1 + 0.2, Math.sin(golden * k) * rr * 0.62);
    // keep the camera side clear (nothing between the lens and the face)
    if (new Vector3(dir.x, 0, dir.z).normalize().dot(b) < -0.15 && dir.y < 1.1) continue;
    if (hash1(k, 41) < 0.5) continue;
    out.push({ dir, r: S.rMin + (S.rMax - S.rMin) * hash1(k, 5), at: S.at + S.spread * hash1(i, 9), icon: i % ICONS.length, size: 0.05 + 0.05 * hash1(k, 13), seed: k });
    i++;
  }
  return out;
};

const Swarm: React.FC<{ S: SwarmDef; t: number; center: V3; back: V3; cam: V3; mood: FieldMood; C: ReturnType<typeof colors> }> = ({ S, t, center, back, cam, mood, C }) => {
  const data = useMemo(() => swarmData(S, back), [S, back]);
  const meshes = useMemo(() => {
    const discs = new InstancedMesh(cached("cfswarmdisc", () => new PlaneGeometry(2.04, 2.04)), new MeshBasicMaterial({ map: bubbleTex(), transparent: true, depthWrite: false, side: DoubleSide }), data.length);
    const icons = ICONS.map((name, k) => new InstancedMesh(unit(), new MeshBasicMaterial({ map: iconTex(name), transparent: true, depthWrite: false, side: DoubleSide }), data.filter((d) => d.icon === k).length));
    const glow = new InstancedMesh(unit(), new MeshBasicMaterial({ map: haloTex(), transparent: true, depthWrite: false, blending: AdditiveBlending, toneMapped: false }), data.length);
    for (const m of [discs, glow, ...icons]) { m.frustumCulled = false; m.layers.set(FX_LAYER); }
    return { discs, icons, glow };
  }, [data]);
  const q = billboard(center, cam);
  const counters = ICONS.map(() => 0);
  const mx = new Matrix4(), mg = new Matrix4();
  data.forEach((d, i) => {
    const age = t - d.at;
    const k = popIn(age / 0.5);
    const sp = mood.phase * (0.6 + 0.5 * hash1(d.seed, 2));
    const dir = d.dir.clone().applyAxisAngle(new Vector3(0, 1, 0), sp);
    // swirled round to the camera side: lift away from the face, push out and shrink
    const hb = new Vector3(dir.x, 0, dir.z).normalize().dot(new Vector3(...back).setY(0).normalize());
    const front = Math.max(0, -hb);
    dir.y += 0.7 * front * Math.sign(dir.y + 0.05);
    const r = d.r * mood.tight * (1 + 0.5 * front) * (0.35 + 0.65 * Math.min(1, Math.max(0, age / 0.6)));
    const bob = 0.03 * Math.sin(mood.tm * (1.3 + 2.5 * mood.agitation) + d.seed) * (1 + 2 * mood.agitation);
    const up = mood.dissolve * (0.25 + 0.5 * hash1(d.seed, 17));
    const p = new Vector3(center[0] + dir.x * r, center[1] + dir.y * r + bob + up, center[2] + dir.z * r);
    const s = d.size * k * (1 - mood.dissolve) * mood.show * (1 + 0.15 * mood.pulse * mood.alarm) * nearFade(p, cam) * (1 - 0.6 * front);
    mx.compose(p, q, new Vector3(s, s, s));
    mg.compose(p.clone().addScaledVector(new Vector3(...back).normalize(), 0.01), q, new Vector3(s * 3, s * 3, s * 3));
    meshes.discs.setMatrixAt(i, mx);
    meshes.glow.setMatrixAt(i, mg);
    const ic = meshes.icons[d.icon];
    const p2 = p.clone().addScaledVector(new Vector3(...cam).sub(p).normalize(), 0.004);
    mx.compose(p2, q, new Vector3(s * 1.2, s * 1.2, s * 1.2));
    ic.setMatrixAt(counters[d.icon]++, mx);
  });
  for (const m of [meshes.discs, meshes.glow, ...meshes.icons]) m.instanceMatrix.needsUpdate = true;
  const flick = 1 - mood.ghost * 0.35;
  (meshes.discs.material as MeshBasicMaterial).color.set(C.disc);
  (meshes.discs.material as MeshBasicMaterial).opacity = 0.9 * flick;
  (meshes.glow.material as MeshBasicMaterial).color.set(C.glow);
  (meshes.glow.material as MeshBasicMaterial).opacity = 0.22 + 0.2 * mood.alarm * mood.pulse;
  for (const ic of meshes.icons) {
    (ic.material as MeshBasicMaterial).color.set(C.ink);
    (ic.material as MeshBasicMaterial).opacity = flick;
  }
  return (
    <>
      <primitive object={meshes.glow} />
      <primitive object={meshes.discs} />
      {meshes.icons.map((m, k) => <primitive key={k} object={m} />)}
    </>
  );
};

// ── motes: what the thoughts dissolve into ──────────────────────────────────
const Motes: React.FC<{ t: number; from: V3[]; t0: number; cam: V3 }> = ({ t, from, t0, cam }) => {
  const N = 140;
  const mesh = useMemo(() => {
    const m = new InstancedMesh(cached("cfmote", () => new SphereGeometry(1, 8, 6)), new MeshBasicMaterial({ color: new Color("#ffe3a8"), transparent: true, depthWrite: false, blending: AdditiveBlending, toneMapped: false }), N);
    m.frustumCulled = false;
    m.layers.set(FX_LAYER);
    return m;
  }, []);
  const age = t - t0;
  const mx = new Matrix4();
  for (let i = 0; i < N; i++) {
    const src = from[i % Math.max(1, from.length)] ?? [0, 0, 0];
    const life = 2.2 + 1.6 * hash1(i, 3);
    const a = Math.max(0, age - 0.25 * hash1(i, 7));
    const k = Math.min(1, a / life);
    const p = new Vector3(src[0] + (hash1(i, 11) - 0.5) * 0.3 + 0.1 * Math.sin(a * 1.3 + i), src[1] + (hash1(i, 13) - 0.5) * 0.3 + a * (0.12 + 0.12 * hash1(i, 19)), src[2] + (hash1(i, 23) - 0.5) * 0.3 + 0.1 * Math.cos(a * 1.1 + i));
    const s = age <= 0 ? 0 : 0.006 * (1 + hash1(i, 29)) * Math.sin(Math.PI * k);
    mx.compose(p, new Quaternion(), new Vector3(s, s, s));
    mesh.setMatrixAt(i, mx);
  }
  mesh.instanceMatrix.needsUpdate = true;
  void cam;
  return <primitive object={mesh} />;
};

export const ThoughtField: React.FC<{
  defs: ThoughtDef[];
  swarm?: SwarmDef;
  t: number;
  center: V3;
  head: V3;
  back: V3;
  cam: V3;
  mood: FieldMood;
  txt: Record<string, VignetteText>;
  motesAt?: number;
}> = ({ defs, swarm, t, center, head, back, cam, mood, txt, motesAt }) => {
  const C = colors(mood);
  const miniFade = (1 - 0.5 * mood.ghost) * (1 - mood.dissolve);
  for (const s of [RAKA, RAKA_OUT, BIMA, FRIEND, BOSS, NAYA]) setMiniFade(s, miniFade);
  const live = defs.filter((d) => t >= d.at);
  const positions = live.map((d, i) => slotPos(d.slot, center, back, mood, i + 1));
  return (
    <group>
      {live.map((d, i) => (
        <Bubble key={d.id} d={d} i={i} t={t} pos={positions[i]} cam={cam} head={head} mood={mood} txt={txt[d.id] ?? {}} C={C} />
      ))}
      {swarm && t >= swarm.at ? <Swarm S={swarm} t={t} center={center} back={back} cam={cam} mood={mood} C={C} /> : null}
      {motesAt !== undefined && t >= motesAt ? <Motes t={t} from={positions} t0={motesAt} cam={cam} /> : null}
    </group>
  );
};

// Imagined situations: tiny staged scenes that live inside thought bubbles. They are cast from the
// SAME character system (spec + presets + performance) in the "mini" material variant, so the lead is
// recognisable inside his own thoughts. Each vignette is built in a 1-unit stage (feet at y = 0,
// facing the viewer at +Z) and scaled into the bubble by Thoughts.tsx.
// Text on screens/phones comes from the topic (props), so the vignettes are reusable.

import React from "react";
import { PlaneGeometry } from "three";
import { Character } from "../character/Character.tsx";
import { perform, type Beat, type Performance } from "../character/performance.ts";
import { BIMA, BOSS, FRIEND, NAYA, RAKA, RAKA_OUT, type CharacterSpec } from "../character/spec.ts";

const perfs = new Map<string, Performance>();
const perf = (key: string, spec: CharacterSpec, x: number, z: number, yaw: number, beats: Omit<Beat, "at">[]): Performance => {
  let p = perfs.get(key);
  if (!p) {
    p = { spec, anchor: { x, y: 0, z, yaw }, beats: beats.map((b, i) => ({ at: i * 0.9, dur: 0.4, ...b })) };
    perfs.set(key, p);
  }
  return p;
};

const Mini: React.FC<{ p: Performance; t: number; seed: number }> = ({ p, t, seed }) => {
  const { pose } = perform(p, t, seed);
  return <Character pose={pose} spec={p.spec} t={t} variant="mini" shadows={false} />;
};

const plane = (w: number, h: number) => cached(`cfvplane${w},${h}`, () => new PlaneGeometry(w, h));
