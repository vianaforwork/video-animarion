// Word timings for a voice-over → JSON [[word, start, end], …] (VO seconds).
//   node scripts/transcribe.mjs <vo.mp3> --out=stories/<id>/words.json [--script=script.txt] [--lang=id]
//   node scripts/transcribe.mjs <any/path/vo.mp3> --id=<id> [--script=…]   copies the VO to public/audio/<id>.<ext>
//                                                                          and writes stories/<id>/words.json
//
// 1. whisper.cpp when available (WHISPER_CLI + WHISPER_MODEL, or `whisper-cli` on PATH with a model in
//    ~/.cache/whisper.cpp or ./models) — real word timings.
// 2. otherwise, with the script text: the VO's speech segments (silence detection) are matched to the
//    script's sentences and words are spread by length inside each segment — close enough for beats
//    and captions, not exact per word.
// Without whisper and without a script there is nothing to align: the script text must be provided.

import { spawnSync } from "node:child_process";
import fs from "node:fs";
import os from "node:os";
import path from "node:path";

const args = process.argv.slice(2);
const file = args.find((a) => !a.startsWith("--"));
const opt = (k, d) => (args.find((a) => a.startsWith(`--${k}=`)) ?? `=${d ?? ""}`).split("=").slice(1).join("=");
const id = opt("id");
if (id && !/^[a-z0-9-]+$/.test(id)) { console.error(`--id "${id}" must use only a-z, 0-9 and "-"`); process.exit(1); }
const out = opt("out") || (id ? `stories/${id}/words.json` : "");
if (file && fs.existsSync(file) && id) {
  const dst = path.join("public", "audio", `${id}${path.extname(file).toLowerCase() || ".mp3"}`);
  fs.mkdirSync(path.dirname(dst), { recursive: true });
  if (path.resolve(dst) !== path.resolve(file)) fs.copyFileSync(file, dst);
  console.log(`vo: ${dst}  →  story.json "vo": { "file": "${path.relative("public", dst)}", "offset": 0.4 }`);
}
if (!file || !fs.existsSync(file) || !out) {
  console.error("usage: node scripts/transcribe.mjs <vo.mp3> (--id=<id> | --out=stories/<id>/words.json) [--script=script.txt] [--lang=id]");
  process.exit(1);
}
const run = (cmd, a) => spawnSync(cmd, a, { encoding: "utf8", maxBuffer: 1 << 28 });
const duration = Number(run("ffprobe", ["-v", "error", "-show_entries", "format=duration", "-of", "csv=p=0", file]).stdout.trim());

const findWhisper = () => {
  const home = os.homedir();
  const cands = [process.env.WHISPER_CLI, run("sh", ["-c", "command -v whisper-cli"]).stdout.trim(), "/opt/homebrew/bin/whisper-cli", "/usr/local/bin/whisper-cli"].filter(Boolean);
  const cli = cands.find((c) => fs.existsSync(c));
  const modelDirs = [path.join(home, ".cache/whisper.cpp"), path.resolve("models"), "/opt/homebrew/share/whisper-cpp"];
  let model = process.env.WHISPER_MODEL && fs.existsSync(process.env.WHISPER_MODEL) ? process.env.WHISPER_MODEL : null;
  if (!model) for (const d of modelDirs) { if (!fs.existsSync(d)) continue; const m = fs.readdirSync(d).filter((f) => f.endsWith(".bin")).sort((a, b) => (b.includes("large") ? 1 : 0) - (a.includes("large") ? 1 : 0))[0]; if (m) { model = path.join(d, m); break; } }
  return cli && model ? { cli, model } : null;
};

let words = null, method = "";
const W = findWhisper();
if (W) {
  const tmp = fs.mkdtempSync(path.join(os.tmpdir(), "cf-vo-"));
  const wav = path.join(tmp, "vo16k.wav");
  run("ffmpeg", ["-v", "error", "-y", "-i", file, "-ar", "16000", "-ac", "1", wav]);
  run(W.cli, ["-m", W.model, "-f", wav, "-l", opt("lang", "id"), "-ml", "1", "-sow", "-ojf", "-of", path.join(tmp, "w")]);
  try {
    const j = JSON.parse(fs.readFileSync(path.join(tmp, "w.json"), "utf8"));
    words = j.transcription.map((s) => [s.text.trim(), s.offsets.from / 1000, s.offsets.to / 1000]).filter((w) => w[0]);
    method = `whisper.cpp (${path.basename(W.model)})`;
  } catch { words = null; }
  fs.rmSync(tmp, { recursive: true, force: true });
}

if (!words) {
  const scriptFile = opt("script");
  if (!scriptFile || !fs.existsSync(scriptFile)) {
    console.error("No whisper.cpp found and no --script given. Provide the narration text (e.g. --script=stories/<id>/script.txt) or install whisper.cpp (see the skill's references/setup.md).");
    process.exit(2);
  }
  const text = fs.readFileSync(scriptFile, "utf8").replace(/\s+/g, " ").trim();
  // speech segments from silence detection
  const log = run("ffmpeg", ["-v", "info", "-i", file, "-af", "silencedetect=n=-38dB:d=0.18", "-f", "null", "-"]).stderr;
  const st = [...log.matchAll(/silence_start: ([\d.]+)/g)].map((m) => Number(m[1]));
  const en = [...log.matchAll(/silence_end: ([\d.]+)/g)].map((m) => Number(m[1]));
  const segs = [];
  let cur = 0;
  st.forEach((s, i) => { if (s - cur > 0.12) segs.push([cur, s]); cur = en[i] ?? duration; });
  if (duration - cur > 0.12) segs.push([cur, duration]);
  const toks = text.split(" ").filter(Boolean);
  // sentences / clauses = groups ending in punctuation; map groups to segments by cumulative length
  const total = toks.reduce((s, w) => s + w.length + 1, 0);
  const speech = segs.reduce((s, [a, b]) => s + (b - a), 0);
  // speech-time position for a fraction of the text, then into real time through the segments
  const toTime = (frac) => {
    let need = frac * speech;
    for (const [a, b] of segs) { if (need <= b - a) return a + need; need -= b - a; }
    return segs.length ? segs[segs.length - 1][1] : duration;
  };
  // snap clause ends to the nearest segment end when one is close
  let acc = 0;
  words = toks.map((w) => {
    const a = acc / total;
    acc += w.length + 1;
    const b = (acc - 1) / total;
    let t0 = toTime(a), t1 = toTime(b);
    if (/[.,?!;:]$/.test(w)) { const near = segs.map((s) => s[1]).reduce((best, e) => (Math.abs(e - t1) < Math.abs(best - t1) ? e : best), t1); if (Math.abs(near - t1) < 0.6) t1 = near; }
    return [w, +t0.toFixed(2), +Math.max(t0 + 0.08, t1).toFixed(2)];
  });
  // keep times monotonic
  for (let i = 1; i < words.length; i++) if (words[i][1] < words[i - 1][2]) words[i][1] = words[i - 1][2];
  method = `script aligned to ${segs.length} speech segments (approximate word timing)`;
}

fs.mkdirSync(path.dirname(out), { recursive: true });
fs.writeFileSync(out, JSON.stringify(words.map(([w, a, b]) => [w, +a.toFixed(2), +b.toFixed(2)])));
console.log(`words: ${out} — ${words.length} words, ${duration.toFixed(2)} s, via ${method}`);
console.log(words.map(([w, a]) => `${a.toFixed(2)} ${w}`).join("  "));

// Full-frame 2D cards for storytelling beats: a cartoon "time skip" title card and a VHS-style rewind
// overlay. Laid out at 1080 × 1920 (a 16:9 film passes its own `top` for the 1920 × 1080 canvas); text comes from the topic.

import React from "react";
import { ease, remap } from "../../../core/motion/easing.ts";
import { FONT } from "../../../core/type/fonts.ts";

/** Bouncy title card over the shot (e.g. "5 DETIK KEMUDIAN…"). */
export const TitleCard: React.FC<{ t: number; t0: number; t1: number; text: string; sub?: string; top?: number }> = ({ t, t0, t1, text, sub, top = 300 }) => {
  if (t < t0 || t > t1) return null;
  const inK = ease.out(remap(t, t0, t0 + 0.3));
  const outK = 1 - ease.in(remap(t, t1 - 0.25, t1));
  const pop = 0.7 + 0.3 * inK + 0.06 * Math.sin(Math.min(1, (t - t0) / 0.5) * Math.PI);
  return (
    <div style={{ position: "absolute", left: 0, right: 0, top, display: "flex", justifyContent: "center", opacity: outK }}>
      <div style={{ transform: `scale(${pop}) rotate(${-3 + 1.5 * Math.sin(t * 2)}deg)`, background: "#ffd166", color: "#1d2a4d", fontFamily: FONT, fontWeight: 600, fontSize: 78, letterSpacing: "0.02em", padding: "22px 46px 26px", borderRadius: 30, border: "8px solid #1d2a4d", boxShadow: "0 14px 0 #1d2a4d, 0 24px 60px rgba(0,0,0,0.25)", textAlign: "center" }}>
        {text}
        {sub ? <div style={{ fontSize: 38, fontWeight: 500, marginTop: 6 }}>{sub}</div> : null}
      </div>
    </div>
  );
};

/** VHS rewind: scan lines, a rolling bar, ◀◀ icon. */
export const RewindOverlay: React.FC<{ t: number; t0: number; t1: number; label?: string }> = ({ t, t0, t1, label }) => {
  if (t < t0 || t > t1) return null;
  const a = ease.out(remap(t, t0, t0 + 0.12)) * (1 - ease.in(remap(t, t1 - 0.15, t1)));
  const bar = ((t - t0) * 1.7) % 1;
  return (
    <div style={{ position: "absolute", inset: 0, opacity: a, pointerEvents: "none" }}>
      <div style={{ position: "absolute", inset: 0, background: "repeating-linear-gradient(0deg, rgba(0,0,0,0.18) 0px, rgba(0,0,0,0.18) 3px, transparent 3px, transparent 7px)" }} />
      <div style={{ position: "absolute", left: 0, right: 0, top: `${bar * 100}%`, height: 90, background: "linear-gradient(180deg, transparent, rgba(255,255,255,0.22), transparent)" }} />
      <div style={{ position: "absolute", left: 70, top: 120, fontFamily: FONT, fontWeight: 600, fontSize: 96, color: "#ffffff", textShadow: "4px 4px 0 #ff4fa3, -4px -2px 0 #4fd8ff", opacity: Math.floor(t * 4) % 2 ? 1 : 0.65 }}>
        ◀◀ {label ?? ""}
      </div>
    </div>
  );
};

// One material set per (character, variant). "hero" materials are opaque; "mini" materials (the
// little imagined versions of the cast inside thought bubbles) can be faded and tinted together
// without touching the lead's look.

import { Color, DoubleSide, type MeshPhysicalMaterial, type Material, MeshBasicMaterial } from "three";
import { canvasTex, eyeTex, mulberry32, phys, shade, weaveTex } from "../lib/mats.ts";
import type { CharacterSpec } from "./spec.ts";

export type Variant = "hero" | "mini";

export type CharMats = {
  skin: MeshPhysicalMaterial; hair: MeshPhysicalMaterial; scarf: MeshPhysicalMaterial; top: MeshPhysicalMaterial; topTrim: MeshPhysicalMaterial;
  bottom: MeshPhysicalMaterial; skirt: MeshPhysicalMaterial; foot: MeshPhysicalMaterial; sole: MeshPhysicalMaterial;
  eye: MeshPhysicalMaterial; lash: MeshPhysicalMaterial; brow: MeshPhysicalMaterial;
  mouth: MeshPhysicalMaterial; lipLine: MeshPhysicalMaterial; teeth: MeshPhysicalMaterial; tongue: MeshPhysicalMaterial;
  blush: MeshBasicMaterial; glint: MeshBasicMaterial; frame: MeshPhysicalMaterial; nail: MeshPhysicalMaterial;
  all: Material[];
};

const sets = new Map<string, CharMats>();

/** Soft knit for t-shirts: fine rib + a few slubs. */
const knitTex = (key: string, color: string) =>
  canvasTex(`knit${key}`, 256, 256, (g, w, h) => {
    const r = mulberry32(17);
    g.fillStyle = color;
    g.fillRect(0, 0, w, h);
    for (let x = 0; x < w; x += 3) {
      g.fillStyle = `rgba(${x % 6 ? "255,255,255" : "0,0,0"},0.035)`;
      g.fillRect(x, 0, 1.5, h);
    }
    for (let k = 0; k < 90; k++) {
      g.fillStyle = `rgba(${r() > 0.5 ? "255,255,255" : "60,40,20"},${0.03 + r() * 0.04})`;
      g.fillRect(r() * w, r() * h, 1 + r() * 2, 6 + r() * 18);
    }
  }, [5, 5]);

export const charMats = (spec: CharacterSpec, variant: Variant = "hero"): CharMats => {
  const key = `${spec.id}:${variant}`;
  let s = sets.get(key);
  if (s) return s;
  const k = (n: string) => `${key}:${n}`;
  const fabric = { roughness: 0.82, sheen: 0.55, sheenRoughness: 0.55 };
  const skin = phys(k("skin"), { color: spec.skin, roughness: 0.48, sheen: 0.45, sheenColor: "#ffb196", sheenRoughness: 0.45, clearcoat: 0.12, clearcoatRoughness: 0.5 });
  const hair = phys(k("hair"), { color: spec.hair, roughness: 0.46, sheen: 1, sheenColor: spec.hairSheen, sheenRoughness: 0.35, clearcoat: 0.12, clearcoatRoughness: 0.5 });
  const scarf = phys(k("scarf"), { color: spec.scarf ?? spec.hair, roughness: 0.85, sheen: 0.7, sheenColor: shade(spec.scarf ?? spec.hair, 1.5), sheenRoughness: 0.5, side: DoubleSide });
  const top = phys(k("top"), { color: "#ffffff", map: spec.top.kind === "hoodie" ? weaveTex(`${spec.id}top`, spec.top.color, [6, 6]) : knitTex(spec.id, spec.top.color), sheenColor: shade(spec.top.color, 1.3), ...fabric });
  const topTrim = phys(k("trim"), { color: spec.top.accent, sheenColor: "#ffffff", ...fabric });
  const bottom = phys(k("bottom"), { color: spec.bottom.color, sheenColor: shade(spec.bottom.color, 2.2), ...fabric, roughness: 0.78 });
  const skirt = phys(k("skirt"), { color: spec.bottom.color, sheenColor: shade(spec.bottom.color, 2.2), ...fabric, roughness: 0.78, side: DoubleSide });
  const foot = phys(k("foot"), { color: spec.feet.kind === "bare" ? spec.skin : spec.feet.color, ...fabric, roughness: spec.feet.kind === "shoes" ? 0.5 : 0.85 });
  const sole = phys(k("sole"), { color: "#4a4a50", roughness: 0.8 });
  const eye = phys(k("eye"), { map: eyeTex(spec.id, spec.iris), roughness: 0.04, clearcoat: 1, clearcoatRoughness: 0.02 });
  const lash = phys(k("lash"), { color: shade(spec.hair, 0.6), roughness: 0.6 });
  const brow = phys(k("brow"), { color: spec.brow, roughness: 0.75, sheen: 0.4, sheenColor: spec.hairSheen });
  const decal = { polygonOffset: true, polygonOffsetFactor: -4, polygonOffsetUnits: -4, side: DoubleSide };
  const mouth = phys(k("mouth"), { color: "#5a2226", roughness: 0.7, ...decal });
  const lipLine = phys(k("lip"), { color: shade(spec.lip, 0.62), roughness: 0.55, ...decal, polygonOffsetFactor: -6, polygonOffsetUnits: -6 });
  const teeth = phys(k("teeth"), { color: "#f6f1e8", roughness: 0.3, ...decal, polygonOffsetFactor: -5, polygonOffsetUnits: -5 });
  const tongue = phys(k("tongue"), { color: "#c4636a", roughness: 0.5, ...decal, polygonOffsetFactor: -5, polygonOffsetUnits: -5 });
  const blush = new MeshBasicMaterial({ color: new Color("#ff6f6a"), transparent: true, opacity: 0.16 * spec.blush, depthWrite: false, polygonOffset: true, polygonOffsetFactor: -3, polygonOffsetUnits: -3, map: blushTex() });
  const glint = new MeshBasicMaterial({ color: "#ffffff", toneMapped: false });
  const frame = phys(k("frame"), { color: "#2b2522", roughness: 0.35, clearcoat: 0.6 });
  const nail = phys(k("nail"), { color: shade(spec.skin, 1.18), roughness: 0.3, clearcoat: 0.5 });
  const all: Material[] = [skin, hair, scarf, top, topTrim, bottom, skirt, foot, sole, eye, lash, brow, mouth, lipLine, teeth, tongue, frame, nail];
  s = { skin, hair, scarf, top, topTrim, bottom, skirt, foot, sole, eye, lash, brow, mouth, lipLine, teeth, tongue, blush, glint, frame, nail, all };
  sets.set(key, s);
  return s;
};

const blushTex = () =>
  canvasTex("blush", 128, 128, (g, w, h) => {
    const gr = g.createRadialGradient(w / 2, h / 2, 0, w / 2, h / 2, w / 2);
    gr.addColorStop(0, "rgba(255,255,255,1)");
    gr.addColorStop(1, "rgba(255,255,255,0)");
    g.fillStyle = gr;
    g.fillRect(0, 0, w, h);
  });

/** Fade every mini material of a character (thought-bubble imagery). */
export const setMiniFade = (spec: CharacterSpec, opacity: number) => {
  const s = charMats(spec, "mini");
  for (const m of s.all) {
    m.opacity = opacity;
    m.transparent = opacity < 0.999;
  }

// Character pose: a plain number tree, computed per frame from film time (no state, no clocks).
// Angles in radians. Conventions (character faces +Z, his left is +X):
//   hips/spine/chest/neck/head [pitch fwd +, yaw to his left +, roll to his left +]
//   legs [swing fwd +, abduct out +, twist], knee bend +, ankle toes-up +
//   arms: FK [swing fwd +, abduct out +, twist] + elbow bend, optionally pulled to an IK target.
// IK targets are written in "chest", "head" or "world" space by presets; the performance resolves
// them to chest space (skeleton.ts) before poses are blended, so blending is always like-for-like.

import { Quaternion } from "three";
import type { V3 } from "./spec.ts";

export type E3 = [number, number, number];
export type IKSpace = "chest" | "head" | "world";

export type Face = {
  /** both brows: +1 raised, −1 lowered */
  browUp: number;
  /** inner brow ends: +1 raised (worry), −1 pulled down (focus/anger) */
  browIn: number;
  /** his left brow up / right down */
  browAsym: number;
  /** upper lids: 1 open, 0 closed, >1 wide */
  lid: number;
  /** lower lid raise (smile squint / tension) 0…1 */
  squint: number;
  /** mouth corners −1 frown … 1 smile */
  smile: number;
  /** mouth open 0…1 */
  open: number;
  /** mouth width −1 pucker … 1 wide/grimace */
  wide: number;
  /** gaze yaw (to his left +) and pitch (up +), radians */
  eyeX: number;
  eyeY: number;
  /** extra blush 0…1 */
  blush: number;
};

export type Hand = { curl: number; index: number; thumb: number; spread: number };
/** `q` optionally fixes the hand orientation (same space as `t`; quaternion x, y, z, w). */
export type IK = { t: V3; space: IKSpace; pole: V3; w: number; q?: [number, number, number, number] };
export type Arm = { sw: number; ab: number; tw: number; el: number; wr: E3; ik?: IK };
/** Legs are FK, or (ik.w ≥ 0.5) planted by IK on a world ankle target with a world foot orientation. */
export type LegIK = { t: V3; q: [number, number, number, number]; w: number };
export type Leg = { sw: number; ab: number; tw: number; kn: number; an: number; ik?: LegIK };

export type Pose = {
  x: number; y: number; z: number; yaw: number;
  hipY: number; hips: E3;
  spine: E3; chest: E3; neck: E3; head: E3;
  /** shoulder shrug 0…1 (both), raises the arm roots */
  shrug: number;
  armL: Arm; armR: Arm;
  legL: Leg; legR: Leg;
  handL: Hand; handR: Hand;
  face: Face;
  /** breathing amplitude (0…1), breaths per second, hand tremble (0…1) */
  breath: number; breathRate: number; tremble: number;
};

export const FACE0: Face = { browUp: 0, browIn: 0, browAsym: 0, lid: 1, squint: 0.1, smile: 0.08, open: 0, wide: 0, eyeX: 0, eyeY: 0, blush: 0 };
export const HAND0: Hand = { curl: 0.35, index: 0.3, thumb: 0.25, spread: 0.1 };

export const arm0 = (): Arm => ({ sw: 0.04, ab: 0.1, tw: 0, el: 0.2, wr: [0, 0, 0] });
export const leg0 = (): Leg => ({ sw: 0, ab: 0.02, tw: 0, kn: 0.04, an: 0 });

export const restPose = (): Pose => ({
  x: 0, y: 0, z: 0, yaw: 0,
  hipY: 0, hips: [0, 0, 0],
  spine: [0, 0, 0], chest: [0, 0, 0], neck: [0, 0, 0], head: [0, 0, 0],
  shrug: 0,
  armL: arm0(), armR: arm0(),
  legL: leg0(), legR: leg0(),
  handL: { ...HAND0 }, handR: { ...HAND0 },
  face: { ...FACE0 },
  breath: 1, breathRate: 0.24, tremble: 0,
});

export const clonePose = (p: Pose): Pose => ({
  ...p,
  hips: [...p.hips], spine: [...p.spine], chest: [...p.chest], neck: [...p.neck], head: [...p.head],
  armL: cloneArm(p.armL), armR: cloneArm(p.armR),
  legL: { ...p.legL }, legR: { ...p.legR },
  handL: { ...p.handL }, handR: { ...p.handR },
  face: { ...p.face },
});
export const cloneArm = (a: Arm): Arm => ({ ...a, wr: [...a.wr], ik: a.ik ? { ...a.ik, t: [...a.ik.t], pole: [...a.ik.pole] } : undefined });

export const lerpN = (a: number, b: number, k: number) => a + (b - a) * k;
export const lerp3 = (a: E3 | V3, b: E3 | V3, k: number): E3 => [lerpN(a[0], b[0], k), lerpN(a[1], b[1], k), lerpN(a[2], b[2], k)];

const slerpQ = (a: IK["q"], b: IK["q"], k: number): IK["q"] => {
  if (!a || !b) return k < 0.5 ? a : b;
  const q = new Quaternion(...a).slerp(new Quaternion(...b), k);
  return [q.x, q.y, q.z, q.w];
};

/** Blend arms. Both IK targets must already be in the same space (the performance resolves to chest). */
export const lerpArm = (a: Arm, b: Arm, k: number): Arm => {
  if (k <= 0) return a;
  if (k >= 1) return b;
  const ia = a.ik, ib = b.ik;
  return {
    sw: lerpN(a.sw, b.sw, k), ab: lerpN(a.ab, b.ab, k), tw: lerpN(a.tw, b.tw, k), el: lerpN(a.el, b.el, k), wr: lerp3(a.wr, b.wr, k),
    ik: ia || ib ? {
      t: lerp3((ia ?? ib)!.t, (ib ?? ia)!.t, k),
      pole: lerp3((ia ?? ib)!.pole, (ib ?? ia)!.pole, k),
      space: (ib ?? ia)!.space,
      w: lerpN(ia?.w ?? 0, ib?.w ?? 0, k),
      q: slerpQ(ia?.q, ib?.q, k),
    } : undefined,
  };
};
export const lerpLeg = (a: Leg, b: Leg, k: number): Leg => {
  const ia = a.ik, ib = b.ik;
  const ik = ia && ib ? { t: lerp3(ia.t, ib.t, k) as V3, q: slerpQ(ia.q, ib.q, k)!, w: lerpN(ia.w, ib.w, k) } : k < 0.5 ? ia : ib;
  return { sw: lerpN(a.sw, b.sw, k), ab: lerpN(a.ab, b.ab, k), tw: lerpN(a.tw, b.tw, k), kn: lerpN(a.kn, b.kn, k), an: lerpN(a.an, b.an, k), ik };
};
export const lerpHand = (a: Hand, b: Hand, k: number): Hand => ({ curl: lerpN(a.curl, b.curl, k), index: lerpN(a.index, b.index, k), thumb: lerpN(a.thumb, b.thumb, k), spread: lerpN(a.spread, b.spread, k) });
export const lerpFace = (a: Face, b: Face, k: number): Face => {
  const o = { ...a };
  for (const key of Object.keys(a) as (keyof Face)[]) o[key] = lerpN(a[key], b[key], k);
  return o;
};

/** Blend two whole poses (yaw the short way round). */
export const blend = (a: Pose, b: Pose, k: number): Pose => {
  if (k <= 0) return a;
  if (k >= 1) return b;
  let dy = b.yaw - a.yaw;
  while (dy > Math.PI) dy -= Math.PI * 2;
  while (dy < -Math.PI) dy += Math.PI * 2;
  return {
    x: lerpN(a.x, b.x, k), y: lerpN(a.y, b.y, k), z: lerpN(a.z, b.z, k), yaw: a.yaw + dy * k,
    hipY: lerpN(a.hipY, b.hipY, k), hips: lerp3(a.hips, b.hips, k),
    spine: lerp3(a.spine, b.spine, k), chest: lerp3(a.chest, b.chest, k), neck: lerp3(a.neck, b.neck, k), head: lerp3(a.head, b.head, k),
    shrug: lerpN(a.shrug, b.shrug, k),
    armL: lerpArm(a.armL, b.armL, k), armR: lerpArm(a.armR, b.armR, k),
    legL: lerpLeg(a.legL, b.legL, k), legR: lerpLeg(a.legR, b.legR, k),
    handL: lerpHand(a.handL, b.handL, k), handR: lerpHand(a.handR, b.handR, k),
    face: lerpFace(a.face, b.face, k),
    breath: lerpN(a.breath, b.breath, k), breathRate: lerpN(a.breathRate, b.breathRate, k), tremble: lerpN(a.tremble, b.tremble, k),
  };
};

const TAU = Math.PI * 2;
const D = Math.PI / 180;

/** Walk/run cycle at gait phase `p` (cycles). `amt` scales the stride (1 walk, 1.6 run). */
export const walkCycle = (pose: Pose, p: number, amt = 1): Pose => {
  const o = clonePose(pose);
  const run = Math.max(0, amt - 1);
  const legAt = (ph: number, l: Leg) => {
    const s = Math.sin(TAU * ph), c = Math.cos(TAU * ph);
    l.sw = (22 * D * s) * amt;
    l.kn = (6 * D + (52 + 40 * run) * D * Math.pow(Math.max(0, c), 1.4)) * (0.55 + 0.45 * Math.min(1, amt));
    l.an = (10 * D * Math.max(0, c) - 14 * D * Math.max(0, -Math.sin(TAU * ph + 0.9))) * amt;
  };
  legAt(p, o.legL);
  legAt(p + 0.5, o.legR);
  o.hipY = pose.hipY - (0.012 + 0.012 * amt) * (0.5 - 0.5 * Math.cos(TAU * 2 * p - Math.PI));
  const sway = Math.sin(TAU * p);
  o.hips = [pose.hips[0] + run * 8 * D, pose.hips[1] + 5 * D * sway * amt, pose.hips[2] + 2 * D * Math.cos(TAU * p) * amt];
  o.chest = [pose.chest[0] + run * 6 * D, pose.chest[1] - 7 * D * sway * amt, pose.chest[2]];
  if (!o.armL.ik) { o.armL.sw = pose.armL.sw - 0.32 * sway * amt; o.armL.el = pose.armL.el + 0.12 * amt + run * 0.9; }
  if (!o.armR.ik) { o.armR.sw = pose.armR.sw + 0.32 * sway * amt; o.armR.el = pose.armR.el + 0.12 * amt + run * 0.9; }
  return o;
};

// Forward kinematics + 2-bone arm IK. The renderer (Character.tsx) nests its groups with exactly these
// local matrices, so anything a performance asks about the body — where the eyes are, where the right
// hand is, which way the chest faces — is answered by the same maths that draws it. Props held in a
// hand, cameras framing a face, and look-at / reach targets all go through here.

import { Euler, Matrix4, Quaternion, Vector3 } from "three";
import type { Arm, IK, Leg, Pose } from "./pose.ts";
import { DIMS, type CharacterSpec, type V3 } from "./spec.ts";

const m4 = () => new Matrix4();
const T = (x: number, y: number, z: number) => m4().makeTranslation(x, y, z);
const R = (e: Euler) => m4().makeRotationFromEuler(e);
const Q = (q: Quaternion) => m4().makeRotationFromQuaternion(q);
const mul = (...ms: Matrix4[]) => ms.reduce((a, b) => a.multiply(b), m4());

export type ArmRig = { shoulder: Matrix4; elbow: Matrix4; wrist: Matrix4; bend: number };
export type LegRig = { hip: Matrix4; knee: Matrix4; ankle: Matrix4 };
export type Skeleton = {
  /** local matrices (parent → child), in rig units; `root` carries world placement + spec scale */
  root: Matrix4; hips: Matrix4; spine: Matrix4; chest: Matrix4; neck: Matrix4; head: Matrix4; headGeo: Matrix4;
  armL: ArmRig; armR: ArmRig; legL: LegRig; legR: LegRig;
  /** world matrices of the main joints */
  W: { hips: Matrix4; chest: Matrix4; head: Matrix4; headGeo: Matrix4; wristL: Matrix4; wristR: Matrix4; kneeL: Matrix4; kneeR: Matrix4; ankleL: Matrix4; ankleR: Matrix4; elbowL: Matrix4; elbowR: Matrix4; shoulderL: Matrix4; shoulderR: Matrix4 };
};

const eul = (e: [number, number, number], extraPitch = 0) => new Euler(e[0] + extraPitch, e[1], -e[2], "YXZ");

/** Shoulder joint position in chest space. */
export const shoulderPos = (spec: CharacterSpec, s: number, shrug: number) =>
  new Vector3(s * DIMS.shoulder[0] * spec.build, DIMS.shoulder[1] + shrug * 0.032, DIMS.shoulder[2]);

// ── 2-bone IK (root joint → end joint), result in the parent's space ─────────
const ik2 = (S: Vector3, Tg: Vector3, pole: Vector3, UA: number, FA: number) => {
  const d0 = new Vector3().subVectors(Tg, S);
  const d = Math.min(Math.max(d0.length(), Math.abs(UA - FA) + 1e-3), UA + FA - 1e-3);
  const u = d0.normalize();
  const cosA = (UA * UA + d * d - FA * FA) / (2 * UA * d);
  const sinA = Math.sqrt(Math.max(0, 1 - cosA * cosA));
  const w = pole.clone().addScaledVector(u, -pole.dot(u));
  if (w.lengthSq() < 1e-8) w.set(0, 0, -1).addScaledVector(u, -u.z);
  w.normalize();
  const upper = u.clone().multiplyScalar(cosA).addScaledVector(w, sinA).normalize();
  const E = S.clone().addScaledVector(upper, UA);
  const fore = new Vector3().subVectors(S.clone().addScaledVector(u, d), E).normalize();
  const bend = Math.acos(Math.min(1, Math.max(-1, upper.dot(fore))));
  const Y = upper.clone().negate();
  let Z = fore.clone().addScaledVector(upper, -fore.dot(upper));
  if (Z.lengthSq() < 1e-8) Z = w.clone().negate();
  Z.normalize();
  const X = new Vector3().crossVectors(Y, Z).normalize();
  Z.crossVectors(X, Y).normalize();
  return { q: new Quaternion().setFromRotationMatrix(m4().makeBasis(X, Y, Z)), bend };
};

const armRig = (spec: CharacterSpec, s: number, a: Arm, shrug: number): ArmRig => {
  const S = shoulderPos(spec, s, shrug);
  const fk = new Quaternion().setFromEuler(new Euler(-a.sw, s * a.tw, s * a.ab, "XZY"));
  let q = fk, el = a.el;
  if (a.ik && a.ik.w > 0) {
    if (a.ik.space !== "chest") throw new Error("IK target must be resolved to chest space before FK");
    const tg = new Vector3(a.ik.t[0], a.ik.t[1], a.ik.t[2]);
    const pole = new Vector3(a.ik.pole[0] * s, a.ik.pole[1], a.ik.pole[2]);
    const r = ik2(S, tg, pole, DIMS.upperArm, DIMS.foreArm);
    q = fk.clone().slerp(r.q, a.ik.w);
    el = el + (r.bend - el) * a.ik.w;
  }
  let wq = new Quaternion().setFromEuler(new Euler(a.wr[0], s * a.wr[1], s * a.wr[2], "XYZ"));
  if (a.ik && a.ik.q && a.ik.w > 0) {
    // hand orientation given in chest space: wrist = inverse(forearm) × hand
    const fore = q.clone().multiply(new Quaternion().setFromEuler(new Euler(-el, 0, 0)));
    const want = fore.invert().multiply(new Quaternion(...a.ik.q));
    wq = wq.slerp(want, a.ik.w);
  }
  return {
    shoulder: mul(T(S.x, S.y, S.z), Q(q)),
    elbow: mul(T(0, -DIMS.upperArm, 0), R(new Euler(-el, 0, 0))),
    wrist: mul(T(0, -DIMS.foreArm, 0), Q(wq)),
    bend: el,
  };
};

const legRig = (spec: CharacterSpec, s: number, l: Leg, Whips: Matrix4): LegRig => {
  const H = new Vector3(s * DIMS.hip[0] * spec.build, DIMS.hip[1], DIMS.hip[2]);
  if (l.ik && l.ik.w >= 0.5) {
    // planted leg: ankle on a world target, knee forward, foot turned to a world orientation
    const tg = new Vector3(...l.ik.t).applyMatrix4(Whips.clone().invert());
    const r = ik2(H, tg, new Vector3(s * 0.12, -0.05, 1), DIMS.thigh, DIMS.shin);
    const hq = new Quaternion();
    Whips.decompose(new Vector3(), hq, new Vector3());
    const kneeWorld = hq.clone().multiply(r.q).multiply(new Quaternion().setFromEuler(new Euler(-r.bend, 0, 0)));
    const ankle = kneeWorld.invert().multiply(new Quaternion(...l.ik.q));
    return {
      hip: mul(T(H.x, H.y, H.z), Q(r.q)),
      knee: mul(T(0, -DIMS.thigh, 0), R(new Euler(-r.bend, 0, 0))),
      ankle: mul(T(0, -DIMS.shin, 0), Q(ankle)),
    };
  }
  return {
    hip: mul(T(H.x, H.y, H.z), R(new Euler(-l.sw, s * l.tw, s * l.ab, "XZY"))),
    knee: mul(T(0, -DIMS.thigh, 0), R(new Euler(l.kn, 0, 0))),
    ankle: mul(T(0, -DIMS.shin, 0), R(new Euler(-l.an, 0, 0))),
  };
};

/** Torso-only chain (root → chest, neck, head) — used to resolve IK and look targets. */
const torso = (p: Pose, spec: CharacterSpec, breathPitch = 0) => {
  const root = mul(T(p.x, p.y, p.z), R(new Euler(0, p.yaw, 0)), m4().makeScale(spec.scale, spec.scale, spec.scale));
  const hips = mul(T(0, DIMS.pelvisH + p.hipY, 0), R(eul(p.hips)));
  const spine = mul(T(0, DIMS.spineY, 0), R(eul(p.spine)));
  const chest = mul(T(0, DIMS.chestY, 0), R(eul(p.chest, breathPitch)));
  const neck = mul(T(0, DIMS.neckY, -0.01), R(eul(p.neck)));
  const head = mul(T(0, DIMS.headY, 0.01), R(eul(p.head)));
  const hc = DIMS.headCenter;
  const headGeo = mul(T(hc[0], hc[1], hc[2]), m4().makeScale(spec.headScale, spec.headScale, spec.headScale));
  const Whips = mul(root, hips);
  const Wchest = mul(Whips, spine, chest);
  const Whead = mul(Wchest, neck, head);
  return { root, hips, spine, chest, neck, head, headGeo, Whips, Wchest, Whead, WheadGeo: mul(Whead, headGeo) };
};

/** Breathing as a chest pitch (rad) at time t — shared by the renderer and the skeleton. */
export const breathPitch = (p: Pose, t: number) => p.breath * 0.014 * Math.sin(t * p.breathRate * Math.PI * 2);

export const skeleton = (p: Pose, spec: CharacterSpec, t = 0): Skeleton => {
  const b = torso(p, spec, breathPitch(p, t));
  const armL = armRig(spec, 1, p.armL, p.shrug), armR = armRig(spec, -1, p.armR, p.shrug);
  const legL = legRig(spec, 1, p.legL, b.Whips), legR = legRig(spec, -1, p.legR, b.Whips);
  const hs = spec.handScale;
  const handS = m4().makeScale(hs, hs, hs);
  const wShL = mul(b.Wchest, armL.shoulder), wShR = mul(b.Wchest, armR.shoulder);
  const wElL = mul(wShL, armL.elbow), wElR = mul(wShR, armR.elbow);
  const wKL = mul(b.Whips, legL.hip, legL.knee), wKR = mul(b.Whips, legR.hip, legR.knee);
  return {
    root: b.root, hips: b.hips, spine: b.spine, chest: b.chest, neck: b.neck, head: b.head, headGeo: b.headGeo,
    armL, armR, legL, legR,
    W: {
      hips: b.Whips, chest: b.Wchest, head: b.Whead, headGeo: b.WheadGeo,
      shoulderL: wShL, shoulderR: wShR, elbowL: wElL, elbowR: wElR,
      wristL: mul(wElL, armL.wrist, handS), wristR: mul(wElR, armR.wrist, handS),
      kneeL: wKL, kneeR: wKR, ankleL: mul(wKL, legL.ankle), ankleR: mul(wKR, legR.ankle),
    },
  };
};

/** Turn an IK target given in head or world space into chest space for this pose. */
export const resolveIK = (ik: IK, p: Pose, spec: CharacterSpec, side: 1 | -1): IK => {
  if (ik.space === "chest") return { ...ik, t: [ik.t[0] * side, ik.t[1], ik.t[2]] };
  const b = torso(p, spec);
  const v = new Vector3(...ik.t);
  if (ik.space === "head") {
    v.x *= side;
    v.applyMatrix4(mul(b.neck, b.head, b.headGeo));
  } else {
    v.applyMatrix4(b.Wchest.clone().invert());
  }
  let q = ik.q;
  if (q && ik.space === "world") {
    const cq = new Quaternion();
    b.Wchest.decompose(new Vector3(), cq, new Vector3());
    const r = cq.invert().multiply(new Quaternion(...q));
    q = [r.x, r.y, r.z, r.w];
  }
  return { ...ik, t: [v.x, v.y, v.z], space: "chest", q };
};

/** Direction from the head to a world point, in chest space → [yaw, pitch] for neck+head+eyes. */
export const lookAngles = (p: Pose, spec: CharacterSpec, w: V3): [number, number] => {
  const b = torso(p, spec);
  const inv = b.Wchest.clone().invert();
  const tgt = new Vector3(...w).applyMatrix4(inv);
  const eye = new Vector3(0, 0, 0.06).applyMatrix4(mul(b.neck, b.head, b.headGeo));
  const d = tgt.sub(eye);
  const yaw = Math.atan2(d.x, d.z);
  const pitch = Math.atan2(-d.y, Math.hypot(d.x, d.z));
  return [yaw, pitch];
};

export const point = (m: Matrix4, local: V3 = [0, 0, 0]): V3 => {
  const v = new Vector3(...local).applyMatrix4(m);
  return [v.x, v.y, v.z];
};

/** World position of a point on the face (head units, e.g. [0, 0.01, 0.1] = between the eyes). */
export const facePoint = (sk: Skeleton, local: V3 = [0, 0.005, 0.1]) => point(sk.W.headGeo, local);
/** World forward direction of the face. */
export const faceDir = (sk: Skeleton): V3 => {
  const a = new Vector3(...facePoint(sk, [0, 0, 0])), b = new Vector3(...facePoint(sk, [0, 0, 1]));
  const d = b.sub(a).normalize();
  return [d.x, d.y, d.z];
};

// mattress from the character's body capsules (pelvis, torso, thighs, shins, feet — taken from the

import { BufferAttribute, BufferGeometry, type Material, Vector3 } from "three";
import type { Skeleton } from "../character/skeleton.ts";
import type { V3 } from "../character/spec.ts";

export type Capsule = { a: V3; b: V3; r: number };
export type BedFrame = { x: number; z0: number; z1: number; halfW: number; top: number };

/** Body capsules for the blanket, from a skeleton (world space). Arms are left out: they rest on top. */
export const bodyCapsules = (sk: Skeleton, scale: number): Capsule[] => {
  const P = (m: { elements: number[] }, l: V3 = [0, 0, 0]) => {
    const v = new Vector3(...l).applyMatrix4(m as never);
    return [v.x, v.y, v.z] as V3;
  };
  const W = sk.W;
  const hip = P(W.hips), chestTop = P(W.chest, [0, 0.12, 0]), chestLow = P(W.chest, [0, -0.2, 0]);
  const caps: Capsule[] = [{ a: P(W.hips, [0.1, -0.02, 0]), b: P(W.hips, [-0.1, -0.02, 0]), r: 0.15 * scale }];
  // the torso only lifts the duvet while it lies down (an upright torso is above the top edge)
  const horiz = (a: V3, b: V3) => Math.abs(b[1] - a[1]) < 0.45 * Math.hypot(b[0] - a[0], b[1] - a[1], b[2] - a[2]);
  if (horiz(hip, chestTop)) caps.push({ a: chestLow, b: chestTop, r: 0.17 * scale }, { a: hip, b: chestLow, r: 0.16 * scale });
  for (const [knee, ankle] of [[W.kneeL, W.ankleL], [W.kneeR, W.ankleR]] as const) {
    const k = P(knee), an = P(ankle);
    const h = P(knee, [0, 0.43, 0]);
    caps.push({ a: h, b: k, r: 0.1 * scale });
    caps.push({ a: k, b: an, r: 0.08 * scale });
    caps.push({ a: P(knee, [0, -0.36, 0]), b: P(ankle, [0, -0.05, 0.2]), r: 0.075 * scale });
  }
  return caps;
};

        for (let dj = -1; dj <= 1; dj++) for (let di = -1; di <= 1; di++) {

          const ww = di === 0 && dj === 0 ? 2 : 1;
          s += H[jj * NX + ii] * ww; w += ww;
        }
        T[j * NX + i] = s / w;
      }
    for (let n = 0; n < NX * NZ; n++) H[n] = Math.max(T[n], B[n]);
  }
  const p = geo.attributes.position as BufferAttribute;
  for (let j = 0; j < NZ; j++)

      const n = j * NX + i;
      // top edge: the first row tucks down to give the duvet thickness
      const tuck = j === 0 ? -0.045 : j === 1 ? -0.008 : 0;
      p.setXYZ(n, X[n], H[n] + tuck, Z[n] + (j === 0 ? 0.012 : 0));
    }
  p.needsUpdate = true;
  geo.computeVertexNormals();
  geo.computeBoundingSphere();
  return (
    <mesh geometry={geo} material={materials} castShadow receiveShadow />
  );
};

/**
 * A scoreboard on two posts that springs up from the ground at `upAt`, swaps to `face2` at `swapAt`
 * (flip), and sinks at `downAt`. Facing +Z at `p` (rotate with `yaw`).
 */
export const PopBoard: React.FC<{ p: V3; yaw?: number; scale?: number; t: number; upAt: number; swapAt: number; downAt: number; face1: BoardFace; face2: BoardFace }> = ({ p, yaw = 0, scale = 1, t, upAt, swapAt, downAt, face1, face2 }) => {
  const up = springUp((t - upAt) / 0.9) * (1 - Math.min(1, Math.max(0, (t - downAt) / 0.45)) ** 2);
  if (up <= 0.001) return null;
  const flip = Math.min(1, Math.max(0, (t - swapAt) / 0.35));
  const face = flip < 0.5 ? face1 : face2;
  const rot = Math.PI * (flip < 0.5 ? flip : flip - 1);
  const W = 2.2, H = 1.375;
  const sink = (1 - up) * 3.4;
  const post = phys("cfboardpost2", { color: "#2d3340", roughness: 0.4, metalness: 0.5 });
  const gold = phys("cftrophy", { color: "#f4c542", metalness: 1, roughness: 0.25 });
  return (
    <group position={[p[0], p[1] - sink * scale, p[2]]} rotation={[0, yaw, 0]} scale={scale}>
      {[-1, 1].map((s) => <mesh key={s} geometry={cyl(0.06, 0.06, 2.6, 12)} material={post} position={[s * 0.85, 1.3, -0.04]} castShadow />)}
      <group position={[0, 2.35, 0]} rotation={[rot, 0, 0]}>
        <mesh geometry={box(W + 0.12, H + 0.12, 0.12, 0.05)} material={phys("cfboardframe", { color: "#f2b544", roughness: 0.4, clearcoat: 0.5 })} castShadow />
        <mesh geometry={cached("cfpopboard", () => new PlaneGeometry(W, H))} material={phys(`cfpopface${face.title}${face.left}`, { map: boardTex(face), emissive: "#ffffff", emissiveMap: boardTex(face), emissiveIntensity: 0.35, roughness: 0.4 })} position={[0, 0, 0.065]} />
      </group>
      {/* trophy on top */}

/** Tumbleweed rolling from `a` to `b` between t0 and t1, bouncing. */
export const Tumbleweed: React.FC<{ a: V3; b: V3; t: number; t0: number; t1: number; r?: number }> = ({ a, b, t, t0, t1, r = 0.32 }) => {
  if (t < t0 || t > t1) return null;
  const k = (t - t0) / (t1 - t0);
  const x = a[0] + (b[0] - a[0]) * k, z = a[2] + (b[2] - a[2]) * k;
  const dist = Math.hypot(b[0] - a[0], b[2] - a[2]) * k;
  const bounce = Math.abs(Math.sin(k * Math.PI * 5)) * 0.25 * (1 - 0.5 * k);
  const roll = dist / r;
  const dir = Math.atan2(b[0] - a[0], b[2] - a[2]);
  const twig = phys("cftwig", { color: "#b08a55", roughness: 0.85 });
  return (
    <group position={[x, a[1] + r + bounce, z]} rotation={[0, dir, 0]}>
      <group rotation={[roll, 0, 0]}>
        {Array.from({ length: 14 }, (_, i) => (
          <mesh key={i} geometry={torus(r * (0.6 + 0.4 * hash1(i, 3)), 0.012, 4, 24)} material={twig} rotation={[hash1(i, 5) * 6.3, hash1(i, 7) * 6.3, hash1(i, 9) * 6.3]} castShadow />
        ))}
        <mesh geometry={sphere(r * 0.35, 8, 6)} material={twig} />
      </group>
    </group>
  );
};

// Small geometry kit for architecture: struts between points, merged lattices, lathe and
// arch profiles, geodesic frames. Geometries are cached by key so a film builds each once.

import {
  BufferGeometry,
  CylinderGeometry,
  IcosahedronGeometry,
  LatheGeometry,
  Matrix4,
  Quaternion,
  Shape,
  SphereGeometry,
  TorusGeometry,
  Vector2,
  Vector3,
} from "three";
import { RoundedBoxGeometry } from "three/examples/jsm/geometries/RoundedBoxGeometry.js";
import { mergeGeometries } from "three/examples/jsm/utils/BufferGeometryUtils.js";

const cache = new Map<string, BufferGeometry>();
export const cached = <G extends BufferGeometry>(key: string, make: () => G): G => {
  let g = cache.get(key);
  if (!g) {
    g = make();
    cache.set(key, g);
  }
  return g as G;
};

export const cyl = (rt: number, rb: number, h: number, seg = 32, open = false, ts = 0, tl = Math.PI * 2) =>
  cached(`cyl${rt},${rb},${h},${seg},${open},${ts},${tl}`, () => new CylinderGeometry(rt, rb, h, seg, 1, open, ts, tl));
export const box = (w: number, h: number, d: number, r = 0.04, seg = 2) =>
  cached(`box${w},${h},${d},${r}`, () => new RoundedBoxGeometry(w, h, d, seg, Math.min(r, w / 2.01, h / 2.01, d / 2.01)));
export const sphere = (r: number, ws = 32, hs = 16, phiLen = Math.PI * 2, thetaLen = Math.PI) =>
  cached(`sph${r},${ws},${hs},${phiLen},${thetaLen}`, () => new SphereGeometry(r, ws, hs, 0, phiLen, 0, thetaLen));
export const torus = (R: number, r: number, rs = 8, ts = 48, arc = Math.PI * 2) =>

const UP = new Vector3(0, 1, 0);
/** A cylinder from p to q (world-free; positions in the caller's frame). */
export const strutGeo = (p: Vector3, q: Vector3, r: number, seg = 6) => {
  const d = new Vector3().subVectors(q, p);
  const len = d.length();
  const g = new CylinderGeometry(r, r, len, seg, 1, false);
  const m = new Matrix4().compose(
    new Vector3().addVectors(p, q).multiplyScalar(0.5),
    new Quaternion().setFromUnitVectors(UP, d.normalize()),
    new Vector3(1, 1, 1),
  );
  g.applyMatrix4(m);
  return g;
};

export const merged = (key: string, make: () => BufferGeometry[]) =>
  cached(key, () => {
    const parts = make().map((g) => (g.index ? g.toNonIndexed() : g));
    for (const p of parts) {
      for (const name of Object.keys(p.attributes)) if (name !== "position" && name !== "normal" && name !== "uv") p.deleteAttribute(name);
    }
    return mergeGeometries(parts, false)!;
  });

/** Geodesic frame over the upper hemisphere (edges of a subdivided icosahedron). */
export const geodesicFrame = (R: number, detail: number, strutR: number) =>
  merged(`geo${R},${detail},${strutR}`, () => {
    const ico = new IcosahedronGeometry(R, detail);
    const p = ico.attributes.position;
    const key = (v: Vector3) => `${v.x.toFixed(3)},${v.y.toFixed(3)},${v.z.toFixed(3)}`;
    const edges = new Map<string, [Vector3, Vector3]>();
    for (let i = 0; i < p.count; i += 3) {
      const tri = [0, 1, 2].map((k) => new Vector3(p.getX(i + k), p.getY(i + k), p.getZ(i + k)));
      for (let k = 0; k < 3; k++) {
        const a = tri[k], b = tri[(k + 1) % 3];
        if (a.y < -0.01 || b.y < -0.01) continue;
        const ka = key(a), kb = key(b);
        const id = ka < kb ? ka + "|" + kb : kb + "|" + ka;
        if (!edges.has(id)) edges.set(id, [a, b]);
      }
    }
    return [...edges.values()].map(([a, b]) => strutGeo(a, b, strutR, 5));
  });

// Deterministic randomness (Node-safe).
export const mulberry32 = (seed: number) => {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
};
/** Integer hash → [0, 1). */
export const hash1 = (x: number, seed = 0) => {
  let h = (Math.imul(x | 0, 374761393) + Math.imul(seed | 0, 668265263)) | 0;
  h = Math.imul(h ^ (h >>> 13), 1274126177);
  h ^= h >>> 16;
  return (h >>> 0) / 4294967296;
};
/** Smooth 1-D value noise in [-1, 1]. */
export const noise1 = (x: number, seed = 0) => {
  const i = Math.floor(x), f = x - i;
  const u = f * f * (3 - 2 * f);
  return (hash1(i, seed) * (1 - u) + hash1(i + 1, seed) * u) * 2 - 1;
};

// Standing and sitting blocking with planted feet (leg IK), for any character.
//
//   standBody — standing on a mark: living idle (slow weight shift), turns in place with real steps
//               (each foot lifts and re-plants at the new heading, staggered), optional base preset.
//   seatBody  — a chair/bench/bed-edge seat: sits down and stands up through a forward-lean crouch
//               while the feet stay planted in front of the seat.
// Pure maths — the director and the sound department (footsteps on turns) share it.

import { BODY, type BodyName } from "../character/presets/bodies.ts";
import { blend, clonePose, type Pose } from "../character/pose.ts";
import { DIMS, type CharacterSpec, type V3 } from "../character/spec.ts";

const D = Math.PI / 180;
const smooth = (x: number) => { const k = Math.min(1, Math.max(0, x)); return k * k * (3 - 2 * k); };
const ANKLE_H = 0.087;

export type Turn = { at: number; dur: number; to: number };
const wrap = (a: number) => { while (a > Math.PI) a -= 2 * Math.PI; while (a < -Math.PI) a += 2 * Math.PI; return a; };

/** Heading at time t after a list of turns (shortest way round). */
export const yawAt = (yaw0: number, turns: Turn[], t: number, lag = 0) => {
  let y = yaw0;
  for (const tr of turns) {
    if (t < tr.at) break;
    const k = smooth((t - tr.at - lag * tr.dur) / tr.dur);
    y = y + wrap(tr.to - y) * k;
  }
  return y;
};

const quatYXZ = (yaw: number, pitch: number): [number, number, number, number] => {
  const cy = Math.cos(yaw / 2), sy = Math.sin(yaw / 2), cx = Math.cos(pitch / 2), sx = Math.sin(pitch / 2);
  return [cy * sx, sy * cx, -sy * sx, cy * cx];
};

/** Foot (ankle target + orientation) standing at `pos` with heading `yaw`; side −1 = left, +1 = right. */
const footAt = (pos: V3, yaw: number, side: number, spec: CharacterSpec, lift = 0, back = 0.02): { t: V3; q: [number, number, number, number] } => {
  const gap = 0.1 * spec.scale;
  const fx = Math.sin(yaw), fz = Math.cos(yaw);
  const rx = -fz, rz = fx; // his right
  return {
    t: [pos[0] + rx * gap * side - fx * back, pos[1] + ANKLE_H * spec.scale + lift, pos[2] + rz * gap * side - fz * back],
    q: quatYXZ(yaw - side * 0.07, 0),
  };
};

/** Turn progress of one foot inside a turn: left steps first, right follows. */
const stepK = (turns: Turn[], t: number, first: boolean) => {
  for (const tr of turns) {
    const u = (t - tr.at) / tr.dur;
    if (u > 0 && u < 1) {
      const [a, b] = first ? [0.05, 0.55] : [0.45, 0.95];
      const k = smooth((u - a) / (b - a));
      return { k, lift: 0.06 * Math.sin(Math.PI * Math.min(1, Math.max(0, (u - a) / (b - a)))) };
    }
  }
  return null;
};

export const standBody = (spec: CharacterSpec, pos: V3, yaw0: number, turns: Turn[] = [], base: BodyName = "stand", mod?: (p: Pose, t: number) => void) => (t: number): Pose => {
  const yaw = yawAt(yaw0, turns, t);
  const p = clonePose(BODY[base]({ x: pos[0], y: pos[1], z: pos[2], yaw }, spec, t));
  // idle: slow weight shift between the feet
  const w = Math.sin(t * 2 * Math.PI * 0.11 + spec.scale * 10);
  p.hips = [p.hips[0], p.hips[1] + 1.2 * D * w, p.hips[2] + 1.8 * D * w];
  p.chest = [p.chest[0], p.chest[1] - 0.8 * D * w, p.chest[2] - 1.0 * D * w];
  p.x += Math.cos(yaw) * 0.012 * w;
  p.z += -Math.sin(yaw) * 0.012 * w;
  p.hipY -= 0.008;
  // turning: dip a little while stepping round
  for (const tr of turns) { const u = (t - tr.at) / tr.dur; if (u > 0 && u < 1) p.hipY -= 0.018 * Math.sin(Math.PI * u); }
  // feet: each re-plants at the new heading on its own step
  const prevYaw = (side: -1 | 1) => {
    const s = stepK(turns, t, side === -1);
    if (!s) return { yaw, lift: 0 };
    // heading this foot has reached
    let y = yaw0;
    for (const tr of turns) { if (t < tr.at) break; if (t > tr.at + tr.dur) y = y + wrap(tr.to - y); else y = y + wrap(tr.to - y) * s.k; }
    return { yaw: y, lift: s.lift };
  };
  const L = prevYaw(-1), R = prevYaw(1);
  const fl = footAt(pos, L.yaw, -1, spec, L.lift), fr = footAt(pos, R.yaw, 1, spec, R.lift);
  p.legL = { ...p.legL, ik: { t: fl.t, q: fl.q, w: 1 } };
  p.legR = { ...p.legR, ik: { t: fr.t, q: fr.q, w: 1 } };
  if (mod) mod(p, t);
  return p;
};

// Locomotion: characters really walk through the set — no treadmill, no sliding feet.
//
// A Walker integrates a speed profile (m/s keys) into distance and gait phase. Step length follows
// speed, so cadence and stride change together; when the speed drops to zero the gait finishes its
// step and both feet come to rest. Each foot is planted at the body's position at mid-stance and
// swings to its next plant with a lift arc, toe-off and heel strike; the legs reach the plants with
// IK (skeleton.ts), so foot contact is exact. walkBody() layers the upper-body motion on any body
// preset: pelvis bob and yaw, counter-rotating chest, arm swing (unless a gesture owns the arm).
//
// Pure maths (no rendering), so the sound department can place footsteps on the same plants.

import { BODY, type BodyName } from "../character/presets/bodies.ts";
import { clonePose, type Pose } from "../character/pose.ts";
import { DIMS, type CharacterSpec, type V3 } from "../character/spec.ts";

export type SpeedKey = [t: number, v: number];
export type WalkerDef = {
  /** ground point where distance 0 is */
  start: V3;
  /** horizontal travel direction (normalised internally) */
  dir: V3;
  /** speed keys (m/s), eased between keys */
  speed: SpeedKey[];
  spec: CharacterSpec;
  /** distance already covered at speed[0][0] */
  d0?: number;
  /** gait phase at speed[0][0] (cycles) — continue another walker's stride */
  phi0?: number;
  /** sideways offset from the line (m, + = his right) over time — lane changes */
  lane?: (t: number) => number;
};
export type Foot = { t: V3; q: [number, number, number, number]; lift: number; planted: boolean };
export type WalkState = { d: number; v: number; phi: number; pos: V3; yaw: number; amt: number };

const DT = 1 / 240;
const smooth = (x: number) => { const k = Math.min(1, Math.max(0, x)); return k * k * (3 - 2 * k); };
const smoother = (x: number) => { const k = Math.min(1, Math.max(0, x)); return k * k * k * (k * (k * 6 - 15) + 10); };

/** Step length (m) at speed v for a character of this scale. */
export const stepLength = (v: number, scale: number) => Math.min(0.8, Math.max(0.3, 0.3 + 0.38 * v)) * (scale / 0.94);
const ankleH = (spec: CharacterSpec) => 0.087 * spec.scale;

const quatYXZ = (yaw: number, pitch: number): [number, number, number, number] => {
  const cy = Math.cos(yaw / 2), sy = Math.sin(yaw / 2), cx = Math.cos(pitch / 2), sx = Math.sin(pitch / 2);
  // q = qy * qx
  return [cy * sx, sy * cx, -sy * sx, cy * cx];
};

export class Walker {
  def: WalkerDef;
  t0: number;
  dir: V3;
  right: V3;
  yaw: number;
  private d: Float64Array;
  private phi: Float64Array;
  private v: Float64Array;

  constructor(def: WalkerDef, t1: number) {
    this.def = def;
    this.t0 = def.speed[0][0];
    const l = Math.hypot(def.dir[0], def.dir[2]);
    this.dir = [def.dir[0] / l, 0, def.dir[2] / l];
    this.right = [-this.dir[2], 0, this.dir[0]];
    this.yaw = Math.atan2(this.dir[0], this.dir[2]);
    const n = Math.ceil((t1 - this.t0) / DT) + 2;
    this.d = new Float64Array(n);
    this.phi = new Float64Array(n);
    this.v = new Float64Array(n);
    let d = def.d0 ?? 0, phi = def.phi0 ?? 0;
    for (let i = 0; i < n; i++) {
      const t = this.t0 + i * DT;
      const v = this.speedAt(t);
      this.d[i] = d;
      this.phi[i] = phi;
      this.v[i] = v;
      const L = stepLength(v, def.spec.scale);
      let rate = v / (2 * L);
      // stopping: finish the current step so both feet end on the ground (rest at φ ≡ 0.06 mod 0.5)
      if (v < 0.04) {
        const u = (((phi - 0.06) % 0.5) + 0.5) % 0.5;
        rate = u > 0.012 ? Math.max(rate, 0.9) : 0;
      }
      d += v * DT;
      phi += rate * DT;
    }
  }

  speedAt(t: number) {
    const k = this.def.speed;
    if (t <= k[0][0]) return k[0][1];
    for (let i = 1; i < k.length; i++) if (t <= k[i][0]) return k[i - 1][1] + (k[i][1] - k[i - 1][1]) * smooth((t - k[i - 1][0]) / (k[i][0] - k[i - 1][0]));
    return k[k.length - 1][1];
  }

  private sample(arr: Float64Array, t: number) {
    const x = Math.min(arr.length - 1.001, Math.max(0, (t - this.t0) / DT));
    const i = Math.floor(x), f = x - i;
    return arr[i] * (1 - f) + arr[i + 1] * f;
  }
  /** distance at the moment the phase reaches `p` (binary search; phase is monotonic) */
  private dAtPhase(p: number) {
    const a = this.phi;
    if (p <= a[0]) return this.d[0] - (a[0] - p) * 2 * stepLength(this.v[0], this.def.spec.scale);
    let lo = 0, hi = a.length - 1;
    if (p >= a[hi]) return this.d[hi];
    while (hi - lo > 1) { const m = (lo + hi) >> 1; if (a[m] < p) lo = m; else hi = m; }
    const f = (p - a[lo]) / Math.max(1e-9, a[hi] - a[lo]);
    return this.d[lo] + (this.d[hi] - this.d[lo]) * f;
  }
  private at3(d: number, lateral = 0, t = this.t0): V3 {
    const s = this.def.start;
    lateral += this.def.lane ? this.def.lane(t) : 0;
    return [s[0] + this.dir[0] * d + this.right[0] * lateral, s[1], s[2] + this.dir[2] * d + this.right[2] * lateral];
  }

  at(t: number): WalkState {
    const d = this.sample(this.d, t), v = this.sample(this.v, t), phi = this.sample(this.phi, t);
    return { d, v, phi, pos: this.at3(d, 0, t), yaw: this.yaw, amt: Math.min(1.2, v / 1.0) };
  }

  /** Ankle targets + foot orientations for both feet at time t. */
  feet(t: number): { L: Foot; R: Foot } {
    const st = this.at(t);
    const spec = this.def.spec;
    const gap = 0.1 * spec.scale;
    const foot = (off: number, side: 1 | -1): Foot => {
      const pf = st.phi + off;
      const n = Math.floor(pf), u = pf - n;
      const plant = (k: number) => this.dAtPhase(k + 0.3 - off) - 0.04;
      let dd: number, lift = 0, pitch = 0, planted = true;
      if (u < 0.6) {
        dd = plant(n);
        pitch = 0.16 * smooth((u - 0.42) / 0.18) * Math.min(1, st.v / 0.4); // heel rises late in stance
      } else {
        const k = smoother((u - 0.6) / 0.4);
        dd = plant(n) + (plant(n + 1) - plant(n)) * k;
        const amp = Math.min(1, 0.25 + st.v / 0.6);
        lift = 0.075 * Math.sin(Math.PI * k) * amp;
        pitch = 0.5 * Math.cos(Math.PI * k) * Math.sqrt(Math.sin(Math.PI * k)) * amp;
        planted = false;
      }
      const p = this.at3(dd, side * gap, t);
      const y = p[1] + ankleH(spec) + lift + Math.max(0, pitch) * 0.12 + Math.max(0, -pitch) * 0.05;
      return { t: [p[0], y, p[2]], q: quatYXZ(this.yaw - side * 0.07, pitch), lift, planted };
    };
    // `right` points to his right: left foot −gap, right foot +gap; toes turn slightly out
    return { L: foot(0, -1), R: foot(0.5, 1) };
  }

  /** Times (s) of heel strikes in [a, b] — for footstep sounds. */
  heelStrikes(a: number, b: number): { t: number; side: "L" | "R" }[] {
    const out: { t: number; side: "L" | "R" }[] = [];
    let prevL = this.sample(this.phi, a), prevR = prevL + 0.5;
    for (let t = a + DT; t <= b; t += DT) {
      const p = this.sample(this.phi, t);
      if (Math.floor(p) > Math.floor(prevL) && this.sample(this.v, t) > 0.02) out.push({ t, side: "L" });
      if (Math.floor(p + 0.5) > Math.floor(prevR) && this.sample(this.v, t) > 0.02) out.push({ t, side: "R" });
      prevL = p; prevR = p + 0.5;
    }
    return out;
  }
}

/** Distance covered at time t by a speed profile (no walker needed). */
export const distanceAt = (speed: SpeedKey[], t: number, d0 = 0) => {
  const w = { speedAt: Walker.prototype.speedAt, def: { speed } } as unknown as Walker;
  let d = d0;
  for (let x = speed[0][0]; x < t; x += DT) d += w.speedAt(x) * DT;
  return d;
};

/**
 * Scale the cruise speed of a profile so the walk covers exactly `dist` by `tEnd`
 * (`keys(v)` builds the profile for a cruise speed v; bisection, deterministic).
 */
export const fitCruise = (keys: (v: number) => SpeedKey[], dist: number, tEnd: number) => {
  let lo = 0.05, hi = 3;
  for (let i = 0; i < 40; i++) {
    const m = (lo + hi) / 2;
    if (distanceAt(keys(m), tEnd) < dist) lo = m; else hi = m;
  }
  return keys((lo + hi) / 2);
};

const D = Math.PI / 180;

/**
 * A walking body: `base` preset at the walker's position, plus gait motion and planted legs.
 * `amtOverride` scales the upper-body swing (e.g. smaller while holding a phone).
 */
export const walkBody = (w: Walker, base: BodyName = "stand", mod?: (p: Pose, t: number) => void) => (t: number): Pose => {
  const st = w.at(t);
  const spec = w.def.spec;
  const p = clonePose(BODY[base]({ x: st.pos[0], y: st.pos[1], z: st.pos[2], yaw: st.yaw }, spec, t));
  const a = Math.min(1, st.amt);
  const c = Math.cos(2 * Math.PI * st.phi);
  p.hipY = p.hipY - 0.008 - 0.024 * a + 0.011 * a * Math.cos(4 * Math.PI * (st.phi - 0.3));
  p.hips = [p.hips[0] + 3 * D * a, p.hips[1] - 6 * D * a * c, p.hips[2] + 2.2 * D * a * Math.sin(2 * Math.PI * (st.phi - 0.3))];
  p.chest = [p.chest[0] + 1.5 * D * a, p.chest[1] + 8 * D * a * c, p.chest[2]];
  p.neck = [p.neck[0] - 1 * D * a, p.neck[1] - 2 * D * a * c, p.neck[2]];
  p.armL = { ...p.armL, sw: p.armL.sw - 0.34 * a * c, el: p.armL.el + 0.1 * a + 0.12 * a * Math.max(0, -c) };
  p.armR = { ...p.armR, sw: p.armR.sw + 0.34 * a * c, el: p.armR.el + 0.1 * a + 0.12 * a * Math.max(0, c) };
  const f = w.feet(t);
  p.legL = { ...p.legL, ik: { t: f.L.t, q: f.L.q, w: 1 } };
  p.legR = { ...p.legR, ik: { t: f.R.t, q: f.R.q, w: 1 } };
  if (mod) mod(p, t);
  return p;
};

// Prop attach maths (pure, Node-safe): a prop's world matrix is a chain of states — resting
// somewhere, or held by a hand. When a hand takes a prop the grip offset is measured at that instant
// (prop rest × inverse hand), so the prop never jumps; when let go it settles onto its next rest pose.

import { Euler, Matrix4, Quaternion, Vector3 } from "three";
import type { V3 } from "../character/spec.ts";

export const mat = (p: V3, yaw = 0, pitch = 0, roll = 0) => {
  const q = new Quaternion().setFromEuler(new Euler(pitch, yaw, roll, "YXZ"));
  return new Matrix4().compose(new Vector3(...p), q, new Vector3(1, 1, 1));
};
export const lerpMat = (a: Matrix4, b: Matrix4, k: number) => {
  if (k <= 0) return a.clone();
  if (k >= 1) return b.clone();
  const pa = new Vector3(), qa = new Quaternion(), sa = new Vector3(), pb = new Vector3(), qb = new Quaternion(), sb = new Vector3();
  a.decompose(pa, qa, sa);
  b.decompose(pb, qb, sb);
  return new Matrix4().compose(pa.lerp(pb, k), qa.slerp(qb, k), sa.lerp(sb, k));
};
/** Offset that keeps `prop` where it is when attached to `hand`. */
export const gripOffset = (hand: Matrix4, prop: Matrix4) => hand.clone().invert().multiply(prop);
export const held = (hand: Matrix4, offset: Matrix4) => hand.clone().multiply(offset);
export const toPQ = (m: Matrix4) => {
  const p = new Vector3(), q = new Quaternion(), s = new Vector3();
  m.decompose(p, q, s);
  return { position: [p.x, p.y, p.z] as V3, quaternion: [q.x, q.y, q.z, q.w] as [number, number, number, number] };
};
export const xform = (m: Matrix4, local: V3): V3 => {
  const v = new Vector3(...local).applyMatrix4(m);
  return [v.x, v.y, v.z];
};

export const BAG = { w: 0.3, h: 0.21, d: 0.085 };

/** Phone lying in a RIGHT palm (hand space): back on the palm (+X side), top toward the fingertips. */
export const PHONE_IN_HAND = (() => {
  const yAxis = new Vector3(0, -1, 0), zAxis = new Vector3(1, 0, 0);
  const xAxis = new Vector3().crossVectors(yAxis, zAxis);
  const q = new Quaternion().setFromRotationMatrix(new Matrix4().makeBasis(xAxis, yAxis, zAxis));
  return new Matrix4().compose(new Vector3(0.024, -0.085, 0.005), q, new Vector3(1, 1, 1));
})();

/**
 * Wrist target + hand orientation (world) that put `propInHand` at the world pose `prop`.
 * `handScale` = character scale × spec.handScale (the wrist frame is scaled by both).
 */
export const gripFor = (prop: Matrix4, propInHand: Matrix4, handScale: number) => {
  const pp = new Vector3(), pq = new Quaternion(), ps = new Vector3();
  prop.decompose(pp, pq, ps);
  const lp = new Vector3(), lq = new Quaternion(), ls = new Vector3();
  propInHand.decompose(lp, lq, ls);
  const hq = pq.clone().multiply(lq.clone().invert());
  const wrist = pp.clone().sub(lp.clone().multiplyScalar(handScale).applyQuaternion(hq));
  return { wrist: [wrist.x, wrist.y, wrist.z] as V3, q: [hq.x, hq.y, hq.z, hq.w] as [number, number, number, number] };
};

/** Phone held up to read: in front of and below the eyes, screen turned to them. */
export const phoneReadPose = (eye: V3, forward: V3, right: V3, lower = 0.27, out = 0.27, side = -0.04) => {
  const f = new Vector3(...forward).setY(0).normalize();
  const r = new Vector3(...right).normalize();
  const p = new Vector3(...eye).addScaledVector(f, out).add(new Vector3(0, -lower, 0)).addScaledVector(r, side);
  const z = new Vector3(...eye).sub(p).normalize();
  const y = new Vector3(0, 1, 0).addScaledVector(z, -z.y).normalize();
  const x = new Vector3().crossVectors(y, z).normalize();
  const q = new Quaternion().setFromRotationMatrix(new Matrix4().makeBasis(x, y, z));
  return new Matrix4().compose(p, q, new Vector3(1, 1, 1));
};

    d -= 160;

    if (d <= l.w) {
      const wob = Math.sin(d * 0.21) * 9 + Math.sin(d * 0.057) * 6;
      return [(l.x0 + d) / CW, (l.y - 14 + wob) / CH];
    }
    d -= l.w;
  }
  const last = L[L.length - 1];
  return [(last.x0 + last.w + 6) / CW, (last.y - 10) / CH];

import React, { useLayoutEffect, useMemo } from "react";
import { useFrame, useThree } from "@react-three/fiber";
import { Color, HalfFloatType, NeutralToneMapping, PerspectiveCamera, Vector2, Vector3, WebGLRenderTarget } from "three";
import { EffectComposer } from "three/examples/jsm/postprocessing/EffectComposer.js";

export type Cam = { pos: V3; target: V3; fov: number; roll?: number; focus?: number; aperture?: number; maxBlur?: number; near?: number; far?: number };

export type Grade = {
  exposure: number;
  /** additive lift in the shadows, multiplicative gain */
  lift: V3; gain: V3;
  sat: number; contrast: number;
  vignette: number; grain: number;
  bloom: number; bloomRadius: number; bloomThreshold: number;
  /** 0..1 flash toward `flashColor`, 0..1 fade to black */

export const GRADE0: Grade = {

export const CameraRig: React.FC<{ cam: Cam }> = ({ cam }) => {
  const { camera } = useThree();
  useLayoutEffect(() => {
    const c = camera as PerspectiveCamera;
    c.position.set(...cam.pos);
    const f = new Vector3(...cam.target).sub(c.position).normalize();
    const r = cam.roll ?? 0;
    // roll about the view axis
    const up = new Vector3(0, 1, 0);
    const side = new Vector3().crossVectors(f, up).normalize();
    const u2 = new Vector3().crossVectors(side, f).normalize();
    c.up.copy(u2.multiplyScalar(Math.cos(r)).addScaledVector(side, Math.sin(r)));
    c.lookAt(...cam.target);
    c.fov = cam.fov;
    c.near = cam.near ?? 0.05;
    c.far = cam.far ?? 900;
    c.updateProjectionMatrix();
    c.updateMatrixWorld();
    c.layers.enable(FX_LAYER);
  }, [cam, camera]);
  return null;
};

const NanGuardShader = {
  uniforms: { tDiffuse: { value: null } },
  vertexShader: /* glsl */ `varying vec2 vUv; void main(){ vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }`,
  fragmentShader: /* glsl */ `uniform sampler2D tDiffuse; varying vec2 vUv;
    void main(){ vec4 c = texture2D(tDiffuse, vUv);
      if (any(isnan(c)) || any(isinf(c))) c = vec4(0.0, 0.0, 0.0, 1.0);
      gl_FragColor = clamp(c, 0.0, 64.0); }`,
};

const GradeShader = {
  uniforms: {
    tDiffuse: { value: null },
    uTime: { value: 0 },
    uRes: { value: new Vector2(1080, 1920) },
    uLift: { value: new Vector3() },
    uGain: { value: new Vector3(1, 1, 1) },
    uSat: { value: 1 },
    uContrast: { value: 1 },
    uVignette: { value: 0.2 },
    uGrain: { value: 0.015 },
    uFlash: { value: 0 },
    uFlashColor: { value: new Color("#fff") },
    uBlack: { value: 0 },
  },
  vertexShader: /* glsl */ `varying vec2 vUv; void main(){ vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }`,
  fragmentShader: /* glsl */ `
    uniform sampler2D tDiffuse; uniform float uTime; uniform vec2 uRes;
    uniform vec3 uLift; uniform vec3 uGain; uniform float uSat; uniform float uContrast;
    uniform float uVignette; uniform float uGrain; uniform float uFlash; uniform vec3 uFlashColor; uniform float uBlack;
    varying vec2 vUv;
    float hash(vec2 p){ p = fract(p * vec2(443.897, 441.423)); p += dot(p, p.yx + 19.19); return fract((p.x + p.y) * p.x); }
    void main(){
      vec4 c = texture2D(tDiffuse, vUv);
      vec3 x = clamp(c.rgb, 0.0, 1.0);
      x = x * uGain + uLift * (1.0 - x);
      float l = dot(x, vec3(0.2126, 0.7152, 0.0722));
      x = mix(vec3(l), x, uSat);
      x = clamp((x - 0.5) * uContrast + 0.5, 0.0, 1.0);
      x = mix(x, x * x * (3.0 - 2.0 * x), 0.18);
      vec2 d = (vUv - 0.5) * vec2(1.0, 1.15);
      x *= 1.0 - uVignette * smoothstep(0.3, 0.95, length(d));
      float g = hash(vUv * uRes + uTime * 17.0) - 0.5;
      x += g * uGrain * (0.5 + 0.5 * (1.0 - l));
      x = mix(x, uFlashColor, clamp(uFlash, 0.0, 1.0));
      x *= 1.0 - clamp(uBlack, 0.0, 1.0);
      gl_FragColor = vec4(x, 1.0);
    }`,
};

export const FilmPost: React.FC<{ cam: Cam; grade: Grade; frame: number }> = ({ cam, grade, frame }) => {
  const { gl, scene, camera, size } = useThree();
  const chain = useMemo(() => {
    const w = size.width, h = size.height;
    const rt = new WebGLRenderTarget(w, h, { type: HalfFloatType, samples: 4 });
    const composer = new EffectComposer(gl, rt);
    composer.setPixelRatio(1);
    composer.setSize(w, h);
    composer.addPass(new RenderPass(scene, camera));
    composer.addPass(new ShaderPass(NanGuardShader));
    const gtao = new GTAOPass(scene, camera, w, h);
    gtao.blendIntensity = 1.0;
    gtao.updateGtaoMaterial({ radius: 0.35, distanceExponent: 1.6, thickness: 0.6, scale: 1.0, samples: 16, distanceFallOff: 1.0, screenSpaceRadius: false });
    gtao.updatePdMaterial({ lumaPhi: 10, depthPhi: 2, normalPhi: 3, radius: 5, rings: 2, samples: 16 });
    composer.addPass(withoutFx(gtao, camera));
    const dof = withoutFx(new BokehPass(scene, camera, { focus: 3, aperture: 0.002, maxblur: 0.008 }), camera);
    composer.addPass(dof);
    const bloom = new UnrealBloomPass(new Vector2(w, h), 0.25, 0.5, 0.85);
    composer.addPass(bloom);
    composer.addPass(new OutputPass());
    const g = new ShaderPass(GradeShader);
    g.uniforms.uRes.value.set(w, h);
    composer.addPass(g);
    return { composer, dof, bloom, gtao, g };
  }, [gl, scene, camera, size.width, size.height]);

  useLayoutEffect(() => {
    gl.toneMapping = NeutralToneMapping;
    gl.toneMappingExposure = grade.exposure;
    const u = chain.dof.uniforms as unknown as Record<string, { value: number }>;
    const dist = new Vector3(...cam.pos).distanceTo(new Vector3(...cam.target));
    u.focus.value = cam.focus ?? dist;
    u.aperture.value = cam.aperture ?? 0;
    u.maxblur.value = cam.maxBlur ?? 0;
    chain.dof.enabled = (cam.aperture ?? 0) > 0 && (cam.maxBlur ?? 0) > 0;
    chain.bloom.strength = grade.bloom;
    chain.bloom.radius = grade.bloomRadius;
    chain.bloom.threshold = grade.bloomThreshold;
    chain.gtao.blendIntensity = grade.ao;
    chain.gtao.enabled = grade.ao > 0.01;
    const U = chain.g.uniforms;
    U.uTime.value = frame % 997;
    U.uLift.value.set(...grade.lift);
    U.uGain.value.set(...grade.gain);
    U.uSat.value = grade.sat;
    U.uContrast.value = grade.contrast;
    U.uVignette.value = grade.vignette;
    U.uGrain.value = grade.grain;
    U.uFlash.value = grade.flash;
    U.uFlashColor.value.set(grade.flashColor);
    U.uBlack.value = grade.black;
  }, [frame, chain, cam, grade, gl]);

  useFrame(() => {
    chain.composer.render();
  }, 1);
  return null;
};

import { Color } from "three";
import type { Grade } from "./FilmPost.tsx";
import type { LightState, PointL } from "./Lights.tsx";
import type { V3 } from "../character/spec.ts";

const mixN = (a: number, b: number, k: number) => a + (b - a) * k;
const mix3 = (a: V3, b: V3, k: number): V3 => [mixN(a[0], b[0], k), mixN(a[1], b[1], k), mixN(a[2], b[2], k)];
const _a = new Color(), _b = new Color();
const mixC = (a: string, b: string, k: number) => `#${_a.set(a).lerp(_b.set(b), k).getHexString()}`;
const mixP = (a: PointL, b: PointL, k: number): PointL => {
  // a light fading in/out keeps its position (no sliding across the room)
  const pos = a.i === 0 ? b.pos : b.i === 0 ? a.pos : mix3(a.pos, b.pos, k);
  return { pos, color: mixC(a.color, b.color, k), i: mixN(a.i, b.i, k), dist: mixN(a.dist, b.dist, k) };
};

export const mixLight = (a: LightState, b: LightState, k: number): LightState => {
  if (k <= 0) return a;
  if (k >= 1) return b;
  return {
    bg: mixC(a.bg, b.bg, k),
    fog: { color: mixC(a.fog.color, b.fog.color, k), density: mixN(a.fog.density, b.fog.density, k) },
    env: mixN(a.env, b.env, k),
    hemi: { sky: mixC(a.hemi.sky, b.hemi.sky, k), ground: mixC(a.hemi.ground, b.hemi.ground, k), i: mixN(a.hemi.i, b.hemi.i, k) },
    moon: { dir: mix3(a.moon.dir, b.moon.dir, k), box: a.moon.box, color: mixC(a.moon.color, b.moon.color, k), i: mixN(a.moon.i, b.moon.i, k) },
    spot: a.spot.i === 0 ? { ...b.spot, i: b.spot.i * k } : b.spot.i === 0 ? { ...a.spot, i: a.spot.i * (1 - k) } : { ...b.spot, i: mixN(a.spot.i, b.spot.i, k) },
    pts: [0, 1, 2, 3].map((i) => mixP(a.pts[i], b.pts[i], k)) as LightState["pts"],
  };
};

export const mixGrade = (a: Grade, b: Grade, k: number): Grade => {
  if (k <= 0) return a;
  if (k >= 1) return b;
  const o = { ...a };
  for (const key of Object.keys(a) as (keyof Grade)[]) {
    const va = a[key], vb = b[key];
    if (typeof va === "number" && typeof vb === "number") (o as Record<string, unknown>)[key] = mixN(va, vb, k);
    else if (Array.isArray(va) && Array.isArray(vb)) (o as Record<string, unknown>)[key] = mix3(va as V3, vb as V3, k);
    else if (typeof va === "string" && typeof vb === "string") (o as Record<string, unknown>)[key] = mixC(va, vb, k);
  }
  return o;
};

/** Piecewise timeline of keyed states with smooth blends: [[t, state, blendSeconds], …]. */
export const keyed = <S,>(keys: [number, S, number][], mix: (a: S, b: S, k: number) => S, t: number): S => {
  let cur = keys[0][1];
  for (let i = 1; i < keys.length; i++) {
    const [t0, s, d] = keys[i];
    if (t < t0) break;
    const k = d <= 0 ? 1 : Math.min(1, (t - t0) / d);
    const e = k * k * (3 - 2 * k);
    cur = mix(cur, s, e);
  }
  return cur;
};

// Voice-over cues: scenes refer to words of the transcript ("the lamp clicks off on *mati*") instead of
// seconds, so a re-recorded voice-over only needs a new transcript — the acting re-times itself.

export type Word = [text: string, start: number, end: number];

const norm = (s: string) => s.toLowerCase().replace(/[^\p{L}\p{N}]+/gu, "");

export type Cues = {
  /** film time of a word (n = occurrence, 0-based); throws if the word is missing */
  at: (word: string, n?: number, edge?: "start" | "end") => number;
  end: (word: string, n?: number) => number;
  /** film time of the k-th word overall */
  word: (k: number) => number;
  words: Word[];
  offset: number;
  voEnd: number;
};

export const makeCues = (words: Word[], offset: number): Cues => {
  const find = (w: string, n = 0) => {
    const key = norm(w);
    let k = -1;
    for (let i = 0, seen = 0; i < words.length; i++)
      if (norm(words[i][0]) === key) {
        if (seen === n) { k = i; break; }
        seen++;
      }
    if (k < 0) throw new Error(`cue "${w}" #${n} not in transcript`);
    return words[k];
  };
  return {
    at: (w, n = 0, edge = "start") => offset + find(w, n)[edge === "start" ? 1 : 2],
    end: (w, n = 0) => offset + find(w, n)[2],
    word: (k) => offset + words[k][1],
    words,
    offset,
    voEnd: offset + words[words.length - 1][2],
  };
};
