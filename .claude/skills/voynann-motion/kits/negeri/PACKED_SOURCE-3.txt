// NEGERI camera maths for the browser: the shot type the Stage reads and the projection the 2D overlay uses to pin
// bubbles / name tags to 3D points. Pure easing helpers live in math.ts (node + browser).
import { PerspectiveCamera, Vector3 } from "three";
import type { V3 } from "./math.ts";

export type { V3 } from "./math.ts";
export { backOut, clamp01, hash, lin, mix, mix3, outCubic, rng, sm, smoother } from "./math.ts";
export type Shot = { pos: V3; target: V3; fov: number; roll?: number; shadowAt?: V3; shadowBox?: number };

const tmpCam = new PerspectiveCamera();
/** World point → overlay space (0..1, y down); vis = in front of the camera and inside the frame. */
export const project = (shot: Shot, aspect: number, p: V3) => {
  tmpCam.fov = shot.fov;
  tmpCam.aspect = aspect;
  tmpCam.near = 0.08;
  tmpCam.far = 2000;
  tmpCam.position.set(...shot.pos);
  tmpCam.up.set(Math.sin(shot.roll ?? 0), Math.cos(shot.roll ?? 0), 0);
  tmpCam.lookAt(...shot.target);
  tmpCam.updateProjectionMatrix();
  tmpCam.updateMatrixWorld();
  const v = new Vector3(...p).project(tmpCam);
  return { x: (v.x + 1) / 2, y: (1 - v.y) / 2, vis: v.z < 1 && v.z > -1 && Math.abs(v.x) < 1 && Math.abs(v.y) < 1 };
};

==== 9964 project/src/kits/negeri/chibi/face.ts
// NEGERI faces — a 2D anime-chibi face (big glossy eyes, blush, tiny nose) drawn on a canvas and projected onto a
// spherical patch on the front of the head (the CERDIK decal technique, new drawing). Expressions swap like a 2D
// cartoon; every combination is drawn once and cached.

import { CanvasTexture, LinearFilter, SRGBColorSpace } from "three";
import type { ChibiSpec } from "./spec.ts";

export const PATCH = { lon: (75 * Math.PI) / 180, lat: (75 * Math.PI) / 180 };

import type { Expr } from "./exprs.ts";
export type { Expr } from "./exprs.ts";
export { EXPRS } from "./exprs.ts";
export type FaceState = { expr: Expr; open?: number; blink?: number; look?: [number, number] };

type P2 = [number, number];
const projector = (W: number, H: number) => (p: P2): P2 => {
  const [x, y] = p;
  const r2 = Math.min(0.995, x * x + y * y);
  const z = Math.sqrt(1 - r2);
  const lon = Math.atan2(x, z), lat = Math.asin(Math.max(-1, Math.min(1, y)));
  return [((lon + PATCH.lon) / (2 * PATCH.lon)) * W, ((PATCH.lat - lat) / (2 * PATCH.lat)) * H];
};
const ellipse = (cx: number, cy: number, rx: number, ry: number, n = 40): P2[] =>
  Array.from({ length: n + 1 }, (_, i) => { const a = (i / n) * Math.PI * 2; return [cx + Math.cos(a) * rx, cy + Math.sin(a) * ry] as P2; });
const quad = (a: P2, c: P2, b: P2, n = 20): P2[] =>
  Array.from({ length: n + 1 }, (_, i) => { const t = i / n, u = 1 - t; return [u * u * a[0] + 2 * u * t * c[0] + t * t * b[0], u * u * a[1] + 2 * u * t * c[1] + t * t * b[1]] as P2; });
const lens = (upper: P2[], lower: P2[]): P2[] => [...upper, ...lower.slice().reverse()];

type Ctx = { g: CanvasRenderingContext2D; P: (p: P2) => P2; k: number };
const path = (C: Ctx, pts: P2[]) => { C.g.beginPath(); pts.forEach((p, i) => { const q = C.P(p); if (i) C.g.lineTo(q[0], q[1]); else C.g.moveTo(q[0], q[1]); }); };
const fill = (C: Ctx, pts: P2[], color: string) => { path(C, pts); C.g.closePath(); C.g.fillStyle = color; C.g.fill(); };
const stroke = (C: Ctx, pts: P2[], w: number, color: string) => {
  path(C, pts); C.g.lineWidth = w * C.k; C.g.lineCap = "round"; C.g.lineJoin = "round"; C.g.strokeStyle = color; C.g.stroke();
};

const INK = "#24170f";
const IRIS = "#3a2216";
const MOUTH = "#6b2421";
const TONGUE = "#e0646a";
const TEETH = "#fffdf8";
const EX = 0.3, EY = -0.06, MY = -0.36;

const openEye = (C: Ctx, cx: number, cy: number, s: ChibiSpec, look: P2, lid = 0) => {
  const rx = 0.105, ry = 0.14;
  const lx = cx + look[0], ly = cy + look[1];
  let pts = ellipse(lx, ly, rx, ry);
  const top = ly + ry - lid * 2 * ry;
  if (lid > 0.01) pts = pts.filter((p) => p[1] <= top);
  if (pts.length > 2) fill(C, pts, IRIS);
  if (lid < 0.45) {
    fill(C, ellipse(lx + 0.034, ly + 0.052, 0.038, 0.042), "#ffffff");
    fill(C, ellipse(lx - 0.036, ly - 0.06, 0.018, 0.018), "#ffffff");
  }
  // upper lash line (thick), lids follow it down
  const lt = Math.min(top, ly + ry) + 0.012;
  stroke(C, quad([lx - rx * 1.2, lt - 0.035 + lid * 0.03], [lx, lt + 0.03 - lid * 0.03], [lx + rx * 1.2, lt - 0.035 + lid * 0.03]), 0.038, INK);
  if (s.face?.lashes) {
    const sd = Math.sign(cx);
    stroke(C, [[lx + sd * rx * 1.15, lt - 0.03], [lx + sd * rx * 1.6, lt + 0.015]], 0.026, INK);
  }
};

export const faceTexture = (spec: ChibiSpec, st: FaceState, size = 1024): CanvasTexture => {
  const open = Math.round(Math.max(0, Math.min(1, st.open ?? 0)) * 3) / 3;
  const blink = Math.round(Math.max(0, Math.min(1, st.blink ?? 0)) * 2) / 2;
  const look: P2 = [Math.round((st.look?.[0] ?? 0) * 40) / 40, Math.round((st.look?.[1] ?? 0) * 40) / 40];
  const key = `${spec.skin}|${spec.hair.color}|${JSON.stringify(spec.face ?? {})}|${st.expr}|${open}|${blink}|${look}|${size}`;
  const hit = cache.get(key);
  if (hit) return hit;
  const cv = document.createElement("canvas");
  cv.width = size; cv.height = size;
  const g = cv.getContext("2d")!;
  g.clearRect(0, 0, size, size);
  const C: Ctx = { g, P: projector(size, size), k: size / (2 * PATCH.lon) };
  const f = spec.face ?? {};
  const brow = spec.hair.style === "bald" ? "#4a3a30" : spec.hair.color;
  const e = st.expr;

  if (f.blush || e === "proud" || e === "happy" || e === "laugh") {
    for (const sd of [-1, 1]) fill(C, ellipse(sd * 0.48, -0.24, 0.1, 0.05), "rgba(232,104,96,0.42)");
  }
  // ---- eyes
  for (const sd of [-1, 1] as const) {
    const cx = sd * EX, cy = EY;
    if (blink >= 0.99 && e !== "shocked" && e !== "sleep" && e !== "proud" && e !== "laugh") {
      stroke(C, quad([cx - 0.085, cy - 0.005], [cx, cy - 0.04], [cx + 0.085, cy - 0.005]), 0.036, INK);
      continue;
    }
    switch (e) {
      case "proud":
      case "laugh":
        stroke(C, quad([cx - 0.09, cy - 0.03], [cx, cy + 0.08], [cx + 0.09, cy - 0.03]), e === "laugh" ? 0.048 : 0.042, INK);
        break;
      case "sad":
        openEye(C, cx, cy, spec, [look[0], look[1] - 0.025], 0.28);
        break;
      case "angry":
        openEye(C, cx, cy, spec, look, 0.42);
        break;
      case "sleep":
        stroke(C, quad([cx - 0.09, cy + 0.0], [cx, cy - 0.065], [cx + 0.09, cy + 0.0]), 0.036, INK);
        break;
      case "shocked": {
        fill(C, ellipse(cx, cy + 0.01, 0.125, 0.15), "#ffffff");
        stroke(C, ellipse(cx, cy + 0.01, 0.125, 0.15), 0.026, INK);
        fill(C, ellipse(cx + look[0] * 0.5, cy + 0.0, 0.04, 0.048), IRIS);
        break;
      }
      case "annoyed":
        openEye(C, cx, cy, spec, look, 0.5);
        break;
      case "smug":
        openEye(C, cx, cy, spec, look, 0.35);
        break;
      default:
        openEye(C, cx, cy, spec, look, blink * 0.6);
    }
  }
  // ---- brows
  const bw = f.brows === "thick" ? 0.055 : 0.036;
  const browAt = (sd: number, dy: number, a: number) => {
    const cx = sd * EX, by = 0.22 + dy;
    const inner: P2 = [cx - sd * 0.08, by - a * 0.08], outer: P2 = [cx + sd * 0.08, by + a * 0.08];
    stroke(C, quad(inner, [cx, by + 0.025 + dy * 0.1], outer), bw, brow);
  };
  for (const sd of [-1, 1]) {
    if (e === "shocked") browAt(sd, 0.1, -0.3);
    else if (e === "annoyed") browAt(sd, -0.02, 0.45);
    else if (e === "proud" || e === "happy" || e === "laugh") browAt(sd, 0.03, -0.15);
    else if (e === "sad") browAt(sd, 0.03, -0.45);
    else if (e === "angry") browAt(sd, -0.035, 0.65);
    else if (e === "sleep") browAt(sd, -0.02, -0.25);
    else if (e === "smug") browAt(sd, sd > 0 ? 0.04 : -0.01, 0.1);
    else browAt(sd, 0, 0);
  }
  if (f.glasses) {
    for (const sd of [-1, 1]) stroke(C, ellipse(sd * EX, EY, 0.14, 0.14), 0.024, "#2b2b2b");
    stroke(C, quad([-EX + 0.14, EY + 0.03], [0, EY + 0.06], [EX - 0.14, EY + 0.03]), 0.022, "#2b2b2b");
  }
  // ---- nose
  stroke(C, quad([-0.025, -0.155], [0, -0.18], [0.025, -0.155]), 0.022, "rgba(110,52,34,0.55)");
  if (f.mustache) {
    for (const sd of [-1, 1]) fill(C, lens(quad([0, -0.2], [sd * 0.12, -0.15], [sd * 0.21, -0.25]), quad([0, -0.27], [sd * 0.12, -0.3], [sd * 0.21, -0.25])), brow);
  }
  if (f.beard) {
    const bc = spec.hair.style === "bald" ? "#4a3a30" : spec.hair.color;
    fill(C, lens(quad([-0.38, -0.18], [0, -0.64], [0.38, -0.18]), quad([-0.38, -0.18], [0, -1.08], [0.38, -0.18])), bc);
  }
  // ---- mouth (any face can talk: the open value flaps the mouth, the eyes / brows keep the expression)
  const talkOpen = e === "talk" ? Math.max(open, 0.34) : open;
  const talking = talkOpen > 0.05 && e !== "sleep" && e !== "shocked";
  if (e === "shocked") {
    const h = 0.1 + open * 0.03;
    fill(C, ellipse(0, MY - 0.03, 0.075, h), MOUTH);
    fill(C, ellipse(0, MY - 0.03 - h * 0.5, 0.045, h * 0.35), TONGUE);
    stroke(C, ellipse(0, MY - 0.03, 0.075, h), 0.026, INK);
  } else if (e === "sleep") {
    fill(C, ellipse(0.02, MY - 0.01, 0.045, 0.04 + open * 0.02), MOUTH);
    stroke(C, ellipse(0.02, MY - 0.01, 0.045, 0.04 + open * 0.02), 0.022, INK);
  } else if (e === "annoyed" && !talking) {
    stroke(C, [[-0.1, MY], [-0.05, MY + 0.018], [0, MY - 0.005], [0.05, MY + 0.018], [0.1, MY]], 0.03, INK);
  } else if (e === "sad" && !talking) {
    stroke(C, quad([-0.09, MY - 0.035], [0, MY + 0.035], [0.09, MY - 0.035]), 0.03, INK);
  } else if (e === "angry" && !talking) {
    const up = quad([-0.11, MY + 0.02], [0, MY + 0.035], [0.11, MY + 0.02]), lo = quad([-0.11, MY + 0.02], [0, MY - 0.11], [0.11, MY + 0.02]);
    fill(C, lens(up, lo), TEETH);
    stroke(C, [[-0.1, MY - 0.02], [0.1, MY - 0.02]], 0.016, INK);
    stroke(C, lens(up, lo), 0.026, INK);
  } else if (e === "proud" || e === "happy" || e === "laugh" || talkOpen > 0.05) {
    const w = e === "proud" ? 0.17 : e === "laugh" ? 0.18 : e === "happy" ? 0.15 : 0.11;
    const h = e === "proud" ? 0.13 + talkOpen * 0.06 : e === "laugh" ? 0.18 + talkOpen * 0.05 : e === "happy" ? 0.1 + talkOpen * 0.07 : 0.05 + talkOpen * 0.11;
    const upper = quad([-w, MY + 0.03], [0, MY + 0.045], [w, MY + 0.03]);
    const lower = quad([-w, MY + 0.03], [0, MY + 0.03 - h * 2], [w, MY + 0.03]);
    fill(C, lens(upper, lower), MOUTH);
    const bottom = MY + 0.03 - h;
    if (h > 0.08) fill(C, ellipse(0, bottom + h * 0.28, w * 0.5, h * 0.3), TONGUE);
    if (e === "proud" || e === "happy" || e === "laugh") fill(C, lens(upper, quad([-w * 0.9, MY + 0.015], [0, MY - 0.02], [w * 0.9, MY + 0.015])), TEETH);
    stroke(C, lens(upper, lower), 0.026, INK);
  } else if (e === "smug" && !talking) {
    stroke(C, quad([-0.11, MY + 0.0], [0.03, MY - 0.04], [0.13, MY + 0.05]), 0.03, INK);
  } else {
    stroke(C, quad([-0.09, MY + 0.02], [0, MY - 0.05], [0.09, MY + 0.02]), 0.03, INK);
  }
  if (e === "annoyed") {
    const drop = [...quad([0.6, 0.34], [0.53, 0.2], [0.6, 0.16]), ...quad([0.6, 0.16], [0.67, 0.2], [0.6, 0.34])];
    fill(C, drop, "#9fd8f5");
    stroke(C, drop, 0.016, "#3b7ea6");
  }
  const t = new CanvasTexture(cv);
  t.colorSpace = SRGBColorSpace;
  t.minFilter = LinearFilter;
  t.generateMipmaps = false;
  t.anisotropy = 4;
  cache.set(key, t);
  return t;
};

// NEGERI type (both SIL OFL, licences in public/negeri/):
//   Fredoka — rounded display face for the chip, counter, punch words and CTA (game-UI feel)
//   Plus Jakarta Sans — handles, comment text, labels
export const F_DISPLAY = "NegeriFredoka";
export const F_TEXT = "NegeriJakarta";

let ready: Promise<void> | null = null;
export const fontsReady = (): Promise<void> => {
  if (ready) return ready;
  if (typeof document === "undefined") return (ready = Promise.resolve());
  const handle = delayRender("negeri fonts");
  const faces = [
    new FontFace(F_DISPLAY, `url(${staticFile("negeri/Fredoka.ttf")}) format("truetype")`, { weight: "300 700" }),
    new FontFace(F_TEXT, `url(${staticFile("negeri/PlusJakartaSans.ttf")}) format("truetype")`, { weight: "200 800" }),
  ];
  ready = Promise.all(faces.map(async (f) => { await f.load(); (document.fonts as unknown as { add(f: FontFace): void }).add(f); })).then(
    () => continueRender(handle),
    (e) => { continueRender(handle); throw e; },
  );
  return ready;
};

export type V3 = [number, number, number];

export const clamp01 = (x: number) => (x < 0 ? 0 : x > 1 ? 1 : x);
export const lin = (s: number, a: number, b: number) => clamp01((s - a) / (b - a));
export const sm = (t: number) => { const x = clamp01(t); return x * x * (3 - 2 * x); };
export const smoother = (t: number) => { const x = clamp01(t); return x * x * x * (x * (x * 6 - 15) + 10); };
/** ease-out with a small overshoot (cards, pop-ins); c = overshoot amount */
export const backOut = (t: number, c = 1.4) => { const x = clamp01(t) - 1; return 1 + (c + 1) * x * x * x + c * x * x; };
export const outCubic = (t: number) => 1 - Math.pow(1 - clamp01(t), 3);
export const mix = (a: number, b: number, t: number) => a + (b - a) * t;
export const mix3 = (a: V3, b: V3, t: number): V3 => [mix(a[0], b[0], t), mix(a[1], b[1], t), mix(a[2], b[2], t)];
export const add3 = (a: V3, b: V3, k = 1): V3 => [a[0] + b[0] * k, a[1] + b[1] * k, a[2] + b[2] * k];
export const wrapA = (a: number) => Math.atan2(Math.sin(a), Math.cos(a));
/** Deterministic hash 0..1 */
export const hash = (n: number) => { const x = Math.sin(n * 127.1 + 311.7) * 43758.5453; return x - Math.floor(x); };
/** Seeded PRNG */
export const rng = (seed: number) => { let s = seed >>> 0 || 1; return () => { s = (s * 1664525 + 1013904223) >>> 0; return s / 4294967296; }; };
/** Local (x, z) of a frame at `at` turned by `yaw` (0 = front toward +z) → world. */
export const toWorld = (p: V3, at: V3, yaw: number): V3 => [at[0] + p[0] * Math.cos(yaw) + p[2] * Math.sin(yaw), at[1] + p[1], at[2] - p[0] * Math.sin(yaw) + p[2] * Math.cos(yaw)];

export const rightOf = (yaw: number): V3 => [-Math.cos(yaw), 0, Math.sin(yaw)];

  let free: Band[] = [[lo, hi]];

// NEGERI world geometry: low-poly pieces baked into single vertex-coloured, flat-shaded geometries.
// Every prop is built from boxes / prisms / cylinders with a colour each, merged once, then drawn (or instanced).

import {
  BoxGeometry, BufferGeometry, Color, ConeGeometry, CylinderGeometry, Euler, Float32BufferAttribute, IcosahedronGeometry, Matrix4,
  Quaternion, Vector3,
} from "three";
import { mergeGeometries } from "three/examples/jsm/utils/BufferGeometryUtils.js";
import { rng, type V3 } from "../math.ts";

export type Part = BufferGeometry;

export const colorize = (g: BufferGeometry, color: string, jitter = 0, seed = 1) => {
  const ng = g.index ? g.toNonIndexed() : g;
  const n = ng.getAttribute("position").count;
  const c = new Color(color);
  const r = rng(seed);
  const arr = new Float32Array(n * 3);
  for (let i = 0; i < n; i += 3) {
    const k = 1 + (r() - 0.5) * jitter;
    for (let v = 0; v < 3; v++) { arr[(i + v) * 3] = c.r * k; arr[(i + v) * 3 + 1] = c.g * k; arr[(i + v) * 3 + 2] = c.b * k; }
  }
  ng.setAttribute("color", new Float32BufferAttribute(arr, 3));
  ng.deleteAttribute("uv");
  return ng;
};
const place = (g: BufferGeometry, pos: V3, rot: V3 = [0, 0, 0], scale: V3 = [1, 1, 1]) => {
  g.applyMatrix4(new Matrix4().compose(new Vector3(...pos), new Quaternion().setFromEuler(new Euler(...rot)), new Vector3(...scale)));
  return g;
};

/** box: size, centre position */
export const box = (size: V3, pos: V3, color: string, rot: V3 = [0, 0, 0], jitter = 0) =>

/** box resting on y = pos[1] (its bottom at pos[1]) */
export const boxOn = (size: V3, pos: V3, color: string, rot: V3 = [0, 0, 0]) => box(size, [pos[0], pos[1] + size[1] / 2, pos[2]], color, rot);
export const cyl = (rTop: number, rBot: number, h: number, pos: V3, color: string, seg = 8, rot: V3 = [0, 0, 0]) =>
  place(colorize(new CylinderGeometry(rTop, rBot, h, seg, 1), color), pos, rot);
/** cylinder between two points */
export const rod = (a: V3, b: V3, r: number, color: string, seg = 6) => {
  const A = new Vector3(...a), B = new Vector3(...b);
  const d = B.clone().sub(A);
  const L = d.length();
  const g = colorize(new CylinderGeometry(r, r, L, seg, 1), color);
  g.applyMatrix4(new Matrix4().compose(A.clone().add(B).multiplyScalar(0.5), new Quaternion().setFromUnitVectors(new Vector3(0, 1, 0), d.normalize()), new Vector3(1, 1, 1)));
  return g;
};
export const cone = (r: number, h: number, pos: V3, color: string, seg = 6, rot: V3 = [0, 0, 0]) => place(colorize(new ConeGeometry(r, h, seg, 1), color), pos, rot);
export const blob = (r: number, pos: V3, color: string, scale: V3 = [1, 1, 1], detail = 0, seed = 1) =>
  place(colorize(new IcosahedronGeometry(r, detail), color, 0.12, seed), pos, [0, 0, 0], scale);
/** triangular prism along z: base width w (x), height h, length l; bottom at y = 0 */
export const prism = (w: number, h: number, l: number, pos: V3, color: string, rot: V3 = [0, 0, 0]) => {
  const x = w / 2, z = l / 2;
  const v = [
    // left slope
    -x, 0, z, 0, h, z, 0, h, -z, -x, 0, z, 0, h, -z, -x, 0, -z,
    // right slope
    x, 0, z, x, 0, -z, 0, h, -z, x, 0, z, 0, h, -z, 0, h, z,
    // front, back triangles
    -x, 0, z, x, 0, z, 0, h, z,
    x, 0, -z, -x, 0, -z, 0, h, -z,
  ];
  const g = new BufferGeometry();
  g.setAttribute("position", new Float32BufferAttribute(v, 3));
  g.computeVertexNormals();
  return place(colorize(g, color), pos, rot);
};
export const merge = (parts: BufferGeometry[]) => {
  const g = mergeGeometries(parts.map((p) => (p.index ? p.toNonIndexed() : p)), false)!;
  g.computeVertexNormals();
  g.computeBoundingSphere();
  return g;
};

// NEGERI series continuity — the country between episodes. world.json keeps every finished day (population, builds
// with their plots, named houses with their slots); the state at the start of Day N is all days before N folded
// together, so a day can be re-rendered (or re-committed) without duplicating anything. Pure TypeScript.

import { ALIASES, BUILDINGS, type BuildingDef } from "./buildings.ts";
import { PLOTS, SLOTS, SPECIAL, type Plot } from "../kits/negeri/world/map.ts";
import type { BuildSpec, PlacedBuild, PlacedPlate, SeriesWorld, Story, WorldDay } from "./types.ts";

export const DEFAULT_FLAG: [string, string] = ["#d7262e", "#f7f5ef"];

/** kind (or alias) → canonical kind, its definition and its default display name */
export const resolveKind = (kind: string, custom: Record<string, BuildingDef> = {}): { kind: string; def: BuildingDef; name: string } | null => {
  const k = kind.toLowerCase().trim().replace(/\s+/g, "-");
  if (custom[k]) return { kind: k, def: custom[k], name: custom[k].name };
  if (BUILDINGS[k]) return { kind: k, def: BUILDINGS[k], name: BUILDINGS[k].name };
  const a = ALIASES[k];
  if (a && BUILDINGS[a[0]]) return { kind: a[0], def: BUILDINGS[a[0]], name: a[1] };
  return null;
};

export const newWorld = (story: Story): SeriesWorld => ({
  voynann: "negeri-world/1",
  country: story.country ?? "Republik Komen",
  flag: story.flag ?? DEFAULT_FLAG,
  host: story.host ?? { preset: "presiden" },
  voice: story.voice ?? "aku",
  custom: {},
  days: [],

export type State = { population: number; builds: PlacedBuild[]; plates: PlacedPlate[]; lastDay: number };
/** The country at the start of `day` (every other story's days before it). */
export const stateBefore = (w: SeriesWorld | null, day: number, storyId: string): State => {
  const days = (w?.days ?? []).filter((d) => d.day < day && d.story !== storyId).sort((a, b) => a.day - b.day);
  const builds: PlacedBuild[] = [], plates: PlacedPlate[] = [];
  let population = 1, lastDay = 0;
  for (const d of days) {
    population = d.population;
    lastDay = d.day;
    for (const b of d.builds) {
      // a build on a plot replaces whatever stood there
      const i = builds.findIndex((x) => x.plot === b.plot && b.plot !== "road");
      if (i >= 0) builds.splice(i, 1);
      builds.push(b);
    }
    plates.push(...d.plates);
  }
  return { population, builds, plates, lastDay };
};
/** plots / slots taken by any other story (earlier or later days) */
const takenElsewhere = (w: SeriesWorld | null, storyId: string, day: number) => {
  const plots = new Set<string>(), slots = new Set<string>(), handles = new Set<string>();
  for (const d of w?.days ?? []) {
    if (d.story === storyId) continue;
    for (const b of d.builds) if (b.kind !== "rumah") plots.add(b.plot); else slots.add(b.plot);
    for (const p of d.plates) { slots.add(p.slot); if (d.day < day) handles.add(p.handle.toLowerCase()); }
    for (const b of d.builds) if (b.kind === "rumah" && b.by && d.day < day) handles.add(b.by.toLowerCase());
  }
  return { plots, slots, handles };
};

const CLS = { S: 0, M: 1, L: 2, XL: 3 };
const fits = (def: BuildingDef, p: Plot) => def.size[0] <= p.w + 0.05 && def.size[1] <= p.d + 0.05;

export type Placed = { builds: (PlacedBuild & { def: BuildingDef })[]; plates: PlacedPlate[]; errors: string[]; warnings: string[] };
/**
 * This episode's builds and new named houses → plots / slots. A day that was committed before keeps its places
 * (re-renders look the same); a new day takes the nearest free plot that fits each building.
 */
export const placeEpisode = (story: Story, w: SeriesWorld | null, custom: Record<string, BuildingDef>, byOf: (b: BuildSpec) => string | undefined): Placed => {
  const errors: string[] = [], warnings: string[] = [];
  const prev = w?.days.find((d) => d.story === story.id);
  const st = stateBefore(w, story.day, story.id);
  const taken = takenElsewhere(w, story.id, story.day);
  const usedPlots = new Set(taken.plots), usedSlots = new Set(taken.slots);
  let no = 2 + st.plates.length + st.builds.filter((b) => b.kind === "rumah").length;
  const out: Placed["builds"] = [];
  for (const b of story.builds ?? []) {
    const r = resolveKind(b.kind, custom);
    if (!r) { errors.push(`build "${b.id}": kind "${b.kind}" is not in the library (have: ${Object.keys(BUILDINGS).join(", ")}; aliases: ${Object.keys(ALIASES).join(", ")}) — or define it in buildings.json`); continue; }
    const name = (b.name ?? r.name).toUpperCase();
    const by = byOf(b);
    const old = prev?.builds.find((x) => x.id === b.id && x.kind === r.kind);
    let plot: string | undefined = old?.plot;
    if (plot && (r.def.special === "slot" ? usedSlots.has(plot) : usedPlots.has(plot) && !b.plot)) plot = undefined;
    if (!plot && b.plot) {
      const p = PLOTS.find((x) => x.id === b.plot) ?? (r.def.special === "slot" ? undefined : undefined);
      const s = SLOTS.find((x) => x.id === b.plot);
      if (r.def.special === "slot" ? !s : !p) errors.push(`build "${b.id}": plot "${b.plot}" does not exist (plots: ${PLOTS.map((x) => x.id).join(", ")}; house slots R1…R${SLOTS.length})`);
      else if (p && !fits(r.def, p)) errors.push(`build "${b.id}": ${r.kind} (${r.def.size.join(" × ")} m) does not fit plot ${p.id} (${p.w} × ${p.d} m)`);
      else plot = b.plot;
      if (plot && st.builds.some((x) => x.plot === plot)) warnings.push(`build "${b.id}": plot ${plot} already has ${st.builds.find((x) => x.plot === plot)!.name} — it is replaced`);
    }
    if (!plot) {
      if (r.def.special === "slot") plot = SLOTS.find((s) => !usedSlots.has(s.id))?.id;
      else if (r.def.special) {
        const sp = SPECIAL[r.def.special === "station" ? "station" : r.def.special === "field" ? "field" : r.def.special];
        const already = st.builds.find((x) => x.plot === sp.id && x.kind === r.kind);
        if (already) errors.push(`build "${b.id}": the ${r.kind} already stands (day ${already.day}) — pick another building`);
        plot = sp.id;
      } else {
        const cands = PLOTS.filter((p) => !usedPlots.has(p.id) && !st.builds.some((x) => x.plot === p.id) && fits(r.def, p));
        cands.sort((a, c) => (c.pref?.includes(r.kind) ? 1 : 0) - (a.pref?.includes(r.kind) ? 1 : 0) || CLS[a.cls] - CLS[c.cls] || PLOTS.indexOf(a) - PLOTS.indexOf(c));
        // a preferred plot is kept for its building while others are free
        const pick = cands.find((p) => !p.pref || p.pref.includes(r.kind) || cands.every((q) => q.pref && !q.pref.includes(r.kind)));
        plot = pick?.id;
      }
    }
    if (!plot) { errors.push(`build "${b.id}": no free plot fits ${r.kind} (${r.def.size.join(" × ")} m) — the map is full for this size; give "plot": "<id>" to replace an old building`); continue; }
    if (r.def.special === "slot") usedSlots.add(plot); else usedPlots.add(plot);
    out.push({ id: b.id, kind: r.kind, plot, name, by, colors: b.colors, day: story.day, def: r.def, no: r.def.special === "slot" ? no++ : undefined });
  }
  const plates: PlacedPlate[] = [];
  for (const raw of story.plates ?? []) {
    const p = typeof raw === "string" ? { handle: raw } : raw;
    if (taken.handles.has(p.handle.toLowerCase())) { warnings.push(`plate ${p.handle} already has a house (an earlier day) — left out`); continue; }
    const old = prev?.plates.find((x) => x.handle.toLowerCase() === p.handle.toLowerCase());
    let slot = old && !usedSlots.has(old.slot) ? old.slot : undefined;
    if (!slot) slot = SLOTS.find((s) => !usedSlots.has(s.id))?.id;
    if (!slot) { errors.push(`plate ${p.handle}: all ${SLOTS.length} named-house slots are used`); continue; }
    usedSlots.add(slot);
    plates.push({ handle: p.handle, no: no++, slot, colors: (p as { colors?: Record<string, string> }).colors, day: story.day });
  }
  return { builds: out, plates, errors, warnings };
};

/** world.json after this episode: its day entry added or replaced (other days untouched). */
export const commitDay = (w: SeriesWorld, story: Story, placed: Placed, citizens: WorldDay["citizens"]): SeriesWorld => {
  const day: WorldDay = {
    day: story.day, story: story.id, population: story.population,
    builds: placed.builds.map(({ def: _def, ...b }) => b), plates: placed.plates, citizens,
  };
  const days = [...w.days.filter((d) => d.story !== story.id), day].sort((a, b) => a.day - b.day);
  return { ...w, days };
};
