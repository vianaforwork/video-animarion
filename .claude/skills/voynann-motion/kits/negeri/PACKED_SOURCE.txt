// NEGERI — "bikin negara dari nol" episode → film.
//   node scripts/make.mjs --library                what exists: beats, buildings (+ aliases), plots, cast + card codes, actions, expressions, gags
//   node scripts/make.mjs <id> --plan              compile stories/<id>/story.json (+ the series world): shot list + warnings
//   node scripts/make.mjs <id> --stills auto       one still per shot → out/<id>/stills/_sheet.jpg (or --stills 2s,9.5s)
//   node scripts/make.mjs <id> --preview           plan + sound + 540×960 preview (rendered at 2×, downscaled) + QA report
//   node scripts/make.mjs <id> --final             plan + sound + QA gate + 1080×1920 @ 60 + delivery scan, then the day is
//                                                  written into series/<name>/world.json (Day N+1 starts from it)
//   node scripts/make.mjs <id> --commit            write the day into world.json without rendering (--no-commit skips it on --final)
//   node scripts/make.mjs --look [kind,kind]       building look-dev sheet → out/_look/_sheet.jpg (library or a story's own buildings: <id> --look)
// Characters (tokoh):
//   node scripts/make.mjs --menu                   the buyer's character cards → out/_menu/kartu-tokoh-1|2|3.jpg
//   node scripts/make.mjs <id> --cast              this episode's host + citizens → out/<id>/tokoh.jpg (validated)
//   node scripts/make.mjs <id> --save-tokoh host[,budi] [--nama "Pak Pres"]   → tokoh/tokoh-<nama>.json
//   story.json "host": { "tokoh": "pak-pres" } / "cast": { "budi": { "tokoh": "budi" } } use a saved file (+ "look" on top)
// Voice: stories/<id>/vo.(mp3|wav|m4a) → word timings with whisper.cpp; without a voice the lines are paced.
// Own buildings: stories/<id>/buildings.json or series/<name>/buildings.json (same format as the library; kept in world.json).
// Flags: --silent (no sound), --no-music (voice + SFX only), --wide (16:9: only when asked), --no-qa, --force-qa.

const renderSeq = (comp, props, dir) => {
  fs.rmSync(dir, { recursive: true, force: true });
  fs.mkdirSync(dir, { recursive: true });
  const pf = path.join(dir, "_props.json");
  fs.writeFileSync(pf, JSON.stringify(props));
  sh("npx", ["remotion", "render", "src/index.ts", comp, dir, "--sequence", "--image-format=jpeg", "--jpeg-quality=92", `--props=${pf}`, "--concurrency=2", "--log=error"]);
  fs.rmSync(pf, { force: true });
  return fs.readdirSync(dir).filter((f) => /^element-\d+\.jpe?g$/.test(f)).sort((a, b) => parseInt(a.slice(8)) - parseInt(b.slice(8))).map((f) => path.join(dir, f));

const sheetOf = (files, W, H, out, labels) => {
  const cols = Math.min(4, files.length), n = cols * Math.ceil(files.length / cols);
  if (n < 1) return;
  const make = (withText) => {
    const pads = Array.from({ length: n - files.length }, () => ["-f", "lavfi", "-i", `color=c=black:s=${W}x${H}`]).flat();
    const filter = Array.from({ length: n }, (_, i) => `[${i}:v]scale=${W}:${H},setsar=1${withText && labels && i < files.length ? `,drawtext=text='${labels[i]}':x=10:y=10:fontsize=34:fontcolor=yellow:box=1:boxcolor=black@0.6` : ""}[v${i}]`).join(";") + ";" +
      Array.from({ length: n }, (_, i) => `[v${i}]`).join("") + `xstack=inputs=${n}:layout=` + Array.from({ length: n }, (_, i) => `${(i % cols) * W}_${Math.floor(i / cols) * H}`).join("|") + `,scale='min(2160,iw)':-2`;
    execFileSync("ffmpeg", ["-v", "error", "-y", ...files.flatMap((f) => ["-i", f]), ...pads, "-filter_complex", filter, "-frames:v", "1", out], { stdio: ["ignore", "ignore", "ignore"] });
  };
  if (n === 1) { fs.copyFileSync(files[0], out); return; }
  try { make(true); } catch { try { make(false); } catch { console.log("(contact sheet skipped: ffmpeg not available)"); } }
};
const renderCards = (groups, h, outDir, names) => {
  const files = renderSeq("NegeriCard", { strips: groups.flat(), h }, path.join(outDir, "_strips"));
  let k = 0;
  const made = groups.map((g, gi) => {
    const part = files.slice(k, (k += g.length)), to = path.join(outDir, names[gi]);
    if (part.length === 1) fs.copyFileSync(part[0], to);
    else execFileSync("ffmpeg", ["-v", "error", "-y", ...part.flatMap((f) => ["-i", f]), "-filter_complex", part.map((_, i) => `[${i}:v]`).join("") + `vstack=inputs=${part.length}`, "-q:v", "3", to]);
    return to;
  });
  fs.rmSync(path.join(outDir, "_strips"), { recursive: true, force: true });
  return made;
};

// NEGERI building library — every building is DATA: labelled build STEPS (shown on the build card as stages), each a
// list of PARTS (simple low-poly shapes) or MACROS (walls with a door and windows, a roof, posts, a fence, stairs,
// a scaffold, trees, lamps, benches). The same format works in JSON, so a buyer's own building (DPR, waterboom,
// menara pandang…) is a buildings.json next to the story — no engine change. Pure TypeScript (node + browser).
//
// Local frame: origin = centre of the footprint on the ground, FRONT = +z (it faces the road / the plaza), metres.
// pos = BOTTOM centre for box / cyl / cone / prism / hip; centre for sphere / blob; rod = a → b. rot in degrees.
// Colours may be tokens "{wall}", "{roof}", "{accent}" — filled from the build entry's "colors" (or the defaults in
// "params"). Sign text may use {name} (the build's display name), {by} (the requester), {no} (house number),
// {country} (the country).

import type { V3 } from "../kits/negeri/math.ts";

export type Anim = "drop" | "grow" | "pop" | "slide";
type Common = { color?: string; anim?: Anim; axis?: "x" | "y" | "z"; temp?: boolean; glass?: boolean; water?: boolean; rot?: V3 };
export type PartSpec = Common & (
  | { shape: "box"; size: V3; pos: V3 }
  | { shape: "cyl"; r: number; r2?: number; h: number; pos: V3; seg?: number }
  | { shape: "cone"; r: number; h: number; pos: V3; seg?: number }
  | { shape: "sphere"; r: number; pos: V3; half?: boolean; scale?: V3 }
  | { shape: "blob"; r: number; pos: V3; scale?: V3 }
  | { shape: "prism"; w: number; h: number; l: number; pos: V3 }
  | { shape: "hip"; w: number; d: number; h: number; ridge?: number; pos: V3 }
  | { shape: "rod"; a: V3; b: V3; r: number }
  | { shape: "sign"; size: V3; pos: V3; text: string; sub?: string; bg?: string; ink?: string; subInk?: string }
);
export type MacroSpec =
  | { macro: "walls"; w: number; d: number; h: number; y?: number; at?: [number, number]; color: string; base?: { h: number; color: string }; trim?: string;
      door?: { x?: number; w?: number; h?: number; color?: string; side?: "front" | "back" }; windows?: { n?: number; w?: number; h?: number; y?: number; sides?: ("front" | "back" | "left" | "right")[]; frame?: string; glass?: string } }

// ------------------------------------------------------------------ expansion: macros → pieces
export type Piece = (PartSpec & { anim: Anim; axis: "x" | "y" | "z"; step: number; color: string; tpl?: string });
const D2R = Math.PI / 180;

const sideFrame = (side: string, w: number, d: number) => {
  // returns origin of the face, along-axis, length, and rotation (deg) so a box's local z points out of the face
  switch (side) {
    case "back": return { o: [0, -d / 2] as [number, number], along: [1, 0] as [number, number], len: w, rot: 180 };
    case "left": return { o: [w / 2, 0] as [number, number], along: [0, -1] as [number, number], len: d, rot: 90 };
    case "right": return { o: [-w / 2, 0] as [number, number], along: [0, 1] as [number, number], len: d, rot: -90 };
    default: return { o: [0, d / 2] as [number, number], along: [1, 0] as [number, number], len: w, rot: 0 };
  }
};

const shift = (parts: PartSpec[], at?: [number, number]): PartSpec[] => {
  if (!at || (!at[0] && !at[1])) return parts;
  return parts.map((q) => (q.shape === "rod" ? { ...q, a: [q.a[0] + at[0], q.a[1], q.a[2] + at[1]] as V3, b: [q.b[0] + at[0], q.b[1], q.b[2] + at[1]] as V3 } : { ...q, pos: [q.pos[0] + at[0], q.pos[1], q.pos[2] + at[1]] as V3 }));
};
const expandMacro = (m: MacroSpec): PartSpec[] => {
  if ((m.macro === "walls" || m.macro === "roof" || m.macro === "posts") && m.at) return shift(expandMacro({ ...m, at: undefined } as MacroSpec), m.at);
  switch (m.macro) {
    case "walls": {
      const y = m.y ?? 0, out: PartSpec[] = [];
      let by = y;
      if (m.base) { out.push({ shape: "box", size: [m.w + 0.2, m.base.h, m.d + 0.2], pos: [0, y, 0], color: m.base.color, anim: "pop" }); by += m.base.h; }
      out.push({ shape: "box", size: [m.w, m.h, m.d], pos: [0, by, 0], color: m.color, anim: "grow" });
      if (m.trim) out.push({ shape: "box", size: [m.w + 0.08, 0.16, m.d + 0.08], pos: [0, by + m.h - 0.16, 0], color: m.trim, anim: "drop" });
      const door = m.door;
      if (door) {
        const f = sideFrame(door.side ?? "front", m.w, m.d), x = door.x ?? 0, dw = door.w ?? 0.9, dh = door.h ?? 1.9;
        out.push({ shape: "box", size: [dw, dh, 0.06], pos: [f.o[0] + f.along[0] * x, by, f.o[1] + f.along[1] * x + (f.rot === 0 ? 0.02 : f.rot === 180 ? -0.02 : 0)], rot: [0, f.rot, 0], color: door.color ?? C.door, anim: "pop" });
      }
      const W = m.windows;
      if (W) {
        const n = W.n ?? 2, ww = W.w ?? 0.8, wh = W.h ?? 0.7, wy = by + (W.y ?? 1.0);

/** Validate a building definition (buyer's buildings.json), plain messages. */
export const buildingProblems = (kind: string, def: unknown): string[] => {
  const p: string[] = [];
  const d = def as Partial<BuildingDef>;
  if (!d || typeof d !== "object") return [`building "${kind}" must be an object`];
  if (typeof d.name !== "string") p.push(`building "${kind}": "name" (the sign / card text) is missing`);
  if (!Array.isArray(d.size) || d.size.length !== 2) p.push(`building "${kind}": "size": [width, depth] in metres is missing`);
  if (typeof d.h !== "number") p.push(`building "${kind}": "h" (height in metres) is missing`);
  if (!Array.isArray(d.steps) || !d.steps.length) p.push(`building "${kind}": "steps" (at least one { "label", "parts": [...] }) is missing`);
  const SH = ["box", "cyl", "cone", "sphere", "blob", "prism", "hip", "rod", "sign"], MA = ["walls", "roof", "posts", "fence", "stairs", "scaffold", "tree", "lamp", "bench"];
  (d.steps ?? []).forEach((st, i) => {
    if (typeof st.label !== "string") p.push(`building "${kind}" step ${i + 1}: "label" missing (shown on the build card, e.g. "ATAP")`);
    (st.parts ?? []).forEach((q, j) => {
      const where = `building "${kind}" step ${i + 1} part ${j + 1}`;
      if ("macro" in q) { if (!MA.includes(q.macro)) p.push(`${where}: macro "${q.macro}" unknown (have: ${MA.join(", ")})`); }
      else if (!SH.includes((q as PartSpec).shape)) p.push(`${where}: shape "${(q as PartSpec).shape}" unknown (have: ${SH.join(", ")})`);
      else if ((q as PartSpec).shape !== "rod" && !Array.isArray((q as { pos?: V3 }).pos)) p.push(`${where}: "pos" [x, y, z] missing`);
    });
  });
  return p;
};

export const paceWords = (text: string, t0: number, t1: number): Word[] => {
  const raw = text.split(/\s+/).filter(Boolean);
  const weights = raw.map((w) => 0.55 + w.replace(/[,.?!"“”]/g, "").length * 0.085 + (/[,.?!]["”]?$/.test(w) ? 0.5 : 0));
  const tot = weights.reduce((x, y) => x + y, 0);
  let acc = 0;
  return raw.map((w, i) => { const t = t0 + ((t1 - t0) * acc) / tot; acc += weights[i]; return { w, t }; });
};

export const paceLines = (texts: string[], wps = 3.0, gap = 0.3, start = 0.3): Timed[] => {

    const n = text.split(/\s+/).length;

    const line = { text, t0: t, t1: t + d, words: paceWords(text, t, t + d) };
    t += d + gap;
    return line;
  });

  if (!showPlates.length) { const old = stateBefore(world, story.day, story.id).plates.slice(-2); showPlates.push(...old.map((p) => ({ slot: p.slot, handle: p.handle }))); }
  const ids = new Set<string>();
  for (const b of story.builds ?? []) { if (ids.has(b.id)) errors.push(`build id "${b.id}" is used twice`); ids.add(b.id); if (castIds.includes(b.id)) errors.push(`build id "${b.id}" is also a cast id — use different ids`); }

  // ---------------------------------------------------------------- words
  const words = lines.flatMap((l, li) => l.words.map((w, i) => ({ w: w.w, t0: w.t, t1: l.words[i + 1]?.t ?? l.t1, line: li })));
  const wordTime = (li: number, at: string | number, from: number, where: string): { t: number; i: number } => {
    const L = lines[li];
    if (typeof at === "number") return { t: L.t0 + at, i: from };
    const [txt, nth] = at.split("#");
    let seen = 0;
    for (let i = Math.max(0, from); i < L.words.length; i++) if (norm(L.words[i].w).startsWith(norm(txt)) && ++seen >= (nth ? +nth : 1)) return { t: L.words[i].t, i };
    errors.push(`${where}: the word "${at}" is not in line ${li + 1} ("${L.text.slice(0, 60)}…")`);
    return { t: L.t0, i: from };
  };

  // ---------------------------------------------------------------- beats (times)
  type B = { spec: BeatSpec; kind: BeatKind; line: number; t0: number; wi: number; auto?: boolean; sub: number };
  const beats: B[] = [];
  story.lines.forEach((ls: LineSpec, li) => {
    const L = lines[li];
    if (!L) { errors.push(`line ${li + 1} has no timing (does the voice-over match the script?)`); return; }
    const { text: _t, beats: bs, ...lineDefaults } = ls;
    void _t;
    const specs: BeatSpec[] = bs?.length ? bs.map((b, i) => (i === 0 ? { ...lineDefaults, ...b, cast: { ...(ls.cast ?? {}), ...(b.cast ?? {}) } } : { beat: b.beat ?? ls.beat, ...b })) : [lineDefaults];
    let from = 0;
    specs.forEach((b, i) => {
      const kind = (b.beat ?? ls.beat ?? "host") as BeatKind;
      if (i === 0) { beats.push({ spec: b, kind, line: li, t0: li === 0 ? 0 : L.t0 - LEAD, wi: 0, sub: 0 }); return; }
      if (b.at === undefined) { errors.push(`line ${li + 1}, beat ${i + 1}: give "at" (the word the cut lands on)`); return; }
      const w = wordTime(li, b.at, from, `line ${li + 1}, beat ${i + 1}`);
      from = w.i + 1;
      beats.push({ spec: b, kind, line: li, t0: w.t - LEAD, wi: w.i, sub: 0 });
    });
    // automatic cuts inside a long single-beat line (the genre cuts every ~2 s)
    const k0 = (ls.beat ?? "host") as BeatKind;
    // a plates line: one shot per new name plate (when each gets ≥ 0.95 s), cut on the nearest words
    if (!bs?.length && k0 === "plates") {
      const nPl = Math.min(4, showPlates.length), dur = L.t1 + 0.4 - L.t0;
      if (nPl > 1 && dur / nPl >= 0.95) {
        let prevI = 0;
        for (let k = 1; k < nPl; k++) {
          const tt = L.t0 + (dur * k) / nPl;
          let bi = -1, bd = 9;
          L.words.forEach((w, i) => { if (i > prevI && Math.abs(w.t - tt) < bd) { bd = Math.abs(w.t - tt); bi = i; } });
          if (bi < 0) continue;
          prevI = bi;
          beats.push({ spec: {}, kind: "plates", line: li, t0: L.words[bi].t - LEAD, wi: bi, auto: true, sub: k });
        }
      }
    }
    if (!bs?.length && k0 !== "counter" && k0 !== "plates" && k0 !== "flyover") {
      const dur = L.t1 - L.t0, n = Math.round(dur / 2.1);
      if (n >= 2) {
        let prevI = 0;
        for (let k = 1; k < n; k++) {
          const tt = L.t0 + (dur * k) / n;
          let bi = -1, bd = 9;
          L.words.forEach((w, i) => { if (i > prevI + 1 && Math.abs(w.t - tt) < bd && L.t1 - w.t > 0.9) { bd = Math.abs(w.t - tt); bi = i; } });
          if (bi < 0) continue;
          prevI = bi;
          beats.push({ spec: {}, kind: k0, line: li, t0: L.words[bi].t - LEAD, wi: bi, auto: true, sub: k });
        }
      }
    }
  });
  beats.sort((a, b) => a.t0 - b.t0);
  const endsCta = beats.length && beats[beats.length - 1].kind === "cta";
  const duration = +((lines[lines.length - 1]?.t1 ?? 1) + (opts.tail ?? (endsCta ? 1.6 : 1.1))).toFixed(3);
  for (let i = beats.length - 1; i > 0; i--) {
    const end = beats[i + 1]?.t0 ?? duration;
    if (end - beats[i].t0 < 0.85 || beats[i].t0 - beats[i - 1].t0 < 0.85) {
      if (!beats[i].auto && beats[i - 1].auto && beats[i].t0 - beats[i - 1].t0 < 0.85) { beats.splice(i - 1, 1); continue; }
      if (!beats[i].auto) warnings.push(`line ${beats[i].line + 1}: a cut at ${beats[i].t0.toFixed(2)} s makes a shot shorter than 0.85 s — merged into the shot before`);
      const keep = beats[i - 1];
      keep.spec = { ...beats[i].spec, ...keep.spec, cast: { ...(keep.spec.cast ?? {}), ...(beats[i].spec.cast ?? {}) } };
      beats.splice(i, 1);
    }
  }

  // ---------------------------------------------------------------- validation of beats
  const buildIds = new Set((story.builds ?? []).map((b) => b.id));
  beats.forEach((b) => {
    const where = `line ${b.line + 1}${b.auto ? "" : b.spec.at !== undefined ? ` (cut at "${b.spec.at}")` : ""}`;
    const s = b.spec;
    if (!BEATS.includes(b.kind)) errors.push(`${where}: beat "${b.kind}" does not exist (have: ${BEATS.join(", ")})`);
    if (s.shot && !SHOTS.includes(s.shot)) errors.push(`${where}: shot "${s.shot}" does not exist (have: ${SHOTS.join(", ")})`);
    if (s.time && !TIMES.includes(s.time)) errors.push(`${where}: time "${s.time}" does not exist (have: ${TIMES.join(", ")})`);
    if (s.do && !ACTIONS.includes(s.do as PoseName)) errors.push(`${where}: host action "${s.do}" does not exist (have: ${ACTIONS.join(", ")})`);
    if (s.expr && !EXPRS.includes(s.expr as never)) errors.push(`${where}: host expression "${s.expr}" does not exist (have: ${EXPRS.join(", ")})`);
    for (const th of s.then ?? []) { if (th.do && !ACTIONS.includes(th.do as PoseName)) errors.push(`${where}: then → action "${th.do}" does not exist`); if (th.expr && !EXPRS.includes(th.expr as never)) errors.push(`${where}: then → expression "${th.expr}" does not exist`); }
    for (const [id, a] of Object.entries(s.cast ?? {})) {
      if (!castIds.includes(id)) errors.push(`${where}: "${id}" is not in "cast" (have: ${castIds.join(", ") || "none"})`);
      if (a.do && !ACTIONS.includes(a.do as PoseName)) errors.push(`${where}: ${id} — action "${a.do}" does not exist`);
      if (a.expr && !EXPRS.includes(a.expr as never)) errors.push(`${where}: ${id} — expression "${a.expr}" does not exist`);
    }
    if (b.kind === "comment" && !b.auto) {
      const refs = ([] as (string | number)[]).concat(s.comment ?? 1);
      for (const r of refs) if (!commentOf(r)) errors.push(`${where}: comment "${r}" is not in "comments" (have: ${comments.map((c) => c.id).join(", ") || "none"})`);
    }
    if ((b.kind === "build" || b.kind === "reveal") && !b.auto) {
      if (!s.build && (story.builds?.length ?? 0) !== 1) errors.push(`${where}: give "build": "<id>" (have: ${[...buildIds].join(", ") || "none"})`);
      else if (s.build && !buildIds.has(s.build)) errors.push(`${where}: build "${s.build}" is not in "builds" (have: ${[...buildIds].join(", ") || "none"})`);
    }
    if (b.kind === "gag" && !b.auto) {
      if (!s.gag) errors.push(`${where}: a gag beat needs "gag" (have: ${GAGS.join(", ")})`);
      else if (!GAGS.includes(s.gag)) errors.push(`${where}: gag "${s.gag}" does not exist (have: ${GAGS.join(", ")})`);
      for (const w of s.who ?? []) if (!castIds.includes(w)) errors.push(`${where}: who → "${w}" is not in "cast"`);
      if (!(s.who?.length) && !s.n) errors.push(`${where}: a gag needs people: "who": [cast ids] and/or "n": number of extra citizens`);
    }
  });
  if (errors.length) throw new Error("the episode cannot be built yet:\n  - " + errors.join("\n  - "));

  // ---------------------------------------------------------------- the world at the start
  const st = stateBefore(world, story.day, story.id);
  if (world && st.lastDay && story.day > st.lastDay + 1) warnings.push(`the series has no day ${st.lastDay + 1}…${story.day - 1}: this episode starts from day ${st.lastDay}'s country`);
  const popFrom = Math.max(1, st.population), popTo = story.population;
  if (popTo < popFrom) warnings.push(`population ${popTo} is lower than yesterday's ${popFrom} — the counter will roll down`);
  type LiveB = { id: string; kind: string; def: BuildingDef; pos: V3; yaw: number; plot: string; name: string; by?: string; vars: Record<string, string>; day: number; start?: number; end?: number; isNew: boolean };

    const slotI = SLOTS.findIndex((s) => s.id === b.plot);

    return { ...(def.params ?? {}), ...dflt, ...(b.colors ?? {}), name: b.name, by: b.by ?? "", no: String(b.no ?? 0).padStart(3, "0"), country: country.toUpperCase(), flag1: flag[0], flag2: flag[1] };
  };
  const live: LiveB[] = [];
  for (const b of st.builds) {
    const r = resolveKind(b.kind, custom);
    if (!r) { warnings.push(`world: building "${b.kind}" (day ${b.day}) is unknown here — copy its buildings.json; it is left out`); continue; }

    live.push({ id: `${b.id}@${b.day}`, kind: r.kind, def: r.def, pos: pl.pos, yaw: pl.yaw, plot: b.plot, name: b.name, by: b.by, vars: varsFor(b, r.def), day: b.day, isNew: false });
  }
  for (const b of placed.builds) {
    // a new build on an occupied plot replaces the old one
    const i = live.findIndex((x) => x.plot === b.plot && b.plot !== "road");
    if (i >= 0) live.splice(i, 1);
    const pl = placeOfPlot(b.plot)!;
    live.push({ id: b.id, kind: b.kind, def: b.def, pos: pl.pos, yaw: pl.yaw, plot: b.plot, name: b.name, by: b.by, vars: varsFor(b, b.def), day: story.day, isNew: true });
  }
  const liveOf = (id: string) => live.find((x) => x.id === id) ?? live.find((x) => x.kind === id || x.name.toLowerCase() === id.toLowerCase() || x.plot === id);

  // ---------------------------------------------------------------- runs → overlays, windows
  type Run = { kind: BeatKind; i0: number; i1: number; t0: number; t1: number; spec: BeatSpec };
  const runs: Run[] = [];
  const sameRun = (r: Run, b: B) => {
    if (r.kind !== b.kind || b.kind === "gag" && !b.auto || b.kind === "host" || b.kind === "establish") return b.auto && r.kind === b.kind;
    if (b.auto) return true;
    if (b.kind === "build" || b.kind === "reveal") return (b.spec.build ?? r.spec.build) === r.spec.build;
    if (b.kind === "comment") return b.spec.comment === undefined || JSON.stringify(b.spec.comment) === JSON.stringify(r.spec.comment);
    return b.kind !== "night";
  };
  beats.forEach((b, i) => {
    const t1 = beats[i + 1]?.t0 ?? duration;
    const last = runs[runs.length - 1];
    if (last && sameRun(last, b)) { last.i1 = i; last.t1 = t1; b.sub = i - last.i0; }
    else { runs.push({ kind: b.kind, i0: i, i1: i, t0: b.t0, t1, spec: b.spec }); b.sub = 0; }
  });
  const overlays: POverlay[] = [];
  const counterRun = runs.find((r) => r.kind === "counter");
  const pop = { from: popFrom, to: popTo, t0: counterRun ? counterRun.t0 + 0.15 : 0, t1: counterRun ? Math.max(counterRun.t0 + 0.9, counterRun.t1 - 0.3) : 0 };
  if (!counterRun) pop.from = popTo;
  else if (counterRun.spec.count && counterRun.spec.count !== popTo) warnings.push(`counter beat: "count" ${counterRun.spec.count} differs from "population" ${popTo} — the story's population is used (it is what the series remembers)`);
  const tags: { t0: number; t1: number; text: string }[] = [];
  for (const r of runs) {
    const s = r.spec;
    if (r.kind === "rules") overlays.push({ kind: "rules", t0: r.t0 + 0.12, t1: r.t1 - 0.04, rules: [
      { icon: "follow", small: s.rules?.[0]?.small ?? (voice === "gue" ? "LO FOLLOW" : "KAMU FOLLOW"), big: s.rules?.[0]?.big ?? "= jadi WARGA" },
      { icon: "comment", small: s.rules?.[1]?.small ?? "KOMEN TERATAS", big: s.rules?.[1]?.big ?? (voice === "gue" ? "= gue BANGUN" : "= aku BANGUN") },
    ] });
    if (r.kind === "comment") {
      const refs = ([] as (string | number)[]).concat(s.comment ?? 1).slice(0, 3);
      overlays.push({ kind: "comment", t0: r.t0 + 0.1, t1: r.t1 - 0.04, comments: refs.map((ref, k) => { const c = commentOf(ref)!; return { handle: c.handle, text: c.text, likes: String(fmtNum(c.likes ?? "")), ago: c.ago ?? `${1 + k} jam`, avatar: c.avatar ?? AVATARS[(comments.indexOf(c) + k) % AVATARS.length] }; }) });
    }
    if (r.kind === "build") overlays.push({ kind: "build", t0: r.t0 + 0.1, t1: r.t1 - 0.04, build: s.build ?? story.builds![0].id });
    if (r.kind === "counter") overlays.push({ kind: "bigcounter", t0: r.t0, t1: r.t1 });
    if (r.kind === "cta") overlays.push({ kind: "cta", t0: r.t0 + 0.12, t1: duration, line: s.cta?.line ?? "Besok bangun apa?", button: s.cta?.button ?? "KOMEN!", sub: s.cta?.sub ?? `komen teratas = HARI ${story.day + 1}` });
    if (r.kind === "flyover") overlays.push({ kind: "title", t0: r.t0 + 0.2, t1: r.t1 - 0.05, text: country.toUpperCase(), sub: `HARI ${story.day} · ${fmtNum(popTo)} WARGA` });
    if (r.kind === "night") tags.push({ t0: r.t0 + 0.35, t1: r.t1 - 0.05, text: s.tag ?? ({ malam: "MALAM HARINYA…", pagi: "SEMALAM KEMUDIAN…", siang: "SIANGNYA…", sore: "SORENYA…" } as const)[s.time ?? "pagi"] });
  }
  beats.forEach((b, i) => { if (b.spec.tag && b.kind !== "night") tags.push({ t0: b.t0 + 0.12, t1: (beats[i + 1]?.t0 ?? duration) - 0.05, text: b.spec.tag }); });
  for (const tg of tags) overlays.push({ kind: "tag", ...tg });
  // build windows
  for (const b of live.filter((x) => x.isNew)) {
    const r = runs.find((x) => x.kind === "build" && (x.spec.build ?? story.builds![0].id) === b.id);
    if (r) { b.start = r.t0 + 0.15; b.end = Math.max(b.start + 1.2, r.t1 - 0.35); }
    else warnings.push(`build "${b.id}" has no build beat — it stands from the first shot (add a line with "beat": "build" to show it rising)`);
  }

  // ---------------------------------------------------------------- plates
  const platesRun = runs.find((r) => r.kind === "plates");
  const plates: PPlate[] = [];
  for (const p of st.plates) { const s = SLOTS.find((x) => x.id === p.slot)!; plates.push({ handle: p.handle, no: p.no, slot: p.slot, pos: s.pos, yaw: s.yaw, colors: varsFor({ kind: "rumah", name: "", plot: p.slot, colors: p.colors }, resolveKind("rumah")!.def), appear: -1 }); }
  placed.plates.forEach((p, k) => {
    const s = SLOTS.find((x) => x.id === p.slot)!;
    const appear = counterRun && (!platesRun || counterRun.t0 < platesRun.t0) ? pop.t0 + (pop.t1 - pop.t0) * 0.35 + k * 0.1 : platesRun ? platesRun.t0 : 0;
    plates.push({ handle: p.handle, no: p.no, slot: p.slot, pos: s.pos, yaw: s.yaw, colors: varsFor({ kind: "rumah", name: "", plot: p.slot, colors: p.colors }, resolveKind("rumah")!.def), appear });
  });
  if (platesRun && !placed.plates.length && !live.some((b) => b.isNew && b.kind === "rumah")) warnings.push(`a plates beat but no new "plates" this episode — it shows the newest named houses`);

  // ---------------------------------------------------------------- dwellings over time (camera blockers)
  const popAt = (t: number) => (t <= pop.t0 ? pop.from : t >= pop.t1 ? pop.to : pop.from + (pop.to - pop.from) * ((t - pop.t0) / Math.max(0.01, pop.t1 - pop.t0)));
  const up = upgradeFor(pop.from);

  // ---------------------------------------------------------------- actors
  const D2R = Math.PI / 180;
  type AS = { pos?: V3; yaw: number; pose: PoseName; expr: string; target?: V3; prop?: "phone" | "bowl"; chatter?: boolean; gone: boolean; seat?: string };
  const host: AS = { pos: MAP.home.pos, yaw: MAP.home.yaw, pose: "talk", expr: "happy", gone: false };
  const W: Record<string, AS> = Object.fromEntries(castIds.map((id) => [id, { yaw: 0, pose: "stand", expr: "neutral", gone: false } as AS]));
  const extras: { id: string; spec: ReturnType<typeof extraLook> }[] = [];
  const keys: Record<string, AKey[]> = { host: [] };
  const props: PProp[] = [];
  const frontSpot = (b: LiveB, side = 1, out = 1.5): { pos: V3; yaw: number } => {
    const sp = b.def.special;
    const local: [number, number, number] =
      sp === "road" ? [-2.9 * side, 6.5 - b.pos[2] + 0 * out, -0.9] :
      sp === "bridge" ? [2.4 * side, 9.6, 0.25] :
      sp === "field" ? [-4 + 1.5 * side, 6, 0.2] :
      sp === "station" ? [8 + 1.2 * side, 2.6, 0.2] :
      sp === "gate" ? [3.6 * side, 3.2, 0.3] :
      [Math.min(b.def.size[0] / 2 - 0.3, 1.7) * side, b.def.size[1] / 2 + out, 0.35];
    const pos = toWorld([local[0], 0, local[1]], b.pos, b.yaw);
    return { pos, yaw: b.yaw + local[2] * side };
  };
  const placeHost = (name: string | undefined, where: string): { pos: V3; yaw: number } | null => {
    if (!name || name === "plaza" || name === "home" || name === "rumah-presiden") return { pos: MAP.home.pos, yaw: MAP.home.yaw };
    if (name === "lawn" || name === "lapangan-depan") return { pos: [0, 0, 12.5], yaw: 0 };
    if (name === "jalan") return { pos: [10.75, 0, 4], yaw: 0 };
    if (castIds.includes(name) && W[name].pos) { const p = W[name].pos!; return { pos: add3(p, rightOf(W[name].yaw), -1.1), yaw: W[name].yaw - 0.4 }; }
    const b = liveOf(name);
    if (b) return frontSpot(b, 1);
    warnings.push(`${where}: place "${name}" is not a building, a citizen or plaza / lawn / jalan — the host stays where they are`);
    return null;
  };
  const pushKey = (id: string, s: AS, t: number) => {
    const list = (keys[id] ??= []);
    if (s.gone) { if (list.length && !list[list.length - 1].gone) list.push({ t, pos: s.pos ?? [0, 0, 0], yaw: s.yaw, pose: s.pose, expr: s.expr as AKey["expr"], gone: true }); return; }
    if (!s.pos) return;
    list.push({ t, pos: s.pos, yaw: s.yaw, pose: s.pose, expr: s.expr as AKey["expr"], target: s.target, prop: s.prop, chatter: s.chatter });
  };
  /** a point (person / building / place) in an actor's own frame, for point / present / argue */
  const targetLocal = (from: AS, to: string | undefined): V3 | undefined => {
    if (!to || !from.pos) return undefined;
    let p: V3 | undefined;
    if (to === "host") p = host.pos ? add3(host.pos, [0, 0.9, 0]) : undefined;
    else if (W[to]?.pos) p = add3(W[to].pos!, [0, 0.9, 0]);
    else { const b = liveOf(to); if (b) p = add3(b.pos, [0, Math.min(2.5, b.def.h * 0.5), 0]); }
    if (!p) return undefined;
    const x = p[0] - from.pos[0], y = p[1] - from.pos[1], z = p[2] - from.pos[2], c = Math.cos(from.yaw), s = Math.sin(from.yaw);
    return [x * c - z * s, y, x * s + z * c];
  };
  const HOST_DEFAULT: Record<BeatKind, { do: PoseName; expr: string }> = {
    host: { do: "talk", expr: "happy" }, establish: { do: "wave", expr: "happy" }, rules: { do: "pointCam", expr: "happy" }, counter: { do: "shocked", expr: "shocked" },
    comment: { do: "read", expr: "smug" }, build: { do: "proud", expr: "proud" }, reveal: { do: "present", expr: "happy" }, plates: { do: "talk", expr: "happy" },
    gag: { do: "facepalm", expr: "annoyed" }, night: { do: "stand", expr: "neutral" }, cta: { do: "wave", expr: "happy" }, flyover: { do: "proud", expr: "proud" },
  };

  // gag layout: people around a building (seats, a queue at its front, a crowd) or on a mat on the plaza
  const gagLayout = (gag: GagKind, ids: string[], placeName: string | undefined, t: number) => {
    const b = placeName && placeName !== "plaza" && placeName !== "lawn" ? liveOf(placeName) : undefined;
    if (placeName && !b && placeName !== "plaza" && placeName !== "lawn") warnings.push(`gag: place "${placeName}" is not a standing building — the gag happens on the plaza`);
    const anchor = b ? { pos: b.pos, yaw: b.yaw } : placeName === "lawn" ? { pos: [-1.5, 0, 13.5] as V3, yaw: 0.15 } : { pos: [-2.6, 0, 4.4] as V3, yaw: 0.25 };
    const seats = b?.def.seats ?? [];
    const needSeat = gag === "sleep" || gag === "sit" || gag === "chat" || gag === "eat";
    const frontL: [number, number] = b ? (b.def.front ?? [0, b.def.size[1] / 2 + 1.2]) : [0, 0];
    const at = (x: number, z: number): V3 => toWorld([x, 0, z], anchor.pos, anchor.yaw);
    const res: { id: string; pos: V3; yaw: number; pose: PoseName; expr: string; prop?: "phone" | "bowl"; chatter?: boolean; target?: string }[] = [];
    let mat = false;
    ids.forEach((id, k) => {
      if (needSeat && b && seats[k]) {
        const s = seats[k], yaw = anchor.yaw + s.yaw * D2R, hang = s.legs === "hang";

      if (bb?.def.special === "station") (subj as { c: V3 }).c = toWorld([0, 0, -3.5], bb.pos, bb.yaw);
      if (bb?.def.special === "field") { Object.assign(subj, { c: toWorld([-6, 0, -1], bb.pos, bb.yaw), w: 34, d: 14, h: 6 }); }
      const skip = new Set([subj.id, ...sp.on]);
      sol = solveBuilding(subj as Extract<Subj, { type: "building" }>, envAt(tm, skip, (subj as { c: V3 }).c), o, sp.subjKind === "area" ? 8 : 12, []);
    } else if (sp.subjKind === "sign") {
      const bb = liveOf(sp.on[0])!;
      const pcs = expand(bb.def, bb.vars);
      const sg = pcs.find((p) => p.shape === "sign" && /\{(name|country|by)\}/.test(p.tpl ?? "")) ?? pcs.find((p) => p.shape === "sign");
      if (!sg || sg.shape !== "sign") { sol = solveBuilding({ id: bb.id, type: "building", c: bb.pos, yaw: bb.yaw, w: bb.def.size[0], d: bb.def.size[1], h: bb.def.h }, envAt(tm, new Set([bb.id]), bb.pos), o); }
      else {
        const ry = (sg.rot?.[1] ?? 0) * D2R;
        const c = toWorld([sg.pos[0] + Math.sin(ry) * sg.size[2] / 2, sg.pos[1] + sg.size[1] / 2, sg.pos[2] + Math.cos(ry) * sg.size[2] / 2], bb.pos, bb.yaw);
        sol = solveSign({ id: bb.id, type: "sign", c, yaw: bb.yaw + ry, w: sg.size[0], h: sg.size[1] }, envAt(tm, new Set([bb.id]), c), o);
      }
    } else if (sp.subjKind === "plate") {
      const handle = sp.on[0].slice(6), sh = showPlates.find((p) => p.handle === handle)!, slot = SLOTS.find((x) => x.id === sh.slot)!;
      const c = toWorld([0.67 + Math.sin(54 * D2R) * 0.03, 0.95 + 0.23, 2.65 + Math.cos(54 * D2R) * 0.03], slot.pos, slot.yaw);
      sol = solveSign({ id: sp.on[0], type: "sign", c, yaw: slot.yaw + 54 * D2R, w: 1.08, h: 0.46 }, envAt(tm, new Set(sp.on), c), { ...o, move: b.spec.move ?? "push" });
    } else if (sp.subjKind === "aerial") {
      // over the growing settlement
      const n0 = dwellingsFor(pop.from), n1 = Math.max(n0 + 1, dwellingsFor(pop.to));
      const pick = LOTS.slice(n1 - n0 > 30 ? n0 : 0, n1);
      const cx = pick.reduce((s, l) => s + l.pos[0], 0) / Math.max(1, pick.length), cz = pick.reduce((s, l) => s + l.pos[2], 0) / Math.max(1, pick.length);
      const ds = pick.map((l) => Math.hypot(l.pos[0] - cx, l.pos[2] - cz)).sort((a, c) => a - c);
      const R = Math.max(14, ds[Math.floor(ds.length * 0.85)] ?? 14);
      const c: V3 = [cx * 0.7, 0, cz * 0.75];
      const az = -0.35 + (b.sub % 2) * 0.5;
      const at = (k: number, el: number): V3 => [c[0] + Math.sin(az) * Math.cos(el) * R * k, R * k * Math.sin(el), c[2] + Math.cos(az) * Math.cos(el) * R * k];
      const fr = frame;
      const posA = at(wide ? 1.0 : 1.25, 0.5), posB = at(wide ? 1.45 : 1.8, 0.66);
      const tg = (p: V3): V3 => { const d = Math.hypot(p[0] - c[0], p[1] - c[1], p[2] - c[2]), tv = Math.tan((42 * D2R) / 2); return [c[0], c[1] + d * tv * (2 * fr.y - 1) * 0.9, c[2]]; };
      sol = { a: { pos: posA, target: tg(posA), fov: 42 }, b: { pos: posB, target: tg(posB), fov: 42 }, vis: 1, az, box: Math.min(70, R * 1.8) };
    } else if (sp.subjKind === "flyover") {
      const pts = [...live.map((x) => x.pos), ...LOTS.slice(0, dwellingsFor(popTo)).map((l) => l.pos), [0, 0, 0] as V3];
      const cx = pts.reduce((s, p) => s + p[0], 0) / pts.length, cz = pts.reduce((s, p) => s + p[2], 0) / pts.length;
      const R = Math.max(26, Math.max(...pts.map((p) => Math.hypot(p[0] - cx, p[2] - cz))) * 0.85);
      const c: V3 = [cx, 0, cz], fov = 42;
      const az0 = -0.55, az1 = -0.05, r = R * (wide ? 1.15 : 1.5), h = R * 0.85;
      const P = (az: number): V3 => [c[0] + Math.sin(az) * r, h, c[2] + Math.cos(az) * r];
      const tv = Math.tan((fov * D2R) / 2), dd = Math.hypot(r, h);
      const tgt: V3 = [c[0], dd * tv * (2 * frame.y - 1) * 0.8, c[2]];
      sol = { a: { pos: P(az0), target: tgt, fov }, b: { pos: P(az1), target: tgt, fov }, vis: 1, az: az0, box: Math.min(90, R * 1.6), orbit: { c, r, h, az0, az1, fov, target: [0, 0, 0] } };
    } else {
      // plates: a constant-speed truck past the newest named houses, each plate crossing the centre
      const newest = plates.filter((p) => p.appear >= 0);
      const row = (newest.length ? newest : plates.slice(-3));
      const first = row[0] ?? { pos: SLOTS[0].pos, yaw: SLOTS[0].yaw };
      const sameRow = row.filter((p) => p.yaw === first.yaw);
      const plate = (p: { pos: V3; yaw: number }) => toWorld([0.67, 1.18, 2.65], p.pos, p.yaw);
      const nl: V3 = [Math.sin(54 * D2R), 0, Math.cos(54 * D2R)];
      const nW: V3 = [nl[0] * Math.cos(first.yaw) + nl[2] * Math.sin(first.yaw), 0, nl[2] * Math.cos(first.yaw) - nl[0] * Math.sin(first.yaw)];
      const look: V3 = [-nW[0], 0, -nW[2]];
      const ordered = [...sameRow].sort((p, q) => (plate(p)[0] - plate(q)[0]) * look[0] + (plate(p)[2] - plate(q)[2]) * look[2]);
      const pA = plate(ordered[0] ?? first), pB = plate(ordered[ordered.length - 1] ?? first);
      const dist = wide ? 4.2 : 3.4;
      const camA = add3(add3(pA, nW, dist), [0, 0.12, 0]);
      const travel = Math.hypot(pB[0] - pA[0], pB[2] - pA[2]);
      const dir: V3 = travel > 0.1 ? [(pB[0] - pA[0]) / travel, 0, (pB[2] - pA[2]) / travel] : [-look[2], 0, look[0]];
      const span = Math.max(travel + 1.6, (t1 - t0) * 1.4);
      const startPos = add3(camA, dir, -0.8), endPos = add3(startPos, dir, span);
      const tA = add3(startPos, look, dist), tB = add3(endPos, look, dist);
      tA[1] -= 0.12; tB[1] -= 0.12;
      sol = { a: { pos: startPos, target: tA, fov: 38 }, b: { pos: endPos, target: tB, fov: 38 }, vis: 1, az: 0, box: 12 };
      linear = true;
      void sameRow;
    }
    if (sol.note || sol.vis < 0.75) warnings.push(`${where}: the camera could not see clearly${sol.note ? ` (${sol.note})` : ""} — move people apart, pick another place or shot`);
    prevAz = sol.az; prevId = sp.on[0];
    const tgt = sol.a.target;
    shots.push({
      i: shots.length, t0: +t0.toFixed(3), t1: +t1.toFixed(3), line: b.line, beat: b.kind, kind: sp.kind, a: sol.a, b: sol.b, linear, orbit: sol.orbit,
      shadowAt: [tgt[0], 0, tgt[2]], shadowBox: sol.box, time: sp.time, dissolve: sp.dissolve, vis: +sol.vis.toFixed(2), on: sp.on,
    });
  }

  // ---------------------------------------------------------------- builds for the plan
  const pb: PBuild[] = live.map((b) => ({ id: b.id, kind: b.kind, name: b.name, by: b.by, plot: b.plot, pos: b.pos, yaw: b.yaw, vars: b.vars, start: b.start, end: b.end, day: b.day }));
  for (const b of live.filter((x) => x.isNew && x.kind === "rumah")) {
    const slot = SLOTS.find((s) => s.id === b.plot)!;
    plates.push({ handle: b.by ?? "@warga", no: Number(b.vars.no), slot: b.plot, pos: slot.pos, yaw: slot.yaw, colors: b.vars, appear: b.start ?? 0, byBuild: b.id });
  }
  const customUsed = Object.fromEntries(Object.entries(custom).filter(([k]) => live.some((b) => b.kind === k)));
  return {
    id: story.id, title: story.title ?? `Hari ${story.day}`, day: story.day, country, flag, voice, duration, aspect: opts.aspect, pop,
    builds: pb, plates, props, custom: customUsed, actors, shots, overlays, words, lines, audio: opts.audio, warnings,
  };
};
