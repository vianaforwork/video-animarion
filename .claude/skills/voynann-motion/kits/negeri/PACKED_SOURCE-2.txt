// NEGERI sound from a Plan — original synth, no samples, no meme sounds.
// Music: a bright "city builder" bed, 112 bpm in C (I–V–vi–IV): plucked bass, soft kick + finger snap, a marimba
// melody on the pentatonic, a bamboo-ish shaker, a light pad; hi-hats join while something is being built; night
// shots keep only the pad + crickets; a landing chord at the end.
// SFX follow the plan: a soft air move on cuts, a "bloop" when a card pops in, a two-tone notification for comments,
// a riser while the counter rolls + a ding when it lands, wooden knocks per build stage + a little fanfare when it is
// done, ticks as name plates pass, a snore / claps / bonks for gags, a whoosh on time jumps, a chime on the CTA.
// Mix: −14 LUFS / −1.5 dBTP; the music ducks under the voice.
import fs from "node:fs";
import path from "node:path";
import { execFileSync, spawnSync } from "node:child_process";

export const buildSound = (plan, { voice, outDir, mixFile, music = true }) => {
  const DUR = plan.duration, SR = 48000, N = Math.ceil(DUR * SR);
  const mk = () => [new Float32Array(N), new Float32Array(N)];
  let seed = 1607;
  const rnd = () => (seed = (seed * 16807) % 2147483647) / 2147483647;
  const add = (buf, t, pan, len, fn) => {
    const i0 = Math.max(0, Math.round(t * SR)), n = Math.min(N - i0, Math.round(len * SR));
    const gl = Math.cos(((pan + 1) * Math.PI) / 4), gr = Math.sin(((pan + 1) * Math.PI) / 4);
    for (let i = 0; i < n; i++) { const v = fn(i / SR, i); buf[0][i0 + i] += v * gl; buf[1][i0 + i] += v * gr; }
  };
  const hz = (m) => 440 * Math.pow(2, (m - 69) / 12);
  const shotAt = (t) => plan.shots.find((s) => t >= s.t0 && t < s.t1) ?? plan.shots[plan.shots.length - 1];
  const nightAt = (t) => shotAt(t).time === "malam";
  const building = (t) => plan.builds.some((b) => b.start !== undefined && t >= b.start && t <= b.end);

  // ---------------------------------------------------------------- music
  const mus = mk();
  if (music) {
    const beat = 60 / 112, bar = beat * 4;
    const CH = [[48, 52, 55, 60], [43, 47, 50, 55], [45, 48, 52, 57], [41, 45, 48, 53]];
    const PENTA = [72, 74, 76, 79, 81, 84, 86];
    const fade = (t) => Math.min(1, Math.max(0, (DUR - 0.35 - t) / 1.4)) * Math.min(1, t / 0.4);
    const bars = Math.ceil(DUR / bar);
    for (let b = 0; b < bars; b++) {
      const c = CH[b % 4], t0 = b * bar;
      const night = nightAt(t0 + 0.1);
      const full = night ? 0 : 1;
      // pluck bass: 1, 2&, 3, 4&
      if (!night) for (const [bt, m] of [[0, c[0] - 12], [1.5, c[0] - 12], [2, c[0] - 5], [3.5, c[0] - 10]]) {
        const f = hz(m);
        add(mus, t0 + bt * beat, -0.05, 0.4, (t) => (Math.sin(2 * Math.PI * f * t) + 0.4 * Math.sin(4 * Math.PI * f * t)) * 0.08 * Math.min(1, t / 0.004) * Math.exp(-t * 8) * fade(t0 + t));
      }
      // soft kick on 1 and 3, snap on 2 and 4
      for (const bt of [0, 2]) if (full) add(mus, t0 + bt * beat, 0, 0.25, (t) => Math.sin(2 * Math.PI * (48 + 70 * Math.exp(-t * 30)) * t) * Math.exp(-t * 16) * 0.11 * fade(t0 + t));
      for (const bt of [1, 3]) if (full) { let lp = 0; add(mus, t0 + bt * beat, 0.15, 0.09, (t) => { const n = rnd() * 2 - 1; lp += (n - lp) * 0.55; return (n - lp) * 0.07 * Math.exp(-t * 55) * fade(t0 + t); }); }
      // marimba melody (8ths, with rests), sparser at night
      for (let e = 0; e < 8; e++) {
        if (rnd() > (night ? 0.18 : 0.62)) continue;
        const pool = PENTA.filter((p) => c.some((x) => (p - x) % 12 === 0) || rnd() < 0.3);
        const m = pool[Math.floor(rnd() * pool.length)] ?? 76;
        const f = hz(m), vel = (e % 4 === 0 ? 0.055 : 0.04) * (0.8 + rnd() * 0.35), pan = (rnd() - 0.5) * 0.7;
        add(mus, t0 + (e * beat) / 2 + (rnd() - 0.5) * 0.008, pan, 0.7, (t) => (Math.sin(2 * Math.PI * f * t) * Math.exp(-t * 9) + 0.3 * Math.sin(2 * Math.PI * f * 3.95 * t) * Math.exp(-t * 34)) * vel * Math.min(1, t / 0.002) * fade(t0 + t));
      }
      // pad
      for (const [k, m] of c.entries()) add(mus, t0, (k - 1.5) * 0.25, bar + 0.4, (t) => (Math.sin(2 * Math.PI * hz(m + 12) * t) + 0.3 * Math.sin(2 * Math.PI * hz(m + 12) * 1.003 * t)) * 0.0075 * Math.min(1, t / 0.6) * Math.min(1, (bar + 0.4 - t) / 0.5) * fade(t0 + t));
      // bamboo shaker on the off-beats; hi-hats while building
      for (let e = 1; e < 8; e += 2) if (full) { let lp = 0; add(mus, t0 + (e * beat) / 2, 0.35, 0.07, (t) => { const n = rnd() * 2 - 1; lp += (n - lp) * 0.75; return (n - lp) * 0.02 * Math.exp(-t * 60) * fade(t0 + t); }); }
      for (let e = 0; e < 16; e++) { const tt = t0 + (e * beat) / 4; if (!building(tt)) continue; let hp = 0, prev = 0; add(mus, tt, -0.3, 0.04, (t) => { const n = rnd() * 2 - 1; hp = 0.9 * (hp + n - prev); prev = n; return hp * 0.012 * (e % 2 ? 0.6 : 1) * Math.exp(-t * 90); }); }
    }
    const endT = Math.max(0, DUR - 1.9);
    for (const m of [48, 55, 60, 64, 67, 72]) add(mus, endT, 0, 1.9, (t) => Math.sin(2 * Math.PI * hz(m) * t) * 0.026 * Math.min(1, t / 0.02) * Math.exp(-t * 1.6));
  }

  // ---------------------------------------------------------------- SFX
  const sfx = mk();
  const events = [];
  const ev = (t, what) => events.push(`${t.toFixed(2)} ${what}`);
  const noise = (t, len, gain, lpK, hpK, env, pan = 0) => { let lp = 0, hp = 0; add(sfx, t, pan, len, (x) => { const n = rnd() * 2 - 1; lp += (n - lp) * lpK; hp += (lp - hp) * hpK; return (lp - hp) * gain * env(x); }); };
  const air = (t) => noise(t - 0.1, 0.26, 0.05, 0.05, 0.002, (x) => Math.sin(Math.PI * Math.min(1, x / 0.26)) ** 2);
  const whoosh = (t) => noise(t - 0.2, 0.8, 0.13, 0.08, 0.01, (x) => Math.sin(Math.PI * Math.min(1, x / 0.8)) ** 2, 0.2);
  const bloop = (t, m = 76) => add(sfx, t, 0.1, 0.16, (x) => Math.sin(2 * Math.PI * hz(m) * x * (1 + 0.6 * x / 0.16)) * 0.05 * Math.min(1, x / 0.004) * Math.exp(-x * 22));
  const notif = (t) => { for (const [k, m] of [[0, 84], [1, 91]]) add(sfx, t + k * 0.09, 0.2, 0.5, (x) => Math.sin(2 * Math.PI * hz(m) * x) * 0.04 * Math.min(1, x / 0.003) * Math.exp(-x * 9)); };
  const ding = (t, m = 88) => { for (const [k, mm] of [m, m + 7, m + 12].entries()) add(sfx, t + k * 0.035, 0.15, 1.1, (x) => Math.sin(2 * Math.PI * hz(mm) * x) * (0.035 - k * 0.008) * Math.min(1, x / 0.003) * Math.exp(-x * 4.5)); };
  const knock = (t, f = 190, g = 0.12, pan = 0) => add(sfx, t, pan, 0.3, (x) => (Math.sin(2 * Math.PI * f * x * (1 + 0.6 * Math.exp(-x * 60))) * Math.exp(-x * 24) + (rnd() * 2 - 1) * 0.25 * Math.exp(-x * 140)) * g * Math.min(1, x / 0.002));
  const fanfare = (t) => { for (const [k, m] of [[0, 72], [1, 76], [2, 79], [3, 84]]) add(sfx, t + k * 0.08, 0.1, 0.6, (x) => (Math.sin(2 * Math.PI * hz(m) * x) + 0.25 * Math.sin(4 * Math.PI * hz(m) * x)) * 0.03 * Math.min(1, x / 0.004) * Math.exp(-x * 5)); };
  const tick = (t) => add(sfx, t, -0.1, 0.12, (x) => Math.sin(2 * Math.PI * 1900 * x) * Math.exp(-x * 60) * 0.035);
  const riser = (t, len) => { let lp = 0; add(sfx, t, 0, len, (x) => { const n = rnd() * 2 - 1; lp += (n - lp) * (0.02 + 0.2 * (x / len)); const f = 300 + 900 * (x / len) ** 2; return (lp * 0.06 + Math.sin(2 * Math.PI * f * x) * 0.012) * Math.min(1, x / 0.2) * Math.min(1, (len - x) / 0.08); }); };
  const snore = (t) => { for (let k = 0; k < 2; k++) noise(t + k * 1.1, 0.7, 0.07, 0.03, 0.01, (x) => Math.sin(Math.PI * x / 0.7) ** 2 * (0.6 + 0.4 * Math.sin(x * 2 * Math.PI * 28)), -0.2); };
  const claps = (t, n) => { for (let k = 0; k < n; k++) { let lp = 0; add(sfx, t + k * 0.21 + rnd() * 0.04, (rnd() - 0.5) * 0.8, 0.08, (x) => { const z = rnd() * 2 - 1; lp += (z - lp) * 0.5; return (z - lp) * 0.05 * Math.exp(-x * 50); }); } };
  const bonk = (t, m) => add(sfx, t, 0, 0.25, (x) => Math.sin(2 * Math.PI * hz(m) * x * (1 - 0.3 * x)) * Math.exp(-x * 14) * 0.06);
  const chime = (t) => { for (const [k, m] of [[0, 79], [1, 86]]) add(sfx, t + k * 0.12, 0, 1.2, (x) => Math.sin(2 * Math.PI * hz(m) * x) * 0.035 * Math.min(1, x / 0.004) * Math.exp(-x * 3.5)); };
  const chirp = (t) => { for (let k = 0; k < 3; k++) add(sfx, t + k * 0.13, 0.4, 0.09, (x) => Math.sin(2 * Math.PI * (3200 + 1800 * Math.sin(Math.PI * x / 0.09)) * x) * 0.012 * Math.sin(Math.PI * x / 0.09)); };

  for (const o of plan.overlays) {
    if (o.kind === "rules") { o.rules.forEach((_, k) => { bloop(o.t0 + k * 0.3, 74 + k * 3); ev(o.t0 + k * 0.3, "card"); }); }
    if (o.kind === "comment") { notif(o.t0 + 0.02); ev(o.t0, "comment"); o.comments.slice(1).forEach((_, k) => { bloop(o.t0 + 0.22 * (k + 1), 72); ev(o.t0, "card"); }); }
    if (o.kind === "build" || o.kind === "tag" || o.kind === "title") { bloop(o.t0, o.kind === "tag" ? 79 : 74); ev(o.t0, `${o.kind} card`); }
    if (o.kind === "cta") { bloop(o.t0, 76); chime(o.t0 + 0.32); ev(o.t0, "cta"); }
  }
  if (plan.pop.t1 > plan.pop.t0) { riser(plan.pop.t0, plan.pop.t1 - plan.pop.t0); ding(plan.pop.t1, 91); ev(plan.pop.t0, "counter riser"); ev(plan.pop.t1, "ding"); }
  for (const b of plan.builds) {
    if (b.start === undefined) continue;
    const len = b.end - b.start, n = Math.max(2, Math.min(5, Math.round(len / 0.9)));
    for (let k = 0; k < n; k++) { knock(b.start + 0.12 + (len * k) / n, 160 + (k % 3) * 30, 0.11, 0.3 - k * 0.1); ev(b.start + (len * k) / n, "knock"); }
    fanfare(b.end + 0.05); ev(b.end, "fanfare");
  }
  for (const S of plan.shots) if (S.beat === "plates" && S.kind === "truck") {
    const n = Math.max(1, Math.min(5, plan.plates.filter((p) => p.appear >= 0).length || 3));
    for (let k = 0; k < n; k++) { tick(S.t0 + ((S.t1 - S.t0) * (k + 0.6)) / (n + 0.4)); ev(S.t0, "plate tick"); }
  }
  // gags: what the citizens do in the first shot of each gag
  const seenGag = new Set();
  for (const S of plan.shots) {
    if (S.beat !== "gag") continue;
    const people = plan.actors.filter((a) => !a.host && a.keys.some((k) => Math.abs(k.t - S.t0) < 0.02));
    if (!people.length || seenGag.has(S.line)) continue;
    seenGag.add(S.line);
    const poses = people.map((a) => a.keys.find((k) => Math.abs(k.t - S.t0) < 0.02).pose);
    if (poses.some((p) => p.startsWith("sleep"))) { snore(S.t0 + 0.35); ev(S.t0, "snore"); }
    if (poses.some((p) => p === "cheer" || p === "clap")) { claps(S.t0 + 0.2, 6); fanfare(S.t0 + 0.1); ev(S.t0, "claps"); }
    if (poses.some((p) => p === "argue")) { bonk(S.t0 + 0.3, 62); bonk(S.t0 + 0.75, 58); ev(S.t0, "bonks"); }
  }
  // cuts: a whoosh on time jumps, a soft air move on the others unless another cue already marks the cut
  const cueT = events.map((e) => parseFloat(e));
  plan.shots.forEach((S, i) => {
    if (!i) return;
    if (S.dissolve) { whoosh(S.t0); ev(S.t0, "whoosh"); }
    else if (!cueT.some((t) => Math.abs(t - S.t0) < 0.35)) { air(S.t0); ev(S.t0, "air"); }
  });
  // night: crickets while it is night; morning after a time jump: a few bird chirps
  const nightOn = new Float32Array(N);
  { let on = 0; const step = 1 / (0.6 * SR); for (let i = 0; i < N; i += 1) { const want = nightAt(i / SR) ? 1 : 0; on += Math.max(-step, Math.min(step, want - on)); nightOn[i] = on; } }
  if (nightOn.some((v) => v > 0)) {
    ev(0, "crickets");
    for (let t = 0; t < DUR; t += 0.55 + rnd() * 0.4) {
      const i = Math.round(t * SR);
      if (!nightOn[i]) continue;
      const f = 4300 + rnd() * 500, g = 0.009 * nightOn[i];
      add(sfx, t, (rnd() - 0.5) * 1.2, 0.18, (x) => Math.sin(2 * Math.PI * f * x) * (0.5 + 0.5 * Math.sin(2 * Math.PI * 38 * x)) * g * Math.sin(Math.PI * x / 0.18));
    }
  }
  for (const S of plan.shots) if (S.beat === "night" && S.dissolve && S.time === "pagi") { chirp(S.t0 + 0.6); chirp(S.t0 + 1.3); ev(S.t0, "birds"); }

  // ---------------------------------------------------------------- write + mix
  const wav = (file, buf) => {
    const data = Buffer.alloc(N * 4);
    for (let i = 0; i < N; i++) for (let c = 0; c < 2; c++) data.writeInt16LE(Math.round(Math.max(-1, Math.min(1, buf[c][i])) * 32767), i * 4 + c * 2);
    const h = Buffer.alloc(44);
    h.write("RIFF", 0); h.writeUInt32LE(36 + data.length, 4); h.write("WAVEfmt ", 8); h.writeUInt32LE(16, 16); h.writeUInt16LE(1, 20); h.writeUInt16LE(2, 22);
    h.writeUInt32LE(SR, 24); h.writeUInt32LE(SR * 4, 28); h.writeUInt16LE(4, 32); h.writeUInt16LE(16, 34); h.write("data", 36); h.writeUInt32LE(data.length, 40);
    fs.writeFileSync(file, Buffer.concat([h, data]));
  };
  fs.mkdirSync(outDir, { recursive: true });
  fs.mkdirSync(path.dirname(mixFile), { recursive: true });
  const mf = path.join(outDir, "music.wav"), sf = path.join(outDir, "sfx.wav");
  wav(mf, mus);
  wav(sf, sfx);
  const tail = `apad=whole_dur=${DUR},atrim=0:${DUR},loudnorm=I=-14:TP=-1.5:LRA=11,aresample=48000[out]`;
  const filter = voice
    ? ["[0:a]highpass=f=70,acompressor=threshold=-20dB:ratio=2.5:attack=8:release=140:makeup=3,asplit=2[v][vk]", "[1:a]volume=1.0[m]", "[m][vk]sidechaincompress=threshold=0.03:ratio=6:attack=25:release=420[md]", "[2:a]volume=0.9[s]", `[v][md][s]amix=inputs=3:normalize=0:duration=longest,${tail}`].join(";")
    : ["[0:a]volume=1.4[m]", "[1:a]volume=1.0[s]", `[m][s]amix=inputs=2:normalize=0:duration=longest,${tail}`].join(";");
  execFileSync("ffmpeg", ["-y", "-v", "error", ...(voice ? ["-i", voice] : []), "-i", mf, "-i", sf, "-filter_complex", filter, "-map", "[out]", "-ac", "2", mixFile]);
  // one-pass loudnorm can land short of −14 when peaks limit it (no voice, punchy SFX): a measured second pass fixes it
  const measure = () => spawnSync("ffmpeg", ["-hide_banner", "-i", mixFile, "-af", "loudnorm=I=-14:TP=-1.5:LRA=11:print_format=json", "-f", "null", "-"], { encoding: "utf8" }).stderr;
  const j1 = JSON.parse((measure().match(/\{[\s\S]*\}/) ?? ["{}"])[0]);
  if (j1.input_i && Math.abs(parseFloat(j1.input_i) + 14) > 0.4) {
    const tmp = mixFile.replace(/\.wav$/, ".pass1.wav");
    fs.renameSync(mixFile, tmp);
    const ln = `loudnorm=I=-14:TP=-1.5:LRA=11:measured_I=${j1.input_i}:measured_TP=${j1.input_tp}:measured_LRA=${j1.input_lra}:measured_thresh=${j1.input_thresh}:offset=${j1.target_offset}:linear=true`;
    execFileSync("ffmpeg", ["-y", "-v", "error", "-i", tmp, "-af", `${ln},alimiter=limit=0.84:level=false,aresample=48000`, "-ac", "2", mixFile]);
    fs.rmSync(tmp, { force: true });
  }
  const m = spawnSync("ffmpeg", ["-hide_banner", "-i", mixFile, "-af", "loudnorm=print_format=summary", "-f", "null", "-"], { encoding: "utf8" }).stderr;
  return { lufs: (m.match(/Input Integrated:\s+(-?[\d.]+)/) ?? [])[1], peak: (m.match(/Input True Peak:\s+(-?[\d.]+)/) ?? [])[1], events: events.length, perMin: Math.round((events.length / DUR) * 60) };
};

  torsoCache.set(key, g);
  return g;
};

// ------------------------------------------------------------------ hair shell (head-local: x left of figure = +x, y up, z front)
const D2R = Math.PI / 180;
type Edge = { top: (lon: number) => number; bottom: (lon: number) => number };
const lerpT = (a: number, b: number, t: number) => a + (b - a) * Math.min(1, Math.max(0, t));
const edges = (style: HairStyle): Edge | null => {
  const back = (front: number, side: number, rear: number) => (lon: number) => {
    const a = Math.abs(lon);
    return a < 35 ? front : a < 100 ? lerpT(front, side, (a - 35) / 65) : lerpT(side, rear, (a - 100) / 80);
  };
  switch (style) {
    case "bald": return { top: (lon) => { const a = Math.abs(lon); return a < 80 ? -24 : lerpT(-24, 8, (a - 80) / 30); }, bottom: () => -30 };
    case "short": return { top: () => 90, bottom: back(36, 2, -36) };
    case "spiky": return { top: () => 90, bottom: back(34, 4, -34) };
    case "fringe": return { top: () => 90, bottom: (lon) => { const a = Math.abs(lon); const base = back(25, 0, -38)(lon); return a < 60 ? base + Math.abs(((lon / 9) % 2) - 1) * 6 - 2 : base; } };
    case "bun": return { top: () => 90, bottom: back(34, -6, -40) };
    case "long": return { top: () => 90, bottom: back(34, -48, -72) };
  }
};
const hairCache = new Map<string, BufferGeometry>();
const hairGeo = (style: HairStyle, R: number) => {
  const key = style + R.toFixed(4);
  const hit = hairCache.get(key);
  if (hit) return hit;
  const e = edges(style)!;
  const NL = 96, NV = 14;
  const pos: number[] = [];
  const idx: number[] = [];
  const at = (lonDeg: number, latDeg: number, k: number) => {
    const lo = lonDeg * D2R, la = latDeg * D2R;
    return [Math.sin(lo) * Math.cos(la) * k, Math.sin(la) * k, Math.cos(lo) * Math.cos(la) * k];
  };
  const rows = NV + 2; // outer rows + the inward lip
  for (let i = 0; i <= NL; i++) {
    const lon = -180 + (360 * i) / NL;
    const top = e.top(lon), bot = e.bottom(lon);
    for (let j = 0; j <= NV; j++) {
      const lat = top + ((bot - top) * j) / NV;
      const puff = 1.045 + 0.025 * (1 - j / NV);
      pos.push(...at(lon, lat, R * puff));
    }
    // lip: tuck the edge back to the scalp so the shell has thickness and no see-through rim
    pos.push(...at(lon, bot, R * 0.995));
  }
  for (let i = 0; i < NL; i++) for (let j = 0; j < rows - 1; j++) {
    const a = i * rows + j, b = (i + 1) * rows + j;
    idx.push(a, a + 1, b, b, a + 1, b + 1);
  }
  const g = new BufferGeometry();
  g.setAttribute("position", new Float32BufferAttribute(pos, 3));
  g.setIndex(idx);
  g.computeVertexNormals();
  hairCache.set(key, g);
  return g;
};

/** Hand-to-face poses: put the right hand on the face (palm), the chin or the brow of the solved head. */
export const withHead = (spec: ChibiSpec, pose: Pose, where: "palm" | "chin" | "brow"): Pose => {
  const j = solveChibi(spec, pose);
  const s = spec.scale ?? 1, hr = DIM.handR * s, R = j.headR;
  let hand: V3;
  if (where === "palm") hand = add(add(add(j.head, j.headFwd, R + hr * 0.75), j.headUp, 0.03 * s), j.headRight, -0.07 * s);
  else {
    const d = where === "chin" ? nrm(add(add([0, 0, 0], j.headFwd, 0.62), j.headUp, -0.8)) : nrm(add(add(add([0, 0, 0], j.headFwd, 0.7), j.headUp, 0.42), j.headRight, -0.6));
    hand = add(j.head, d, where === "chin" ? R + hr * 0.8 : R * 1.08 + hr * 0.9);
  }
  return { ...pose, handR: [hand[0] / s, hand[1] / s, hand[2] / s] };
};

const kerudungCache = new Map<string, BufferGeometry>();
const kerudungGeo = (R: number) => {
  const key = R.toFixed(4);
  const hit = kerudungCache.get(key);
  if (hit) return hit;
  const NL = 96, NV = 16, pos: number[] = [], idx: number[] = [];
  const bottom = (lon: number) => { const a = Math.abs(lon); return a < 46 ? 28 : a < 78 ? lerpT(28, -62, (a - 46) / 32) : -66; };
  const at = (lonDeg: number, latDeg: number, k: number) => { const lo = lonDeg * D2R, la = latDeg * D2R; return [Math.sin(lo) * Math.cos(la) * k, Math.sin(la) * k, Math.cos(lo) * Math.cos(la) * k]; };
  const rows = NV + 2;
  for (let i = 0; i <= NL; i++) {
    const lon = -180 + (360 * i) / NL, bot = bottom(lon);
    for (let j = 0; j <= NV; j++) { const lat = 90 + ((bot - 90) * j) / NV; pos.push(...at(lon, lat, R * (1.075 + 0.02 * (1 - j / NV)))); }
    pos.push(...at(lon, bot, R * 0.99));
  }
  for (let i = 0; i < NL; i++) for (let j = 0; j < rows - 1; j++) { const a = i * rows + j, b = (i + 1) * rows + j; idx.push(a, a + 1, b, b, a + 1, b + 1); }
  const g = new BufferGeometry();
  g.setAttribute("position", new Float32BufferAttribute(pos, 3));
  g.setIndex(idx);
  g.computeVertexNormals();
  kerudungCache.set(key, g);
  return g;
};
const drapeCache = new Map<string, BufferGeometry>();
/** the kerudung's drape over the neck and shoulders (torso frame, y from the neck down to the shoulders) */
const drapeGeo = (top: number, bot: number, r0: number, r1: number) => {
  const key = [top, bot, r0, r1].map((x) => x.toFixed(4)).join(",");
  const hit = drapeCache.get(key);
  if (hit) return hit;
  const pts: Vector2[] = [];
  for (let i = 0; i <= 8; i++) { const t = i / 8; pts.push(new Vector2(r0 + (r1 - r0) * Math.pow(t, 0.8), top + (bot - top) * t)); }
  pts.push(new Vector2(r1 * 0.94, bot - 0.004));
  const g = new LatheGeometry(pts.reverse(), 32);
  drapeCache.set(key, g);
  return g;
};

const add = (a: V3, b: V3): V3 => [a[0] + b[0], a[1] + b[1], a[2] + b[2]];
const sub = (a: V3, b: V3): V3 => [a[0] - b[0], a[1] - b[1], a[2] - b[2]];
const mul = (a: V3, s: number): V3 => [a[0] * s, a[1] * s, a[2] * s];

const cross = (a: V3, b: V3): V3 => [a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0]];

export const ik2 = (root: V3, target: V3, a: number, b: number, pole: V3): [V3, V3] => {
  let d = sub(target, root);
  let L = len(d);
  const max = (a + b) * 0.999, min = Math.abs(a - b) + 1e-3;
  if (L < 1e-6) { d = [0, -1, 0]; L = 1e-6; }
  const dir = mul(d, 1 / L);
  const Lc = Math.min(max, Math.max(min, L));
  const x = (a * a - b * b + Lc * Lc) / (2 * Lc);
  const h = Math.sqrt(Math.max(0, a * a - x * x));
  let bend = sub(pole, mul(dir, dot(pole, dir)));
  if (len(bend) < 1e-5) bend = cross(dir, [1, 0, 0]);
  bend = norm(bend);
  const mid = add(add(root, mul(dir, x)), mul(bend, h));
  return [mid, add(root, mul(dir, Lc))];
};

export const beanR = (v: number) => {
  const p = 2.4;
  let r = Math.pow(Math.max(0, 1 - Math.pow(Math.abs(v), p)), 1 / p);

const SkyShader = {
  uniforms: {
    uZenith: { value: new Color() }, uHorizon: { value: new Color() }, uSunDir: { value: new Vector3(0, 1, 0) },
    uCover: { value: 0.5 }, uCloud: { value: new Color() }, uShade: { value: new Color() }, uTime: { value: 0 },
  },
  vertexShader: /* glsl */ `varying vec3 vDir; void main(){ vDir = normalize(position); vec4 p = projectionMatrix * modelViewMatrix * vec4(position, 1.0); gl_Position = p.xyww; }`,
  fragmentShader: /* glsl */ `
    precision highp float;
    uniform vec3 uZenith; uniform vec3 uHorizon; uniform vec3 uSunDir; uniform float uCover; uniform vec3 uCloud; uniform vec3 uShade; uniform float uTime;
    varying vec3 vDir;
    float hash(vec2 p){ vec3 q = fract(vec3(p.xyx) * vec3(0.1031, 0.1030, 0.0973)); q += dot(q, q.yzx + 33.33); return fract((q.x + q.y) * q.z); }
    float vn(vec2 p){ vec2 i = floor(p), f = fract(p); vec2 u = f * f * (3.0 - 2.0 * f);
      return mix(mix(hash(i), hash(i + vec2(1, 0)), u.x), mix(hash(i + vec2(0, 1)), hash(i + vec2(1, 1)), u.x), u.y); }
    float dens(vec2 p){
      float base = 0.58 * vn(p * vec2(0.5, 0.62)) + 0.30 * vn(p * 1.05 + 7.3) + 0.12 * vn(p * 2.3 + 3.1);
      float bil = 0.0, a = 0.5; vec2 q = p * 2.6;
      for (int i = 0; i < 3; i++){ bil += a * abs(2.0 * vn(q) - 1.0); q = q * 2.07 + vec2(3.7, 1.3); a *= 0.5; }
      return base - bil * 0.2 + 0.07;
    }
    void main(){
      vec3 d = normalize(vDir);
      float h = max(d.y, 0.0);
      vec3 c = mix(uHorizon, uZenith, 1.0 - exp(-h * 3.2));
      if (d.y > 0.0) {
        vec2 p = d.xz / (d.y + 0.3) * 2.2 + vec2(uTime * 0.01, uTime * 0.003);
        float n = dens(p);
        float cov = 1.0 - uCover;
        // crisp-edged (game-like) cloud masses, two flat tones: lit top, shaded belly
        float a = smoothstep(cov, cov + 0.03, n);
        if (a > 0.0) {
          vec2 toSun = normalize(uSunDir.xz + vec2(1e-4)) * 0.25;
          float lit = step(dens(p + toSun), n + 0.02);
          vec3 cc = mix(uShade, uCloud, lit);
          a *= smoothstep(0.02, 0.12, d.y);
          c = mix(c, cc, a);
        }
      }
      gl_FragColor = vec4(c, 1.0);
    }`,
};

const GradeShader = {
  uniforms: { tDiffuse: { value: null }, uTint: { value: new Color("#fff") }, uContrast: { value: 0.1 }, uSat: { value: 1 } },
  vertexShader: /* glsl */ `varying vec2 vUv; void main(){ vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }`,
  fragmentShader: /* glsl */ `
    uniform sampler2D tDiffuse; uniform vec3 uTint; uniform float uContrast; uniform float uSat; varying vec2 vUv;
    void main(){
      vec4 c = texture2D(tDiffuse, vUv);
      c.rgb *= uTint;
      c.rgb = mix(c.rgb, c.rgb * c.rgb * (3.0 - 2.0 * c.rgb), uContrast);
      float l = dot(c.rgb, vec3(0.2126, 0.7152, 0.0722));
      c.rgb = mix(vec3(l), c.rgb, uSat);
      gl_FragColor = vec4(clamp(c.rgb, 0.0, 1.0), 1.0);
    }`,
};

const SanitizeShader = {
  uniforms: { tDiffuse: { value: null } },
  vertexShader: `varying vec2 vUv; void main() { vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0); }`,
  fragmentShader: `uniform sampler2D tDiffuse; varying vec2 vUv;
    void main() { vec4 c = texture2D(tDiffuse, vUv); if (any(isnan(c)) || any(isinf(c))) c = vec4(0.0, 0.0, 0.0, 1.0); gl_FragColor = vec4(clamp(c.rgb, 0.0, 32.0), 1.0); }`,
};

export const Stage: React.FC<{ shot: Shot; look: Look; t: number; children: React.ReactNode }> = ({ shot, look, t, children }) => {
  const { gl, scene, camera, size } = useThree();

  const rig = useMemo(() => {
    const target = new Object3D();
    const sun = new DirectionalLight("#fff", 3);
    sun.target = target;
    sun.castShadow = true;
    sun.shadow.mapSize.set(4096, 4096);
    sun.shadow.radius = 3;
    sun.shadow.bias = -0.0004;
    sun.shadow.normalBias = 0.025;
    const hemi = new HemisphereLight("#fff", "#444", 1);
    const uniforms = Object.fromEntries(Object.entries(SkyShader.uniforms).map(([k, v]) => [k, { value: (v.value as { clone?: () => unknown }).clone ? (v.value as { clone: () => unknown }).clone() : v.value }]));
    const sky = new Mesh(new SphereGeometry(1, 48, 24), new ShaderMaterial({ ...SkyShader, uniforms, side: BackSide, depthWrite: false, fog: false }));
    sky.frustumCulled = false;
    sky.renderOrder = -10;
    return { target, sun, hemi, sky };
  }, []);

  const chain = useMemo(() => {
    const w = size.width, h = size.height;
    const rt = new WebGLRenderTarget(w, h, { type: HalfFloatType, samples: 8 });
    const composer = new EffectComposer(gl, rt);
    composer.setPixelRatio(1);
    composer.setSize(w, h);
    composer.addPass(new RenderPass(scene, camera));
    composer.addPass(new ShaderPass(SanitizeShader));
    const gtao = new GTAOPass(scene, camera, w, h);
    gtao.updateGtaoMaterial({ radius: 0.45, distanceExponent: 1.3, thickness: 1.0, scale: 1.0, samples: 16, distanceFallOff: 1.0, screenSpaceRadius: false });
    gtao.updatePdMaterial({ lumaPhi: 10, depthPhi: 2, normalPhi: 3, radius: 6, rings: 2, samples: 16 });
    composer.addPass(gtao);
    composer.addPass(new OutputPass());
    const grade = new ShaderPass(GradeShader);
    composer.addPass(grade);
    return { composer, gtao, grade };
  }, [gl, scene, camera, size.width, size.height]);

  useLayoutEffect(() => {
    gl.toneMapping = NeutralToneMapping;
    gl.shadowMap.autoUpdate = false;
  }, [gl]);

  useFrame(() => {
    const L = look;
    const cam = camera as PerspectiveCamera;
    cam.position.set(...shot.pos);
    cam.up.set(Math.sin(shot.roll ?? 0), Math.cos(shot.roll ?? 0), 0);
    cam.lookAt(...shot.target);
    cam.fov = shot.fov;
    cam.near = 0.08;
    cam.far = 2000;
    cam.updateProjectionMatrix();
    cam.updateMatrixWorld();

    const at = shot.shadowAt ?? shot.target;
    const box = shot.shadowBox ?? 10;
    const dir = new Vector3(...L.sun.dir).normalize();
    rig.target.position.set(...at);
    rig.target.updateMatrixWorld();
    rig.sun.position.set(at[0] + dir.x * 80, at[1] + dir.y * 80, at[2] + dir.z * 80);
    rig.sun.color.set(L.sun.color);
    rig.sun.intensity = L.sun.intensity;
    const sc = rig.sun.shadow.camera;
    sc.left = -box; sc.right = box; sc.top = box; sc.bottom = -box;
    sc.near = 1;
    sc.far = 200;
    sc.updateProjectionMatrix();
    rig.sun.updateMatrixWorld();
    rig.hemi.color.set(L.hemi.sky);
    rig.hemi.groundColor.set(L.hemi.ground);
    rig.hemi.intensity = L.hemi.intensity;

    const fog = scene.fog as FogExp2 | null;
    if (!fog) scene.fog = new FogExp2(L.fog.color, L.fog.density);
    else { fog.color.set(L.fog.color); fog.density = L.fog.density; }

    rig.sky.position.copy(cam.position);
    rig.sky.scale.setScalar(cam.far * 0.9);
    rig.sky.updateMatrixWorld();
    const su = (rig.sky.material as ShaderMaterial).uniforms;
    su.uZenith.value.set(L.sky.zenith);
    su.uHorizon.value.set(L.sky.horizon);
    su.uSunDir.value.copy(dir);
    su.uCover.value = L.sky.cover;
    su.uCloud.value.set(L.sky.cloud);
    su.uShade.value.set(L.sky.shade);
    su.uTime.value = t;

    gl.toneMappingExposure = L.exposure;
    chain.gtao.blendIntensity = L.ao;
    const g = chain.grade.uniforms;
    g.uTint.value.set(L.grade.tint);
    g.uContrast.value = L.grade.contrast;
    g.uSat.value = L.grade.sat;
    gl.shadowMap.needsUpdate = true;
    chain.composer.render();
  }, 1);

  return (
    <>
      <primitive object={rig.target} />
      <primitive object={rig.sun} />
      <primitive object={rig.hemi} />
      <primitive object={rig.sky} />
      {children}
    </>
  );
};

  const g = cv.getContext("2d")!;
  g.fillStyle = o.bg; g.fillRect(0, 0, W, H);
  const bw = Math.round(H * 0.06);
  g.strokeStyle = o.ink; g.lineWidth = bw; g.strokeRect(bw / 2, bw / 2, W - bw, H - bw);

  const fit = (txt: string, font: (s: number) => string, max: number, maxW: number) => { let s = max; g.font = font(s); while (s > 12 && g.measureText(txt).width > maxW) { s -= 2; g.font = font(s); } return s; };

  }

  t.colorSpace = SRGBColorSpace;
  t.anisotropy = 8;
  texCache.set(key, t);

const D2R = Math.PI / 180;
const hipGeo = (w: number, d: number, h: number, ridge: number, color: string) => {
  const x = w / 2, z = d / 2, r = Math.min(ridge, w) / 2;
  const v = [
    -x, 0, z, x, 0, z, r, h, 0, -x, 0, z, r, h, 0, -r, h, 0, // front
    x, 0, -z, -x, 0, -z, -r, h, 0, x, 0, -z, -r, h, 0, r, h, 0, // back
    -x, 0, -z, -x, 0, z, -r, h, 0, // left
    x, 0, z, x, 0, -z, r, h, 0, // right
    -x, 0, -z, x, 0, -z, x, 0, z, -x, 0, -z, x, 0, z, -x, 0, z, // bottom
  ];
  const g = new BufferGeometry();
  g.setAttribute("position", new Float32BufferAttribute(v, 3));
  g.computeVertexNormals();
  return colorize(g, color);
};
/** geometry of one piece, centred on its pivot (bottom centre / centre), plus the pivot's transform */
type PG = { geo: BufferGeometry | null; pos: V3; quat: Quaternion; len: number; sign?: { size: V3; tex: CanvasTexture; edge: string } };
const pieceGeo = (p: Piece): PG => {
  const q = new Quaternion();
  if (p.rot) q.setFromEuler(new Euler(p.rot[0] * D2R, p.rot[1] * D2R, p.rot[2] * D2R));
  switch (p.shape) {
    case "box": return { geo: box(p.size, [0, p.size[1] / 2, 0], p.color), pos: p.pos, quat: q, len: Math.max(...p.size) };
    case "cyl": return { geo: cyl(p.r, p.r2 ?? p.r, p.h, [0, p.h / 2, 0], p.color, p.seg ?? 10), pos: p.pos, quat: q, len: p.h };
    case "cone": return { geo: cone(p.r, p.h, [0, p.h / 2, 0], p.color, p.seg ?? 8), pos: p.pos, quat: q, len: p.h };
    case "sphere": {
      const g = blob(p.r, [0, 0, 0], p.color, p.scale ?? [1, 1, 1], 1, 3);
      if (p.half) { const pa = g.getAttribute("position"); for (let i = 0; i < pa.count; i++) if (pa.getY(i) < 0) pa.setY(i, 0); g.computeVertexNormals(); }
      return { geo: g, pos: p.pos, quat: q, len: p.r * 2 };
    }
    case "blob": return { geo: blob(p.r, [0, 0, 0], p.color, p.scale ?? [1, 1, 1], 1, 5), pos: p.pos, quat: q, len: p.r * 2 };
    case "prism": return { geo: prism(p.w, p.h, p.l, [0, 0, 0], p.color), pos: p.pos, quat: q, len: Math.max(p.w, p.l) };
    case "hip": return { geo: hipGeo(p.w, p.d, p.h, p.ridge ?? 0, p.color), pos: p.pos, quat: q, len: Math.max(p.w, p.d) };
    case "rod": {
      const A = new Vector3(...p.a), B = new Vector3(...p.b);
      const d = B.clone().sub(A);
      const L = d.length();
      q.setFromUnitVectors(new Vector3(0, 1, 0), d.normalize());
      return { geo: rod([0, 0, 0], [0, L, 0], p.r, p.color, 7), pos: p.a, quat: q, len: L };
    }
    case "sign": {
      const tex = signTexture({ text: p.text, sub: p.sub, bg: p.bg ?? "#fffaf0", ink: p.ink ?? "#1b1b1b", subInk: p.subInk, w: p.size[0], h: p.size[1] });
      return { geo: null, pos: p.pos, quat: q, len: p.size[0], sign: { size: p.size, tex, edge: p.ink ?? "#1b1b1b" } };
    }
  }
};

// ------------------------------------------------------------------ finished building (merged)
type Finished = { solid: BufferGeometry | null; glass: BufferGeometry | null; signs: { pos: V3; quat: Quaternion; size: V3; tex: CanvasTexture; edge: string }[] };
const finishedCache = new Map<string, Finished>();
const finished = (key: string, pieces: Piece[]): Finished => {
  const hit = finishedCache.get(key);
  if (hit) return hit;
  const solid: BufferGeometry[] = [], glass: BufferGeometry[] = [], signs: Finished["signs"] = [];
  for (const p of pieces) {
    if (p.temp) continue;
    const g = pieceGeo(p);
    if (g.sign) { signs.push({ pos: [g.pos[0], g.pos[1] + g.sign.size[1] / 2, g.pos[2]], quat: g.quat, ...g.sign }); continue; }
    if (!g.geo) continue;
    g.geo.applyMatrix4(new Matrix4().compose(new Vector3(...g.pos), g.quat, new Vector3(1, 1, 1)));
    (p.glass ? glass : solid).push(g.geo);
  }
  const f = { solid: solid.length ? merge(solid) : null, glass: glass.length ? merge(glass) : null, signs };
  finishedCache.set(key, f);
  return f;
};

export type BuildingProps = {
  def: BuildingDef;
  vars: Record<string, string>;
  pos: V3;
  yaw: number;
  t: number;
  /** build window (film seconds); undefined = standing */
  start?: number;
  end?: number;
  /** grow-in time for a house that appears (named houses) */
  appear?: number;
  night: boolean;
};
export const Building: React.FC<BuildingProps> = ({ def, vars, pos, yaw, t, start, end, appear, night }) => {
  const key = useMemo(() => JSON.stringify([def.name, def.size, def.steps.length, vars]), [def, vars]);
  const pieces = useMemo(() => expand(def, vars), [key]); // eslint-disable-line react-hooks/exhaustive-deps
  const strike = def.strike ?? 0.35;
  const building = start !== undefined && end !== undefined && t < end + strike + 0.05;
  const timeline = useMemo(() => {
    if (start === undefined || end === undefined) return null;
    const total = Math.max(0.6, end - start);
    const wsum = def.steps.reduce((s, x) => s + (x.dur ?? 1), 0);
    let acc = start;
    const stepAt = def.steps.map((x) => { const d = ((x.dur ?? 1) / wsum) * total; const r = { t0: acc, d }; acc += d; return r; });
    const byStep = new Map<number, number>();
    const counts = new Map<number, number>();
    for (const p of pieces) counts.set(p.step, (counts.get(p.step) ?? 0) + 1);
    return pieces.map((p) => {
      const k = byStep.get(p.step) ?? 0;
      byStep.set(p.step, k + 1);
      const S = stepAt[p.step], n = counts.get(p.step) ?? 1;
      const t0 = S.t0 + (k / n) * S.d * 0.72;
      const A = Math.min(0.42, Math.max(0.2, S.d * 0.32));
      return { p, g: pieceGeo(p), t0, A, tOut: p.temp ? end + (k / n) * strike * 0.6 : undefined };
    });
  }, [pieces, start, end]); // eslint-disable-line react-hooks/exhaustive-deps
  if (start !== undefined && t < start) return null;
  const gIn = appear !== undefined ? Math.min(1, Math.max(0, (t - appear) / 0.35)) : 1;
  if (gIn <= 0) return null;
  if (!building || !timeline) {
    const f = finished(key, pieces);

        {f.glass && <mesh geometry={f.glass} material={glassMat(night)} castShadow={false} receiveShadow />}
        {f.signs.map((s, i) => <group key={i} position={s.pos} quaternion={s.quat}><Board size={s.size} tex={s.tex} edge={s.edge} /></group>)}
      </group>
    );
  }

    <group position={pos} rotation={[0, yaw, 0]}>
      {timeline.map(({ p, g, t0, A, tOut }, i) => {

        u = Math.min(1, u);

        if (tOut !== undefined && t > tOut) { out = 1 - outCubic(Math.min(1, (t - tOut) / 0.25)); if (out <= 0.001) return null; }

        else if (p.anim === "pop") { const k = Math.max(1e-4, backOut(u, 1.6)); sx = sy = sz = k; }
        else if (p.anim === "slide") slide = -(1 - e) * 9;
        sy *= out;
        const [px, py, pz] = g.pos;
        if (g.sign) return (
          <group key={i} position={[px + slide, py + dy, pz]} quaternion={g.quat} scale={[sx, Math.max(1e-4, sy), sz]}>
            <group position={[0, g.sign.size[1] / 2, 0]}><Board size={g.sign.size} tex={g.sign.tex} edge={g.sign.edge} /></group>
          </group>
        );
        if (!g.geo) return null;
        // grow along x / z keeps one end fixed: the offset is along the piece's own axis
        const off = new Vector3(dx, 0, dz).applyQuaternion(g.quat);
        return (
          <group key={i} position={[px + off.x + slide, py + dy + off.y, pz + off.z]} quaternion={g.quat} scale={[sx, Math.max(1e-4, sy), sz]}>
            <mesh geometry={g.geo} material={p.glass ? glassMat(night) : WMAT} castShadow={!p.glass} receiveShadow />
          </group>
        );
      })}
    </group>
  );
};

type Inst = { pos: V3; rotY: number; s?: number; color?: string; grow?: number };
const Instanced: React.FC<{ geo: BufferGeometry; items: Inst[]; mat?: MeshStandardMaterial | THREE_Material; shadow?: boolean; cap?: number }> = ({ geo, items, mat = WMAT, shadow = true, cap }) => {
  const ref = useRef<InstancedMesh>(null);
  const tmp = useMemo(() => new Object3D(), []);
  useLayoutEffect(() => {
    const im = ref.current!;
    im.count = items.length;
    items.forEach((it, i) => {
      const g = it.grow ?? 1;
      const gy = g >= 1 ? 1 : g <= 0 ? 0.0001 : backOut(g, 1.5);
      const gx = g >= 1 ? 1 : g <= 0 ? 0.0001 : 0.55 + 0.45 * outCubic(g * 1.3);
      tmp.position.set(...it.pos);
      tmp.rotation.set(0, it.rotY, 0);
      const s = it.s ?? 1;
      tmp.scale.set(s * gx, s * gy, s * gx);
      tmp.updateMatrix();
      im.setMatrixAt(i, tmp.matrix);
      if (it.color) im.setColorAt(i, new Color(it.color));
    });
    im.instanceMatrix.needsUpdate = true;
    if (im.instanceColor) im.instanceColor.needsUpdate = true;
    im.computeBoundingSphere();
  });
  return <instancedMesh ref={ref} args={[geo, mat as MeshStandardMaterial, Math.max(1, cap ?? items.length)]} castShadow={shadow} receiveShadow frustumCulled={false} />;
};

// NEGERI camera solver — a camera for every shot, chosen automatically. For a shot kind and its subjects (people,
// a building, a sign) it tries a fan of angles × distances × heights, rejects any position inside a building / tent
// / tree or where something hides the subject (rays against boxes and spheres), fits the subject into the part of
// the frame the cards leave free (9:16: a vertical band; 16:9: a side column), prefers faces toward the lens, and
// a different angle from the previous shot of the same subject. Pure TypeScript.

import type { PCam, V3 } from "./types.ts";

/** an axis-aligned box, or (with `yaw`) a box turned about its centre `c` — then min / max are in its own frame around c */
export type Box = { min: V3; max: V3; yaw?: number; c?: V3 };
export type Sphere = { c: V3; r: number; id?: string };
export type Env = { boxes: (Box & { id?: string })[]; spheres: Sphere[] };
export type Subj =
  | { id: string; type: "person"; p: V3; face: number; s: number; skip?: string[]; noFace?: boolean }
  | { id: string; type: "building"; c: V3; yaw: number; w: number; d: number; h: number }
  | { id: string; type: "sign"; c: V3; yaw: number; w: number; h: number };
/** free part of the frame for the subject: centre (0..1, y down) and size (fraction of the frame) */
export type Frame = { x: number; y: number; fx: number; fy: number };
export type SolveOpts = { aspect: number; frame: Frame; prevAz?: number; prevId?: string; side?: "left" | "right"; move?: string; bounds?: Box };
export type Solve = { a: PCam; b: PCam; vis: number; az: number; box: number; note?: string; orbit?: { c: V3; r: number; h: number; az0: number; az1: number; fov: number; target: V3 } };

const toLocal = (p: V3, b: Box, dir = false): V3 => {
  if (b.yaw === undefined) return p;
  const c = Math.cos(b.yaw), s = Math.sin(b.yaw);
  const x = dir ? p[0] : p[0] - b.c![0], z = dir ? p[2] : p[2] - b.c![2];
  return [x * c - z * s, dir ? p[1] : p[1], x * s + z * c];
};
const rayBox = (o0: V3, d0: V3, b: Box) => {
  const o = toLocal(o0, b), d = toLocal(d0, b, true);
  let t0 = 0, t1 = 1;
  for (let i = 0; i < 3; i++) {
    if (Math.abs(d[i]) < 1e-9) { if (o[i] < b.min[i] || o[i] > b.max[i]) return Infinity; continue; }
    let a = (b.min[i] - o[i]) / d[i], c = (b.max[i] - o[i]) / d[i];
    if (a > c) [a, c] = [c, a];
    t0 = Math.max(t0, a); t1 = Math.min(t1, c);
    if (t0 > t1) return Infinity;
  }
  return t0;
};
const raySphere = (o: V3, d: V3, c: V3, r: number) => {
  const m = sub(o, c), b = dot(m, d), cc = dot(m, m) - r * r, a = dot(d, d);
  const disc = b * b - a * cc;
  if (disc < 0) return Infinity;
  const t = (-b - Math.sqrt(disc)) / a;
  return t >= 0 && t <= 1 ? t : cc < 0 ? 0 : Infinity;
};
export const inside = (p0: V3, b: Box, pad = 0) => { const p = toLocal(p0, b); return insideL(p, b, pad); };
const insideL = (p: V3, b: Box, pad = 0) => p[0] > b.min[0] - pad && p[0] < b.max[0] + pad && p[1] > b.min[1] - pad && p[1] < b.max[1] + pad && p[2] > b.min[2] - pad && p[2] < b.max[2] + pad;
/** fraction of the way a → b that is clear */
const clear = (a: V3, b: V3, E: Env, skip: Set<string>) => {
  const d = sub(b, a);
  let t = 1;
  for (const c of E.boxes) if (!c.id || !skip.has(c.id)) t = Math.min(t, rayBox(a, d, c));
  for (const s of E.spheres) if (!s.id || !skip.has(s.id)) t = Math.min(t, raySphere(a, d, s.c, s.r));
  return t;
};

  const skip0 = new Set([s0.id, ...(s0.type === "person" ? s0.skip ?? [] : [])]);
  for (const q of mainPts) vMain += clear(pos, q, E, skip0) >= 0.985 ? 1 : 0;
  vMain /= mainPts.length || 1;
  let vAll = 0, nAll = 0;
  allPts.forEach((pts, i) => { for (const q of pts) { vAll += clear(pos, q, E, new Set([subs[i].id])) >= 0.985 ? 1 : 0; nAll++; } });
  return { vMain, vAll: vAll / (nAll || 1) };
};
const ptsOf = (s: Subj): V3[] => {
  if (s.type === "person") return [headOf(s.p, s.s), chestOf(s.p, s.s), [s.p[0], s.p[1] + 0.25 * s.s, s.p[2]]];
  if (s.type === "sign") return [s.c];
  const f: V3 = [Math.sin(s.yaw), 0, Math.cos(s.yaw)], r: V3 = [Math.cos(s.yaw), 0, -Math.sin(s.yaw)];
  const front = add(s.c, f, s.d / 2);
  return [add(front, [0, s.h * 0.5, 0]), add(add(front, r, s.w * 0.35), [0, s.h * 0.35, 0]), add(add(front, r, -s.w * 0.35), [0, s.h * 0.35, 0]), add(s.c, [0, s.h * 0.85, 0])];
};
const okPos = (pos: V3, E: Env, o: SolveOpts, skip: Set<string>) => {
  if (pos[1] < 0.28) return false;
  const B = o.bounds;
  if (B && (pos[0] < B.min[0] || pos[0] > B.max[0] || pos[2] < B.min[2] || pos[2] > B.max[2])) return false;
  for (const b of E.boxes) if (inside(pos, b, 0.25)) return false;
  // keep a head's width away from other people's heads (a giant ear in the foreground reads as a mistake)
  for (const s of E.spheres) if ((!s.id || !skip.has(s.id)) && len(sub(pos, s.c)) < s.r + (s.id ? 0.7 : 0.3)) return false;
  return true;
};

/** A camera move's end key. */
export const moved = (a: PCam, move: string, E: Env): PCam => {
  const v = sub(a.target, a.pos), d = len(v);
  let b: PCam = a;
  if (move === "push") b = { ...a, pos: add(a.pos, v, 0.08), fov: a.fov - 1 };
  else if (move === "pull") b = { ...a, pos: add(a.pos, v, -0.1), fov: a.fov + 1 };
  else if (move === "orbit") { const r = sub(a.pos, a.target), c = Math.cos(8 * D), s = Math.sin(8 * D); b = { ...a, pos: add(a.target, [r[0] * c + r[2] * s, r[1], -r[0] * s + r[2] * c]) }; }
  else if (move === "rise") b = { ...a, pos: add(a.pos, [0, d * 0.14, 0]), target: add(a.target, [0, d * 0.04, 0]) };
  else if (move === "track") { const rgt = nrm([v[2], 0, -v[0]]); b = { ...a, pos: add(a.pos, rgt, d * 0.06), target: add(a.target, rgt, d * 0.06) }; }
  if (E.boxes.some((c) => inside(b.pos, c, 0.15))) return a;
  return b;
};

/** People shots (close / medium / low / wide), one or several subjects (subs[0] = main). */
export const solvePeople = (kind: string, subs: Subj[], E: Env, o: SolveOpts): Solve => {
  const K0 = KIND[kind] ?? KIND.medium;
  // landscape frames are short: a close / medium shot needs a little more height to keep the hat in
  const K: K = o.aspect > 1 && (kind === "close" || kind === "medium") ? { ...K0, ext: (s) => K0.ext(s) * 1.18 } : K0;
  const skip = new Set(subs.map((s) => s.id));
  const main = subs[0] as Extract<Subj, { type: "person" }>;
  const group = subs.length > 1;
  const aim: V3 = group ? (() => { const c = subs.map(K.aim).reduce((x, y) => add(x, y), [0, 0, 0] as V3); return [c[0] / subs.length, c[1] / subs.length, c[2] / subs.length]; })() : K.aim(main);
  const hf = hfovOf(K.fov, o.aspect);
  const mainPts = ptsOf(main), allPts = subs.map(ptsOf);
  let best: (Solve & { score: number }) | null = null;
  for (const dk of K.dists) for (const el of K.els) for (const off of K.offs) {
    const A = main.face + off * D;
    let d = K.ext(main) / (2 * Math.tan((K.fov * D) / 2) * Math.max(0.35, o.frame.fy));
    if (group) {
      const right: V3 = [Math.cos(A), 0, -Math.sin(A)];
      const xs = subs.map((s) => dot(sub(s.type === "person" ? s.p : s.c, aim), right));
      const spread = Math.max(...xs) - Math.min(...xs) + 0.9;
      d = Math.max(d, spread / 2 / Math.tan((hf * D) / 2) / Math.max(0.35, o.frame.fx));
    } else d = Math.max(d, 0.9 / 2 / Math.tan((hf * D) / 2) / Math.max(0.35, o.frame.fx));
    d *= dk;
    const dir: V3 = [Math.cos(el * D) * Math.sin(A), Math.sin(el * D), Math.cos(el * D) * Math.cos(A)];
    const pos = add(aim, dir, d);
    if (kind === "low") pos[1] = Math.max(0.3, Math.min(pos[1], 0.5 * main.s));
    if (!okPos(pos, E, o, skip)) continue;
    if (subs.some((q) => q.type === "person" && len(sub(pos, headOf(q.p, q.s))) < K.minD * (q === main ? 0.6 : 1))) continue;
    const { vMain, vAll } = scoreView(pos, subs, E, mainPts, allPts);
    // faces toward the lens
    const people = subs.filter((q): q is Extract<Subj, { type: "person" }> => q.type === "person" && !q.noFace);
    const faceOf = (q: Extract<Subj, { type: "person" }>) => Math.cos(wrapA(Math.atan2(pos[0] - q.p[0], pos[2] - q.p[2]) - q.face));
    const front = people.length ? people.reduce((acc, q, i) => acc + faceOf(q) * (i === 0 ? 2 : 1), 0) / (people.length + 1) : 0;
    // something behind the subject (a building, trees) rather than empty sky / grass far away
    const fwd = nrm(sub(aim, pos));
    const hitT = clear(aim, add(aim, fwd, 40), E, skip);
    const back = hitT < 1 ? Math.min(1, (hitT * 40) / 10) * 0.8 + 0.2 : 0.35;
    let score = 4 * vMain + 1.5 * vAll + 1.3 * front + 0.4 * back - 0.25 * Math.abs(dk - 1) - 0.015 * Math.abs(el - K.els[0]);
    if (o.prevAz !== undefined && o.prevId === main.id && Math.abs(wrapA(A - o.prevAz)) < 16 * D) score -= 0.8;
    if (o.side) score += (Math.sign(off) === (o.side === "left" ? 1 : -1) ? 0.35 : 0);
    if (!best || score > best.score) {
      const target = framed(pos, aim, K.fov, o.aspect, o.frame);
      const a: PCam = { pos, target, fov: K.fov };
      best = { a, b: a, vis: vMain, az: A, box: Math.max(4, Math.min(24, d * 1.3 + 3)), score };
    }
  }
  if (!best) {
    const A = main.face, d = K.ext(main) / (2 * Math.tan((K.fov * D) / 2)) * 1.2;
    const pos = add(aim, [Math.sin(A) * d, 0.4, Math.cos(A) * d]);
    const a: PCam = { pos, target: aim, fov: K.fov };
    return { a, b: a, vis: 0, az: A, box: 8, note: "no clear camera position" };
  }
  best.b = moved(best.a, o.move ?? "push", E);
  return best;
};

/** A building: a slow orbit (wide) around it, sized to fit, from its front side. */
export const solveBuilding = (s: Extract<Subj, { type: "building" }>, E: Env, o: SolveOpts, sweep = 12, extra: Subj[] = []): Solve => {
  const skip = new Set([s.id, ...extra.map((x) => x.id)]);
  const fov = 40, hf = hfovOf(fov, o.aspect);
  const pts = ptsOf(s);
  let best: (Solve & { score: number }) | null = null;
  for (const off of [28, -28, 40, -40, 16, -16, 55, -55, 0]) for (const el of [17, 24, 11, 32]) for (const dk of [1, 1.15, 0.92, 1.35]) {
    const A = s.yaw + off * D;
    // horizontal extent of the footprint seen from A, vertical extent = height
    const ca = Math.abs(Math.cos(A - s.yaw)), sa = Math.abs(Math.sin(A - s.yaw));
    const across = s.w * ca + s.d * sa + 1.2, depth = s.w * sa + s.d * ca;
    let d = Math.max(across / 2 / Math.tan((hf * D) / 2) / Math.max(0.35, o.frame.fx), (s.h + 1.2) / 2 / Math.tan((fov * D) / 2) / Math.max(0.35, o.frame.fy)) + depth * 0.5;
    d = Math.max(d, 5) * dk;
    const aim: V3 = [s.c[0], s.c[1] + s.h * 0.42, s.c[2]];
    const dir: V3 = [Math.cos(el * D) * Math.sin(A), Math.sin(el * D), Math.cos(el * D) * Math.cos(A)];
    const pos = add(aim, dir, d);
    if (!okPos(pos, E, o, skip)) continue;
    // the whole sweep must stay clear too
    const endA = A + sweep * D * (off >= 0 ? -1 : 1);
    const pos2 = add(aim, [Math.cos(el * D) * Math.sin(endA) * d, Math.sin(el * D) * d, Math.cos(el * D) * Math.cos(endA) * d]);
    if (!okPos(pos2, E, o, skip)) continue;
    let v = 0;
    for (const q of pts) v += clear(pos, q, E, skip) >= 0.985 ? 1 : 0;
    let v2 = 0;
    for (const q of pts) v2 += clear(pos2, q, E, skip) >= 0.985 ? 1 : 0;
    const vis = (v + v2) / (pts.length * 2);
    let score = 5 * vis + 0.6 * Math.cos(off * D) - 0.3 * Math.abs(dk - 1) - 0.01 * Math.abs(el - 17);
    if (o.prevAz !== undefined && o.prevId === s.id && Math.abs(wrapA(A - o.prevAz)) < 14 * D) score -= 0.6;
    if (!best || score > best.score) {
      const target = framed(pos, aim, fov, o.aspect, o.frame);
      const h = pos[1] - aim[1], r = Math.hypot(pos[0] - aim[0], pos[2] - aim[2]);
      const a: PCam = { pos, target, fov };
      const tgt2 = framed(pos2, aim, fov, o.aspect, o.frame);
      best = { a, b: { pos: pos2, target: tgt2, fov }, vis, az: A, box: Math.max(10, Math.min(60, Math.max(s.w, s.d) * 1.6 + 6)), score, orbit: { c: aim, r, h, az0: A, az1: endA, fov, target: sub(target, aim) } };
    }
  }
  if (!best) {
    const A = s.yaw, d = Math.max(s.w, s.h * 1.6) * 2.4;
    const aim: V3 = [s.c[0], s.c[1] + s.h * 0.4, s.c[2]];
    const pos = add(aim, [Math.sin(A) * d, d * 0.35, Math.cos(A) * d]);
    const a: PCam = { pos, target: aim, fov };
    return { a, b: a, vis: 0, az: A, box: 30, note: "no clear view of the building" };
  }
  return best;
};

/** A sign / name board: straight-ish in front, close enough to read. */
export const solveSign = (s: Extract<Subj, { type: "sign" }>, E: Env, o: SolveOpts): Solve => {
  const fov = 36, hf = hfovOf(fov, o.aspect);
  let best: (Solve & { score: number }) | null = null;
  for (const off of [14, -14, 24, -24, 0, 34, -34]) for (const el of [6, 12, 0]) for (const dk of [1, 0.85, 1.2]) {
    const A = s.yaw + off * D;
    const want = Math.max(s.w / 0.62 / 2 / Math.tan((hf * D) / 2) / Math.max(0.35, o.frame.fx), (s.h * 3.2) / 2 / Math.tan((fov * D) / 2));
    const d = Math.max(1.6, Math.min(want, 22)) * dk;
    const dir: V3 = [Math.cos(el * D) * Math.sin(A), Math.sin(el * D), Math.cos(el * D) * Math.cos(A)];
    const pos = add(s.c, dir, d);
    if (!okPos(pos, E, o, new Set([s.id]))) continue;
    const vis = clear(pos, s.c, E, new Set([s.id])) >= 0.985 ? 1 : 0;
    let score = 5 * vis + Math.cos(off * D) - 0.3 * Math.abs(dk - 1);
    if (o.prevAz !== undefined && o.prevId === s.id && Math.abs(wrapA(A - o.prevAz)) < 10 * D) score -= 0.3;
    if (!best || score > best.score) {
      const target = framed(pos, s.c, fov, o.aspect, o.frame);
      const a: PCam = { pos, target, fov };
      best = { a, b: a, vis, az: A, box: Math.max(5, d * 1.5 + 4), score };
    }
  }
  if (!best) {
    const pos = add(s.c, [Math.sin(s.yaw) * 4, 0.4, Math.cos(s.yaw) * 4]);
    const a: PCam = { pos, target: s.c, fov };
    return { a, b: a, vis: 0, az: s.yaw, box: 8, note: "no clear view of the sign" };
  }
  best.b = moved(best.a, o.move ?? "push", E);
  return best;
};

// ------------------------------------------------------------------ merge
const isObj = (x: unknown): x is Record<string, unknown> => !!x && typeof x === "object" && !Array.isArray(x);
const deep = (a: Record<string, unknown>, b: Record<string, unknown>): Record<string, unknown> => {
  const o: Record<string, unknown> = { ...a };
  for (const [k, v] of Object.entries(b)) {
    if (v === false || v === null) { delete o[k]; continue; }
    o[k] = isObj(v) && isObj(a[k]) ? deep(a[k] as Record<string, unknown>, v) : v;
  }
  return o;
};
export const mergeLook = (a?: Record<string, unknown>, b?: Record<string, unknown>): Record<string, unknown> => deep(a ?? {}, b ?? {});
export const resolveLook = (preset: string, over?: Record<string, unknown>): ChibiSpec => {
  const base = CAST[preset];
  if (!base) throw new Error(`cast preset "${preset}" does not exist (have: ${Object.keys(CAST).join(", ")})`);
  return (over ? deep(base.look as unknown as Record<string, unknown>, over) : base.look) as unknown as ChibiSpec;
};

/** Problems with a final look, in plain words (empty = fine). */
export const lookProblems = (look: Record<string, unknown>): string[] => {
  const p: string[] = [];
  const known = ["scale", "belly", "skin", "hair", "shirt", "pants", "shoes", "hat", "face", "lanyard", "sarung", "vest", "apron", "sash"];
  for (const k of Object.keys(look)) if (!known.includes(k)) p.push(`unknown look field "${k}" (have: ${known.join(", ")})`);
  const hex = (v: unknown, where: string) => { if (typeof v !== "string" || !HEX.test(v)) p.push(`${where} must be a colour like "#3a74c4" (got ${JSON.stringify(v)})`); };
  const num = (v: unknown, where: string, a: number, b: number) => { if (v !== undefined && (typeof v !== "number" || v < a || v > b)) p.push(`${where} must be a number ${a}–${b} (got ${JSON.stringify(v)})`); };
  num(look.scale, "scale", 0.6, 1.15);
  num(look.belly, "belly", 0.9, 1.3);
  hex(look.skin, "skin");
  hex(look.pants, "pants");
  hex(look.shoes, "shoes");
  const h = look.hair as Record<string, unknown> | undefined;
  if (!isObj(h)) p.push(`hair must be { "style": …, "color": "#…" }`);
  else { if (!HAIR.includes(String(h.style))) p.push(`hair.style "${h.style}" unknown (have: ${HAIR.join(", ")})`); hex(h.color, "hair.color"); }
  const sh = look.shirt as Record<string, unknown> | undefined;
  if (!isObj(sh)) p.push(`shirt must be { "color": "#…", … }`);
  else for (const [k, v] of Object.entries(sh)) {
    if (k === "motif") { if (!MOTIF.includes(String(v))) p.push(`shirt.motif "${v}" unknown (have: ${MOTIF.join(", ")})`); }
    else if (k === "long" || k === "tunic") { if (typeof v !== "boolean") p.push(`shirt.${k} must be true or false`); }
    else if (["color", "motifColor", "collar", "buttons"].includes(k)) hex(v, `shirt.${k}`);
    else p.push(`unknown shirt field "${k}" (have: color, motif, motifColor, collar, buttons, long, tunic)`);
  }
  const t = look.hat as Record<string, unknown> | undefined;
  if (t !== undefined) {
    if (!isObj(t)) p.push(`hat must be { "kind": … }`);
    else { if (!HATS.includes(String(t.kind))) p.push(`hat.kind "${t.kind}" unknown (have: ${HATS.join(", ")})`); if (t.color !== undefined) hex(t.color, "hat.color"); if (t.color2 !== undefined) hex(t.color2, "hat.color2"); }
  }
  const f = look.face as Record<string, unknown> | undefined;
  if (f !== undefined) {
    if (!isObj(f)) p.push(`face must be an object`);
    else for (const [k, v] of Object.entries(f)) {
      if (k === "brows") { if (!["thin", "thick"].includes(String(v))) p.push(`face.brows "${v}" unknown (have: thin, thick)`); }
      else if (["lashes", "blush", "mustache", "beard", "glasses"].includes(k)) { if (typeof v !== "boolean") p.push(`face.${k} must be true or false`); }
      else p.push(`unknown face field "${k}" (have: brows, lashes, blush, mustache, beard, glasses)`);
    }
  }
  const l = look.lanyard as Record<string, unknown> | undefined;
  if (l !== undefined) {
    if (!isObj(l)) p.push(`lanyard must be { "strap": "#…", "text": "PRESIDEN" } (or false)`);
    else { hex(l.strap, "lanyard.strap"); if (l.card !== undefined) hex(l.card, "lanyard.card"); if (typeof l.text !== "string" || l.text.length > 10) p.push(`lanyard.text must be a short word (≤ 10 letters), e.g. "PRESIDEN"`); }
  }
  const sa = look.sarung as Record<string, unknown> | undefined;
  if (sa !== undefined) { if (!isObj(sa)) p.push(`sarung must be { "color": "#…", "check": "#…" } (or false)`); else { hex(sa.color, "sarung.color"); hex(sa.check, "sarung.check"); } }
  for (const k of ["vest", "apron", "sash"]) if (look[k] !== undefined) hex(look[k], k);
  return p;
};

export const blinkAt = (t: number, seed = 1) => { const period = 2.6 + (seed % 5) * 0.41; const ph = (t + seed * 1.37) % period; return ph < 0.05 ? 0.5 : ph < 0.11 ? 1 : ph < 0.16 ? 0.5 : 0; };
/** lip flap on the spoken words (open while a word is spoken, pulsing per syllable) */
export const mouthAt = (t: number, words: { t0: number; t1: number; w: string }[]) => {
  for (const w of words) {
    if (t < w.t0 - 0.02 || t > w.t1 + 0.02) continue;
    const syl = Math.max(1, (w.w.match(/[aiueo]/gi) ?? []).length);
    const k = (t - w.t0) / Math.max(0.05, w.t1 - w.t0);
    return 0.25 + 0.75 * (0.5 - 0.5 * Math.cos(2 * Math.PI * k * syl));
  }
  return 0;
};
const poseOf = (k: AKey, t: number, seed: number): Pose => {
  const ph = t + seed * 1.7;
  switch (k.pose) {
    case "talk": return POSES.talk(ph * 4.2);
    case "wave": return POSES.wave(ph * 7);
    case "cheer": return POSES.cheer(ph * 5.5);
    case "clap": return POSES.clap(ph * 13);
    case "queue": return POSES.queue(ph * 2);
    case "eat": return POSES.eat(ph * 3);
    case "sitTalk": return POSES.sitTalk(ph * 4);
    case "sitEat": return POSES.sitEat(ph * 3);
    case "argue": return POSES.argue(k.target ?? [0, 1, 1.5], ph * 3);
    case "point": return POSES.point(k.target ?? [-1.4, 0.9, 0.8]);
    case "present": return POSES.present(k.target ?? [-1.4, 0.9, 0.8]);
    default: return (POSES[k.pose] as () => Pose)();

type Live = { pose: Pose; pos: V3; yaw: number; expr: Expr; open: number; key: AKey; seated: boolean };
const liveActor = (a: PActor, t: number, words: Plan["words"]): Live | null => {
  let ki = -1;
  for (let i = 0; i < a.keys.length; i++) if (t >= a.keys[i].t - 1e-4) ki = i;
  if (ki < 0) return null;
  const k = a.keys[ki];
  if (k.gone) return null;
  const withHand = (p: Pose, kk: AKey) => (HEAD_HAND[kk.pose] ? withHead(a.spec, p, HEAD_HAND[kk.pose]!) : p);
  let pose = withHand(poseOf(k, t, a.seed), k);
  const prev = a.keys[ki - 1];
  // blend from the previous pose over 0.25 s when the person did not move (a move happens on a cut)
  if (prev && !prev.gone && t - k.t < 0.25 && Math.hypot(prev.pos[0] - k.pos[0], prev.pos[1] - k.pos[1], prev.pos[2] - k.pos[2]) < 0.01 && SEATED_POSES.has(prev.pose) === SEATED_POSES.has(k.pose))
    pose = mixPose(withHand(poseOf(prev, t, a.seed), prev), pose, smoother((t - k.t) / 0.25));
  const seated = SEATED_POSES.has(k.pose);
  if (!seated) pose = { ...pose, pelvis: [pose.pelvis[0], pose.pelvis[1] + 0.004 * Math.sin(t * 2.4 + a.seed), pose.pelvis[2]] };
  const open = a.host ? mouthAt(t, words) : k.chatter ? 0.5 + 0.5 * Math.sin(t * 11 + a.seed * 2) : k.expr === "sleep" ? 0.5 + 0.5 * Math.sin(t * 2.2 + a.seed) : 0;
  return { pose, pos: k.pos, yaw: k.yaw, expr: k.expr, open, key: k, seated };
};

};

const chunkWords = (words: Plan["words"], maxChars = 22) => {
  const out: { text: string; t0: number; t1: number; start: boolean }[] = [];
  let cur: typeof words = [];
  const flush = () => { if (cur.length) out.push({ text: cur.map((w) => w.w).join(" "), t0: cur[0].t0, t1: cur[cur.length - 1].t1, start: false }); cur = []; };
  for (const w of words) {
    const txt = [...cur, w].map((x) => x.w).join(" ");
    if (cur.length && (txt.length > maxChars || cur.length >= 4 || cur[0].line !== w.line)) flush();
    cur.push(w);
    if (/[.!?,]["”]?$/.test(w.w) && cur.map((x) => x.w).join(" ").length > 8) flush();
  }
  flush();
  out.forEach((c, i) => { const prev = out[i - 1]; c.start = !prev || c.t0 - prev.t1 > 0.35; if (out[i + 1]) c.t1 = Math.min(out[i + 1].t0, c.t1 + 0.4); else c.t1 += 0.5; });
  return out.map((c) => ({ ...c, text: c.text.replace(/["“”]/g, "") }));
};

const Overlay: React.FC<{ plan: Plan; S: PShot; t: number; L: Lay; chunks: ReturnType<typeof chunkWords> }> = ({ plan, S, t, L, chunks }) => {
  const aspect = L.w / L.h;
  const act = plan.overlays.filter((o) => t >= o.t0 && t <= o.t1);
  // speech: a bubble over the host when the host is on screen, otherwise a caption card in the free part of the frame
  const cam = camAt(S, t);
  const hostA = plan.actors.find((a) => a.host);
  const hostL = hostA ? liveActor(hostA, t, plan.words) : null;
  const ch = chunks.find((c) => t >= c.t0 - 0.02 && t < c.t1);
  let speech: React.ReactNode = null;
  if (ch) {
    const size = L.wide ? 54 : 60;
    const w = Math.min(L.wide ? 760 : 860, Math.max(380, ch.text.length * size * 0.52 + 70));
    const perLine = (w - 60) / (size * 0.5), lines = Math.ceil(ch.text.length / perLine), bh = lines * size * 1.06 + 44;
    const free = freeOf(0.235, 0.84, act.map((o) => bandP(o)).filter(Boolean) as [number, number][]);
    const xBand = L.wide ? freeOf(0.03, 0.97, act.map((o) => colW(o)).filter(Boolean) as [number, number][]) : ([0.03, 0.97] as [number, number]);
    const age = t - ch.t0, pop = ch.start && age < 0.18 ? Math.max(0.2, Math.min(1, 0.35 + age / 0.18 * 0.65 + Math.sin(Math.min(1, age / 0.18) * Math.PI) * 0.06)) : 1;
    const sc = hostA?.spec.scale ?? 1;
    const head = hostL ? project(cam, aspect, [hostL.pos[0], hostL.pos[1] + 1.27 * sc, hostL.pos[2]]) : null;
    const y0 = L.wide ? 0.09 * L.h : free[0] * L.h + 8, y1 = L.wide ? 0.84 * L.h : free[1] * L.h;
    const bx0 = xBand[0] * L.w + 10, bx1 = xBand[1] * L.w - 10;
    if (head && head.vis && S.on.includes("host") && head.x > 0.06 && head.x < 0.94 && head.y > 0.12) {
      const ax = head.x * L.w, ay = head.y * L.h;
      let x = Math.max(bx0, Math.min(bx1 - w, ax - w * 0.35));
      const y = Math.max(y0, ay - 64 - bh);
      // landscape: the day chip + counter own the top-left corner
      if (L.wide && y < 400 && x < 620) x = Math.min(Math.max(620, x), bx1 - w);
      if (y + bh < ay - 30) speech = <Bubble text={ch.text} x={x} y={y} w={w} tail={Math.max(x + 60, Math.min(x + w - 60, ax))} scale={pop} size={size} />;
    }
    if (!speech) {
      const x = (bx0 + bx1) / 2 - w / 2, y = Math.max(y0, y1 - bh - (L.wide ? 20 : 40));
      speech = <Bubble text={ch.text} x={x} y={y} w={w} tail={null} scale={pop} size={size} />;
    }
  }
  // build card
  const bo = act.find((o) => o.kind === "build");
  let buildCard: React.ReactNode = null;
  if (bo && bo.kind === "build") {
    const b = plan.builds.find((x) => x.id === bo.build);
    const r = b ? resolveKind(b.kind, plan.custom) : null;
    if (b && r) {
      const steps = r.def.steps.map((s) => s.label);
      const wsum = r.def.steps.reduce((s, x) => s + (x.dur ?? 1), 0);
      const p = b.start === undefined || b.end === undefined ? 1 : clamp01((t - b.start) / (b.end - b.start));
      let acc = 0, label = "SELESAI";
      if (p < 1) for (const s of r.def.steps) { acc += (s.dur ?? 1) / wsum; if (p <= acc) { label = s.label; break; } }
      // the bar fills step by step (each stage its own share)
      let shown = 0, a2 = 0;
      for (let i = 0; i < r.def.steps.length; i++) { const w = (r.def.steps[i].dur ?? 1) / wsum; if (p >= a2 + w) shown += 1 / steps.length; else { shown += ((p - a2) / w) / steps.length; break; } a2 += w; }
      buildCard = <BuildCard t={t} t0={bo.t0} t1={bo.t1} name={b.name} by={b.by} label={label} p={p >= 1 ? 1 : shown} steps={steps} L={L} />;
    }
  }
  const bigO = plan.overlays.find((o) => o.kind === "bigcounter" && t >= o.t0 - 0.4 && t <= o.t1 + 0.4);
  const big = bigO ? Math.min(sm((t - bigO.t0) / 0.35), 1 - sm((t - (bigO.t1 - 0.3)) / 0.3)) : 0;
  // name tags + Zzz for the citizens on screen
  const marks: React.ReactNode[] = [];
  const tags: { id: string; text: string; x: number; y: number; a: number }[] = [];
  for (const a of plan.actors) {
    if (a.host) continue;
    const lv = liveActor(a, t, plan.words);
    if (!lv) continue;
    const sc = a.spec.scale ?? 1, top = lv.pos[1] + (lv.seated ? 0.95 : 1.32) * sc;
    const pr = project(cam, aspect, [lv.pos[0], top, lv.pos[2]]);
    if (!pr.vis || pr.y < 0.1 || pr.y > 0.86) continue;
    const fade = clamp01((t - S.t0 - 0.15) / 0.25);
    const nearCounter = L.wide ? pr.x < 0.34 && pr.y < 0.37 : pr.y * L.h < 500;
    if (a.label && !nearCounter && (S.beat === "gag" || S.on.includes(a.id))) tags.push({ id: a.id, text: a.label, x: Math.max(14 + a.label.length * 9, Math.min(L.w - 14 - a.label.length * 9, pr.x * L.w)), y: pr.y * L.h, a: fade });
    if (lv.expr === "sleep" && (S.beat === "gag" || S.on.includes(a.id))) marks.push(<Zzz key={a.id + "z"} t={t} x={pr.x * L.w + (a.label ? 190 : 80)} y={pr.y * L.h + (a.label ? 40 : -10)} a={fade} />);
  }
  // name tags never overlap: a tag that would cover another one moves up a row
  tags.sort((p, q) => q.y - p.y);
  const placedTags: typeof tags = [];
  for (const tg of tags) {
    const w = tg.text.length * 18 + 30;
    for (let guard = 0; guard < 6 && placedTags.some((o) => Math.abs(o.x - tg.x) < (w + o.text.length * 18 + 30) / 2 && Math.abs(o.y - tg.y) < 50); guard++) tg.y -= 52;
    placedTags.push(tg);
    marks.push(<NameTag key={tg.id} text={tg.text} x={tg.x} y={tg.y} a={tg.a} />);
  }
  const topCard = act.some((o) => o.kind === "rules" || o.kind === "comment" || o.kind === "cta" || o.kind === "title");
  return (
    <>
      <DayChip t={t} day={plan.day} country={plan.country} at={0.12} L={L} />
      <Counter t={t} at={0.3} value={stepped((x) => popShown(plan, x), t, 10, 0.08)} big={big} L={L} />
      {marks}
      {act.map((o, i) => {
        if (o.kind === "rules") return <RuleCards key={i} t={t} t0={o.t0} t1={o.t1} rules={o.rules} L={L} />;
        if (o.kind === "comment") return <CommentCards key={i} t={t} t0={o.t0} t1={o.t1} comments={o.comments} L={L} />;
        if (o.kind === "tag") return <Tag key={i} t={t} t0={o.t0} t1={o.t1} text={o.text} top={L.wide ? 800 : topCard ? 1160 : 470} />;
        if (o.kind === "cta") return <CTA key={i} t={t} t0={o.t0} line={o.line} button={o.button} sub={o.sub} L={L} />;
        if (o.kind === "title") return <Title key={i} t={t} t0={o.t0} t1={o.t1} text={o.text} sub={o.sub} L={L} />;
        return null;
      })}
      {buildCard}
      {speech}
    </>
  );
};

export const NegeriFilm: React.FC<FilmProps> = ({ plan, frames }) => {
  const at = useCurrentFrame();
  const frame = frames?.length ? frames[Math.min(frames.length - 1, at)] : at;
  const { fps, width, height } = useVideoConfig();
  const t = frame / (frames?.length ? 30 : fps);
  const [h] = useState(() => delayRender("negeri fonts"));
  const [ok, setOk] = useState(false);
  useEffect(() => { fontsReady().then(() => { setOk(true); continueRender(h); }); }, [h]);
  const W = usePlanWorld(plan);
  const chunks = useMemo(() => chunkWords(plan.words), [plan]);
  if (!plan.shots.length) return <AbsoluteFill style={{ background: "#000" }} />;
  let si = 0;
  for (let i = 0; i < plan.shots.length; i++) if (t >= plan.shots[i].t0) si = i;
  const S = plan.shots[si];
  const prev = si > 0 ? plan.shots[si - 1] : null;
  const fade = S.dissolve && prev ? clamp01((t - S.t0) / DISSOLVE) : 1;
  const wide = width > height, L: Lay = { w: wide ? 1920 : 1080, h: wide ? 1080 : 1920, wide };
  return (
    <AbsoluteFill style={{ background: "#cfe3ee" }}>
      {ok && (
        <>
          <AbsoluteFill><ShotView plan={plan} S={S} t={t} W={W} /></AbsoluteFill>
          {fade < 1 && prev ? <AbsoluteFill style={{ opacity: 1 - smoother(fade) }}><ShotView plan={plan} S={prev} t={Math.min(t, prev.t1 + DISSOLVE)} W={W} /></AbsoluteFill> : null}
          {qaNoOverlay() ? null : (
            <div style={{ position: "absolute", left: 0, top: 0, width: L.w, height: L.h, transform: `scale(${width / L.w})`, transformOrigin: "0 0", pointerEvents: "none" }}>
              <Overlay plan={plan} S={S} t={t} L={L} chunks={chunks} />
            </div>
          )}
        </>
      )}
      {plan.audio && !frames ? <Html5Audio src={staticFile(plan.audio)} /> : null}
    </AbsoluteFill>
  );
};
