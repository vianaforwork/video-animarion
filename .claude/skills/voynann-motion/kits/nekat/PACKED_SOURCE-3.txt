// Visual QA: bundle once, render chosen frames of a composition, tile them into a contact sheet.
//   node scripts/stills.mjs <outDir> <frame,frame,...> <compositionId> [scale]
// Frames may be given in seconds with an "s" suffix (e.g. 4.5s), using the composition's fps.

import path from "node:path";
import fs from "node:fs";
import { execFileSync } from "node:child_process";
import { bundle } from "@remotion/bundler";
import { openBrowser, renderStill, selectComposition } from "@remotion/renderer";
import { glMode, isWebGLError, switchToSoftware, tolerateClosedTargets } from "./gl.mjs";

tolerateClosedTargets();

const [outDir, list, id, scaleArg = "1", propsFile] = process.argv.slice(2);
const rawProps = propsFile ? JSON.parse(fs.readFileSync(propsFile, "utf8")) : {};
// a props file is either { plan, … } or a bare plan
const inputProps = propsFile ? (rawProps.plan ? rawProps : { plan: rawProps }) : {};
if (!outDir || !list || !id) {
  console.error("usage: node scripts/stills.mjs <outDir> <frame,frame,...|4.5s,...> <compositionId> [scale]");
  process.exit(1);
}
fs.mkdirSync(outDir, { recursive: true });

const serveUrl = await bundle({ entryPoint: path.resolve("src/index.ts") });
// bundles are ~50 MB each; never leave them in the temp dir, even when a render throws
const cleanup = () => fs.rmSync(serveUrl, { recursive: true, force: true });
process.on("exit", cleanup);
const composition = await selectComposition({ serveUrl, id, inputProps });
const frames = list.split(",").map((t) => (t.endsWith("s") ? Math.round(parseFloat(t) * composition.fps) : parseInt(t, 10)));
const files = [];
const renderAll = async (gl) => {
  // one browser for all stills: WebGL contexts are expensive to create and tear down
  const puppeteerInstance = await openBrowser("chrome", { chromiumOptions: { gl } });
  try {
    files.length = 0;
    for (const frame of frames) {
      const output = path.join(outDir, `f${String(frame).padStart(4, "0")}.png`);
      await renderStill({ composition, serveUrl, output, frame, inputProps, scale: Number(scaleArg), chromiumOptions: { gl }, timeoutInMilliseconds: 180000, puppeteerInstance });
      files.push(output);
      process.stdout.write(`${frame} `);
    }
  } finally {
    // after a WebGL crash the tab may already be gone; closing must not hide the original error
    await puppeteerInstance.close({ silent: true }).catch(() => {});
  }
};
const gl = glMode();
try {
  await renderAll(gl);
} catch (e) {
  if (gl === "swangle" || !isWebGLError(e)) throw e;
  await renderAll(switchToSoftware());
}
console.log();
cleanup();

// contact sheet (needs ffmpeg; skipped if missing)
try {
  const w = Math.round(composition.width * Number(scaleArg)), h = Math.round(composition.height * Number(scaleArg));
  const cols = Math.min(4, files.length), rows = Math.ceil(files.length / cols), n = cols * rows;
  if (n > 1) {
    const inputs = files.flatMap((f) => ["-i", f]);
    const pads = Array.from({ length: n - files.length }, () => ["-f", "lavfi", "-i", `color=c=black:s=${w}x${h}`]).flat();
    const filter =
      Array.from({ length: n }, (_, i) => `[${i}:v]scale=${w}:${h},setsar=1[v${i}]`).join(";") + ";" +
      Array.from({ length: n }, (_, i) => `[v${i}]`).join("") +
      `xstack=inputs=${n}:layout=` + Array.from({ length: n }, (_, i) => `${(i % cols) * w}_${Math.floor(i / cols) * h}`).join("|") +
      `,scale='min(2160,iw)':-2`;
    execFileSync("ffmpeg", ["-v", "error", "-y", ...inputs, ...pads, "-filter_complex", filter, "-frames:v", "1", path.join(outDir, "_sheet.jpg")]);
    console.log(`sheet: ${path.join(outDir, "_sheet.jpg")}`);
  }
} catch {
  console.log("(contact sheet skipped: ffmpeg not available)");
}
// a render that failed on the GPU path leaves its timeout timer behind; do not wait for it
process.exit(0);

==== 8311 project/src/nekat/camera.ts
// NEKAT camera solver — an automatic camera per shot. For the points a shot must show (the hero's head and feet, the
// tool, the targets, the nearest arrivals) it tries a fan of directions around the set's camera side (+z) and
// elevations, finds for each the closest distance at which every point sits inside the picture's safe box (clear of
// the counter / level chip at the top and the platform UI at the bottom), and keeps the best one: lines of sight to
// the hero and the targets clear of walls / houses, the camera inside the set bounds and outside every solid, a
// different angle from the previous shot. Pure TypeScript (no three).
import type { Box, Cam, SetMeta } from "./types.ts";
import type { V3 } from "./math.ts";

const D = Math.PI / 180;
const add = (a: V3, b: V3, k = 1): V3 => [a[0] + b[0] * k, a[1] + b[1] * k, a[2] + b[2] * k];
const sub = (a: V3, b: V3): V3 => [a[0] - b[0], a[1] - b[1], a[2] - b[2]];
const dot = (a: V3, b: V3) => a[0] * b[0] + a[1] * b[1] + a[2] * b[2];
const cross = (a: V3, b: V3): V3 => [a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0]];
const nrm = (a: V3): V3 => { const l = Math.hypot(a[0], a[1], a[2]) || 1; return [a[0] / l, a[1] / l, a[2] / l]; };
const wrapA = (a: number) => Math.atan2(Math.sin(a), Math.cos(a));

const rayBox = (o: V3, d: V3, b: Box) => {
  let t0 = 0, t1 = 1;
  for (let i = 0; i < 3; i++) {
    if (Math.abs(d[i]) < 1e-9) { if (o[i] < b.min[i] || o[i] > b.max[i]) return Infinity; continue; }
    let a = (b.min[i] - o[i]) / d[i], c = (b.max[i] - o[i]) / d[i];
    if (a > c) [a, c] = [c, a];
    t0 = Math.max(t0, a); t1 = Math.min(t1, c);
    if (t0 > t1) return Infinity;
  }
  return t0;
};
const inside = (p: V3, b: Box, pad = 0) => p[0] > b.min[0] - pad && p[0] < b.max[0] + pad && p[1] > b.min[1] - pad && p[1] < b.max[1] + pad && p[2] > b.min[2] - pad && p[2] < b.max[2] + pad;
/** Is the segment a → b clear of the blockers (blockers holding b itself are ignored)? */
export const clearLine = (a: V3, b: V3, blockers: Box[]) => {
  const d = sub(b, a);
  for (const bx of blockers) { if (inside(b, bx, 0.02)) continue; if (rayBox(a, d, bx) < 0.985) return false; }
  return true;
};

/** Safe box in NDC [x0, x1, y0, y1]: portrait keeps the top ~25 % for the counter + level chip and the bottom 16 %
 *  for the platform UI; landscape keeps a strip at the top. */
export const safeBox = (aspect: number, role: string): [number, number, number, number] => {
  if (aspect > 1) return role === "hook" ? [-0.86, 0.86, -0.7, 0.1] : [-0.86, 0.86, -0.66, 0.5];
  return role === "hook" ? [-0.84, 0.84, -0.66, 0.08] : [-0.84, 0.84, -0.64, 0.4];
};

export type FitOpts = {
  aspect: number;
  S: SetMeta;
  fov: number;
  /** azimuths (deg) of the camera around the subject; 0 = on the +z side */
  azs: number[];
  /** elevations (deg) */
  els: number[];
  box: [number, number, number, number];
  /** points that need a clear line of sight */
  see: V3[];
  prevAz?: number;
  minD?: number;
  maxD?: number;
  /** prefer this azimuth (deg) */
  prefer?: number;
  /** ignore blockers north of this z (pieces the picture hides when the camera stands there) */
  hideZ?: number;
};
export type Fit = { cam: Cam; az: number; el: number; vis: number; visFirst: number; d: number; ok: boolean };

const project = (q: V3, pos: V3, target: V3, fov: number, aspect: number) => {
  const f = nrm(sub(target, pos)), r = nrm(cross(f, [0, 1, 0])), u = cross(r, f);
  const v = sub(q, pos), depth = dot(v, f);
  const tv = Math.tan((fov * D) / 2), th = tv * aspect;
  return { x: dot(v, r) / (depth * th), y: dot(v, u) / (depth * tv), depth };
};

export const projectPt = (q: V3, pos: V3, target: V3, fov: number, aspect: number) => project(q, pos, target, fov, aspect);
/** Best camera that shows every point of `pts` inside the safe box. */
export const fitShot = (pts: V3[], o: FitOpts): Fit => {
  let lo: V3 = [1e9, 1e9, 1e9], hi: V3 = [-1e9, -1e9, -1e9];
  for (const p of pts) for (let i = 0; i < 3; i++) { lo[i] = Math.min(lo[i], p[i]); hi[i] = Math.max(hi[i], p[i]); }
  const aim: V3 = [(lo[0] + hi[0]) / 2, (lo[1] + hi[1]) / 2, (lo[2] + hi[2]) / 2];
  const [bx0, bx1, by0, by1] = o.box;
  const S = o.S;
  let best: (Fit & { score: number }) | null = null;
  for (const el of o.els)
    for (const azd of o.azs) {
      const az = azd * D;
      const dir: V3 = [Math.cos(el * D) * Math.sin(az), Math.sin(el * D), Math.cos(el * D) * Math.cos(az)];
      let target = aim;
      let d = 0;
      for (let it = 0; it < 3; it++) {
        const fits = (dd: number) => {
          const pos = add(target, dir, dd);
          for (const p of pts) {
            const q = project(p, pos, target, o.fov, o.aspect);
            if (q.depth < 0.3 || q.x < bx0 || q.x > bx1 || q.y < by0 || q.y > by1) return false;
          }
          return true;
        };
        let a = o.minD ?? 0.8, b = o.maxD ?? 40;
        if (!fits(b)) { d = b; break; }
        for (let k = 0; k < 28; k++) { const m = (a + b) / 2; if (fits(m)) b = m; else a = m; }
        d = b;
        // re-centre the content vertically / horizontally in the box
        const pos = add(target, dir, d);
        let yl = 9, yh = -9, xl = 9, xh = -9, dep = 0;
        for (const p of pts) { const q = project(p, pos, target, o.fov, o.aspect); yl = Math.min(yl, q.y); yh = Math.max(yh, q.y); xl = Math.min(xl, q.x); xh = Math.max(xh, q.x); dep += q.depth; }
        dep /= pts.length;
        const f = nrm(sub(target, pos)), r = nrm(cross(f, [0, 1, 0])), u = cross(r, f);
        const tv = Math.tan((o.fov * D) / 2), th = tv * o.aspect;
        const dy = ((yl + yh) / 2 - (by0 + by1) / 2) * dep * tv, dx = ((xl + xh) / 2 - (bx0 + bx1) / 2) * dep * th;
        target = add(add(target, u, dy), r, dx);
      }
      {
        // vertical slack: sit the content low in the box (backdrop above, not empty floor below)
        const pos0 = add(target, dir, d);
        let yl = 9, yh = -9, dep = 0;
        for (const p of pts) { const q = project(p, pos0, target, o.fov, o.aspect); yl = Math.min(yl, q.y); yh = Math.max(yh, q.y); dep += q.depth; }
        dep /= pts.length;
        const slack = by1 - by0 - (yh - yl);
        if (slack > 0.06) {
          const f = nrm(sub(target, pos0)), r = nrm(cross(f, [0, 1, 0])), u = cross(r, f);
          const tgt2 = add(target, u, (yl - (by0 + 0.04)) * dep * Math.tan((o.fov * D) / 2));
          const pos2 = add(tgt2, dir, d);
          if (pts.every((p) => { const q = project(p, pos2, tgt2, o.fov, o.aspect); return q.depth > 0.3 && q.y >= by0 - 0.01 && q.y <= by1 && q.x >= bx0 && q.x <= bx1; })) target = tgt2;
        }
      }
      const pos = add(target, dir, d);
      // validity
      const dbg = (w: string) => { if (typeof process !== "undefined" && process.env?.CAMDBG) console.log(w, azd, el, d.toFixed(1), pos.map((v) => v.toFixed(1)).join()); };
      if (pos[1] < 0.35 || pos[1] > S.bounds.y1) { dbg("y"); continue; }
      if (pos[0] < S.bounds.x0 || pos[0] > S.bounds.x1 || pos[2] < S.bounds.z0 || pos[2] > S.bounds.z1) { dbg("xz"); continue; }
      const hiding = o.hideZ !== undefined && pos[2] > o.hideZ && pos[1] < 4;
      const blockers = hiding ? S.blockers.filter((b) => b.min[2] < o.hideZ!) : S.blockers;
      if (blockers.some((b) => inside(pos, b, 0.25))) { dbg("in"); continue; }
      let vis = 0;
      for (const p of o.see) if (clearLine(pos, p, blockers)) vis++;
      vis /= Math.max(1, o.see.length);
      const visFirst = o.see.length && clearLine(pos, o.see[0], blockers) ? 1 : 0;
      let score = 4 * vis - 0.12 * d - 0.01 * Math.abs(el - o.els[0]);
      if (o.prefer !== undefined) score -= 0.012 * Math.abs(azd - o.prefer);
      if (o.prevAz !== undefined && Math.abs(wrapA(az - o.prevAz)) < 14 * D) score -= 0.7;
      if (!best || score > best.score) best = { cam: { pos, target, fov: o.fov }, az, el, vis, visFirst, d, ok: true, score };
    }
  if (!best) {
    const pos = add(aim, [0, 1.6, 5]);
    return { cam: { pos, target: aim, fov: o.fov }, az: 0, el: 15, vis: 0, visFirst: 0, d: 5, ok: false };
  }
  return best;
};

// NEKAT caption layout (pure TypeScript, shared by the HUD and the compiler): how a caption line wraps and sizes, the
// screen boxes it fills (text + speaker chip) and where it may sit. The compiler keeps every caption off the hero's and
// the partner's heads (and the tool's working end during a level) by moving the line to the nearest free height.

/** Break a long caption into two lines at the space nearest the middle (an explicit \n wins). */
export const wrap = (s: string) => {
  if (s.includes("\n") || s.replace(/\*/g, "").length <= 20) return s;
  const mid = s.length / 2;
  let best = -1;
  for (let i = 0; i < s.length; i++) if (s[i] === " " && (best < 0 || Math.abs(i - mid) < Math.abs(best - mid))) best = i;
  return best < 0 ? s : s.slice(0, best) + "\n" + s.slice(best + 1);
};
/** Font size that keeps the longest line inside `width` (Lilita One ≈ 0.52 em per letter). */
export const fitSize = (text: string, size: number, width: number) => {
  const longest = Math.max(...text.split("\n").map((l) => l.replace(/\*/g, "").length));
  const per = size * 0.52;
  return longest * per > width ? Math.max(size * 0.62, width / (longest * 0.52)) : size;
};

/** Design frame of the HUD (the overlay is drawn at this size and scaled to the video). */
export const frameOf = (wide: boolean) => (wide ? { W: 1920, H: 1080 } : { W: 1080, H: 1920 });
export const CAPTION = {
  size: (wide: boolean) => (wide ? 80 : 100),
  width: (wide: boolean) => (wide ? 1500 : 980),
  chip: 82,
};
export type Rect = { x0: number; x1: number; y0: number; y1: number };

// Lilita One advance widths (1/1000 em) for " "…"~", "…", "é", "è" — read from public/nekat/LilitaOne-Regular.ttf (hmtx).
const GLYPHS = " !\"#$%&'()*+,-./0123456789:;<=>?@ABCDEFGHIJKLMNOPQRSTUVWXYZ[\\]^_`abcdefghijklmnopqrstuvwxyz{|}~…éè";
const ADV = [188,287,460,833,531,798,667,301,430,422,557,602,305,525,229,573,639,406,525,528,598,524,575,500,570,565,277,277,484,580,450,584,867,655,594,561,616,493,472,621,644,429,462,620,435,907,648,696,570,706,590,531,502,628,661,953,683,615,563,385,587,385,687,636,350,517,520,424,565,486,375,516,501,273,271,531,272,788,551,513,555,511,396,450,368,544,521,752,557,537,454,362,628,362,459,909,486,486];
/** Rendered width (px) of one line in Lilita One at `size`, with letter-spacing and the outside half of the stroke. */
export const textWidth = (line: string, size: number, spacing = 1, stroke = 0.16) => {
  let em = 0, n = 0;
  for (const ch of line.replace(/\*/g, "")) { const i = GLYPHS.indexOf(ch); em += i < 0 ? 0.6 : ADV[i] / 1000; n++; }
  return em * size + n * spacing + stroke * size;
};
/** Screen boxes (design px) of a caption centred at yc: the speaker chip and the text block (as the HUD lays them out). */
export const captionBoxes = (text: string, speaker: string | undefined, yc: number, wide: boolean): Rect[] => {
  const t = wrap(text);
  const size = fitSize(t, CAPTION.size(wide), CAPTION.width(wide));
  const lines = t.split("\n");
  const tw = Math.max(...lines.map((l) => textWidth(l, size)));
  const th = lines.length * size * 1.08 + size * 0.1;
  const chipH = speaker ? CAPTION.chip : 0;
  const { W } = frameOf(wide);

// NEKAT materials: matte "toy clay" — high roughness, a faint thumbed-clay bump, flat real colours (no gradients,
// no emissive glow). Everything is cached by colour/options so a thousand props share a handful of materials.
import {
  CanvasTexture, DoubleSide, FrontSide, LinearFilter, MeshStandardMaterial, RepeatWrapping, SRGBColorSpace, type Texture,
} from "three";
import { rng } from "../math.ts";

let noise: CanvasTexture | null = null;
/** Soft lumpy value noise (clay surface) used as a bump map. */
export const clayNoise = (): CanvasTexture => {
  if (noise) return noise;
  const N = 256;
  const cv = document.createElement("canvas");
  cv.width = N;
  cv.height = N;
  const g = cv.getContext("2d")!;
  const img = g.createImageData(N, N);
  const r = rng(7);
  const G = 16;
  const grid: number[] = [];
  for (let i = 0; i < G * G; i++) grid.push(r());
  const at = (x: number, y: number) => grid[((y % G) + G) % G * G + (((x % G) + G) % G)];
  const vn = (x: number, y: number) => {
    const xi = Math.floor(x), yi = Math.floor(y), xf = x - xi, yf = y - yi;
    const u = xf * xf * (3 - 2 * xf), v = yf * yf * (3 - 2 * yf);
    return (at(xi, yi) * (1 - u) + at(xi + 1, yi) * u) * (1 - v) + (at(xi, yi + 1) * (1 - u) + at(xi + 1, yi + 1) * u) * v;
  };
  for (let y = 0; y < N; y++)
    for (let x = 0; x < N; x++) {
      const fx = (x / N) * G, fy = (y / N) * G;
      const n = 0.6 * vn(fx, fy) + 0.3 * vn(fx * 2, fy * 2) + 0.1 * vn(fx * 4, fy * 4);
      const c = Math.round(n * 255);
      const o = (y * N + x) * 4;
      img.data[o] = img.data[o + 1] = img.data[o + 2] = c;
      img.data[o + 3] = 255;
    }
  g.putImageData(img, 0, 0);
  noise = new CanvasTexture(cv);
  noise.wrapS = noise.wrapT = RepeatWrapping;
  noise.repeat.set(2, 2);
  return noise;
};

  else if (pat === "spots") st = st === 1 || st === 3 ? 4 : 0;

    g.setAttribute("aLocal", (g.getAttribute("position") as BufferAttribute).clone());

// NEKAT maths: vectors, easing, seeded random, two-bone IK. Pure functions only (same frame in, same pose out).
export type V3 = [number, number, number];

export const clamp = (x: number, lo = 0, hi = 1) => (x < lo ? lo : x > hi ? hi : x);
export const mix = (a: number, b: number, t: number) => a + (b - a) * t;
export const mix3 = (a: V3, b: V3, t: number): V3 => [mix(a[0], b[0], t), mix(a[1], b[1], t), mix(a[2], b[2], t)];
export const lin = (x: number, a: number, b: number) => clamp((x - a) / (b - a));
export const smooth = (t: number) => {
  const x = clamp(t);
  return x * x * (3 - 2 * x);
};
export const smoother = (t: number) => {
  const x = clamp(t);
  return x * x * x * (x * (x * 6 - 15) + 10);
};
export const easeOut = (t: number) => 1 - Math.pow(1 - clamp(t), 3);
export const easeIn = (t: number) => Math.pow(clamp(t), 3);
/** overshoot then settle (pop-in), C1 at the end */
export const easeBack = (t: number, k = 1.9) => {
  const x = clamp(t) - 1;
  return 1 + (k + 1) * x * x * x + k * x * x;
};
/** 0 → 1 → 0 bump without a cusp */
export const bump = (t: number) => (t <= 0 || t >= 1 ? 0 : 0.5 - 0.5 * Math.cos(t * Math.PI * 2));
/** window: 0 before a, ramps to 1 over `r` s, holds, ramps down to 0 ending at b */
export const win = (t: number, a: number, b: number, r = 0.15) => smooth(lin(t, a, a + r)) * (1 - smooth(lin(t, b - r, b)));

export const add = (a: V3, b: V3): V3 => [a[0] + b[0], a[1] + b[1], a[2] + b[2]];
export const sub = (a: V3, b: V3): V3 => [a[0] - b[0], a[1] - b[1], a[2] - b[2]];
export const mul = (a: V3, s: number): V3 => [a[0] * s, a[1] * s, a[2] * s];
export const dot = (a: V3, b: V3) => a[0] * b[0] + a[1] * b[1] + a[2] * b[2];
export const cross = (a: V3, b: V3): V3 => [a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0]];
export const len = (a: V3) => Math.hypot(a[0], a[1], a[2]);
export const norm = (a: V3): V3 => {
  const l = len(a) || 1;
  return [a[0] / l, a[1] / l, a[2] / l];
};
export const rotY = (p: V3, a: number): V3 => {
  const c = Math.cos(a), s = Math.sin(a);
  return [p[0] * c + p[2] * s, p[1], -p[0] * s + p[2] * c];
};

/** Deterministic PRNG (mulberry32). */
export const rng = (seed: number) => {
  let s = seed >>> 0;
  return () => {
    s = (s + 0x6d2b79f5) >>> 0;
    let t = s;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
};
export const hash = (n: number) => {
  const x = Math.sin(n * 127.1 + 311.7) * 43758.5453;
  return x - Math.floor(x);
};

/** Two-bone IK: elbow/knee between root and target (bone lengths a, b) bent toward `pole`. Returns [mid, end]. */
export const ik2 = (root: V3, target: V3, a: number, b: number, pole: V3): [V3, V3] => {
  let d = sub(target, root);
  let L = len(d);
  if (L < 1e-6) {
    d = [0, -1, 0];
    L = 1e-6;
  }
  const dir = mul(d, 1 / L);
  // never fully straight: near full extension the elbow would swing a lot for a tiny change in distance
  const Lc = Math.min((a + b) * 0.97, Math.max(Math.abs(a - b) + 1e-3, L));
  const x = (a * a - b * b + Lc * Lc) / (2 * Lc);
  const h = Math.sqrt(Math.max(0, a * a - x * x));
  let bend = sub(pole, mul(dir, dot(pole, dir)));
  if (len(bend) < 1e-5) bend = cross(dir, [1, 0, 0]);
  bend = norm(bend);
  return [add(add(root, mul(dir, x)), mul(bend, h)), add(root, mul(dir, Lc))];
};

/** Keyframed scalar/vector track: keys [{t, v}], eased with smootherstep between keys. */
export type Key<T> = { t: number; v: T; ease?: (x: number) => number };
export const track = (keys: Key<number[]>[], t: number): number[] => {
  if (t <= keys[0].t) return keys[0].v.slice();
  const last = keys[keys.length - 1];
  if (t >= last.t) return last.v.slice();
  let i = 0;
  while (i < keys.length - 2 && t > keys[i + 1].t) i++;
  const a = keys[i], b = keys[i + 1];
  const u = (b.ease ?? smoother)(lin(t, a.t, b.t));
  return a.v.map((x, j) => mix(x, b.v[j], u));
};

// passing through anything. The truck is drawn around its rider: seat height / depth, backrest, foot well, cab length
// and width follow the rider's hips, thighs, knees and feet (seated "chair" lower body: lap + shin drape for a sarung
// or skirt, legs for trousers / shorts), and the steering wheel is placed where the rider's hands reach it (two-bone IK
// within reach, rim clear of belly, lap and chin) for every lean the ride poses use. A rider too wide for the cab makes
// the whole toy a little bigger (k), never the rider smaller.
//
// Units: truck units (the truck group is scaled by k in the world). Rider-local = Person root space before build.s
// (x = rider's left, y up, z forward). Truck-local: x forward (the rider faces +x), y up, z = −rider-x.
import { BD } from "./rig.ts";
import type { PersonLook } from "./cast.ts";
import { ik2, type V3 } from "./math.ts";

export type Ride = {
  /** truck scale (1 = the base toy) */
  k: number;
  /** rider size in truck units (build.s / k) and belly */
  s: number; b: number;
  /** rider root (feet level) in truck units */
  x0: number; floor: number;
  /** cab: rear / front inner faces (x), inner half-width (z), seat top, seat front, backrest front */
  rear: number; front: number; side: number; seatTop: number; seatFront: number; backFront: number; backTop: number;
  /** steering wheel in truck units: centre, half-width / half-height of the (round or oval) rim, tilt (top toward the
   *  rider), and where the column joins the rim (its lowest point) */
  wheel: { c: V3; rx: number; ry: number; tilt: number; join: V3 };
  /** rider-local hand targets on the rim, elbow poles, feet */
  handL: V3; handR: V3; poleL: V3; poleR: V3; footL: V3; footR: V3;
};

/** Leans the ride poses may use (the fit is checked for all of them). */
export const RIDE_LEAN: [number, number] = [-0.03, 0.08];

// rider-local seated body (Person "chair" lower body; see kit/Person.tsx)
const SEAT_Y = 0.405; // under the thighs / lap
const KNEE_Z = 0.44; // front of the knees / skirt drape
const TOE_Z = 0.56; // front of the feet (foot at z 0.40)
const FOOT_Z = 0.4;
const LAP_TOP = 0.6;
/** column joint radius (rider-local) */
export const HUB = 0.032; // top of lap / hip block over the thighs (pants hip block reaches 0.65 at z ≤ 0.15)

const rotX = (p: V3, a: number, pivotY: number): V3 => {
  const y = p[1] - pivotY, z = p[2];
  return [p[0], pivotY + y * Math.cos(a) - z * Math.sin(a), y * Math.sin(a) + z * Math.cos(a)];
};
/** Distance from p to the rider's torso ellipsoid (approximate, negative inside), at lean a. */
const torsoGap = (p: V3, b: number, a: number) => {
  // into the torso's own frame (undo the lean about the waist), then the ellipsoid test
  const q = rotX(p, -a, BD.waistY), c0: V3 = [0, BD.waistY + BD.torsoC, 0];
  const r: V3 = [BD.torsoR[0] * b, BD.torsoR[1], BD.torsoR[2] * b];
  const u = Math.hypot((q[0] - c0[0]) / r[0], (q[1] - c0[1]) / r[1], (q[2] - c0[2]) / r[2]);
  return (u - 1) * Math.min(r[0], r[1], r[2]);
};
const shoulder = (side: 1 | -1, b: number, a: number): V3 => rotX([side * BD.shX * b, BD.waistY + BD.shY, -0.01], a, BD.waistY);
const segGap = (a: V3, c: V3, b: number, lean: number) => {
  let g = Infinity;
  for (let i = 0; i <= 6; i++) { const k = i / 6; g = Math.min(g, torsoGap([a[0] + (c[0] - a[0]) * k, a[1] + (c[1] - a[1]) * k, a[2] + (c[2] - a[2]) * k], b, lean)); }
  return g;
};

type WheelFit = { yc: number; zc: number; rx: number; ry: number; tilt: number; score: number };
const wheelCache = new Map<string, WheelFit>();
/** Steering wheel (rider-local) that every lean in RIDE_LEAN can hold: hands on the rim's side points within 95 %
 *  reach, rim ≥ 3 cm off the belly (2.5 / 2 cm only when nothing else fits), clear of the lap and under the chin, elbows
 *  and forearms outside the body. Wide bellies get a wider (oval, yoke-like) wheel so the arms pass beside the belly. */
const fitWheel = (b: number, head: number): WheelFit => {
  const key = `${b.toFixed(3)}|${head.toFixed(3)}`;
  const hit = wheelCache.get(key);
  if (hit) return hit;
  const chin = BD.waistY + BD.headC + (head - 1) * BD.headR * 0.9 - BD.headR * head * 0.92;
  const reach = (BD.upper + BD.fore) * 0.95;
  const leans = [RIDE_LEAN[0], 0, (RIDE_LEAN[0] + RIDE_LEAN[1]) / 2, RIDE_LEAN[1]];
  const search = (rimGap: number, hubGap: number): WheelFit | null => {
    let best: WheelFit | null = null;
    for (const tilt of [0.9, 0.6, 0.3, 0])
      for (let rx = 0.13; rx <= 0.4001; rx += 0.03)
        for (const ry of [0.13, 0.11, 0.09]) {
          if (ry > rx + 1e-6) continue;
          for (let yc = 0.78; yc <= 1.0001; yc += 0.02) {
            if (yc + ry * Math.cos(tilt) > chin - 0.03) continue;
            for (let zc = 0.22; zc <= 0.62; zc += 0.01) {
              const rim = (ph: number): V3 => [rx * Math.cos(ph), yc + ry * Math.sin(ph) * Math.cos(tilt), zc - ry * Math.sin(ph) * Math.sin(tilt)];
              let ok = true;
              for (let i = 0; i < 32 && ok; i++) { const p = rim((i / 32) * Math.PI * 2); if (p[1] < LAP_TOP + 0.04 && p[2] < KNEE_Z + 0.04) ok = false; }
              // the column joins the rim at its lowest point and runs forward-down to the dash (≈ z 0.62, y 0.55 lowest)
              const A = rim(-Math.PI / 2);
              const col = [0, 0.25, 0.5, 0.75, 1].map((k): V3 => [0, A[1] + (0.55 - A[1]) * k, A[2] + (0.62 - A[2]) * k]);
              if (col.some((p) => p[1] < LAP_TOP + 0.05 && p[2] < KNEE_Z + 0.05)) ok = false;
              for (const a of leans) {
                if (!ok) break;
                if (col.some((p) => torsoGap(p, b, a) < HUB + hubGap)) ok = false;
                for (let i = 0; i < 32 && ok; i++) if (torsoGap(rim((i / 32) * Math.PI * 2), b, a) < rimGap) ok = false;
                for (const side of [1, -1] as const) {
                  if (!ok) break;
                  const sh = shoulder(side, b, a), h: V3 = [side * rx, yc, zc];
                  const L = Math.hypot(h[0] - sh[0], h[1] - sh[1], h[2] - sh[2]);
                  if (L > reach || L < 0.22 || torsoGap(h, b, a) < BD.handR + 0.01) { ok = false; break; }
                  const [el, ha] = ik2(sh, h, BD.upper, BD.fore, [side, -0.5, -0.2]);
                  if (torsoGap(el, b, a) < BD.armR * 0.6 || segGap(el, ha, b, a) < BD.armR + 0.01) ok = false;
                }
              }
              if (!ok) continue;
              const score = Math.abs(yc - 0.86) + 0.6 * zc + 0.6 * (rx - ry) + 0.1 * Math.abs(tilt - 0.9);
              if (!best || score < best.score) best = { yc, zc, rx, ry, tilt, score };
              break; // the nearest zc for this shape and height
            }
          }
        }
    return best;
  };
  const out = search(0.03, 0.025) ?? search(0.025, 0.015) ?? search(0.02, 0.008) ?? { yc: 0.86, zc: 0.36, rx: 0.13, ry: 0.13, tilt: 0.9, score: 99 };
  wheelCache.set(key, out);
  return out;
};

// NEKAT director — data types. A Story is what an author writes (stories/<id>/story.json); a Plan is the compiled
// film. Both are plain JSON: the same Plan drives the picture (Film.tsx), the sound (scripts/lib/nekat-sound.mjs) and
// the QA gate. Pure TypeScript (no React / three): node runs the compiler directly.
import type { V3 } from "./math.ts";
import type { CreatureLook, PersonLook } from "./cast.ts";
import type { BoxTop, Zone } from "./crowd.ts";
import type { Dress } from "./dressing.ts";

export type Box = { min: V3; max: V3 };
export type Cam = { pos: V3; target: V3; fov: number };

// ------------------------------------------------------------------ sets
export type SlopeSpec = { o: V3; e: [number, number]; downYaw: number; angle: number; cols: number; rows: number; dx: number; dz: number; fill?: number };
export type SetMeta = {
  id: string;
  about: string;
  /** where the hero stands in the levels; he always faces +z (toward the cameras) */
  hero: V3;
  /** the partner's spot (the second human of the duo) and facing yaw */
  watch: { p: V3; yaw: number };
  /** the named creature's spot in the opening (y = the surface it sits on) */
  perch: { p: V3; yaw: number };
  /** areas the crowd fills, one layer first (nearest the action first), y = floor height */
  ground: { x0: number; x1: number; z0: number; z1: number; y?: number }[];
  /** things creatures can stand ON (height map boxes x0, x1, z0, z1, top): fence caps, walls, seats */
  tops: BoxTop[];
  /** strips along ledges where creatures climb up at mid counts (dropped onto `tops`) */
  ledges: { cx: number; cz: number; rx: number; rz: number; facing: number }[];
  /** roofs + flat roofs filled at the ending */
  roofs: SlopeSpec[];
  flats: { x0: number; x1: number; z0: number; z1: number; y: number }[];
  /** the crowd keeps out of these (pots, poles, trees, doors) */
  avoid: Zone[];
  /** solid volumes: cameras stay out and need a clear line of sight past them */
  blockers: Box[];
  /** vehicles enter from this x (left), along z = hero.z + 0.75, and stop left of the stage */
  laneFrom: number;
  /** a giant creature rises here (cliffhanger ending): behind a roof */
  giant: { p: V3; scale: number; over: V3 };
  /** camera bounds (x, z) and max height */
  bounds: { x0: number; x1: number; z0: number; z1: number; y1: number };
  /** authored views: the whole set full (peak / ending), and the high crane end */
  wide: Cam;
  top: Cam;
  /** default light */
  light?: "siang" | "sore";
};

  by?: string;
  /** false = on screen only, not in the voice-over */
  vo?: boolean;
};
