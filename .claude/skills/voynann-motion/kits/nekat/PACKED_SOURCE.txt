const memo = new Map<string, unknown>();
const once = <T,>(key: string, make: () => T): T => {
  if (!memo.has(key)) memo.set(key, make());
  return memo.get(key) as T;
};
const breathe = (t: number) => 0.012 * Math.sin(t * 2.3);
/** 0 → 1 pop with a small overshoot (0 before t0). */
export const popScale = (t: number, t0: number, dur = 0.3) => {
  if (t < t0) return 0;
  if (t >= t0 + dur) return 1;
  const x = (t - t0) / dur - 1, k = 1.6;
  return Math.max(0, 1 + (k + 1) * x * x * x + k * x * x);
};
/** Hop with anticipation squash, stretch in the air, landing squash. */
export const hop = (p0: V3, p1: V3, t: number, t0: number, dur: number, h: number): { p: V3; squash: number; u: number } => {
  const t1 = t0 + dur, u = clamp((t - t0) / dur), e = smooth(u);
  const p: V3 = [mix(p0[0], p1[0], e), mix(p0[1], p1[1], e) + 4 * h * u * (1 - u), mix(p0[2], p1[2], e)];
  let sq = 0;
  if (t < t0 - 0.12) sq = 0;
  else if (t < t0) sq = 0.45 * smooth(lin(t, t0 - 0.12, t0));
  else if (t < t0 + 0.08) sq = mix(0.45, -0.3, smooth(lin(t, t0, t0 + 0.08)));
  else if (t < t1) sq = mix(-0.3, 0, smooth(lin(t, t0 + 0.08, t1)));
  else if (t < t1 + 0.06) sq = 0.45 * smooth(lin(t, t1, t1 + 0.06));
  else sq = 0.45 * (1 - smooth(lin(t, t1 + 0.06, t1 + 0.26)));
  return { p, squash: sq, u };
};
const yawTo = (from: V3, to: V3) => Math.atan2(to[0] - from[0], to[2] - from[2]);

export const tauOf = (t: number, F: number, Ft: number) => (F >= Ft ? Math.max(0, t - (F - Ft)) : t < F ? (t * Ft) / F : Ft + (t - F));
/** Mouth flap while a cast member speaks one of the segment lines. */
const talk = (seg: Seg, who: string, t: number) => {
  for (const l of seg.lines) {
    if (l.by !== who || t < l.t0 || t > l.t1) continue;
    const env = smooth(lin(t, l.t0, l.t0 + 0.08)) * (1 - smooth(lin(t, l.t1 - 0.12, l.t1)));
    if (l.words.length && l.spoken) {
      const w = l.words.find((x) => t >= x.t0 && t < x.t1);
      if (!w) return 0;
      const k = clamp((t - w.t0) / Math.max(0.08, w.t1 - w.t0));
      return env * (0.35 + 0.65 * Math.sin(Math.PI * k));
    }
    return env * (0.55 + 0.45 * (0.5 - 0.5 * Math.cos(t * 2 * Math.PI * 4.2)));
  }
  return 0;
};
const withTalk = (f: Face, m: number): Face => (m > 0 ? { ...f, mouthOpen: Math.max(f.mouthOpen, m * 0.75), mouthW: Math.max(0.8, f.mouthW) } : f);

const MATE_EXPR: Record<Mate, Expr> = { crossed: "smug", laugh: "grin", shock: "shock", point: "smug" };
const partnerActor = (c: Ctx, seg: Seg, t: number): Actor | null => {
  const pl = c.plan.partner;
  if (!pl) return null;
  const W = c.S.watch;
  const flip = seg.level?.flip ?? seg.peak?.flip ?? 1e9;
  // mode at the segment start, and after the flip (blended over 0.3 s: never a one-frame switch)
  let m0: Mate = "crossed", m1: Mate = "crossed";
  if (seg.kind === "level") m1 = seg.level!.n % 2 ? "laugh" : "point";
  else if (seg.kind === "peak") { m0 = "crossed"; m1 = "shock"; }
  else if (seg.kind === "ending") m0 = m1 = "laugh";
  const sw = seg.kind === "peak" ? seg.t0 + 0.05 : flip + 0.1;
  const k = smooth(lin(t, sw, sw + 0.3));
  const pose = poseTrack([{ t: 0, p: MATE[m0] }, { t: 1, p: MATE[m1], ease: (x) => x }], k);
  const laughK = (m1 === "laugh" ? k : 0) + (m0 === "laugh" ? 1 - k : 0);
  const bob = laughK * (0.5 - 0.5 * Math.cos(t * 2 * Math.PI * 2.2));
  pose.lean = (pose.lean ?? 0) + 0.05 * bob + breathe(t + 1);
  pose.hip = add(pose.hip ?? [0, 0, 0], [0, -0.01 * bob, 0]);
  const face = withTalk(faceAt([{ t: -1, e: MATE_EXPR[m0] }, { t: sw, e: MATE_EXPR[m1] }], t, 0.25), talk(seg, pl.id, t));
  return { pos: W.p, yaw: W.yaw, pose, face, blink: blinkAt(t, 2.2) };
};

// ------------------------------------------------------------------ HOOK
const hookScene = (c: Ctx, seg: Seg, t: number): Scene => {
  const u = t - seg.t0;
  const P = c.S.hero;
  const perch = c.S.perch.p;
  const look = yawTo(P, perch);
  const turn = clamp(wrap(look) * 0.55, -1.0, 1.0);
  const pose = poseTrack([
    { t: 0, p: { ...AKIMBO, lean: -0.04, turn: turn * 0.8, nod: 0.02, twist: turn * 0.25 } },
    { t: 1.0, p: { ...AKIMBO, lean: 0.06, turn, nod: 0.06, hip: [0, -0.015, 0.02], twist: turn * 0.3 } },
    { t: 2.3, p: { ...AKIMBO, lean: 0.02, turn: turn * 0.95, nod: 0.04, twist: turn * 0.3 } },
  ], u);
  pose.lean = (pose.lean ?? 0) + breathe(t);
  const face = withTalk({ ...faceAt([{ t: 0, e: "angry" }], u), lookX: clamp(turn * 1.4, -0.9, 0.9) }, talk(seg, c.plan.hero.id, t));
  const n = vis(c, seg.count);
  const crowd = idleCrowd(c, n, new Set([0]));
  crowd.push({ p: perch, yaw: c.S.perch.yaw, pose: 0, seed: 11, eye: 0.42, wag: 1.3 });
  return { hero: { pos: P, yaw: 0, pose, face, blink: blinkAt(t, 1.2) }, partner: partnerActor(c, seg, t), crowd, puffs: [], tools: [] };
};

const targetsOf = (c: Ctx, seg: Seg, T: ToolDef): Target[] => {

  return once(`tg|${c.plan.id}|${seg.i}|${c.S.id}|${c.sp}|${L.from}|${L.tool}`, () => {

    const nFrom = vis(c, L.from);
    const sc = stageCentre(c.S);
    const k = Math.min(T.targets, nFrom);
    const idx = Array.from({ length: nFrom }, (_, i) => i).sort((a, b) => Math.hypot(H[a].p[0] - sc[0], H[a].p[2] - sc[2]) - Math.hypot(H[b].p[0] - sc[0], H[b].p[2] - sc[2])).slice(0, k);
    const P = c.S.hero, z = T.zone;
    const avoid: Zone[] = [{ kind: "circle", c: [P[0], P[2]], r: 0.55 }];
    if (T.act === "vacuum") avoid.push({ kind: "circle", c: [P[0] + 0.82 * c.s, P[2] - 0.12 * c.s], r: 0.5 });
    if (T.act === "kipas") avoid.push({ kind: "circle", c: [P[0] + 0.75, P[2] + 0.3], r: 0.6 });
    if (T.act === "umpan") avoid.push({ kind: "circle", c: [P[0], P[2] + 1.85], r: 0.75 }, { kind: "rect", x0: P[0] - 0.3, x1: P[0] + 0.3, z0: P[2], z1: P[2] + 1.9 });

    return idx.slice(0, pre.length).map((hi, j) => {

  });

/** Small effortful tug when the tool jams ("stuck" backfire): hip + lean wobble after the impact. */
const tug = (x: ToolCtx, pose: Pose) => {
  if (!x.stuck || x.tau < x.T.impact) return pose;
  const k = Math.sin((x.tau - x.T.impact) * 2 * Math.PI * 1.6) * smooth(lin(x.tau, x.T.impact, x.T.impact + 0.15));
  return { ...pose, lean: (pose.lean ?? 0) - 0.09 * Math.max(0, k), hip: add(pose.hip ?? [0, 0, 0], [0, -0.012 * Math.abs(k), -0.04 * Math.max(0, k)]) };
};

export const TRUCK = { cannon: [0.72, 1.02, 0] as V3 };
export const cannonMuzzle = (p: V3, yaw: number, aim: number, cyaw = 0, k = 1): { at: V3; dir: V3 } => {
  const r = 0.31;
  const lx = k * (TRUCK.cannon[0] + Math.cos(aim) * Math.cos(cyaw) * r), ly = k * (TRUCK.cannon[1] + Math.sin(aim) * r), lz = -k * Math.cos(aim) * Math.sin(cyaw) * r;
  const c = Math.cos(yaw), s = Math.sin(yaw);
  const at: V3 = [p[0] + lx * c + lz * s, p[1] + ly, p[2] - lx * s + lz * c];
  const dl: V3 = [Math.cos(aim) * Math.cos(cyaw), Math.sin(aim), -Math.cos(aim) * Math.sin(cyaw)];
  return { at, dir: [dl[0] * c + dl[2] * s, dl[1], -dl[0] * s + dl[2] * c] };
};
export const landing = (from: V3, vel: V3, floor = 0): V3 => {
  const a = -4.9, b = vel[1], cc = from[1] - floor;
  const T = (-b - Math.sqrt(Math.max(0, b * b - 4 * a * cc))) / (2 * a);
  return [from[0] + vel[0] * T, floor, from[2] + vel[2] * T];
};

// ------------------------------------------------------------------ LEVEL scene
const ARRIVE_DUR = 0.9;
const levelScene = (c: Ctx, seg: Seg, t: number): Scene => {
  const L = seg.level!, T = TOOLS[L.tool];
  const u = t - seg.t0, F = L.flip - seg.t0;
  const tau = tauOf(u, F, T.flip);
  const P = c.S.hero;
  const tg = targetsOf(c, seg, T);
  const stuck = L.backfire === "stuck";
  const x: ToolCtx = { tau, t: u, F, c, seg, T, P, tg, stuck };
  const out = ACTS[T.act](x);
  // map tool-time events to film time
  const real = (τ: number) => seg.t0 + (F >= T.flip ? τ + (F - T.flip) : τ < T.flip ? (τ * F) / T.flip : F + (τ - T.flip));
  const flipT = L.flip;
  const H = homes(c.S, c.sp, c.partner);
  const nFrom = vis(c, L.from), nTo = vis(c, L.to);
  const crowd: Inst[] = [];
  const puffs: Puff[] = [];
  const tset = new Set(tg.map((g) => g.i));
  for (let i = 0; i < nFrom; i++) if (!tset.has(i)) crowd.push(H[i]);
  const src = out.src ?? out.point;
  // targets react
  const dodge = L.backfire === "dodge";
  tg.forEach((g, j) => {
    const home = g.home;
    const hit = real(T.impact) + (T.effect === "spray" ? 0.12 + j * 0.08 : j * 0.06);
    const sw = out.swallow?.(j);
    let inst: Inst = { ...home, p: g.pre, yaw: g.yaw, s: home.s, eye: 0.5 };
    if (stuck) {
      // the tool jams: they just watch, then go home with the arrivals
      const h = hop(g.pre, home.p, t, flipT + 0.1 + j * 0.05, 0.4, 0.3);
      inst = { ...home, p: h.p, yaw: mix(g.yaw, home.yaw, smooth(h.u)), squash: h.squash, eye: t < flipT ? 0.42 : 1.05 };
    } else if (sw) {
      const a = real(sw[0]), b = real(sw[1]);
      if (t < a) inst = { ...inst, eye: 0.5 };
      else if (t < b) {
        const e = easeIn(lin(t, a, b));
        inst = { ...home, p: mix3(g.pre, sw[2], e), yaw: g.yaw + e * 2.5, s: (home.s ?? 1) * (1 - e), squash: -0.4 * Math.sin(e * Math.PI), eye: 1.1 };
      } else {
        // re-emitted with the burst
        const t0 = flipT + 0.02 * j;
        if (t < t0) return;
        const h = hop(src, home.p, t, t0, 0.62, 2.0);
        inst = { ...home, p: h.p, s: (home.s ?? 1) * smooth(lin(t, t0, t0 + 0.4)), squash: h.squash, yaw: home.yaw + (1 - smooth(h.u)) * 3, eye: h.u < 1 ? 1.1 : home.eye };
        if (h.u >= 1 && t < t0 + 0.62 + 0.35) puffs.push({ p: home.p, t0: t0 + 0.62, r: 0.2 });
      }
    } else if (T.effect === "lure") {
      const bowl = out.point;
      const a = (j / Math.max(1, tg.length)) * Math.PI * 2 + 0.4;
      const ring: V3 = [bowl[0] + Math.sin(a) * 0.46, 0, bowl[2] + Math.cos(a) * 0.46];
      const t0 = real(T.impact) + j * 0.1;
      const h = hop(g.pre, ring, t, t0, 0.45, 0.25);
      inst = { ...home, p: h.p, yaw: t > t0 ? yawTo(ring, bowl) : g.yaw, squash: h.squash, eye: t > t0 + 0.45 ? 0.42 : 1 };
    } else if (T.effect === "scare" || T.effect === "fly") {
      const t0 = real(T.impact) + j * (T.effect === "fly" ? 0.08 : 0.05);
      const up = hop(g.pre, g.pre, t, t0, T.effect === "fly" ? 0.42 : 0.36, T.effect === "fly" ? 0.55 : 0.4);
      const go = hop(g.pre, home.p, t, flipT + 0.08 + j * 0.05, 0.42, 0.35);
      inst = t < flipT + 0.08 + j * 0.05 ? { ...home, p: up.p, yaw: g.yaw, squash: up.squash, eye: t > t0 ? 1.15 : 0.6 } : { ...home, p: go.p, yaw: mix(g.yaw, home.yaw, smooth(go.u)), squash: go.squash, eye: 1.05 };
    } else if (T.effect === "blow") {
      const t0 = real(T.impact) + j * 0.06;
      const pushed: V3 = add(g.pre, [-0.15, 0, 0.75]);
      const e = smooth(lin(t, t0, t0 + 0.5));
      const go = hop(pushed, home.p, t, flipT + 0.1 + j * 0.05, 0.45, 0.4);
      inst = t < flipT + 0.1 + j * 0.05 ? { ...home, p: mix3(g.pre, pushed, e), yaw: g.yaw, roll: Math.sin(e * Math.PI * 2) * 0.35 * (1 - e * 0.5), eye: t > t0 ? 0.3 : 0.6 } : { ...home, p: go.p, yaw: mix(g.yaw, home.yaw, smooth(go.u)), squash: go.squash, eye: 1 };
    } else if (T.effect === "cover") {
      const t0 = real(T.impact) + j * 0.06;
      const back: V3 = add(g.pre, [0, 0, 0.45]);
      const h1 = hop(g.pre, back, t, t0, 0.32, 0.25);
      const h2 = hop(back, home.p, t, flipT + 0.15 + j * 0.05, 0.42, 0.35);
      inst = t < flipT + 0.15 + j * 0.05 ? { ...home, p: h1.p, yaw: g.yaw, squash: h1.squash, eye: t > t0 ? 1.1 : 0.5 } : { ...home, p: h2.p, yaw: mix(g.yaw, home.yaw, smooth(h2.u)), squash: h2.squash, eye: 1 };
    } else {
      // hit / spray: hop away to their home (dodge = a beat earlier)
      const t0 = dodge ? Math.max(seg.t0 + 0.05, real(Math.max(0.05, T.impact - 0.36)) + j * 0.05) : hit;
      const h = hop(g.pre, home.p, t, t0, 0.42, 0.42);
      inst = { ...home, p: h.p, yaw: mix(g.yaw, home.yaw, smooth(h.u)), squash: h.squash, eye: h.u > 0 && h.u < 1 ? 1.1 : t > t0 ? home.eye : 0.45 };
      if (dodge && j < 3) {
        // the copy left behind
        const ct = t0 + 0.12;
        if (t > ct && j + nFrom < nTo) puffs.push({ p: g.pre, t0: ct, r: 0.22 });
      }
    }
    crowd.push(inst);
  });
  // impact puff
  if (!stuck && (T.effect === "hit" || T.effect === "catch" || T.effect === "trap")) puffs.push({ p: out.point, t0: real(T.impact), r: 0.3 });
  // arrivals
  const style = L.arrive;
  const arr = H.slice(nFrom, nTo);
  const sp = SPECIES[c.sp];
  const d0 = arr.map((a) => Math.hypot(a.p[0] - src[0], a.p[2] - src[2]));
  const dmax = Math.max(1, ...d0);
  const order = once(`ord|${c.plan.id}|${seg.i}|${style}`, () => {
    if (style !== "burst") return arr.map((_, i) => i);
    const ang = arr.map((a, i) => ({ i, a: Math.atan2(a.p[2] - src[2], a.p[0] - src[0]) })).sort((p, q) => p.a - q.a).map((v) => v.i);
    const per = 4, vol = Math.ceil(ang.length / per);
    const o = new Array(ang.length);
    ang.forEach((_, k) => { o[k] = ang[(k % per) * vol + Math.floor(k / per)]; });
    return o.filter((v) => v !== undefined) as number[];
  });
  const rank = new Map<number, number>();
  order.forEach((ai, k) => rank.set(ai, k));
  arr.forEach((a, i) => {
    const stacked = a.p[1] > 0.05 && a.p[1] < 0.6 && style !== "rain";
    const k = rank.get(i) ?? i;
    let t0: number;
    if (style === "burst") t0 = flipT + Math.floor(k / 4) * Math.min(0.12, 0.9 / Math.max(1, arr.length / 4)) + (k % 4) * 0.02;
    else if (style === "rain") t0 = flipT + hash(a.seed * 1.7) * Math.min(ARRIVE_DUR, 0.25 + arr.length * 0.02);
    else t0 = flipT - 0.08 + Math.min(ARRIVE_DUR, (d0[i] / dmax) * Math.min(ARRIVE_DUR, 0.2 + arr.length * 0.03)) + hash(a.seed) * 0.08 + (stacked ? 0.25 : 0);
    if (t < t0) return;
    if (style === "burst") {
      const dur = 0.66 + hash(i) * 0.06;
      const h = hop(src, a.p, t, t0, dur, 2.4 + hash(i * 3) * 0.4);
      crowd.push({ ...a, p: h.p, yaw: a.yaw + (1 - smooth(h.u)) * 3, s: (a.s ?? 1) * smooth(lin(t, t0, t0 + dur * 0.55)), squash: h.squash, eye: h.u < 1 ? 1.1 : a.eye });
      if (t > t0 + dur && t < t0 + dur + 0.35) puffs.push({ p: a.p, t0: t0 + dur, r: 0.22 });
    } else if (style === "rain") {
      const dur = 0.55, fall = clamp((t - t0) / dur), h = 5.0 * (1 - fall * fall);
      const landed = fall >= 1;
      const sq = landed ? 0.45 * (1 - smooth(lin(t, t0 + dur, t0 + dur + 0.22))) : -0.25 * fall;
      crowd.push({ ...a, p: [a.p[0], a.p[1] + h, a.p[2]], squash: sq, eye: landed ? a.eye : 1.15, roll: landed ? 0 : 0.3 * Math.sin(i) * (1 - fall), pitch: landed ? a.pitch : 0, s: (a.s ?? 1) * smooth(lin(t, t0, t0 + 0.14)) });
      if (landed && t < t0 + dur + 0.35 && hash(a.seed * 1.3) < 0.5) puffs.push({ p: a.p, t0: t0 + dur, r: 0.2 });
    } else if (style === "march") {
      const out2 = norm([a.p[0] - P[0], 0, a.p[2] - P[2] + 0.001]);
      const from: V3 = [a.p[0] + out2[0] * 3.2, a.p[1], a.p[2] + out2[2] * 3.2];
      const h = hop(from, a.p, t, t0, 0.65, 0.6);
      crowd.push({ ...a, p: h.p, yaw: h.u < 1 ? yawTo(from, a.p) : a.yaw, squash: h.squash, eye: h.u < 1 ? 1.05 : a.eye, s: (a.s ?? 1) * smooth(lin(t, t0, t0 + 0.16)) });
    } else if (style === "climb" && a.p[1] > 0.3) {
      const from: V3 = [a.p[0], 0, a.p[2] + 0.5];
      const h = hop(from, a.p, t, t0, 0.42, 0.45);
      crowd.push({ ...a, p: h.p, squash: h.squash, eye: h.u < 1 ? 1.05 : a.eye, s: (a.s ?? 1) * smooth(lin(t, t0, t0 + 0.12)) });
    } else {
      crowd.push({ ...a, s: popScale(t, t0) * (a.s ?? 1) });
      if (hash(a.seed * 1.3) < (arr.length > 60 ? 0.35 : 0.8)) puffs.push({ p: a.p, t0, r: arr.length > 60 ? 0.17 : 0.24 });
    }
  });
  // chase: the crowd turns on the hero
  if (L.backfire === "chase" && t > flipT + 0.3) {
    const k = smooth(lin(t, flipT + 0.3, flipT + 0.7));
    for (let i = 0; i < crowd.length; i++) {
      const cI = crowd[i];
      if (Math.hypot(cI.p[0] - P[0], cI.p[2] - P[2]) > 4.5 || cI.p[1] > 0.6) continue;
      const bob = 0.03 * (1 - Math.cos((t - flipT) * 9 + cI.seed)) * k;
      crowd[i] = { ...cI, yaw: cI.yaw + wrap(yawTo(cI.p, P) - cI.yaw) * k, p: [cI.p[0], cI.p[1] + bob, cI.p[2]], eye: 1.1 };
    }
  }
  // hero
  const heroBase = out.hero ?? {};
  let pose = out.pose;
  pose.lean = (pose.lean ?? 0) + breathe(t);
  let faceKeys = out.faceKeys.map((k) => ({ t: real(k.t), e: k.e }));
  if (stuck) faceKeys = [...faceKeys.filter((k) => k.t < real(T.impact)), { t: real(T.impact) + 0.1, e: "angry" as Expr }, { t: real(T.impact) + 0.5, e: "worried" as Expr }];
  const react: Expr = L.backfire === "splash" ? "defeated" : L.backfire === "chase" ? "worried" : "shock";
  faceKeys.push({ t: flipT - 0.1, e: L.backfire === "splash" ? "shock" : react }, ...(L.backfire === "splash" ? [{ t: flipT + 0.55, e: "defeated" as Expr }] : []));
  faceKeys.sort((a, b) => a.t - b.t);
  if (L.backfire === "chase" && !T.vehicle) {
    const k = smooth(lin(t, flipT + 0.25, flipT + 0.75));
    pose = { ...pose, hip: add(pose.hip ?? [0, 0, 0], [0, 0, -0.45 * k]), footL: add(pose.footL ?? [0.1, 0, 0.02], [0, 0, -0.42 * k]), footR: add(pose.footR ?? [-0.1, 0, 0.02], [0, 0, -0.48 * k]), lean: (pose.lean ?? 0) - 0.12 * k };
    if (k > 0) { pose.handL = mix3(pose.handL, [0.24, 1.32, 0.18], k); pose.handR = mix3(pose.handR, [-0.24, 1.32, 0.18], k); pose.poleL = [1, -0.4, -0.2]; pose.poleR = [-1, -0.4, -0.2]; }
  }
  if (L.backfire === "splash") {
    const k = smooth(lin(t, flipT, flipT + 0.25));
    pose = { ...pose, lean: (pose.lean ?? 0) - 0.1 * k };
  }
  let face = faceAt(faceKeys, t);
  face = withTalk(face, talk(seg, c.plan.hero.id, t));
  const tools = out.tools.slice();
  const heroPos = heroBase.pos ?? P;
  if (L.backfire === "splash" && t > flipT) {
    const head = toWorld(heroPos, heroBase.yaw ?? 0, c.s, headTop(pose, c.plan.hero.look));
    if (T.effect === "spray" || T.act === "kipas") tools.push({ kind: "drips", p: head, t0: flipT + 0.15 });
    puffs.push({ p: add(head, [0, -0.35, 0]), t0: flipT, r: 0.45 });
    if (T.effect === "spray") tools.push({ kind: "water", from: add(head, [0.2, 1.4, 0.9]), vel: [-0.3, -1.0, -1.4], on: smooth(lin(t, flipT - 0.05, flipT + 0.05)) * (1 - smooth(lin(t, flipT + 0.25, flipT + 0.4))), n: 40, r: 0.05, seed: 9 });
  }
  if (L.backfire === "chase" && t > flipT + 0.25 && !["lure", "suck", "catch", "trap"].includes(T.effect)) {
    for (const g of tg.slice(0, 3)) {
      const i = crowd.findIndex((q) => q.seed === g.home.seed);
      if (i < 0) continue;
      const near: V3 = add(P, [(hash(g.i) - 0.5) * 0.9, 0, 0.75 + hash(g.i * 3) * 0.3]);
      const h = hop(g.home.p, near, t, flipT + 0.35 + g.i * 0.01, 0.45, 0.35);
      crowd[i] = { ...crowd[i], p: h.p, squash: h.squash, yaw: yawTo(near, P), eye: 1.1 };
    }
  }
  void sp;
  return {
    hero: { pos: heroPos, yaw: heroBase.yaw ?? 0, pose, face, blink: blinkAt(t, 0.4 + seg.i), seated: heroBase.seated },
    partner: partnerActor(c, seg, t), crowd, puffs, tools,
  };
};

// ------------------------------------------------------------------ PEAK: close on the hero, the crowd keeps growing
const peakScene = (c: Ctx, seg: Seg, t: number): Scene => {
  const pk = seg.peak!;
  const u = t - seg.t0;
  const veh = pk.vehicle;
  const ln = laneOf(c.S);
  const tp: V3 = [ln.stopX, 0, ln.z];
  const R = veh ? rideOf(c) : null;
  const pos = R ? riderPos(tp, R) : c.S.hero;
  const yaw = veh ? Math.PI / 2 : 0;
  const base = R ? wheelPose(R) : { handL: [0.3, 0.6, 0.06] as V3, handR: [-0.3, 0.6, 0.06] as V3, poleL: [0.7, -0.4, -0.5] as V3, poleR: [-0.7, -0.4, -0.5] as V3 };
  // in the truck the leans stay inside RIDE_LEAN (the wheel fit holds for that range)
  const lk = R ? 0.45 : 1;
  const pose = poseTrack([
    { t: 0, p: { ...base, lean: -0.07 * lk, nod: -0.1, turn: 0.15 } },
    { t: 0.8, p: { ...base, lean: -0.06 * lk, nod: -0.06 } },
    { t: 1.05, p: { ...base, lean: -0.04 * lk } },
    { t: 1.75, p: { ...base, lean: R ? 0.08 : 0.1, nod: 0.12 } },
  ], u);
  const face0 = faceAt([{ t: 0, e: "shock" }, { t: 1.08, e: "worried" }, { t: 1.55, e: "defeated" }], u);
  const face = withTalk({ ...face0, lookY: u > 1.0 && u < 1.55 ? 0.7 : face0.lookY }, talk(seg, c.plan.hero.id, t));
  const H = homes(c.S, c.sp, c.partner);
  const nFrom = vis(c, pk.from), nTo = vis(c, pk.to);
  const crowd: Inst[] = H.slice(0, nFrom);
  const puffs: Puff[] = [];
  const arr = H.slice(nFrom, nTo);
  arr.forEach((a, i) => {
    const t0 = pk.flip - 0.05 + (i / Math.max(1, arr.length)) * 0.9 + (a.p[1] > 0.05 ? 0.15 : 0);
    if (t < t0) return;
    crowd.push({ ...a, s: popScale(t, t0) * (a.s ?? 1) });
    if (i % 3 === 0) puffs.push({ p: a.p, t0, r: 0.18 });
  });
  const tools: ToolDesc[] = R ? [{ kind: "truck", p: tp, yaw: 0, spin: 0, aim: 0.32, cyaw: 0.3, ride: R }] : [];
  return { hero: { pos, yaw, pose, face, blink: u > 1.55 ? 0 : blinkAt(t, 0.2), seated: R ? "chair" : undefined }, partner: partnerActor(c, seg, t), crowd, puffs, tools };
};

// ------------------------------------------------------------------ ENDINGS
const napSpot = (S: SetMeta) => ({ x: S.hero[0] + 0.4, z: S.hero[2] + 0.9 });
/** The sleeping hero lies on his back along z: feet toward the cameras (+z), head away, face to the sky with a small
 *  nod toward his chest — so every camera in front sees the sleeping face upright, never rotated 90°. */
export const NAP_NOD = 0.42;
export const napBody = (c: Ctx, N: { x: number; z: number }, bed: number) => {
  const s = c.s, belly = c.plan.hero.look.build?.belly ?? 1;
  const rootZ = N.z + 0.72 * s, backY = bed + BD.torsoR[2] * belly * 0.98 * s;
  const headLocalY = BD.waistY + BD.headC;
  const head: V3 = [N.x, backY + 0.02 * s, rootZ - headLocalY * s];
  return { rootZ, backY, head, s, belly };
};
export const napLayout = (c: Ctx, total: number) =>
  once(`nap|${c.plan.id}|${c.S.id}|${c.sp}|${total}`, () => {
    const S = c.S, sp = c.sp, N = napSpot(S);
    const H = homes(S, sp, c.partner);
    const def = SPECIES[sp];
    const n = vis(c, total);
    const moundN = Math.min(70, Math.max(6, Math.round(n * 0.12)));
    const keep0 = H.slice(0, Math.max(0, n - moundN - 1)).filter((h) => Math.hypot((h.p[0] - N.x) / 2.6, (h.p[2] - N.z) / 1.4) > 1 || h.p[1] > 0.6);
    const P = new Pile(sp, S.bounds.x0 - 4, S.bounds.x1 + 4, -14, 16, 0.035).boxes(S.tops);
    for (const h of keep0) P.place(h.pose, h.p[0], h.p[2], h.yaw, h.s ?? 1);
    const capH = def.mound * Math.min(1, 0.35 + n / 900);
    const cap = (x: number, z: number) => { const d = Math.hypot((x - N.x) / 2.4, (z - N.z) / 1.3); return capH * Math.max(0, 1 - d * d); };
    const avoid: Zone[] = c.partner ? [{ kind: "circle", c: [S.watch.p[0], S.watch.p[2]], r: 0.6 }] : [];
    const mound = P.drop({ cx: N.x, cz: N.z, rx: 2.4, rz: 1.3, n: moundN, seed: 102, facing: 0.4, yawJitter: 0.9, best: 2, power: 0.5, s: 0.9, cap, avoid });
    const s = c.s;
    // his body lies along z, the bed is the highest creature under it
    const bed = P.maxIn(N.x - 0.27 * s, N.x + 0.27 * s, N.z - 0.82 * s, N.z + 0.78 * s);
    const B = napBody(c, N, bed);
    // keep his head and face clear: nothing beside the head may rise above the bed (no tails or bodies over the face)
    const topOf = (q: Inst) => q.p[1] + shapeOf(sp, q.pose).height * (q.s ?? 1);
    const clear = (q: Inst) => Math.hypot(q.p[0] - B.head[0], q.p[2] - B.head[2]) > 0.62 * s + 0.15 || topOf(q) < bed + 0.04;
    const crowd = [...keep0, ...mound].filter(clear);
    // Si Oren lies ACROSS his legs (facing +x): its tail curls toward his feet, away from the face
    const thighZ = B.rootZ - 0.36 * s;
    const pest: Inst = { p: [N.x + 0.04 * s, B.backY + 0.235 * B.belly * s - 0.01, thighZ], yaw: Math.PI / 2, pose: 0, seed: 11, eye: 0.42, wag: 1.0, s: Math.min(1, s * 1.05) };
    return { crowd, pest, bed, N, B };
  });
const endingScene = (c: Ctx, seg: Seg, t: number): Scene => {
  const e = seg.ending!;
  const u = t - seg.t0;
  const S = c.S, P = S.hero;
  const n = vis(c, e.count);
  const H = homes(S, c.sp, c.partner);
  const partner = partnerActor(c, seg, t);
  const heroTalk = talk(seg, c.plan.hero.id, t);
  if (e.kind === "nap") {
    const Lay = napLayout(c, e.count);
    const breath = Math.sin(t * 2.0);
    // on his back, hands folded behind the head like a pillow, head nodded toward his chest (face toward the camera)
    const pose: Pose = { lean: 0.01 * breath, nod: NAP_NOD, turn: 0, tilt: 0, handL: [0.2, 1.3, -0.1], handR: [-0.2, 1.3, -0.1], poleL: [1, 0.3, -0.4], poleR: [-1, 0.3, -0.4], footL: [0.12, 0, 0], footR: [-0.12, 0, 0] };
    const B = Lay.B;
    const zzzAt: V3 = [B.head[0] + 0.18 * B.s, B.head[1] + 0.32 * B.s, B.head[2] - 0.12 * B.s];
    const pest = { ...Lay.pest, p: [Lay.pest.p[0], Lay.pest.p[1] + 0.008 * breath, Lay.pest.p[2]] as V3 };

  const starIdx = (text: string) => { const m = text.match(/\*([^*]+)\*/); if (!m) return -1; const before = plain(text.slice(0, m.index)).split(/\s+/).filter(Boolean).length; return before; };
  if (!voiced) {
    let t = 0;

      const ls = d.lines.filter((fi) => !flat[fi].title);
      const lines: PLine[] = [];

        segT.push({ t0: t, t1: t + D, lines });
        t += D;
      } else if (d.kind === "level") {
        const T = TOOLS[story.levels[d.li].tool];
        let punch = ls.findIndex((fi) => /\*[^*]+\*/.test(flat[fi].text) && !flat[fi].by);
        if (punch < 0) punch = ls.findIndex((fi) => /\*[^*]+\*/.test(flat[fi].text));
        if (punch < 0) punch = ls.length - 1;
        let a = 0.12;
        const pre = ls.slice(0, Math.max(0, punch));
        for (const fi of pre) { const dur = readDur(flat[fi].text); lines.push(mkLine(fi, t + a, t + a + dur, false)); a += dur + 0.08; }
        const F = Math.max(T.flip * 0.92, a + 0.12);
        let b = F - 0.06;
        for (const fi of ls.slice(Math.max(0, punch))) { const dur = readDur(flat[fi].text); lines.push(mkLine(fi, t + b, t + b + dur, false)); b += dur + 0.06; }
        const D = Math.max(T.min, b + 0.12, F + 0.95);
        segT.push({ t0: t, t1: t + D, flipRel: F, lines });
        t += D;
      } else if (d.kind === "peak") {
        let a = 0.75;
        for (const fi of ls) { const dur = readDur(flat[fi].text); lines.push(mkLine(fi, t + a, t + a + dur, false)); a += dur + 0.06; }
        const D = Math.max(2.2, a + 0.15);
        segT.push({ t0: t, t1: t + D, flipRel: 0.85, lines });
        t += D;
      } else {
        let a = 0.2;
        for (const fi of ls) { const dur = readDur(flat[fi].text) + 0.25; lines.push(mkLine(fi, t + a, t + a + dur, false)); a += dur + 0.08; }
        const D = Math.max(3.4, a + 1.7);
        segT.push({ t0: t, t1: t + D, lines });
        t += D;
      }

      const sp0 = d.lines.filter((fi) => flat[fi].vo).map((fi) => timed.get(fi)!);
      if (!sp0.length && d.kind !== "hook") errors.push(`${d.kind === "level" ? `level ${d.li + 1}` : d.kind}: no line is in the voice-over — every part needs at least one spoken line`);

    });

    drafts.forEach((d, di) => {
      const s = segT[di];
      let cursor = s.t0 + 0.1;
      for (const fi of d.lines) {
        if (flat[fi].title) continue;
        const tm = timed.get(fi);
        if (tm) { s.lines.push(mkLine(fi, tm.t0, Math.min(s.t1 - 0.04, tm.t1 + 0.25), true, tm.words.map((w, k) => ({ w: w.w, t0: w.t, t1: tm.words[k + 1]?.t ?? tm.t1 })))); cursor = tm.t1 + 0.05; }

      }
    });
  }

  const flipOf = (lines: PLine[], word: string | undefined, dflt: number, where: string, t0: number, t1: number, minGap = 0.55) => {

    if (f > t1 - 0.5) f = t1 - 0.5;
    return f;
  };
  const segs: Seg[] = [];
  let count = start;
  drafts.forEach((d, di) => {
    const s = segT[di];
    const seg: Seg = { kind: d.kind, i: di, t0: +s.t0.toFixed(3), t1: +s.t1.toFixed(3), lines: s.lines, count };
    if (d.kind === "level") {
      const L = story.levels[d.li], T = TOOLS[L.tool];
      const dflt = voiced ? (s.lines.filter((l) => !l.by).slice(-1)[0]?.t0 ?? s.lines.slice(-1)[0]?.t0 ?? s.t0 + T.flip) : s.t0 + (s.flipRel ?? T.flip);

      count = tos[d.li];

      const flip = flipOf(s.lines, undefined, s.t0 + 0.85, "peak", s.t0, s.t1, 0.2);
      const lastTool = story.levels[story.levels.length - 1]?.tool;
      seg.peak = { from: count, to: peakTo, flip: +flip.toFixed(3), vehicle: !!TOOLS[lastTool]?.vehicle };
      count = peakTo;
    } else if (d.kind === "ending") {
      const lastTool = story.levels[story.levels.length - 1]?.tool;
      seg.ending = { kind: story.ending.kind, part: (story.ending.part ?? `PART ${story.part ?? (st ? st.part + 1 : 1)}`).toUpperCase(), next: (story.ending.next ?? "BERSAMBUNG…").toUpperCase(), count, vehicle: !!TOOLS[lastTool]?.vehicle };
    }
    seg.count = d.kind === "level" ? seg.level!.from : d.kind === "peak" ? seg.peak!.from : count;
    segs.push(seg);
  });
  if (errors.length) throw new Error("the story cannot be built yet:\n  - " + errors.join("\n  - "));
  const duration = +segs[segs.length - 1].t1.toFixed(3);
  if (duration > 62) warnings.push(`the episode is ${duration.toFixed(0)} s — the genre lands best at 15–45 s`);
  for (const sg of segs) for (const l of sg.lines) if (plain(l.text).length > 54) warnings.push(`caption "${plain(l.text).slice(0, 40)}…" is long for a phone (≤ 54 letters; split it into two lines)`);
  const keys = [{ t: 0, n: start }];
  for (const sg of segs) {
    if (sg.level && sg.level.to !== sg.level.from) keys.push({ t: sg.level.flip, n: sg.level.to });
    if (sg.peak && sg.peak.to !== sg.peak.from) keys.push({ t: sg.peak.flip, n: sg.peak.to });
  }
  const label = (story.counter?.label ?? SPECIES[sp].id).toUpperCase();
  const plan: Plan = {
    id: story.id, title: titleLines(story), set: S0.id, dressing: story.dressing?.length ? story.dressing : undefined, light: story.time ?? S.light ?? "siang", aspect: o.aspect, duration,
    hero: { id: heroId!, name: nameOf(heroId)!, look: heroLook }, partner: partnerId && partnerLook ? { id: partnerId, name: nameOf(partnerId)!, look: partnerLook } : undefined,
    pest: { id: pestId!, name: nameOf(pestId)!, look: pestLook as CreatureLook },
    counter: { label, keys }, drawn: { end: Math.max(...keys.map((k) => k.n)), cap: capHere }, part: segs[segs.length - 1].ending?.part ?? "PART 1", segs, shots: [], cues: [], audio: o.audio, voiced, warnings,
  };
  // ---------------------------------------------------------------- shots
  const shots: PShot[] = [];
  segs.forEach((sg) => {
    const D = sg.t1 - sg.t0;
    if (sg.kind === "hook" && D > 4.4) { const m = sg.t0 + Math.max(2.2, D * 0.5); shots.push(draft(shots.length, sg.t0, m, sg.i, "hook"), draft(shots.length + 1, m, sg.t1, sg.i, "hook2")); }
    else if (sg.kind === "hook") shots.push(draft(shots.length, sg.t0, sg.t1, sg.i, "hook"));
    else if (sg.kind === "level" && D > 3.7 && sg.level!.flip - 0.3 - sg.t0 > 1.3 && sg.t1 - (sg.level!.flip - 0.3) > 1.3) {
      const m = sg.level!.flip - 0.3;
      shots.push(draft(shots.length, sg.t0, m, sg.i, "attempt"), draft(shots.length + 1, m, sg.t1, sg.i, "result"));
    } else shots.push(draft(shots.length, sg.t0, sg.t1, sg.i, sg.kind === "level" ? "level" : sg.kind === "peak" ? "peak" : "end"));
  });
  plan.shots = shots;
  // cameras (they need the scene, which needs the plan)
  let prevAz: number | undefined;
  for (const sh of shots) {
    const r = solve(plan, sh, prevAz, o.fast);
    sh.a = r.a; sh.b = r.b; sh.hold = r.hold; sh.vis = +r.vis.toFixed(2);
    sh.shadowAt = [r.a.target[0], 0, r.a.target[2]];
    sh.shadowBox = Math.max(5, Math.min(16, Math.hypot(r.a.pos[0] - r.a.target[0], r.a.pos[2] - r.a.target[2]) * 0.9 + 2.5));
    const hz = HIDE_Z[S0.id];
    const napEnd = sh.role === "end" && segs[sh.seg].ending?.kind === "nap";
    if (hz !== undefined && !napEnd && r.a.pos[2] > hz && r.b.pos[2] > hz - 0.3) sh.hide = hz;
    if (r.az !== undefined) prevAz = r.az;
    if (r.vis < 0.75) warnings.push(`shot ${sh.i + 1} (${sh.role}, ${sh.t0.toFixed(1)} s): the camera could not see the hero clearly — check the still`);
  }
  if (plan.segs.some((g) => g.level && TOOLS[g.level.tool]?.vehicle) && !rideOk(plan.hero.look))
    warnings.push(`${plan.hero.name} has no steering-wheel fit in the ride-on truck for this build (belly / head) — use a calmer build or another tool`);
  placeCaptions(plan, warnings);
  plan.cues = cuesOf(plan);
  return plan;
};

const smoothK = (x: number) => { const t = Math.max(0, Math.min(1, x)); return t * t * t * (t * (t * 6 - 15) + 10); };
export const camOf = (sh: PShot, t: number): Cam => {
  const k = smoothK((t - sh.t0 - (sh.hold ?? 0)) / Math.max(0.01, sh.t1 - sh.t0 - (sh.hold ?? 0)));
  const m = (a: V3, b: V3): V3 => [a[0] + (b[0] - a[0]) * k, a[1] + (b[1] - a[1]) * k, a[2] + (b[2] - a[2]) * k];
  return { pos: m(sh.a.pos, sh.b.pos), target: m(sh.a.target, sh.b.target), fov: sh.a.fov + (sh.b.fov - sh.a.fov) * k };
};
/** Screen rectangles (design px) that a caption must not cover at time t: heads (hero, partner), the tool's working end. */
export const keepClear = (plan: Plan, t: number, cam: Cam): { who: string; r: Rect }[] => {
  const wide = plan.aspect > 1, { W, H } = frameOf(wide);
  const f = norm3(sub3(cam.target, cam.pos)), right = norm3(cross3(f, [0, 1, 0])), upv = cross3(right, f);
  const px = (p: V3) => { const q = projectPt(p, cam.pos, cam.target, cam.fov, plan.aspect); return { x: (q.x * 0.5 + 0.5) * W, y: (1 - (q.y * 0.5 + 0.5)) * H, d: q.depth }; };
  const onScreen = (r: Rect) => r.x1 > 0 && r.x0 < W && r.y1 > 0 && r.y0 < H;
  const disc = (c: V3, rad: number, who: string) => {
    const a = px(c);
    if (a.d < 0.15) return null;
    const e = px(add3(c, [upv[0] * rad, upv[1] * rad, upv[2] * rad]));
    const R = Math.hypot(e.x - a.x, e.y - a.y);
    const r = { x0: a.x - R, x1: a.x + R, y0: a.y - R * 1.05, y1: a.y + R * 1.2 };
    return onScreen(r) ? { who, r } : null;
  };
  const sc = sceneAt(plan, t);
  const out: { who: string; r: Rect }[] = [];
  const headOf = (a: NonNullable<Scene["hero"]>, look: PersonLook): V3 => {
    const s = look.build?.s ?? 1;
    if (a.rot) return [a.pos[0], a.pos[1] + 0.02 * s, a.pos[2] - (BD.waistY + BD.headC) * s];
    return toWorld(a.pos, a.yaw, s, headCentre(a.pose, look));
  };
  if (sc.hero) { const l = plan.hero.look; const d = disc(headOf(sc.hero, l), BD.headR * (l.build?.s ?? 1) * (l.build?.head ?? 1) * 1.3, plan.hero.name); if (d) out.push(d); }
  if (sc.partner && plan.partner) { const l = plan.partner.look; const d = disc(headOf(sc.partner, l), BD.headR * (l.build?.s ?? 1) * (l.build?.head ?? 1) * 1.3, plan.partner.name); if (d) out.push(d); }
  const seg = plan.segs[segIdx(plan, t)];
  if (seg.kind === "level") {
    for (const d of sc.tools) {
      const pts = workingEnd(d);
      if (!pts.length) continue;
      const ps = pts.map(px).filter((q) => q.d > 0.15);
      if (!ps.length) continue;
      const r = { x0: Math.min(...ps.map((q) => q.x)) - 60, x1: Math.max(...ps.map((q) => q.x)) + 60, y0: Math.min(...ps.map((q) => q.y)) - 60, y1: Math.max(...ps.map((q) => q.y)) + 60 };
      if (onScreen(r)) out.push({ who: "the tool", r });
    }
  }
  return out;
};
/** The part of a tool a caption must not hide (not long handles or set-size vehicles). */
const workingEnd = (d: ToolDesc): V3[] => {
  switch (d.kind) {
    case "broom": return [[d.top[0] + d.dir[0] * 1.25, d.top[1] + d.dir[1] * 1.25, d.top[2] + d.dir[2] * 1.25]];
    case "net": return [[d.grip[0] + d.dir[0] * 1.6, d.grip[1] + d.dir[1] * 1.6, d.grip[2] + d.dir[2] * 1.6]];
    case "umbrella": return [[d.grip[0] + d.dir[0] * 0.8, d.grip[1] + d.dir[1] * 0.8, d.grip[2] + d.dir[2] * 0.8]];
    case "box": return [[d.p[0] - 0.46, d.p[1], d.p[2] - 0.46], [d.p[0] + 0.46, d.p[1] + 0.62, d.p[2] + 0.46], [d.p[0] - 0.46, d.p[1] + 0.62, d.p[2] + 0.46], [d.p[0] + 0.46, d.p[1], d.p[2] - 0.46]];
    case "vacuum": return [d.can, [d.can[0], d.can[1] + 0.55, d.can[2]], d.handle];
    case "bucket": case "bowl": case "toa": case "drone": return [d.p];
    case "flag": return [[d.grip[0], d.grip[1] + 0.9, d.grip[2]]];
    default: return [];
  }
};
const segIdx = (plan: Plan, t: number) => { let i = 0; for (let k = 0; k < plan.segs.length; k++) if (t >= plan.segs[k].t0 - 1e-6) i = k; return i; };
export const shotIdx = (plan: Plan, t: number) => { let i = 0; for (let k = 0; k < plan.shots.length; k++) if (t >= plan.shots[k].t0 - 1e-6) i = k; return i; };
type Keep = { who: string; r: Rect };
const lineSpan = (seg: Seg, ln: PLine) => { const t0 = Math.max(ln.t0, seg.t0), t1 = Math.min(ln.t1, seg.t1); return { t0, t1 }; };
const lineSamples = (t0: number, t1: number) => [0.04, 0.25, 0.5, 0.75, 0.96].map((k) => t0 + (t1 - t0) * k);
/** Everything a caption line must stay off over its whole span: faces + tool (projected per sample), the hook title, the end card. */
const keepsOf = (plan: Plan, seg: Seg, ln: PLine): Keep[] => {
  const wide = plan.aspect > 1, { t0, t1 } = lineSpan(seg, ln);
  const out = lineSamples(t0, t1).flatMap((t) => keepClear(plan, t, camOf(plan.shots[shotIdx(plan, t)], t)));
  const hook = plan.segs[0];
  if (t0 < Math.min(hook.t1 - 0.05, 2.25)) out.push({ who: "the title", r: titleRect(plan.title, wide) });
  const end = plan.segs[plan.segs.length - 1];
  if (end.ending && t1 > end.t0 + endCardAt(end)) {
    const k = end.ending.kind;
    out.push({ who: "the end card", r: partCardRect(k === "nap" || k === "adopt" ? (wide ? 300 : 430) : wide ? 640 : 1180, wide) });
  }
  return out;
};
/** Nearest free height for a line (cost 0), else the least-covered one. */
const bestY = (plan: Plan, seg: Seg, ln: PLine, keeps: Keep[]) => {
  const wide = plan.aspect > 1;
  const h0 = captionRect(ln.text, ln.who, 0, wide);
  let best = { y: captionHome(wide, seg.kind), cost: Infinity, who: "" };
  for (const y of captionYs(wide, seg.kind, h0.y1 - h0.y0)) {
    const boxes = captionBoxes(ln.text, ln.who, y, wide);
    const hit = keeps.filter((k) => boxes.some((b) => overlaps(b, k.r, 6)));
    if (!hit.length) return { y, cost: 0, who: "" };
    const cost = hit.reduce((acc, k) => acc + boxes.reduce((s0, b) => s0 + overlapArea(b, k.r), 0) + 1, 0);
    if (cost < best.cost) best = { y, cost, who: hit[0].who };
  }
  return best;
};
/** A gentle reframe of one shot: tilt the aim (rad, + looks up) and widen the lens (×fov). Camera position unchanged, so
 *  walls, hides and the lens-clear zone stay as solved. */
const reframed = (c: Cam, tilt: number, zoom: number): Cam => {
  const f = norm3(sub3(c.target, c.pos)), right = norm3(cross3(f, [0, 1, 0])), up = cross3(right, f);
  const d = Math.hypot(c.target[0] - c.pos[0], c.target[1] - c.pos[1], c.target[2] - c.pos[2]);
  const g: V3 = [f[0] * Math.cos(tilt) + up[0] * Math.sin(tilt), f[1] * Math.cos(tilt) + up[1] * Math.sin(tilt), f[2] * Math.cos(tilt) + up[2] * Math.sin(tilt)];
  return { ...c, target: [c.pos[0] + g[0] * d, c.pos[1] + g[1] * d, c.pos[2] + g[2] * d], fov: Math.min(62, c.fov * zoom) };
};
const REFRAMES: [number, number][] = [[0.05, 1], [-0.05, 1], [0, 1.12], [0.1, 1], [-0.1, 1], [0.05, 1.12], [-0.05, 1.12], [0.1, 1.12], [-0.1, 1.12], [0, 1.25], [0.08, 1.25], [-0.08, 1.25]];
/** Per caption line: the nearest height whose boxes (text + speaker chip) cover no face, no tool head (levels), no title
 *  or end card. A shot whose lines find no free height is reframed (small tilt / wider lens) when that frees every line
 *  in it and keeps the hero's head well inside the frame. Still covered → plan warning. */
const placeCaptions = (plan: Plan, warnings: string[]) => {
  const wide = plan.aspect > 1, { W, H } = frameOf(wide);
  const lines = plan.segs.flatMap((seg) => seg.lines.map((ln) => ({ seg, ln }))).filter(({ seg, ln }) => { const s0 = lineSpan(seg, ln); return s0.t1 > s0.t0; });
  const blocked = (L: typeof lines) => L.filter(({ seg, ln }) => bestY(plan, seg, ln, keepsOf(plan, seg, ln)).cost > 0);
  const shotsOf = (seg: Seg, ln: PLine) => { const { t0, t1 } = lineSpan(seg, ln); return [...new Set(lineSamples(t0, t1).map((t) => shotIdx(plan, t)))]; };
  const heroInside = (i: number) => {
    const sh = plan.shots[i];
    for (const k of [0.1, 0.5, 0.9]) {
      const t = sh.t0 + (sh.t1 - sh.t0) * k;
      const hero = keepClear(plan, t, camOf(sh, t)).find((c) => c.who === plan.hero.name);
      if (hero && (hero.r.x0 < 30 || hero.r.x1 > W - 30 || hero.r.y0 < captionTop(wide) || hero.r.y1 > H - 120)) return false;
    }
    return true;
  };
  const todo = [...new Set(blocked(lines).flatMap(({ seg, ln }) => shotsOf(seg, ln)))];
  for (const i of todo) {
    const sh = plan.shots[i], a0 = sh.a, b0 = sh.b;
    const mine = lines.filter(({ seg, ln }) => shotsOf(seg, ln).includes(i));
    const before = blocked(mine).length;
    if (!before) continue;
    const wasInside = heroInside(i);
    let done = false;
    for (const [tilt, zoom] of REFRAMES) {
      sh.a = reframed(a0, tilt, zoom); sh.b = reframed(b0, tilt, zoom);
      if ((!wasInside || heroInside(i)) && blocked(mine).length === 0) { done = true; sh.reframe = { tilt, zoom }; break; }
    }
    if (!done) { sh.a = a0; sh.b = b0; }
  }
  for (const { seg, ln } of lines) {
    const pick = bestY(plan, seg, ln, keepsOf(plan, seg, ln));
    ln.y = pick.y;
    if (pick.cost > 0) warnings.push(`caption "${plain(ln.text).slice(0, 36)}" (${ln.t0.toFixed(1)} s) still covers ${pick.who}${/^the /.test(pick.who) ? "" : "'s face"} at every height — shorten the line or give that beat a calmer camera`);
  }
};

};

// ------------------------------------------------------------------ geometry caches
const G = {
  sphere: new SphereGeometry(1, 32, 20),
  sphereLo: new SphereGeometry(1, 16, 12),
  torso: new SphereGeometry(1, 48, 32),
  cyl: new CylinderGeometry(1, 1, 1, 18, 1),
  hairShort: new SphereGeometry(BD.headR * 1.04, 40, 20, 0, Math.PI * 2, 0, 1.12),
  hairRing: new SphereGeometry(BD.headR * 1.035, 40, 10, Math.PI - 0.35, Math.PI + 0.7, 1.0, 0.62),
  hairCap: new SphereGeometry(BD.headR * 1.04, 40, 20, 0, Math.PI * 2, 0, 1.25),
  kerCap: new SphereGeometry(BD.headR * 1.09, 40, 16, 0, Math.PI * 2, 0, 0.98),
  kerBand: new SphereGeometry(BD.headR * 1.09, 40, 16, Math.PI / 2 + 0.92, Math.PI * 2 - 1.84, 0.95, 1.35),
  topi: new SphereGeometry(BD.headR * 1.06, 40, 16, 0, Math.PI * 2, 0, 1.2),
  caping: new ConeGeometry(0.4, 0.2, 40, 1, true),
  waistRoll: new TorusGeometry(0.232, 0.03, 10, 40),
  ring: new TorusGeometry(0.042, 0.007, 8, 24),
  spike: new ConeGeometry(0.05, 0.11, 10),
};
const rbox = new Map<string, RoundedBoxGeometry>();
export const rb = (w: number, h: number, d: number, r: number) => {
  const k = `${w}|${h}|${d}|${r}`;
  let g = rbox.get(k);
  if (!g) {
    g = new RoundedBoxGeometry(w, h, d, 3, Math.min(r, w / 2 - 1e-3, h / 2 - 1e-3, d / 2 - 1e-3));
    rbox.set(k, g);
  }
  return g;
};
const latheCache = new Map<string, LatheGeometry>();
const lathe = (key: string, pts: [number, number][]) => {
  let g = latheCache.get(key);
  if (!g) {
    g = new LatheGeometry(pts.map(([r, y]) => new Vector2(r, y)), 40);
    latheCache.set(key, g);
  }
  return g;
};
/** Sarung / skirt / shorts tube from hem height `bottom` to the waist. */
const skirtGeo = (bottom: number, flare = 0) =>
  lathe(`sk|${bottom}|${flare}`, [
    [0.205 + flare, bottom], [0.243 + flare, bottom + 0.004], [0.248 + flare * 0.7, bottom + 0.12], [0.262, 0.36], [0.262, 0.47], [0.245, 0.6], [0.228, 0.665],
  ]);
/** Seated hips of a sarung / skirt: from the seat line up to the waist, never dipping below the seat (the standing
 *  profile's knee-level point would fold down through a seat). */
const seatedHipGeo = () =>
  lathe("seatedhip", [[0.2, 0.43], [0.24, 0.436], [0.26, 0.47], [0.262, 0.52], [0.252, 0.58], [0.24, 0.62], [0.228, 0.665]]);
const capeGeo = (b: number) => lathe(`cape|${b}`, [[0.29 * b, 0.2], [0.305 * b, 0.25], [0.275 * b, 0.33], [0.17 * b, 0.4], [0.1, 0.44]]);
/** Hip piece of trousers / shorts (the legs are tubes). */
const hipGeo = () => lathe("hip", [[0.19, 0.5], [0.215, 0.515], [0.238, 0.58], [0.24, 0.62], [0.228, 0.665]]);
const smileGeo = new Map<string, TubeGeometry>();
const smileLine = (smile: number, w: number) => {
  const s = Math.round(clamp(smile, -1, 1) * 6) / 6, ww = Math.round(w * 10) / 10;
  const k = `${s}|${ww}`;
  let g = smileGeo.get(k);
  if (!g) {
    const hw = 0.045 * ww;
    const c = new QuadraticBezierCurve3(new Vector3(-hw, 0.012 * s, 0), new Vector3(0, -0.05 * s, 0), new Vector3(hw, 0.012 * s, 0));
    g = new TubeGeometry(c, 16, 0.0085, 6, false);
    smileGeo.set(k, g);
  }
  return g;
};

// ------------------------------------------------------------------ torso material: top painted in the shader
const bodyMats = new Map<string, MeshStandardMaterial>();
const KIND_ID: Record<string, number> = { singlet: 0, kaos: 1, kemeja: 2, daster: 3 };
const torsoMat = (skin: string, shirt: string, shirt2: string, kind: string, stripes: boolean) => {
  const k = `${skin}|${shirt}|${shirt2}|${kind}|${stripes}`;
  let m = bodyMats.get(k);
  if (m) return m;
  m = clay("#ffffff").clone();
  const u = { uSkin: { value: new Color(skin) }, uShirt: { value: new Color(shirt) }, uShirt2: { value: new Color(shirt2) }, uKind: { value: KIND_ID[kind] ?? 1 }, uStripes: { value: stripes ? 1 : 0 } };
  m.onBeforeCompile = (sh) => {
    Object.assign(sh.uniforms, u);
    sh.vertexShader = sh.vertexShader
      .replace("#include <common>", "#include <common>\nvarying vec3 vObj;")
      .replace("#include <begin_vertex>", "#include <begin_vertex>\nvObj = position;");
    sh.fragmentShader = sh.fragmentShader
      .replace("#include <common>", "#include <common>\nvarying vec3 vObj; uniform vec3 uSkin; uniform vec3 uShirt; uniform vec3 uShirt2; uniform float uKind; uniform float uStripes;")
      .replace(
        "#include <color_fragment>",
        `#include <color_fragment>
        {
          vec3 p = normalize(vObj);
          float ax = abs(p.x);
          float skinA = 0.0;
          if (uKind < 0.5) {
            float scoop = p.z > 0.0 ? 0.42 + 1.5 * ax * ax : 0.66 + 1.0 * ax * ax;
            float neck = (ax < 0.36 && p.y > scoop) ? 1.0 : 0.0;
            float arm = length(vec3(ax - 0.93, p.y - 0.5, p.z * 0.9)) < 0.5 ? 1.0 : 0.0;
            skinA = max(neck, arm);
          } else if (uKind < 2.5) {
            skinA = (ax < 0.34 && p.y > 0.8 + 0.6 * ax * ax && p.z > -0.2) ? 1.0 : 0.0;
          } else {
            skinA = (ax < 0.3 && p.y > 0.84 + 0.6 * ax * ax && p.z > -0.2) ? 1.0 : 0.0;
          }
          vec3 shirt = uShirt;
          if (uStripes > 0.5) {
            float s = sin(p.y * 22.0);
            float aa = fwidth(s) * 1.2;
            shirt = mix(uShirt, uShirt2, smoothstep(0.35 - aa, 0.35 + aa, s));
          }
          if (uKind > 1.5 && uKind < 2.5) {
            float placket = (ax < 0.035 && p.z > 0.0 && p.y < 0.8) ? 1.0 : 0.0;
            shirt = mix(shirt, uShirt2, placket);
          }
          diffuseColor.rgb = mix(shirt, uSkin, skinA);
        }`,
      );
  };
  m.customProgramCacheKey = () => "nekat-torso2";
  bodyMats.set(k, m);
  return m;
};

// ------------------------------------------------------------------ helpers
const UP = new Vector3(0, 1, 0);
const tmpQ = new Quaternion();
const Bone: React.FC<{ a: V3; b: V3; r: number; mat: MeshStandardMaterial; caps?: boolean }> = ({ a, b, r, mat, caps = true }) => {
  const va = new Vector3(...a), vb = new Vector3(...b);
  const d = vb.clone().sub(va);
  const L = Math.max(1e-4, d.length());
  const q = tmpQ.clone().setFromUnitVectors(UP, d.normalize());
  const mid = va.clone().add(vb).multiplyScalar(0.5);
  return (
    <>
      <mesh geometry={G.cyl} material={mat} position={mid} quaternion={q} scale={[r, L, r]} castShadow receiveShadow />
      {caps ? (
        <>
          <mesh geometry={G.sphereLo} material={mat} position={va} scale={r} castShadow receiveShadow />
          <mesh geometry={G.sphereLo} material={mat} position={vb} scale={r} castShadow receiveShadow />
        </>
      ) : null}
    </>
  );
};
const OnHead: React.FC<{ lon: number; lat: number; r: number; spin?: number; children: React.ReactNode }> = ({ lon, lat, r, spin = 0, children }) => (
  <group rotation={new Euler(-lat, lon, 0, "YXZ")}>
    <group position={[0, 0, r]} rotation={[0, 0, spin]}>
      {children}
    </group>
  </group>
);

export type PersonProps = {
  look: PersonLook;
  pose: Pose;
  face: Face;
  blink?: number;
  pos?: V3;
  rot?: V3;
  /** "hidden": everything below the waist hidden (inside a vehicle); "chair": thighs forward, shins down */
  seated?: "hidden" | "chair" | false;
  children?: React.ReactNode;
};

export const Person: React.FC<PersonProps> = ({ look, pose, face, blink = 0, pos = [0, 0, 0], rot = [0, 0, 0], seated = false, children }) => {
  const R = BD.headR;
  const hs = look.build?.head ?? 1;
  const scale = look.build?.s ?? 1;
  const belly = look.build?.belly ?? 1;
  const hairColor = look.hair?.color ?? "#231c18";
  const m = useMemo(() => {
    const b = look.bottom;
    const sh = b.kind === "sarung" ? plaid(b.color, b.color2 ?? "#2c4f69", b.color3 ?? "#e7c66b", [5, 2]) : null;
    return {
      skin: clay(look.skin),
      shade: clay(new Color(look.skin).multiplyScalar(0.86).getStyle()),
      hair: clay(hairColor, { rough: 0.9 }),
      stache: clay(hairColor, { rough: 0.95, bump: 0.8 }),
      torso: torsoMat(look.skin, look.top.color, look.top.color2 ?? "#ffffff", look.top.kind, !!look.top.stripes),
      shirt: clay(look.top.color),
      shirt2: clay(look.top.color2 ?? look.top.color),
      bottom: sh ? clay("#ffffff", { map: sh, rough: 0.92 }) : clay(b.color, { rough: 0.9 }),
      band: clay(b.color2 ?? b.color, { rough: 0.9 }),
      sole: clay(look.feet?.color ?? "#4c8fd0"),
      strap: clay(look.feet?.color2 ?? "#f2f2f0"),
      eye: clay("#1d1816", { rough: 0.35, bump: 0 }),
      white: clay("#fbfaf6", { rough: 0.5, bump: 0 }),
      mouth: clay("#4a1f22", { rough: 0.9, bump: 0 }),
      tongue: clay("#d86a6f", { rough: 0.8, bump: 0 }),
      hat: clay(look.hat?.color ?? (look.hat?.kind === "caping" ? "#d9be7c" : "#222"), { rough: 0.85, double: look.hat?.kind === "caping" }),
      glass: clay("#2b2626", { rough: 0.5, bump: 0 }),
      blush: clay("#ec9a95", { rough: 0.9, bump: 0 }),
      towel: clay(look.towel ?? "#ffffff"),
    };
  }, [look, hairColor]);

  const hip = pose.hip ?? [0, 0, 0];
  const P = new Vector3(hip[0], BD.waistY + hip[1], hip[2]);
  const qT = new Quaternion().setFromEuler(new Euler(pose.lean ?? 0, pose.twist ?? 0, pose.side ?? 0, "YXZ"));
  const tl = (x: number, y: number, z: number): V3 => {
    const v = new Vector3(x, y, z).applyQuaternion(qT).add(P);
    return [v.x, v.y, v.z];
  };
  const tr: V3 = [BD.torsoR[0] * belly, BD.torsoR[1], BD.torsoR[2] * belly];
  const shL = tl(BD.shX * belly, BD.shY, -0.01), shR = tl(-BD.shX * belly, BD.shY, -0.01);
  const head = tl(0, BD.headC + headLift(look), 0.0);
  const qH = qT.clone().multiply(new Quaternion().setFromEuler(new Euler(pose.nod ?? 0, pose.turn ?? 0, pose.tilt ?? 0, "YXZ")));
  const poleL = pose.poleL ?? [0.7, -0.4, -0.5], poleR = pose.poleR ?? [-0.7, -0.4, -0.5];
  const [elL, haL] = ik2(shL, pose.handL, BD.upper, BD.fore, poleL);
  const [elR, haR] = ik2(shR, pose.handR, BD.upper, BD.fore, poleR);
  const footL = pose.footL ?? [BD.footX, 0, 0.02], footR = pose.footR ?? [-BD.footX, 0, 0.02];
  const sleeve = look.top.kind !== "singlet";

  const hand = (at: V3, from: V3, side: 1 | -1) => {
    const d = new Vector3(at[0] - from[0], at[1] - from[1], at[2] - from[2]).normalize();
    const q = new Quaternion().setFromUnitVectors(UP, d);
    return (
      <group position={at} quaternion={q}>
        <mesh geometry={G.sphere} material={m.skin} scale={[BD.handR, BD.handR * 1.05, BD.handR * 0.95]} castShadow receiveShadow />
        <mesh geometry={G.sphereLo} material={m.skin} position={[-side * 0.05, -0.005, 0.035]} scale={[0.028, 0.034, 0.028]} castShadow receiveShadow />
      </group>
    );
  };
  const foot = (f: V3) =>
    look.feet?.kind === "shoes" ? (
      <group position={f}>
        <mesh geometry={rb(0.13, 0.03, 0.26, 0.012)} material={m.strap} position={[0, 0.015, 0.02]} castShadow receiveShadow />
        <mesh geometry={rb(0.12, 0.085, 0.24, 0.04)} material={m.sole} position={[0, 0.06, 0.02]} castShadow receiveShadow />
      </group>
    ) : (
      <group position={f}>
        <mesh geometry={rb(0.13, 0.026, 0.255, 0.012)} material={m.sole} position={[0, 0.013, 0.02]} castShadow receiveShadow />
        <mesh geometry={rb(0.1, 0.065, 0.2, 0.03)} material={m.skin} position={[0, 0.058, 0.015]} castShadow receiveShadow />
        <mesh geometry={rb(0.018, 0.014, 0.1, 0.006)} material={m.strap} position={[0.028, 0.093, 0.04]} rotation={[0.15, 0.62, 0]} castShadow />
        <mesh geometry={rb(0.018, 0.014, 0.1, 0.006)} material={m.strap} position={[-0.028, 0.093, 0.04]} rotation={[0.15, -0.62, 0]} castShadow />
        <mesh geometry={G.sphereLo} material={m.strap} position={[0, 0.09, 0.085]} scale={0.012} />
      </group>
    );

  // lower body
  const bk = look.bottom.kind;
  let lower: React.ReactNode = null;
  if (seated === "chair") {
    const wy = BD.waistY + hip[1];
    const knee = (s: number): V3 => [s * 0.11 + hip[0], wy - 0.16, hip[2] + 0.36];
    const thigh = (s: number): V3 => [s * 0.1 + hip[0], wy - 0.14, hip[2]];
    const cover = bk === "sarung" || bk === "skirt";
    lower = (
      <>
        {/* thighs: covered by the sarung / skirt lap, or legs (pants / shorts) */}
        {cover ? (
          <>
            <group position={[hip[0], 0, hip[2]]} scale={[belly, 1, 0.88 * belly]}>
              <mesh geometry={seatedHipGeo()} material={m.bottom} position={[0, hip[1], 0]} castShadow receiveShadow />
            </group>
            <mesh geometry={rb(0.46 * belly, 0.16, 0.5, 0.07)} material={m.bottom} position={[hip[0], wy - 0.12, hip[2] + 0.17]} castShadow receiveShadow />
            <mesh geometry={rb(0.44 * belly, 0.3, 0.12, 0.05)} material={m.bottom} position={[hip[0], wy - 0.3, hip[2] + 0.38]} castShadow receiveShadow />
          </>
        ) : (
          <>
            <Bone a={thigh(1)} b={knee(1)} r={0.07} mat={bk === "shorts" || bk === "pants" ? m.bottom : m.skin} />
            <Bone a={thigh(-1)} b={knee(-1)} r={0.07} mat={bk === "shorts" || bk === "pants" ? m.bottom : m.skin} />
            <mesh geometry={rb(0.46 * belly, 0.22, 0.34, 0.1)} material={m.bottom} position={[hip[0], wy - 0.08, hip[2] - 0.02]} castShadow receiveShadow />
          </>
        )}
        <Bone a={knee(1)} b={[footL[0], footL[1] + 0.07, footL[2] - 0.035]} r={0.05} mat={bk === "pants" ? m.bottom : m.skin} />
        <Bone a={knee(-1)} b={[footR[0], footR[1] + 0.07, footR[2] - 0.035]} r={0.05} mat={bk === "pants" ? m.bottom : m.skin} />
        {foot(footL)}
        {foot(footR)}
      </>
    );
  } else if (!seated) {
    const legged = bk === "pants" || bk === "shorts";
    const tube = bk === "sarung" ? skirtGeo(BD.hemY) : bk === "skirt" ? skirtGeo(0.075, 0.03) : hipGeo();
    const hipJ = (s: number): V3 => [s * BD.footX * 0.92 + hip[0], 0.56 + hip[1] * 0.6, hip[2]];
    const ankle = (f: V3): V3 => [f[0], f[1] + 0.07, f[2] - 0.035];
    const kneeAt = (s: number, f: V3): V3 => { const a = hipJ(s), b = ankle(f); return [mix(a[0], b[0], 0.48), mix(a[1], b[1], 0.48), mix(a[2], b[2], 0.48) + 0.012]; };
    lower = (
      <>
        <group position={[hip[0], hip[1], hip[2]]} scale={[belly, 1, 0.88 * belly]}>
          <mesh geometry={tube} material={m.bottom} castShadow receiveShadow />
          {bk === "sarung" ? <mesh geometry={G.waistRoll} material={m.bottom} position={[0, 0.655, 0]} rotation={[Math.PI / 2, 0, 0]} castShadow /> : null}
          {look.bottom.color2 && bk !== "sarung" ? <mesh geometry={G.waistRoll} material={m.band} position={[0, 0.645, 0]} rotation={[Math.PI / 2, 0, 0]} scale={[1, 1, 0.6]} castShadow /> : null}
        </group>
        {legged ? (
          <>
            {bk === "pants" ? (
              <>
                <Bone a={hipJ(1)} b={ankle(footL)} r={0.07} mat={m.bottom} />
                <Bone a={hipJ(-1)} b={ankle(footR)} r={0.07} mat={m.bottom} />
              </>
            ) : (
              <>
                <Bone a={hipJ(1)} b={kneeAt(1, footL)} r={0.085} mat={m.bottom} />
                <Bone a={hipJ(-1)} b={kneeAt(-1, footR)} r={0.085} mat={m.bottom} />
                <Bone a={kneeAt(1, footL)} b={ankle(footL)} r={0.048} mat={m.skin} />
                <Bone a={kneeAt(-1, footR)} b={ankle(footR)} r={0.048} mat={m.skin} />
              </>
            )}
          </>
        ) : (
          <>
            <Bone a={[BD.footX * 0.9 + hip[0], 0.42 + hip[1] * 0.5, hip[2]]} b={ankle(footL)} r={0.048} mat={m.skin} />
            <Bone a={[-BD.footX * 0.9 + hip[0], 0.42 + hip[1] * 0.5, hip[2]]} b={ankle(footR)} r={0.048} mat={m.skin} />
          </>
        )}
        {foot(footL)}
        {foot(footR)}
      </>
    );
  }

  const eyeH = face.eyeS * (1 - 0.88 * clamp(blink));
  const eyeLat = 0.06 + face.lookY * 0.06, eyeLon = 0.36;
  const lookOff = face.lookX * 0.06;
  const mouthO = face.mouthOpen;
  const hk = look.hat?.kind ?? "none";
  const hairStyle = hk === "kerudung" ? "none" : look.hair?.style ?? "none";
  const f = look.face ?? {};
  const browR = f.brows === "bushy" ? 0.024 : f.brows === "thin" ? 0.013 : 0.019;

  return (
    <group position={pos} rotation={new Euler(rot[0], rot[1], rot[2], "YXZ")} scale={scale}>
      <mesh geometry={G.torso} material={m.torso} position={tl(0, BD.torsoC, 0)} quaternion={qT} scale={tr} castShadow receiveShadow />
      {sleeve ? (
        <>
          <Bone a={shL} b={[mix(shL[0], elL[0], 0.5), mix(shL[1], elL[1], 0.5), mix(shL[2], elL[2], 0.5)]} r={BD.armR + 0.014} mat={m.shirt} />
          <Bone a={shR} b={[mix(shR[0], elR[0], 0.5), mix(shR[1], elR[1], 0.5), mix(shR[2], elR[2], 0.5)]} r={BD.armR + 0.014} mat={m.shirt} />
        </>
      ) : null}
      {look.top.kind === "kemeja" ? (
        <>
          <mesh geometry={rb(0.12, 0.05, 0.11, 0.02)} material={m.shirt} position={tl(0.075, BD.shY + 0.07, 0.06)} quaternion={qT.clone().multiply(new Quaternion().setFromEuler(new Euler(0.5, 0, -0.5)))} castShadow />
          <mesh geometry={rb(0.12, 0.05, 0.11, 0.02)} material={m.shirt} position={tl(-0.075, BD.shY + 0.07, 0.06)} quaternion={qT.clone().multiply(new Quaternion().setFromEuler(new Euler(0.5, 0, 0.5)))} castShadow />
        </>
      ) : null}
      <Bone a={shL} b={elL} r={BD.armR} mat={m.skin} />
      <Bone a={elL} b={haL} r={BD.armR * 0.92} mat={m.skin} />
      <Bone a={shR} b={elR} r={BD.armR} mat={m.skin} />
      <Bone a={elR} b={haR} r={BD.armR * 0.92} mat={m.skin} />
      {hand(haL, elL, 1)}
      {hand(haR, elR, -1)}
      {look.towel ? <mesh geometry={rb(0.12, 0.02, 0.5, 0.01)} material={m.towel} position={tl(0.13, 0.39, 0)} quaternion={qT} castShadow /> : null}
      {hk === "kerudung" ? <mesh geometry={capeGeo(belly)} material={m.hat} position={[P.x, P.y, P.z]} quaternion={qT} castShadow receiveShadow /> : null}
      {lower}

      {/* head */}
      <group position={head} quaternion={qH} scale={hs}>
        <mesh geometry={G.sphere} material={m.skin} scale={[R * 1.02, R, R * 0.98]} castShadow receiveShadow />
        <OnHead lon={Math.PI / 2} lat={0.02} r={R * 0.96}><mesh geometry={G.sphereLo} material={m.shade} scale={[0.042, 0.058, 0.03]} castShadow /></OnHead>
        <OnHead lon={-Math.PI / 2} lat={0.02} r={R * 0.96}><mesh geometry={G.sphereLo} material={m.shade} scale={[0.042, 0.058, 0.03]} castShadow /></OnHead>
        {hairStyle === "short" ? <mesh geometry={G.hairShort} material={m.hair} rotation={[-0.5, 0, 0]} castShadow /> : null}
        {hairStyle === "ring" ? <mesh geometry={G.hairRing} material={m.hair} castShadow /> : null}
        {hairStyle === "spiky" ? (
          <>
            <mesh geometry={G.hairShort} material={m.hair} rotation={[-0.5, 0, 0]} castShadow />
            {[[-0.07, -0.25], [0.07, -0.25], [0, 0.0], [-0.1, 0.2], [0.1, 0.2], [0, -0.45]].map(([x, a], i) => (
              <mesh key={i} geometry={G.spike} material={m.hair} position={[x, R * 0.95 - Math.abs(x) * 0.4, -0.03 + a * 0.12]} rotation={[a - 0.35, 0, -x * 3]} castShadow />
            ))}
          </>
        ) : null}
        {hairStyle === "bun" ? (
          <>
            <mesh geometry={G.hairCap} material={m.hair} rotation={[-0.45, 0, 0]} castShadow />
            <mesh geometry={G.sphere} material={m.hair} position={[0, R * 0.55, -R * 0.85]} scale={0.08} castShadow />
          </>
        ) : null}

export type HouseSpec = {
  /** centre x of the lot, facade line z (world), +1 faces +z (south row), −1 faces −z (north row) */
  x: number; z: number; face: 1 | -1;
  w: number; h: number; d: number;

const prism = (w: number, depth: number, rise: number) => {
  // triangle (gable end) in the z–y plane, extruded along x
  const s = new Shape();
  s.moveTo(-depth / 2, 0);
  s.lineTo(depth / 2, 0);
  s.lineTo(0, rise);
  s.closePath();
  const g = new ExtrudeGeometry(s, { depth: w, bevelEnabled: false });
  g.rotateY(Math.PI / 2);
  return g;
};
const prismCache = new Map<string, ExtrudeGeometry>();
const gable = (w: number, d: number, rise: number) => {
  const k = `${w}|${d}|${rise}`;
  let g = prismCache.get(k);
  if (!g) {
    g = prism(w, d, rise);
    prismCache.set(k, g);
  }
  return g;
};

export const Dressing: React.FC<{ d: import("../dressing.ts").Dress }> = ({ d }) => (
  <group position={[d.at[0], 0, d.at[1]]} rotation={[0, ((d.yaw ?? 0) * Math.PI) / 180, 0]}>
    {d.parts.map((q, i) => {
      const p = q.pos ?? [0, 0, 0];
      const rot = (q.rot ?? [0, 0, 0]).map((v) => (v * Math.PI) / 180) as V3;
      if (q.shape === "box") return <B key={i} p={p} s={q.size!} c={q.color} r={Math.min(0.04, Math.min(...q.size!) / 4)} rot={rot} />;
      if (q.shape === "cyl") return <mesh key={i} geometry={cylGeoD} material={clay(q.color)} position={p} rotation={rot} scale={[q.r!, q.h ?? q.r! * 2, q.r!]} castShadow receiveShadow />;
      if (q.shape === "cone") return <mesh key={i} geometry={coneGeo} material={clay(q.color)} position={p} rotation={rot} scale={[q.r!, q.h ?? q.r! * 2, q.r!]} castShadow receiveShadow />;
      return <mesh key={i} geometry={sph} material={clay(q.color)} position={p} rotation={rot} scale={q.r} castShadow receiveShadow />;
    })}
  </group>
);
const cylGeoD = new CylinderGeometry(1, 1, 1, 20, 1);
