// NEKAT sound from a Plan — original synth, no samples, no meme sounds.
// Music: a bouncy, mischievous "challenge show" bed (staccato tuba-like bass, plucky marimba hook, woodblock + shaker)
// that ESCALATES: every level adds a layer and steps the key up a semitone; the peak drops to a stop-time hit; the
// ending gets its own mood (nap = music-box lullaby, adopt = warm resolve, run = fast chase, cliffhanger = low
// suspense, menyerah = slow falling marimba).
// SFX follow the plan cues: cut swishes, level-chip ting, each tool's own noise (broom swish, hose hiss, vacuum hum +
// burst, siren, fan, megaphone blare, party horn, umbrella fwump, drone buzz, box thump, bowl clink), hops, the counter
// tick + ×N ding, pop waves for the arrivals, little synth creature voices, splash, scramble, snore, the PART thump.
// Mix: music + SFX (+ voice ducking the music when there is a narrator) → −14 LUFS / −1.5 dBTP.
import fs from "node:fs";
import path from "node:path";
import { execFileSync, spawnSync } from "node:child_process";

export const buildSound = (plan, { voice, outDir, mixFile, music = true }) => {
  const DUR = plan.duration, SR = 48000, N = Math.ceil(DUR * SR);
  const mk = () => [new Float32Array(N), new Float32Array(N)];
  let seed = 9173;
  const rnd = () => (seed = (seed * 16807) % 2147483647) / 2147483647;
  const add = (buf, t, pan, len, fn) => {
    const i0 = Math.max(0, Math.round(t * SR)), n = Math.min(N - i0, Math.round(len * SR));
    const gl = Math.cos(((pan + 1) * Math.PI) / 4), gr = Math.sin(((pan + 1) * Math.PI) / 4);
    for (let i = 0; i < n; i++) { const v = fn(i / SR, i); buf[0][i0 + i] += v * gl; buf[1][i0 + i] += v * gr; }
  };
  const hz = (m) => 440 * Math.pow(2, (m - 69) / 12);
  const segAt = (t) => plan.segs.find((s) => t >= s.t0 && t < s.t1) ?? plan.segs[plan.segs.length - 1];

  // ---------------------------------------------------------------- music
  const mus = mk();
  if (music) {
    const pluck = (t, m, vel, pan, len = 0.5, bright = 0.35) => { const f = hz(m); add(mus, t, pan, len, (x) => (Math.sin(2 * Math.PI * f * x) * Math.exp(-x * 9) + bright * Math.sin(2 * Math.PI * f * 4 * x) * Math.exp(-x * 34)) * vel * Math.min(1, x / 0.002)); };
    const tuba = (t, m, vel, len = 0.22) => { const f = hz(m); add(mus, t, -0.05, len + 0.05, (x) => (Math.sin(2 * Math.PI * f * x) + 0.45 * Math.sin(4 * Math.PI * f * x) + 0.15 * Math.sin(6 * Math.PI * f * x)) * vel * Math.min(1, x / 0.008) * Math.exp(-x * 7) * Math.min(1, (len + 0.05 - x) / 0.04)); };
    const wood = (t, f, vel, pan) => add(mus, t, pan, 0.1, (x) => Math.sin(2 * Math.PI * f * x) * Math.exp(-x * 60) * vel);
    const shaker = (t, vel, pan) => { let lp = 0; add(mus, t, pan, 0.06, () => { const n = rnd() * 2 - 1; lp += (n - lp) * 0.75; return (n - lp) * vel; }); };
    // a hook motif in scale degrees (major), 8 eighths per bar, two bars
    const SCALE = [0, 2, 4, 5, 7, 9, 11, 12, 14];
    const HOOK = [4, -1, 4, 5, 7, -1, 5, 4, 2, -1, 2, 4, 5, 4, 2, 0];
    const CH = [[0, 4, 7], [5, 9, 12], [7, 11, 14], [0, 4, 7]];
    for (const sg of plan.segs) {
      const mood = sg.kind === "ending" ? sg.ending.kind : sg.kind;
      const lvl = sg.kind === "level" ? sg.level.n : sg.kind === "peak" ? 6 : 0;
      const root = 48 + Math.min(lvl, 5) + (sg.kind === "peak" ? 0 : 0);
      const bpm = mood === "run" ? 150 : mood === "nap" ? 76 : mood === "cliffhanger" ? 90 : mood === "menyerah" ? 84 : sg.kind === "level" ? 116 + lvl * 3 : 116;
      const beat = 60 / bpm, eighth = beat / 2;
      const fadeIn = (t) => Math.min(1, (t - sg.t0) / 0.05);
      const endFade = (t) => Math.min(1, Math.max(0, (DUR - 0.25 - t) / 1.4));
      if (sg.kind === "peak") {
        // stop-time: a held low note under the shock, a hit on the flip
        add(mus, sg.t0, 0, Math.min(1.0, sg.peak.flip - sg.t0 + 0.1), (x) => Math.sin(2 * Math.PI * hz(root - 12) * x) * 0.05 * Math.min(1, x / 0.05));
        for (const m of [root, root + 4, root + 7, root + 12]) add(mus, sg.peak.flip, 0, 1.2, (x) => Math.sin(2 * Math.PI * hz(m) * x) * 0.03 * Math.min(1, x / 0.01) * Math.exp(-x * 2.4));
        for (let k = 0; sg.peak.flip + 0.45 + k * eighth < sg.t1; k++) wood(sg.peak.flip + 0.45 + k * eighth, k % 2 ? 900 : 1300, 0.018, 0.2);
        continue;
      }
      if (mood === "nap") {
        const MEL = [7, 4, 2, 4, 0, -1, 2, 4];
        for (let k = 0; sg.t0 + k * beat < sg.t1; k++) {
          const t = sg.t0 + k * beat;
          pluck(t, 72 + SCALE[Math.max(0, MEL[k % 8] < 0 ? 0 : MEL[k % 8])], 0.03 * endFade(t), (k % 2 ? 0.3 : -0.3), 1.2, 0.6);
          if (k % 2 === 0) tuba(t, 36 + CH[Math.floor(k / 2) % 4][0], 0.03 * endFade(t), 0.5);
        }
        continue;
      }
      if (mood === "cliffhanger") {
        add(mus, sg.t0, 0, sg.t1 - sg.t0, (x) => (Math.sin(2 * Math.PI * hz(36) * (sg.t0 + x)) + 0.3 * Math.sin(2 * Math.PI * hz(43) * (sg.t0 + x))) * 0.03 * Math.min(1, x / 0.6) * endFade(sg.t0 + x));
        for (let k = 0; sg.t0 + k * beat < sg.t1; k++) wood(sg.t0 + k * beat, 700, 0.02 * endFade(sg.t0 + k * beat), 0.3);
        continue;
      }
      if (mood === "menyerah") {
        const MEL = [7, 6, 5, 4, 3, 2, 1, 0];
        for (let k = 0; sg.t0 + k * beat < sg.t1; k++) { const t = sg.t0 + k * beat; pluck(t, 64 + SCALE[MEL[Math.min(7, k)]], 0.04 * endFade(t), 0, 0.9); if (k % 2 === 0) tuba(t, 36 + (k % 4 ? 5 : 0), 0.035 * endFade(t), 0.4); }
        continue;
      }
      // hook / levels / adopt / run: the bouncy loop, thicker each level
      const layers = sg.kind === "hook" ? 1 : mood === "adopt" ? 2 : mood === "run" ? 4 : Math.min(4, lvl);
      let k = 0;
      for (let t = sg.t0; t < sg.t1 - 0.02; t += eighth, k++) {
        const bar = Math.floor(k / 8), e = k % 8;
        const ch = CH[bar % 4].map((x) => x + root);
        const g = fadeIn(t) * endFade(t);
        if (e % 2 === 0) tuba(t, ch[e % 4 === 0 ? 0 : 2] - 12, 0.08 * g);
        if (e === 2 || e === 6) wood(t, 1250, 0.035 * g, -0.3);
        if (layers >= 1) { const d = HOOK[k % 16]; if (d >= 0 && (sg.kind !== "hook" || k < 16)) pluck(t, root + 24 + SCALE[d], (e % 4 === 0 ? 0.06 : 0.045) * g, (rnd() - 0.5) * 0.6); }
        if (layers >= 2 && e % 2 === 1) pluck(t, ch[1] + 12, 0.022 * g, 0.4, 0.25);
        if (layers >= 3) shaker(t, 0.022 * g, 0.35);
        if (layers >= 4 && e % 4 === 3) pluck(t, root + 36 + SCALE[(k * 3) % 7], 0.026 * g, -0.4, 0.3, 0.6);
      }
      if (mood === "adopt") for (const m of [root + 12, root + 16, root + 19, root + 24]) add(mus, sg.t0 + 0.4, 0, 2.4, (x) => Math.sin(2 * Math.PI * hz(m) * x) * 0.016 * Math.min(1, x / 0.3) * Math.exp(-x * 0.8) * endFade(sg.t0 + 0.4 + x));
    }
  }

  // ---------------------------------------------------------------- SFX
  const sfx = mk();
  const noise = (t, len, gain, lpK, hpK, env, pan = 0) => { let lp = 0, hp = 0; add(sfx, t, pan, len, (x) => { const n = rnd() * 2 - 1; lp += (n - lp) * lpK; hp += (lp - hp) * hpK; return (lp - hp) * gain * env(x); }); };
  const tone = (t, f0, f1, len, gain, pan = 0, shape = "sin") => { let ph = 0; add(sfx, t, pan, len, (x) => { const f = f0 * Math.pow(f1 / f0, x / len); ph += (2 * Math.PI * f) / SR; const s = shape === "sq" ? Math.sign(Math.sin(ph)) * 0.6 : shape === "saw" ? ((ph / Math.PI) % 2) - 1 : Math.sin(ph); return s * gain * Math.min(1, x / 0.004) * Math.min(1, (len - x) / 0.03); }); };
  const swish = (t, len = 0.26, g = 0.12, pan = 0) => noise(t, len, g, 0.25, 0.04, (x) => Math.sin(Math.PI * Math.min(1, x / len)) ** 2, pan);
  const thud = (t, g = 0.16, f = 80) => add(sfx, t, 0, 0.35, (x) => Math.sin(2 * Math.PI * f * x * (1 + Math.exp(-x * 40))) * Math.exp(-x * 14) * g * Math.min(1, x / 0.003));
  const knock = (t, f = 420, g = 0.1) => add(sfx, t, 0.1, 0.18, (x) => (Math.sin(2 * Math.PI * f * x) * Math.exp(-x * 40) + (rnd() * 2 - 1) * 0.3 * Math.exp(-x * 120)) * g);
  const ding = (t, m = 88, g = 0.04) => { for (const [k, mm] of [m, m + 7, m + 12].entries()) add(sfx, t + k * 0.03, 0.15, 0.9, (x) => Math.sin(2 * Math.PI * hz(mm) * x) * (g - k * 0.008) * Math.min(1, x / 0.003) * Math.exp(-x * 5)); };
  const pop = (t, f = 800, g = 0.06, pan = 0) => add(sfx, t, pan, 0.09, (x) => Math.sin(2 * Math.PI * f * x * (1 + 1.2 * x / 0.09)) * Math.exp(-x * 45) * g * Math.min(1, x / 0.002));
  const boing = (t, f = 220, g = 0.07) => { let ph = 0; add(sfx, t, 0, 0.45, (x) => { ph += (2 * Math.PI * f * (1 + 0.6 * Math.sin(x * 38) * Math.exp(-x * 6)) * (1 + x)) / SR; return Math.sin(ph) * g * Math.exp(-x * 6) * Math.min(1, x / 0.004); }); };
  const hiss = (t, len, g = 0.05, pan = 0.2) => noise(t, len, g, 0.55, 0.12, (x) => Math.min(1, x / 0.08) * Math.min(1, (len - x) / 0.12), pan);
  const voiceOf = (sp, t, pitch) => {
    if (sp === "kucing") { let ph = 0, lp1 = 0, lp2 = 0; add(sfx, t, (rnd() - 0.5) * 0.6, 0.42, (x) => { const f = pitch * (1 + 0.35 * Math.sin(Math.PI * x / 0.42)); ph += (2 * Math.PI * f) / SR; const s = ((ph / Math.PI) % 2) - 1; const k1 = 0.05 + 0.25 * Math.sin(Math.PI * x / 0.42); lp1 += (s - lp1) * k1; lp2 += (lp1 - lp2) * 0.3; return lp2 * 0.11 * Math.sin(Math.PI * Math.min(1, x / 0.42)); }); }
    else if (sp === "ayam") { for (let k = 0; k < 3; k++) tone(t + k * 0.11, pitch * 1.6, pitch * 1.1, 0.07, 0.035, (rnd() - 0.5) * 0.6, "sq"); }
    else if (sp === "bebek") { tone(t, pitch * 1.2, pitch * 0.8, 0.16, 0.04, (rnd() - 0.5) * 0.6, "sq"); tone(t + 0.2, pitch * 1.25, pitch * 0.82, 0.14, 0.035, (rnd() - 0.5) * 0.6, "sq"); }
    else { let ph = 0, lp = 0; add(sfx, t, (rnd() - 0.5) * 0.6, 0.55, (x) => { const f = pitch * 0.75 * (1 + 0.06 * Math.sin(x * 2 * Math.PI * 9)); ph += (2 * Math.PI * f) / SR; const s = ((ph / Math.PI) % 2) - 1; lp += (s - lp) * 0.18; return lp * 0.09 * Math.sin(Math.PI * Math.min(1, x / 0.55)); }); }
  };
  const events = [];
  const ev = (t, k) => events.push(`${t.toFixed(2)} ${k}`);
  const sp = plan.pest.look.species;
  for (const c of plan.cues) {
    const t = c.t;
    if (c.k === "title") { tone(t, 330, 660, 0.18, 0.06); ding(t + 0.12, 84, 0.05); ev(t, "title"); }
    else if (c.k === "cut") swish(t - 0.1, 0.24, 0.07);
    else if (c.k === "chip") { /* the level chip slides in under the cut swish */ }
    else if (c.k.startsWith("tool:")) {
      const kind = c.k.slice(5), len = Math.max(0.3, c.v ?? 0.8);
      if (kind === "swish") { swish(t + 0.12, 0.22, 0.14, -0.2); }
      else if (kind === "water") hiss(t, len, 0.05);
      else if (kind === "splash") { swish(t, 0.2, 0.1); noise(t + 0.25, 0.4, 0.12, 0.6, 0.05, (x) => Math.exp(-x * 7)); thud(t + 0.25, 0.08, 120); }
      else if (kind === "vacuum") { let ph = 0; add(sfx, t, 0.2, len, (x) => { const f = 110 * (1 + 0.5 * Math.min(1, x / len)); ph += (2 * Math.PI * f) / SR; return (((ph / Math.PI) % 2) - 1) * 0.02 * Math.min(1, x / 0.15) * Math.min(1, (len - x) / 0.05); }); hiss(t, len, 0.03); thud(t + len - 0.05, 0.14, 70); pop(t + len - 0.04, 500, 0.08); }
      else if (kind === "siren") { const L = Math.min(1.6, len + 0.6); add(sfx, t - 0.6, 0.3, L, (x) => Math.sin(2 * Math.PI * (Math.floor(x / 0.32) % 2 ? 560 : 720) * x) * 0.03 * Math.min(1, x / 0.1) * Math.min(1, (L - x) / 0.2)); for (let k = 0; k < 8; k++) thud(t - 0.6 + k * 0.12, 0.03, 55); }
      else if (kind === "fan") { add(sfx, t, 0.3, len + 0.4, (x) => Math.sin(2 * Math.PI * 95 * x) * 0.02 * Math.min(1, x / 0.3) * Math.min(1, (len + 0.4 - x) / 0.2)); noise(t + 0.2, len + 0.2, 0.06, 0.12, 0.02, (x) => Math.min(1, x / 0.3) * Math.min(1, (len - x + 0.2) / 0.2), 0.2); }
      else if (kind === "clink") { knock(t + 0.1, 2300, 0.06); noise(t + 0.2, 0.35, 0.04, 0.4, 0.2, (x) => Math.exp(-x * 4)); }
      else if (kind === "toa") { let ph = 0, lp = 0; add(sfx, t + 0.25, 0, Math.min(0.85, len), (x) => { const f = 260 * (1 + 0.02 * Math.sin(x * 40)); ph += (2 * Math.PI * f) / SR; const s = Math.sign(Math.sin(ph)) + 0.5 * Math.sign(Math.sin(ph * 1.5)); lp += (s - lp) * 0.15; return lp * 0.03 * Math.min(1, x / 0.03) * Math.min(1, (Math.min(0.85, len) - x) / 0.1); }); }
      else if (kind === "horn") { tone(t + 0.25, 480, 620, Math.min(0.7, len), 0.035, 0, "saw"); }
      else if (kind === "fwump") { noise(t + 0.25, 0.25, 0.14, 0.08, 0.01, (x) => Math.exp(-x * 12)); thud(t + 0.25, 0.08, 90); }
      else if (kind === "drone") { let ph = 0; add(sfx, t - 0.2, -0.2, len + 0.6, (x) => { ph += (2 * Math.PI * 190) / SR; return (((ph / Math.PI) % 2) - 1) * 0.018 * (0.7 + 0.3 * Math.sin(x * 2 * Math.PI * 23)) * Math.min(1, x / 0.2) * Math.min(1, (len + 0.6 - x) / 0.2); }); }
      else if (kind === "thump") { swish(t + 0.1, 0.25, 0.08); }
      ev(t, kind);
    }
    else if (c.k === "impact") { knock(t, 380, 0.11); thud(t, 0.08, 110); }
    else if (c.k === "slurp") { tone(t, 300, 1400, 0.5, 0.03); }
    else if (c.k === "hop") { boing(t, 300, 0.05); }
    else if (c.k === "stuck") { const n = Math.max(2, Math.min(5, Math.round((c.v ?? 1) / 0.6))); for (let k = 0; k < n; k++) tone(t + k * 0.62, 900, 1100, 0.12, 0.03, 0, "sin"); }
    else if (c.k === "flip") { knock(t, 1600, 0.05); ding(t + 0.06, 86 + Math.min(10, Math.round(Math.log2(Math.max(2, c.v ?? 2)) * 2)), 0.045); ev(t, `flip x${c.v}`); }
    else if (c.k === "arrive") {
      const n = c.n ?? 0, style = c.v ?? 0;
      const audible = Math.min(12, n);
      for (let k = 0; k < audible; k++) {
        const tt = t + (k / Math.max(1, audible)) * Math.min(0.9, 0.15 + n * 0.03) + rnd() * 0.04;
        if (style === 2) { tone(tt, 1400, 500, 0.3, 0.012, (rnd() - 0.5) * 0.8); thud(tt + 0.3, 0.03, 140); }
        else if (style === 3) knock(tt, 700 + rnd() * 300, 0.03);
        else pop(tt + (style === 1 ? 0.25 : 0), 600 + rnd() * 600, 0.045, (rnd() - 0.5) * 0.8);
      }
      if (style === 1) { noise(t, 0.3, 0.12, 0.1, 0.01, (x) => Math.exp(-x * 9)); thud(t, 0.12, 70); }
      if (n > 40) noise(t + 0.1, 1.0, 0.035, 0.3, 0.05, (x) => Math.sin(Math.PI * Math.min(1, x / 1.0)), 0);
      
    }
    else if (c.k.startsWith("voice:")) { const nV = c.n ?? 1; for (let k = 0; k < nV; k++) voiceOf(sp, t + k * 0.16 + rnd() * 0.05, 520 + rnd() * 160 + k * 40); }
    else if (c.k === "splash") { noise(t, 0.35, 0.13, 0.6, 0.05, (x) => Math.exp(-x * 8), -0.1); }
    else if (c.k === "scramble") { for (let k = 0; k < 10; k++) knock(t + k * 0.07 + rnd() * 0.03, 600 + rnd() * 400, 0.025); }
    else if (c.k === "sting") { for (const m of [43, 50, 55]) add(sfx, t, 0, 0.6, (x) => Math.sin(2 * Math.PI * hz(m) * x) * 0.05 * Math.min(1, x / 0.005) * Math.exp(-x * 5)); ev(t, "sting"); }
    else if (c.k === "boing") { boing(t, 240, 0.08); }
    else if (c.k.startsWith("end:")) {
      const kind = c.k.slice(4), len = c.v ?? 3;
      if (kind === "nap") { for (let k = 0; k * 1.5 + 0.4 < len - 0.3; k++) noise(t + 0.4 + k * 1.5, 0.9, 0.06, 0.05, 0.01, (x) => Math.sin(Math.PI * Math.min(1, x / 0.9)) ** 2); }
      else if (kind === "run") { swish(t + 0.3, 0.3, 0.12); for (let k = 0; k < 16; k++) knock(t + 0.4 + k * 0.09, 500 + rnd() * 500, 0.03); }
      else if (kind === "cliffhanger") { noise(t + 1.0, 1.0, 0.1, 0.03, 0.005, (x) => Math.sin(Math.PI * Math.min(1, x / 1.0))); for (const m of [36, 42, 48]) add(sfx, t + 1.75, 0, 1.2, (x) => Math.sin(2 * Math.PI * hz(m) * x) * 0.05 * Math.min(1, x / 0.01) * Math.exp(-x * 2.5)); }
      else if (kind === "menyerah") { for (let k = 0; k < 8; k++) pop(t + 0.5 + k * 0.13, 500 + rnd() * 300, 0.035); swish(t + 1.6, 0.3, 0.08); }
      else if (kind === "adopt") { ding(t + 0.9, 79, 0.03); }
      ev(t, `end ${kind}`);
    }
    else if (c.k === "part") { thud(t, 0.12, 90); ding(t + 0.05, 84, 0.04); ev(t, "part"); }
  }

  // ---------------------------------------------------------------- write + mix
  const wav = (file, buf) => {
    const data = Buffer.alloc(N * 4);
    for (let i = 0; i < N; i++) for (let c = 0; c < 2; c++) data.writeInt16LE(Math.round(Math.max(-1, Math.min(1, buf[c][i])) * 32767), i * 4 + c * 2);
    const h = Buffer.alloc(44);
    h.write("RIFF", 0); h.writeUInt32LE(36 + data.length, 4); h.write("WAVEfmt ", 8); h.writeUInt32LE(16, 16); h.writeUInt16LE(1, 20); h.writeUInt16LE(2, 22);
    h.writeUInt32LE(SR, 24); h.writeUInt32LE(SR * 4, 28); h.writeUInt16LE(4, 32); h.writeUInt16LE(16, 34); h.write("data", 36); h.writeUInt32LE(data.length, 40);
    fs.writeFileSync(file, Buffer.concat([h, data]));
  };
  fs.mkdirSync(outDir, { recursive: true });
  fs.mkdirSync(path.dirname(mixFile), { recursive: true });
  const mf = path.join(outDir, "music.wav"), sf = path.join(outDir, "sfx.wav");
  wav(mf, mus);
  wav(sf, sfx);
  const tail = `apad=whole_dur=${DUR},atrim=0:${DUR},loudnorm=I=-14:TP=-1.5:LRA=11,aresample=48000[out]`;
  const filter = voice
    ? ["[0:a]highpass=f=70,acompressor=threshold=-20dB:ratio=2.5:attack=8:release=140:makeup=3,asplit=2[v][vk]", "[1:a]volume=1.0[m]", "[m][vk]sidechaincompress=threshold=0.03:ratio=6:attack=25:release=420[md]", "[2:a]volume=0.85[s]", `[v][md][s]amix=inputs=3:normalize=0:duration=longest,${tail}`].join(";")
    : ["[0:a]volume=1.35[m]", "[1:a]volume=1.0[s]", `[m][s]amix=inputs=2:normalize=0:duration=longest,${tail}`].join(";");
  execFileSync("ffmpeg", ["-y", "-v", "error", ...(voice ? ["-i", voice] : []), "-i", mf, "-i", sf, "-filter_complex", filter, "-map", "[out]", "-ac", "2", mixFile]);
  const m = spawnSync("ffmpeg", ["-hide_banner", "-i", mixFile, "-af", "loudnorm=print_format=summary", "-f", "null", "-"], { encoding: "utf8" }).stderr;
  return { lufs: (m.match(/Input Integrated:\s+(-?[\d.]+)/) ?? [])[1], peak: (m.match(/Input True Peak:\s+(-?[\d.]+)/) ?? [])[1], events: events.length, perMin: Math.round((events.length / DUR) * 60) };

const renderCards = (groups, h, outDir, names) => {
  fs.mkdirSync(outDir, { recursive: true });
  const tmp = path.join(outDir, "_strips"), pf = path.join(outDir, "_card-props.json");
  fs.rmSync(tmp, { recursive: true, force: true });
  fs.writeFileSync(pf, JSON.stringify({ strips: groups.flat(), h }));
  sh("npx", ["remotion", "render", "src/index.ts", "NekatCard", tmp, "--sequence", "--image-format=jpeg", "--jpeg-quality=92", `--props=${pf}`, "--concurrency=2", "--log=error"]);
  const files = fs.readdirSync(tmp).filter((f) => /^element-\d+\.jpe?g$/.test(f)).sort((a, b) => parseInt(a.slice(8)) - parseInt(b.slice(8))).map((f) => path.join(tmp, f));
  let k = 0;
  const made = groups.map((g, gi) => {
    const part = files.slice(k, (k += g.length)), to = path.join(outDir, names[gi]);
    if (part.length === 1) fs.copyFileSync(part[0], to);
    else execFileSync("ffmpeg", ["-v", "error", "-y", ...part.flatMap((f) => ["-i", f]), "-filter_complex", part.map((_, i) => `[${i}:v]`).join("") + `vstack=inputs=${part.length}`, "-q:v", "3", to]);
    return to;
  });
  fs.rmSync(tmp, { recursive: true, force: true });
  fs.rmSync(pf, { force: true });
  return made;
};

  const frames = list.split(",").map((t) => (t.endsWith("s") ? Math.round(parseFloat(t) * 30) : parseInt(t, 10)));

  const W = wide ? 960 : 540, H = wide ? 540 : 960, cols = Math.min(4, files.length), n = cols * Math.ceil(files.length / cols);
  const sheet = (labels) => {
    const pads = Array.from({ length: n - files.length }, () => ["-f", "lavfi", "-i", `color=c=black:s=${W}x${H}`]).flat();

      Array.from({ length: n }, (_, i) => `[v${i}]`).join("") + `xstack=inputs=${n}:layout=` + Array.from({ length: n }, (_, i) => `${(i % cols) * W}_${Math.floor(i / cols) * H}`).join("|") + `,scale='min(2160,iw)':-2`;

const INKC = "#2b2533", PAPER = "#fff8ec", ACC = "#ff9a2e";
const band = (title: string) => (
  <div style={{ position: "absolute", left: 0, right: 0, top: 0, height: 64, background: INKC, color: PAPER, display: "flex", alignItems: "center", paddingLeft: 32, fontFamily: F_DISPLAY, fontSize: 34, letterSpacing: 1 }}>{title}</div>

    <planeGeometry args={[80, 80]} />
  </mesh>

/** Camera that fits a row of n items (spacing d) and a content height y0…y1 between the title band and the labels. */
const rowShot = (W: number, H: number, n: number, d: number, y0: number, y1: number, tb: number, lb: number): { shot: Shot; vis: number; xs: number[] } => {
  const A = W / H, f01 = (H - tb - lb) / H, V = 22;
  const vis = Math.max((y1 - y0) / f01, (n * d) / (A * 0.97));
  const dist = vis / (2 * Math.tan(((V / 2) * Math.PI) / 180));
  const offPx = tb + (H - tb - lb) / 2 - H / 2;
  const camY = (y0 + y1) / 2 + (offPx / H) * vis;
  const xs = Array.from({ length: n }, (_, i) => (i - (n - 1) / 2) * d);
  return { shot: { pos: [0, camY, dist], target: [0, camY, 0], fov: V, shadowAt: [0, 0, 0], shadowBox: Math.max(6, n * d) }, vis, xs };
};
const Labels: React.FC<{ items: { code?: string; name: string }[]; xs: number[]; vis: number; W: number; H: number; lb: number }> = ({ items, xs, vis, W, H, lb }) => (
  <>
    <div style={{ position: "absolute", left: 0, right: 0, bottom: 0, height: lb, background: PAPER }} />
    {items.map((it, i) => (
      <div key={i} style={{ position: "absolute", bottom: 8, left: W / 2 + (xs[i] / (vis * (W / H))) * W - 95, width: 190, textAlign: "center", fontFamily: F_TEXT, fontWeight: 700, color: INKC, lineHeight: 1.05 }}>
        {it.code ? <div style={{ fontFamily: F_DISPLAY, fontSize: 32, color: ACC }}>{it.code}</div> : null}
        <div style={{ fontSize: it.code ? 20 : 26 }}>{it.name}</div>
      </div>
    ))}
  </>
);

const PeopleStrip: React.FC<{ s: Extract<CardStrip, { kind: "people" }> }> = ({ s }) => {
  const { width: W, height: H } = useVideoConfig();
  const head = s.frame === "head";
  const tb = s.title ? 64 : 0, lb = 84;
  const smax = Math.max(...s.items.map((i) => (i.look.build?.s ?? 1) * (1 + ((i.look.build?.head ?? 1) - 1) * 0.3)));
  const { shot, vis, xs } = rowShot(W, H, s.items.length, head ? 0.62 : 1.0, head ? 0.98 * smax : 0, (head ? 1.62 : 1.66) * smax, tb, lb);
  return (
    <AbsoluteFill style={{ background: PAPER }}>
      <ToyCanvas shot={shot}>
        <Floor />
        {s.items.map((it, i) => (
          <Person key={i} look={it.look} pose={defaultPose()} face={EXPR[it.expr ?? "grin"]} pos={[xs[i], 0, 0]} rot={[0, it.rotY ?? (head ? 0.32 : 0.18), 0]} />
        ))}
      </ToyCanvas>

const CreatureStrip: React.FC<{ s: Extract<CardStrip, { kind: "creatures" }>; f: number }> = ({ s, f }) => {
  const { width: W, height: H } = useVideoConfig();
  const tb = s.title ? 64 : 0, lb = 84;

  return (
    <AbsoluteFill style={{ background: PAPER }}>

        <Floor />

          <Creatures key={i} look={it.look} t={f * 0.37 + i} list={[{ p: [xs[i], 0, 0], yaw: 0.45, pose: it.pose ?? 1, seed: 7 + i, eye: 1 }]} max={2} />
        ))}

      {s.title ? band(s.title) : null}
      <Labels items={s.items} xs={xs} vis={vis} W={W} H={H} lb={lb} />
    </AbsoluteFill>

};

// NEKAT crowd layout — where hundreds of creatures go without passing through each other or the set. Pure TypeScript.
//   scatter(): one layer on a floor (oriented-rectangle footprints, separating-axis test, exclusion zones)
//   Pile:      a height-map "drop": each creature lands on the highest thing under its footprint (ground, set pieces,
//              other creatures, a lying person) and stamps its own top surface in — stacks never interpenetrate.
//   row():     along a ledge (fence top, bench); slope(): on a tilted plane (a roof).
// Footprints and top surfaces come from the species data (species.ts › shapeOf). Everything is seeded: the same layout
// on every frame and in every render tab.
import { hash, rng, type V3 } from "./math.ts";
import { SPECIES, shapeOf } from "./species.ts";

export type Inst = {
  p: V3;
  yaw: number;
  pose: number;
  s?: number;
  pitch?: number;
  roll?: number;
  squash?: number;
  eye?: number;
  seed: number;
  wag?: number;
};

type Rect = { c: [number, number]; ax: [number, number]; az: [number, number]; hx: number; hz: number };
export const rectOf = (sp: string, pose: number, x: number, z: number, yaw: number, s = 1, pad = 0.012): Rect => {
  const f = shapeOf(sp, pose).foot;
  const c = Math.cos(yaw), sn = Math.sin(yaw);
  const ax: [number, number] = [c, -sn], az: [number, number] = [sn, c];
  return { c: [x + (ax[0] * f.cx + az[0] * f.cz) * s, z + (ax[1] * f.cx + az[1] * f.cz) * s], ax, az, hx: f.hx * s + pad, hz: f.hz * s + pad };
};
const overlap = (a: Rect, b: Rect) => {
  const d: [number, number] = [b.c[0] - a.c[0], b.c[1] - a.c[1]];
  for (const axis of [a.ax, a.az, b.ax, b.az]) {
    const proj = (r: Rect) => r.hx * Math.abs(r.ax[0] * axis[0] + r.ax[1] * axis[1]) + r.hz * Math.abs(r.az[0] * axis[0] + r.az[1] * axis[1]);
    if (Math.abs(d[0] * axis[0] + d[1] * axis[1]) > proj(a) + proj(b)) return false;
  }
  return true;
};

export type Zone = { kind: "circle"; c: [number, number]; r: number } | { kind: "rect"; x0: number; x1: number; z0: number; z1: number };
const inZone = (r: Rect, zones: Zone[]) =>
  zones.some((z) => {
    if (z.kind === "circle") return Math.hypot(r.c[0] - z.c[0], r.c[1] - z.c[1]) < z.r + Math.max(r.hx, r.hz) * 0.85;
    const ex = Math.abs(r.ax[0]) * r.hx + Math.abs(r.az[0]) * r.hz;
    const ez = Math.abs(r.ax[1]) * r.hx + Math.abs(r.az[1]) * r.hz;
    return r.c[0] + ex > z.x0 && r.c[0] - ex < z.x1 && r.c[1] + ez > z.z0 && r.c[1] - ez < z.z1;
  });
export const pointInZones = (x: number, z: number, zones: Zone[], pad = 0) =>
  zones.some((q) => (q.kind === "circle" ? Math.hypot(x - q.c[0], z - q.c[1]) < q.r + pad : x > q.x0 - pad && x < q.x1 + pad && z > q.z0 - pad && z < q.z1 + pad));

export type ScatterOpts = {
  sp: string;
  x0: number; x1: number; z0: number; z1: number; n: number; seed: number; y?: number;
  avoid?: Zone[]; facing?: number; yawJitter?: number; altShare?: number; s?: number; tries?: number;
  /** already-placed creatures to keep clear of */
  existing?: Inst[];
  /** sort candidates nearest this point first (fills outward from it) */
  near?: [number, number];
};
/** One non-overlapping layer in a rectangle. Returns as many as fit (≤ n). */
export const scatter = (o: ScatterOpts): Inst[] => {
  const r = rng(o.seed);
  const out: Inst[] = [];
  const rects: Rect[] = (o.existing ?? []).map((c) => rectOf(o.sp, c.pose, c.p[0], c.p[2], c.yaw, c.s ?? 1));
  const tries = o.tries ?? o.n * 40;
  const alt = o.altShare ?? SPECIES[o.sp].altShare;
  const cand: [number, number][] = [];
  for (let k = 0; k < tries; k++) cand.push([o.x0 + r() * (o.x1 - o.x0), o.z0 + r() * (o.z1 - o.z0)]);
  if (o.near) cand.sort((a, b) => Math.hypot(a[0] - o.near![0], a[1] - o.near![1]) - Math.hypot(b[0] - o.near![0], b[1] - o.near![1]));
  for (const [x, z] of cand) {
    if (out.length >= o.n) break;
    const yaw = (o.facing ?? 0) + (r() * 2 - 1) * (o.yawJitter ?? 0.55);
    const pose = r() < alt ? 1 : 0;
    const s = (o.s ?? 1) * (0.92 + 0.12 * r());
    const rc = rectOf(o.sp, pose, x, z, yaw, s);
    if (o.avoid && inZone(rc, o.avoid)) continue;
    if (rects.some((q) => overlap(q, rc))) continue;
    rects.push(rc);
    out.push({ p: [x, o.y ?? 0, z], yaw, pose, s, seed: o.seed * 1000 + out.length, eye: r() < 0.45 ? 0.5 : 1 });
  }
  return out;
};

// ------------------------------------------------------------------ height-map pile
export type BoxTop = [number, number, number, number, number];
export class Pile {
  x0: number; z0: number; res: number; nx: number; nz: number; h: Float32Array; sp: string;
  constructor(sp: string, x0: number, x1: number, z0: number, z1: number, res = 0.035) {
    this.sp = sp;
    this.x0 = x0;
    this.z0 = z0;
    this.res = res;
    this.nx = Math.ceil((x1 - x0) / res) + 1;
    this.nz = Math.ceil((z1 - z0) / res) + 1;
    this.h = new Float32Array(this.nx * this.nz);
  }
  private cells(rc: Rect, fn: (i: number, x: number, z: number) => void) {
    const ex = Math.abs(rc.ax[0]) * rc.hx + Math.abs(rc.az[0]) * rc.hz;
    const ez = Math.abs(rc.ax[1]) * rc.hx + Math.abs(rc.az[1]) * rc.hz;
    const i0 = Math.max(0, Math.floor((rc.c[0] - ex - this.x0) / this.res)), i1 = Math.min(this.nx - 1, Math.ceil((rc.c[0] + ex - this.x0) / this.res));
    const j0 = Math.max(0, Math.floor((rc.c[1] - ez - this.z0) / this.res)), j1 = Math.min(this.nz - 1, Math.ceil((rc.c[1] + ez - this.z0) / this.res));
    for (let j = j0; j <= j1; j++)
      for (let i = i0; i <= i1; i++) {
        const x = this.x0 + i * this.res, z = this.z0 + j * this.res;
        const dx = x - rc.c[0], dz = z - rc.c[1];
        if (Math.abs(dx * rc.ax[0] + dz * rc.ax[1]) <= rc.hx && Math.abs(dx * rc.az[0] + dz * rc.az[1]) <= rc.hz) fn(j * this.nx + i, x, z);
      }
  }
  maxIn(x0: number, x1: number, z0: number, z1: number) {
    let m = 0;
    this.cells({ c: [(x0 + x1) / 2, (z0 + z1) / 2], ax: [1, 0], az: [0, 1], hx: (x1 - x0) / 2, hz: (z1 - z0) / 2 }, (i) => (m = Math.max(m, this.h[i])));
    return m;
  }
  /** Raise an axis-aligned box region to `top` (set pieces, a lying body). */
  box(x0: number, x1: number, z0: number, z1: number, top: number) {
    this.cells({ c: [(x0 + x1) / 2, (z0 + z1) / 2], ax: [1, 0], az: [0, 1], hx: (x1 - x0) / 2, hz: (z1 - z0) / 2 }, (i) => (this.h[i] = Math.max(this.h[i], top)));
  }
  boxes(list: BoxTop[]) { for (const b of list) this.box(b[0], b[1], b[2], b[3], b[4]); return this; }
  /** Drop one creature at (x, z): returns its resting height and stamps it in. */
  place(pose: number, x: number, z: number, yaw: number, s = 1) {
    const rc = rectOf(this.sp, pose, x, z, yaw, s, 0.005);
    let y = 0;
    this.cells(rc, (i) => (y = Math.max(y, this.h[i])));
    const c = Math.cos(yaw), sn = Math.sin(yaw), top = shapeOf(this.sp, pose).top;
    this.cells(rc, (i, wx, wz) => {
      const dx = wx - x, dz = wz - z;
      const lx = (dx * c - dz * sn) / s, lz = (dx * sn + dz * c) / s;
      const tp = top(lx, lz);
      if (tp >= 0) this.h[i] = Math.max(this.h[i], y + tp * s);
    });
    return y;
  }
  /** Drop n creatures around a centre (elliptic, centre-weighted, or uniform in a rect). Order = landing order. */
  drop(o: { cx: number; cz: number; rx: number; rz: number; n: number; seed: number; facing?: number; yawJitter?: number; altShare?: number; best?: number; avoid?: Zone[]; power?: number; rect?: boolean; s?: number; cap?: (x: number, z: number) => number; minY?: number }): Inst[] {
    const r = rng(o.seed);
    const out: Inst[] = [];
    let guard = 0, miss = 0;
    const alt = o.altShare ?? SPECIES[this.sp].altShare * 0.35;
    while (out.length < o.n && guard++ < o.n * 60 && miss < 250) {
      let bestC: { x: number; z: number; y: number; yaw: number; pose: number; s: number } | null = null;
      for (let k = 0; k < (o.best ?? 4); k++) {
        const a = r() * Math.PI * 2, d = Math.pow(r(), o.power ?? 0.6);
        const ux = r() * 2 - 1, uz = r() * 2 - 1;
        const x = o.rect ? o.cx + ux * o.rx : o.cx + Math.cos(a) * d * o.rx, z = o.rect ? o.cz + uz * o.rz : o.cz + Math.sin(a) * d * o.rz;
        const yaw = (o.facing ?? 0) + (r() * 2 - 1) * (o.yawJitter ?? 0.6);
        const pose = r() < alt ? 1 : 0;
        const s = (o.s ?? 1) * (0.92 + 0.12 * r());
        if (o.avoid && inZone(rectOf(this.sp, pose, x, z, yaw, s), o.avoid)) continue;
        const rc = rectOf(this.sp, pose, x, z, yaw, s, 0.005);
        let y = 0;
        this.cells(rc, (i) => (y = Math.max(y, this.h[i])));
        if (o.cap && y > o.cap(x, z)) continue;
        if (o.minY !== undefined && y < o.minY) continue;
        if (!bestC || y < bestC.y) bestC = { x, z, y, yaw, pose, s };
      }
      if (!bestC) { miss++; continue; }
      miss = 0;
      const y = this.place(bestC.pose, bestC.x, bestC.z, bestC.yaw, bestC.s);
      out.push({ p: [bestC.x, y, bestC.z], yaw: bestC.yaw, pose: bestC.pose, s: bestC.s, seed: o.seed * 1000 + out.length, eye: hash(o.seed + out.length) < 0.5 ? 0.5 : 1 });
    }
    return out;
  }
}

/** Creatures in a row along a ledge from a to b (top surface height y), gaps between. */
export const row = (sp: string, a: [number, number], b: [number, number], y: number, n: number, seed: number, facing = 0): Inst[] => {
  const r = rng(seed);
  const out: Inst[] = [];
  for (let i = 0; i < n; i++) {
    const u = n === 1 ? 0.5 : i / (n - 1);
    out.push({ p: [a[0] + (b[0] - a[0]) * u, y, a[1] + (b[1] - a[1]) * u], yaw: facing + (r() * 2 - 1) * 0.25, pose: r() < SPECIES[sp].altShare ? 1 : 0, s: 0.9 + 0.1 * r(), seed: seed * 1000 + i, eye: r() < 0.5 ? 0.5 : 1 });
  }
  return out;
};

/** Creatures on a tilted plane: origin o, along-edge unit vector e (xz), down-slope yaw and slope angle; cols × rows. */
export const slope = (o: V3, e: [number, number], downYaw: number, angle: number, cols: number, rows: number, dx: number, dz: number, seed: number, fill = 0.85): Inst[] => {
  const r = rng(seed);
  const out: Inst[] = [];
  const up: [number, number] = [-Math.sin(downYaw), -Math.cos(downYaw)];
  for (let j = 0; j < rows; j++)
    for (let i = 0; i < cols; i++) {
      if (r() > fill) continue;
      const along = (i + (j % 2) * 0.5) * dx, upd = j * dz;
      const hor = upd * Math.cos(angle);
      out.push({ p: [o[0] + e[0] * along + up[0] * hor, o[1] + upd * Math.sin(angle), o[2] + e[1] * along + up[1] * hor], yaw: downYaw + (r() * 2 - 1) * 0.12, pitch: angle, pose: 0, s: 0.9 + 0.1 * r(), seed: seed * 1000 + out.length, eye: r() < 0.5 ? 0.5 : 1 });
    }
  return out;
};

const c01 = (x: number) => (x < 0 ? 0 : x > 1 ? 1 : x);
const smoother = (x: number) => { const t = c01(x); return t * t * t * (t * (t * 6 - 15) + 10); };
const mix = (a: number, b: number, t: number) => a + (b - a) * t;
const mix3 = (a: V3, b: V3, t: number): V3 => [mix(a[0], b[0], t), mix(a[1], b[1], t), mix(a[2], b[2], t)];

const camAt = (S: PShot, t: number): Shot => {
  const hold = S.hold ?? 0;
  const k = smoother((t - S.t0 - hold) / Math.max(0.01, S.t1 - S.t0 - hold));
  return { pos: mix3(S.a.pos, S.b.pos, k), target: mix3(S.a.target, S.b.target, k), fov: mix(S.a.fov, S.b.fov, k), shadowAt: S.shadowAt, shadowBox: S.shadowBox };
};
/** Creatures this close to the shot's camera path are left out for the whole shot (no giant blur at the lens, no pop). */
const NEAR = 0.85;
const clearOfLens = (S: PShot, p: V3) => {
  const a = S.a.pos, b = S.b.pos, q: V3 = [p[0], p[1] + 0.3, p[2]];
  const ab: V3 = [b[0] - a[0], b[1] - a[1], b[2] - a[2]], aq: V3 = [q[0] - a[0], q[1] - a[1], q[2] - a[2]];
  const L = ab[0] * ab[0] + ab[1] * ab[1] + ab[2] * ab[2];
  const k = L > 1e-6 ? Math.max(0, Math.min(1, (aq[0] * ab[0] + aq[1] * ab[1] + aq[2] * ab[2]) / L)) : 0;
  return Math.hypot(q[0] - a[0] - ab[0] * k, q[1] - a[1] - ab[1] * k, q[2] - a[2] - ab[2] * k) > NEAR;
};
const project = (shot: Shot, p: V3, w: number, h: number) => {
  const c = new PerspectiveCamera(shot.fov, w / h, 0.05, 400);
  c.position.set(...shot.pos);
  c.lookAt(...shot.target);
  c.updateMatrixWorld();
  const v = new Vector3(...p).project(c);
  return { x: (v.x * 0.5 + 0.5) * w, y: (1 - (v.y * 0.5 + 0.5)) * h };
};

/** Orientation whose +y = dir and +x ≈ side. */
export const basis = (dir: V3, side: V3 = [1, 0, 0]): Quaternion => {
  const y = new Vector3(...dir).normalize();
  let x = new Vector3(...side);
  x.sub(y.clone().multiplyScalar(x.dot(y)));
  if (x.lengthSq() < 1e-6) x = new Vector3(0, 0, 1).cross(y);
  x.normalize();
  const z = new Vector3().crossVectors(x, y).normalize();
  return new Quaternion().setFromRotationMatrix(new Matrix4().makeBasis(x, y, z));
};

const dropGeo = new SphereGeometry(1, 12, 8);
const dropMat = new MeshStandardMaterial({ color: "#86cdf2", roughness: 0.35 });
const M4 = new Matrix4(), Qd = new Quaternion(), Sd = new Vector3(), Td = new Vector3();
/** Ballistic jet from `from` with velocity `vel` (m/s); `on` 0..1 ramps the jet in/out (shrinks, never pops). */
export const WaterArc: React.FC<{ from: V3; vel: V3; t: number; on: number; n?: number; r?: number; floor?: number; seed?: number }> = ({ from, vel, t, on, n = 46, r = 0.022, floor = 0, seed = 0 }) => {
  const ref = useRef<InstancedMesh>(null);
  useLayoutEffect(() => {
    const m = ref.current;
    if (!m) return;
    const g = -9.8;
    // flight time to the floor
    const a = 0.5 * g, b = vel[1], c = from[1] - floor;
    const T = (-b - Math.sqrt(Math.max(0, b * b - 4 * a * c))) / (2 * a);
    let k = 0;
    if (on > 0.01) {
      for (let i = 0; i < n; i++) {
        const u = ((i / n + t * 2.2 + seed * 0.37) % 1 + 1) % 1;
        const s = u * T;
        const p: V3 = [from[0] + vel[0] * s, from[1] + vel[1] * s + 0.5 * g * s * s, from[2] + vel[2] * s];
        const v = new Vector3(vel[0], vel[1] + g * s, vel[2]).normalize();
        Qd.setFromUnitVectors(UP, v);
        const sz = r * on * (0.7 + 0.6 * ((i * 7) % 5) / 4) * (u > 0.94 ? (1 - u) / 0.06 : 1) * (0.8 + 0.5 * u);
        // a little spread so it reads as spray, not a rope
        const sp = 0.06 * u;
        const jx = Math.sin(i * 12.9898) * sp, jy = Math.sin(i * 78.233) * sp, jz = Math.sin(i * 37.719) * sp;
        M4.compose(Td.set(p[0] + jx, p[1] + jy, p[2] + jz), Qd, Sd.set(sz, sz * 1.8, sz));
        m.setMatrixAt(k++, M4);
      }
    }
    m.count = k;
    m.instanceMatrix.needsUpdate = true;
  });
  return <instancedMesh ref={ref} args={[dropGeo, dropMat, n]} frustumCulled={false} castShadow={false} />;
};

export const track = (keys: { t: number; v: number[]; ease?: (x: number) => number }[], t: number): number[] => {
  if (t <= keys[0].t) return keys[0].v.slice();
  const last = keys[keys.length - 1];
  if (t >= last.t) return last.v.slice();
  let i = 0;
  while (i < keys.length - 2 && t > keys[i + 1].t) i++;

  | { k: "cone"; c: V3; r: number; h: number; rot: V3; col: Role; flat?: number }

const ry = (v: V3, a: number): V3 => [v[0] * Math.cos(a) + v[2] * Math.sin(a), v[1], -v[0] * Math.sin(a) + v[2] * Math.cos(a)];
const rz = (v: V3, a: number): V3 => [v[0] * Math.cos(a) - v[1] * Math.sin(a), v[0] * Math.sin(a) + v[1] * Math.cos(a), v[2]];

// ------------------------------------------------------------------ surface samples (footprint + top height map)
/** Catmull-Rom point (uniform) on a polyline at u ∈ [0, 1]. */
const crAt = (pts: V3[], u: number): V3 => {
  const n = pts.length - 1, f = Math.min(n - 1e-6, Math.max(0, u * n)), i = Math.floor(f), t = f - i;
  const p0 = pts[Math.max(0, i - 1)], p1 = pts[i], p2 = pts[Math.min(n, i + 1)], p3 = pts[Math.min(n, i + 2)];
  const c = (a: number, b: number, cc: number, d: number) => 0.5 * (2 * b + (-a + cc) * t + (2 * a - 5 * b + 4 * cc - d) * t * t + (-a + 3 * b - 3 * cc + d) * t * t * t);
  return [c(p0[0], p1[0], p2[0], p3[0]), c(p0[1], p1[1], p2[1], p3[1]), c(p0[2], p1[2], p2[2], p3[2])];
};
/** Points on the surface of a part (local creature frame). */
export const samplePart = (p: Part, dense = 1): V3[] => {
  const out: V3[] = [];
  if (p.k === "ell") {
    const nt = Math.round(14 * dense), np = Math.round(24 * dense);
    for (let i = 0; i <= nt; i++)
      for (let j = 0; j < np; j++) {
        const th = (i / nt) * Math.PI, ph = (j / np) * Math.PI * 2;
        let v: V3 = [Math.sin(th) * Math.cos(ph) * p.r[0], Math.cos(th) * p.r[1], Math.sin(th) * Math.sin(ph) * p.r[2]];
        v = add(rotE(v, p.rot ?? [0, 0, 0], "YXZ"), p.c);
        if (p.floor !== undefined && v[1] < p.floor + 0.03) v = [v[0], p.floor + 0.03 - (p.floor + 0.03 - v[1]) * 0.12, v[2]];
        out.push(v);
      }
  } else if (p.k === "cone") {
    const f = p.flat ?? 0.6;
    out.push(add(rotE([0, p.h / 2, 0], p.rot, "XYZ"), p.c));
    for (let j = 0; j < 12; j++) {
      const a = (j / 12) * Math.PI * 2;
      out.push(add(rotE([Math.cos(a) * p.r, -p.h / 2, Math.sin(a) * p.r * f], p.rot, "XYZ"), p.c));
    }
  } else {
    for (let i = 0; i <= 16; i++) {
      const c = crAt(p.pts, i / 16), rr = p.r * (1 - (p.taper ?? 0) * (i / 16) ** 2);
      for (let j = 0; j < 6; j++) { const a = (j / 6) * Math.PI * 2; out.push([c[0] + Math.cos(a) * rr, c[1] + Math.sin(a) * rr, c[2]]); }
      out.push([c[0], c[1] + rr, c[2]]);
    }
  }
  return out;
};

export type Shape = { foot: { cx: number; cz: number; hx: number; hz: number }; top: (x: number, z: number) => number; height: number };
const shapeCache = new Map<string, Shape>();
/** Footprint rectangle (incl. tail, ears) and top-surface height function of a species pose, scale 1. */
export const shapeOf = (species: string, pose: number): Shape => {
  const key = `${species}|${pose}`;
  const hit = shapeCache.get(key);
  if (hit) return hit;
  const P = SPECIES[species].poses[pose];
  const parts = [...P.parts, ...(P.move?.parts ?? []), ...(P.eyes ? shiftParts(P.eyes.parts, P.eyes.at) : [])];
  const pts = parts.flatMap((p) => samplePart(p, p.k === "ell" && Math.max(...p.r) > 0.12 ? 2.2 : 1));
  let x0 = 1e9, x1 = -1e9, z0 = 1e9, z1 = -1e9, h = 0;
  for (const q of pts) { x0 = Math.min(x0, q[0]); x1 = Math.max(x1, q[0]); z0 = Math.min(z0, q[2]); z1 = Math.max(z1, q[2]); h = Math.max(h, q[1]); }
  const res = 0.02, nx = Math.ceil((x1 - x0) / res) + 1, nz = Math.ceil((z1 - z0) / res) + 1;
  const g = new Float32Array(nx * nz).fill(-1);
  for (const q of pts) {
    const i = Math.round((q[0] - x0) / res), j = Math.round((q[2] - z0) / res);
    for (let dj = -1; dj <= 1; dj++) for (let di = -1; di <= 1; di++) {
      const ii = i + di, jj = j + dj;
      if (ii < 0 || jj < 0 || ii >= nx || jj >= nz) continue;
      const k = jj * nx + ii;
      if (q[1] > g[k]) g[k] = q[1];
    }
  }
  const top = (x: number, z: number) => {
    const i = Math.round((x - x0) / res), j = Math.round((z - z0) / res);
    if (i < 0 || j < 0 || i >= nx || j >= nz) return -1;
    return g[j * nx + i];
  };
  const s: Shape = { foot: { cx: (x0 + x1) / 2, cz: (z0 + z1) / 2, hx: (x1 - x0) / 2, hz: (z1 - z0) / 2 }, top, height: h };
  shapeCache.set(key, s);
  return s;
};
