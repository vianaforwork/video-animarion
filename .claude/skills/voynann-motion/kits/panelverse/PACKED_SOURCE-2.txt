// Word timings for PANELVERSE without whisper.cpp: clause ↔ speech-segment alignment.
//   node scripts/pv-align.mjs <vo.mp3> --script=<text file> --out=src/pv-stories/<id>/words.json [--id=<id>]
// With whisper.cpp installed it hands over to scripts/transcribe.mjs (exact word timings).
// Otherwise: the script is split into clauses at punctuation (. , ! ? ; : …), the VO's speech segments are found with
// silence detection (several thresholds are tried until segments and clauses match one-to-one), and each clause's words
// are spread over its segment by syllable count. Typical error ≈ ±0.1 s per word when the clauses match; if no threshold
// matches, it falls back to transcribe.mjs (sentence-level, ≈ ±0.3–0.5 s) and says so.
import { spawnSync } from "node:child_process";
import fs from "node:fs";
import path from "node:path";

const args = process.argv.slice(2);
const vo = args.find((a) => !a.startsWith("--"));
const opt = (k, d) => (args.find((a) => a.startsWith(`--${k}=`)) ?? `=${d ?? ""}`).split("=").slice(1).join("=");
const scriptFile = opt("script"), out = opt("out"), id = opt("id");
if (!vo || !fs.existsSync(vo) || !out) {
  console.error("usage: node scripts/pv-align.mjs <vo.mp3> --script=<text file> --out=src/pv-stories/<id>/words.json [--id=<id>]");
  process.exit(1);
}
const run = (c, a) => spawnSync(c, a, { encoding: "utf8", maxBuffer: 1 << 28 });
const handOver = (why) => {
  console.log(`pv-align: ${why} → scripts/transcribe.mjs`);
  const r = spawnSync("node", [path.join("scripts", "transcribe.mjs"), vo, `--out=${out}`, ...(scriptFile ? [`--script=${scriptFile}`] : []), ...(id ? [`--id=${id}`] : []), "--lang=id"], { stdio: "inherit" });
  process.exit(r.status ?? 1);
};
// whisper.cpp present → exact timings from transcribe.mjs
const hasWhisper = run("sh", ["-c", "command -v whisper-cli"]).stdout.trim() || ["/opt/homebrew/bin/whisper-cli", "/usr/local/bin/whisper-cli"].some((f) => fs.existsSync(f)) || process.env.WHISPER_CLI;
if (hasWhisper) handOver("whisper.cpp found");
if (!scriptFile || !fs.existsSync(scriptFile)) { console.error("pv-align: without whisper.cpp the VO text is needed: --script=<text file>"); process.exit(1); }
if (id) {
  if (!/^[a-z0-9-]+$/.test(id)) { console.error(`--id "${id}" must use only a-z, 0-9 and "-"`); process.exit(1); }
  const dst = path.join("public", "audio", `${id}${path.extname(vo).toLowerCase() || ".mp3"}`);
  fs.mkdirSync(path.dirname(dst), { recursive: true });
  if (path.resolve(dst) !== path.resolve(vo)) fs.copyFileSync(vo, dst);
  console.log(`vo: ${dst}`);
}

// clauses: words grouped until a punctuation mark closes the clause
const tokens = text.split(" ").filter(Boolean);
const clauses = [];
let cur = [];
for (const t of tokens) {
  cur.push(t);
  if (/[.,!?;:…]["')\]]*$/.test(t)) { clauses.push(cur); cur = []; }
}
if (cur.length) clauses.push(cur);
const duration = Number(run("ffprobe", ["-v", "error", "-show_entries", "format=duration", "-of", "csv=p=0", vo]).stdout.trim());
const segmentsFor = (noise, d) => {
  const txt = run("ffmpeg", ["-hide_banner", "-i", vo, "-af", `silencedetect=noise=${noise}dB:d=${d}`, "-f", "null", "-"]).stderr;
  const starts = [...txt.matchAll(/silence_start: ([\d.]+)/g)].map((m) => Number(m[1]));
  const ends = [...txt.matchAll(/silence_end: ([\d.]+)/g)].map((m) => Number(m[1]));
  // speech = gaps between silences
  const segs = [];
  let t = 0;
  for (let i = 0; i < starts.length; i++) {
    if (starts[i] - t > 0.08) segs.push([t, starts[i]]);
    t = ends[i] ?? duration;
  }
  if (duration - t > 0.08) segs.push([t, duration]);
  return segs.filter(([a, b]) => b - a > 0.08);
};
const syl = (w) => Math.max(1, (w.toLowerCase().match(/[aiueoé]+/g) ?? []).length);
const csyl = clauses.map((c) => c.reduce((n, w) => n + syl(w), 0));
const totalSyl = csyl.reduce((a, b) => a + b, 0);
// monotonic alignment: each step pairs 1 clause with 1–4 consecutive segments, or 2–3 clauses with 1 segment.
// cost = squared log ratio of segment speech time to the expected time (syllables × speech rate), weighted by size
const align = (segs) => {
  const C = clauses.length, Sn = segs.length;
  const speech = segs.reduce((n, [a, b]) => n + b - a, 0);
  const rate = speech / totalSyl;
  const segDur = (j0, j1) => segs.slice(j0, j1).reduce((n, [a, b]) => n + b - a, 0);
  const sylOf = (i0, i1) => csyl.slice(i0, i1).reduce((a, b) => a + b, 0);
  const pairCost = (i0, i1, j0, j1) => {
    const exp = sylOf(i0, i1) * rate, got = segDur(j0, j1);
    const r = Math.log(Math.max(got, 0.05) / Math.max(exp, 0.05));
    return r * r * sylOf(i0, i1) + 0.6 * (i1 - i0 - 1) + 0.4 * (j1 - j0 - 1);
  };
  const INF = 1e9;
  const f = Array.from({ length: C + 1 }, () => new Array(Sn + 1).fill(INF));
  const back = Array.from({ length: C + 1 }, () => new Array(Sn + 1).fill(null));
  f[0][0] = 0;
  for (let i = 0; i <= C; i++) for (let j = 0; j <= Sn; j++) {
    if (f[i][j] >= INF) continue;
    for (let k = 1; k <= 4 && j + k <= Sn && i + 1 <= C; k++) {
      const c = f[i][j] + pairCost(i, i + 1, j, j + k);
      if (c < f[i + 1][j + k]) { f[i + 1][j + k] = c; back[i + 1][j + k] = [i, j]; }
    }
    for (let k = 2; k <= 3 && i + k <= C && j + 1 <= Sn; k++) {
      const c = f[i][j] + pairCost(i, i + k, j, j + 1);
      if (c < f[i + k][j + 1]) { f[i + k][j + 1] = c; back[i + k][j + 1] = [i, j]; }
    }
  }
  if (f[C][Sn] >= INF) return null;
  const steps = [];
  for (let i = C, j = Sn; i > 0 || j > 0; ) { const [pi, pj] = back[i][j]; steps.unshift([pi, i, pj, j]); i = pi; j = pj; }
  return { cost: f[C][Sn] / totalSyl, steps };
};
let best = null;
for (const noise of [-30, -32, -35, -38, -40, -45]) for (const d of [0.08, 0.1, 0.12, 0.15, 0.2, 0.25]) {
  const segs = segmentsFor(noise, d);
  if (!segs.length) continue;
  const a = align(segs);
  if (a && (!best || a.cost < best.cost)) best = { ...a, segs, noise, d };
}
if (!best || best.cost > 0.35) handOver(best ? `the VO's pauses do not match the script's punctuation well (fit ${best.cost.toFixed(2)})` : "no speech segments found");
const words = [];
for (const [i0, i1, j0, j1] of best.steps) {
  // the clause run's words spread over the segment run's speech time (pauses between segments skipped)
  const segs = best.segs.slice(j0, j1);
  const ws = clauses.slice(i0, i1).flat();
  const n = ws.map(syl), total = n.reduce((a, b) => a + b, 0);
  const speech = segs.reduce((m, [a, b]) => m + b - a, 0);
  const at = (x) => { // speech-time offset → real time
    for (const [a, b] of segs) { if (x <= b - a) return a + x; x -= b - a; }
    return segs[segs.length - 1][1];
  };
  let acc = 0;
  ws.forEach((w, k) => {
    const dur = (speech * n[k]) / total;
    words.push([w, +at(acc + Math.min(0.01, dur * 0.1)).toFixed(3), +at(acc + dur * 0.94).toFixed(3)]);
    acc += dur;
  });
}
fs.mkdirSync(path.dirname(out), { recursive: true });
fs.writeFileSync(out, JSON.stringify(words));
console.log(`words: ${out} — ${words.length} words, ${clauses.length} clauses aligned to ${best.segs.length} speech segments (silence ${best.noise} dB / ${best.d} s, fit ${best.cost.toFixed(3)}), ≈ ±0.1 s per word`);
console.log(words.map(([w, s]) => `${s.toFixed(2)} ${w}`).join("  "));

// Story lint: validity + safe-zone + anti-monotony rules, run by scripts/pv-make.mjs before rendering.
// Pure TS (Node + browser). Errors stop the make; warnings are printed for the director to judge.
import { samplePanel, boundsOf } from "../engine/panels/geometry.ts";
import { sampleCam } from "../engine/camera.ts";
import type { StoryDef } from "./types.ts";

export type LintContext = { voEndFrame: number; items: string[]; props: string[] };
export type LintResult = { errors: string[]; warnings: string[]; stats: Record<string, string | number> };

const sec = (f: number) => `${(f / 60).toFixed(1)} s`;
/** width of one letter as a share of the font size, per lettering kind (an estimate, also used to place lettering on the wide page) */
export const TEXT_W: Record<string, number> = { impact: 0.46, title: 0.6, sfxText: 0.52, stamp: 0.62, caption: 0.5 };

/** what the viewer sees at frame f, reduced to a layout signature */
const signature = (st: StoryDef, f: number) => {
  const vis = st.panels.filter((p) => f >= p.from && f <= (p.shatter ? Math.max(p.to, p.shatter.at + p.shatter.dur) : p.to));
  if (!vis.length) return "empty";
  const parts = vis.map((p) => {
    const g = samplePanel(p.geom, f);
    const b = boundsOf(g.pts);
    const [cw, ch] = st.wide ? [1920, 1080] : [1080, 1920];
    const area = (Math.min(b.x1, cw) - Math.max(b.x0, 0)) * (Math.min(b.y1, ch) - Math.max(b.y0, 0)) * g.scale * g.scale;
    return area / (cw * ch);
  }).filter((a) => a > 0.02);
  const full = parts.filter((a) => a > 0.92).length;
  const small = parts.length - full;
  if (full && small) return "full+inset";
  if (full) return "full";
  return small === 1 ? "page-1" : `page-${Math.min(small, 4)}`;
};

export const lintStory = (st: StoryDef, ctx: LintContext): LintResult => {
  const E: string[] = [], W: string[] = [];
  const D = st.durationInFrames;
  // ── basics ────────────────────────────────────────────────────────────────────────────────
  if (!/^[a-z0-9-]+$/.test(st.id)) E.push(`id "${st.id}" must use only a-z, 0-9 and "-"`);
  if (D < ctx.voEndFrame + 12) E.push(`durationInFrames ${D} (${sec(D)}) ends before / right on the VO end ${sec(ctx.voEndFrame)} — the narration would be cut`);
  if (D > ctx.voEndFrame + 300) W.push(`${sec(D - ctx.voEndFrame)} after the VO ends — keep the closing hold under ~3 s`);
  // ── references ───────────────────────────────────────────────────────────────────────────
  const shotIds = Object.keys(st.shots);
  const items = new Set([...ctx.items, ...(st.elements ?? [])]);
  const props = new Set(ctx.props);
  for (const p of st.panels) {
    if (!st.shots[p.shot]) E.push(`panel uses unknown shot "${p.shot}" (shots: ${shotIds.join(", ")})`);
    if (p.to <= p.from) E.push(`panel "${p.shot}" ends (${p.to}) before it starts (${p.from})`);
    if (!p.geom.length) E.push(`panel "${p.shot}" has no geometry keys`);
  }
  for (const id of shotIds) if (!st.panels.some((p) => p.shot === id)) W.push(`shot "${id}" is never shown in a panel`);
  for (const [id, s] of Object.entries(st.shots)) {
    if (!s.camera.length) E.push(`shot "${id}" has no camera keys`);
    for (const it of s.set.items ?? []) if (!items.has(it.type)) E.push(`shot "${id}": set item "${it.type}" is not a library item or a declared story element`);
    const allProps = [...(s.props ?? []), ...s.actors.flatMap((a) => [...(a.torso ?? []), ...(a.head ?? []), ...(a.heldR ? [a.heldR] : []), ...(a.heldL ? [a.heldL] : [])])];
    for (const p of allProps) {
      if (!props.has(p.kind)) E.push(`shot "${id}": prop kind "${p.kind}" does not exist (${[...props].join(", ")})`);
      if (p.kind === "item") for (const k of p.keys) if (k.v.type && !items.has(String(k.v.type))) E.push(`shot "${id}": item prop "${k.v.type}" is not a library item or declared element`);
    }
    for (const a of s.actors) if (!st.cast[a.cast]) E.push(`shot "${id}": actor "${a.cast}" is not in cast (${Object.keys(st.cast).join(", ")})`);
  }
  // ── custom elements ("think up elements for THIS script") ────────────────────────────────
  const used = new Set<string>();
  for (const s of Object.values(st.shots)) {
    for (const it of s.set.items ?? []) used.add(it.type);
    for (const p of [...(s.props ?? []), ...s.actors.flatMap((a) => [...(a.torso ?? []), ...(a.head ?? []), ...(a.heldR ? [a.heldR] : []), ...(a.heldL ? [a.heldL] : [])])])
      if (p.kind === "item") for (const k of p.keys) if (k.v.type) used.add(String(k.v.type));
  }
  if (!st.elements?.length) E.push("no story elements: design at least one element for this script in elements.tsx (registerItems) and list it in story.elements");
  for (const el of st.elements ?? []) if (!used.has(el)) W.push(`story element "${el}" is declared but never placed`);
  // ── beat sheet ───────────────────────────────────────────────────────────────────────────
  const beats = st.beats ?? [];
  const minBeats = Math.max(3, Math.floor(D / 60 / 3.5));
  if (beats.length < minBeats) E.push(`beat sheet has ${beats.length} beats — write at least ${minBeats} (one per 2–3.5 s of VO, each with words, idea and device)`);
  const devices = new Set(beats.map((b) => b.device.toLowerCase()));
  if (beats.length && devices.size < Math.min(4, beats.length)) E.push(`only ${devices.size} different devices in the beat sheet — vary them (${[...devices].join(", ")})`);
  for (let i = 1; i < beats.length; i++) if (beats[i].device.toLowerCase() === beats[i - 1].device.toLowerCase()) W.push(`beats ${i} and ${i + 1} use the same device "${beats[i].device}"`);
  // ── variety: layouts / transitions / fx ──────────────────────────────────────────────────
  const sigTime: Record<string, number> = {};
  let empty = 0, emptyRun = 0;
  for (let f = 0; f < D; f += 6) {
    const s = signature(st, f);
    sigTime[s] = (sigTime[s] ?? 0) + 6;
    if (s === "empty") { emptyRun += 6; empty = Math.max(empty, emptyRun); } else emptyRun = 0;
  }
  const layouts = Object.entries(sigTime).filter(([k, t]) => k !== "empty" && t >= 24).map(([k]) => k);
  if (D > 600 && layouts.length < 3) E.push(`only ${layouts.length} screen layout(s) (${layouts.join(", ")}) — use at least 3 (full-bleed, a 2–4 panel page, an inset, a printed page…)`);
  if (empty > 30) W.push(`${sec(empty)} with no panel on screen (blank page)`);
  const moves = st.panels.flatMap((p) => p.moves ?? []).filter((m) => m.type !== "cut" && m.type !== "drift").sort((a, b) => a.at - b.at);
  const moveTypes = new Set(moves.map((m) => m.type));
  if (D > 600 && moveTypes.size < 3) E.push(`only ${moveTypes.size} transition type(s) (${[...moveTypes].join(", ")}) — use at least 3 different panel moves`);
  for (let i = 2; i < moves.length; i++) if (moves[i].type === moves[i - 1].type && moves[i].type === moves[i - 2].type && moves[i].at - moves[i - 2].at > 30) W.push(`three "${moves[i].type}" transitions in a row (${sec(moves[i - 2].at)}–${sec(moves[i].at)})`);
  const fxKinds = new Set(st.fx.map((c) => c.kind));
  if (D > 600 && fxKinds.size < 3) W.push(`only ${fxKinds.size} kinds of comic FX — mix lettering, bursts, speed lines, captions, stamps, shock lines…`);
  // ── lettering ────────────────────────────────────────────────────────────────────────────
  const texts = st.fx.filter((c) => c.kind === "impact" || c.kind === "sfxText" || c.kind === "stamp" || c.kind === "caption") as Extract<StoryDef["fx"][number], { text: string }>[];
  const bigText = texts.filter((c) => c.kind !== "caption");
  if (bigText.length > Math.max(3, Math.ceil(D / 60 / 2.5))) W.push(`${bigText.length} lettering hits in ${sec(D)} — more than one per 2.5 s turns into a slideshow of words`);
  for (const c of texts) {
    const words = c.text.trim().split(/\s+/).length;
    if (words > 3 && c.kind !== "caption") W.push(`"${c.text}" is ${words} words — impact lettering reads best at 1–3 words`);
    const size = "size" in c && c.size ? c.size : 84;
    const k = c.kind === "impact" && "font" in c && c.font === "title" ? TEXT_W.title : TEXT_W[c.kind] ?? 0.55;
    const w = c.text.length * size * k + (c.kind === "caption" ? 80 : 0);
    const x0 = c.kind === "caption" ? c.x : c.x - w / 2, x1 = c.kind === "caption" ? c.x + w : c.x + w / 2;
    // the wide (16:9) page has no phone UI: lettering only has to stay inside the 1920×1080 frame
    if (x0 < (st.wide ? 30 : 50) || x1 > (st.wide ? 1890 : 1030)) E.push(`"${c.text}" (${c.kind} at ${sec(c.at)}) is ≈${Math.round(w)} px wide and leaves the ${st.wide ? "wide " : ""}frame (${Math.round(x0)}–${Math.round(x1)}) — smaller size or shorter text`);
    else if (!st.wide && x1 > 920 && c.kind !== "caption") W.push(`"${c.text}" reaches x=${Math.round(x1)} — TikTok/Reels buttons sit on the right edge (keep ≤ 920)`);
    if (st.wide ? c.y < 30 || c.y > 1050 : c.y < 200 || c.y > 1480) E.push(st.wide ? `"${c.text}" at y=${c.y} is outside the wide frame (30–1050)` : `"${c.text}" at y=${c.y} is outside the mobile safe zone (200–1480)`);
  }
  // ── pacing: long static stretches ────────────────────────────────────────────────────────
  for (let f0 = 0; f0 + 270 < D; f0 += 30) {
    const sig = signature(st, f0);
    if (sig !== "full") continue;
    const p = st.panels.find((x) => f0 >= x.from && f0 <= x.to && signature(st, f0) === "full");
    if (!p) continue;
    const s = st.shots[p.shot];
    const a = sampleCam(s.camera, f0 - s.start), b = sampleCam(s.camera, f0 + 270 - s.start);
    const move = Math.hypot(a.pos[0] - b.pos[0], a.pos[1] - b.pos[1], a.pos[2] - b.pos[2]) + Math.abs(a.fov - b.fov) * 0.05;
    const sameShot = st.panels.filter((x) => x.from <= f0 + 270 && x.to >= f0).length === 1;
    const fxNear = st.fx.some((c) => ("at" in c ? c.at : c.from) > f0 && ("at" in c ? c.at : c.from) < f0 + 270);
    if (sameShot && move < 0.05 && !fxNear) { W.push(`static: one full-screen shot with no camera move and no FX for 4.5 s from ${sec(f0)}`); break; }
  }
  const shotsShown = new Set(st.panels.map((p) => p.shot)).size;
  if (shotsShown < Math.min(4, Math.ceil(D / 60 / 4))) W.push(`only ${shotsShown} shots for ${sec(D)} — a new picture every 3–4 s keeps a comic alive`);
  const setPrints = new Set(Object.values(st.shots).map((s) => JSON.stringify([s.set.kind, s.set.look, s.set.wall, s.set.floor, s.set.time, (s.set.items ?? []).map((i) => i.type).sort()])));
  if (shotIds.length > 2 && setPrints.size < 2) W.push("every shot uses the same set — change place, stage look or colours between beats");
  return {
    errors: E,
    warnings: W,
    stats: {
      duration: sec(D),
      shots: shotIds.length,
      panels: st.panels.length,
      layouts: layouts.join(" · "),
      transitions: [...moveTypes].join(" · "),
      fx: [...fxKinds].join(" · "),
      beats: beats.length,
      devices: [...devices].join(" · "),
      elements: (st.elements ?? []).join(" · "),
      lettering: texts.map((t) => t.text).join(" · "),
    },
  };

// Deterministic keyframe engine. Every animated value in PANELVERSE is a pure function of a frame
// number, so any frame renders identically in any tab, in any order.

export type EaseName =
  | "linear"
  | "in"
  | "out"
  | "inOut"
  | "outBack"
  | "outBackBig"
  | "inBack"
  | "outElastic"
  | "hold"
  | "expoOut"
  | "expoIn";

const c1 = 1.70158;
export const EASES: Record<EaseName, (t: number) => number> = {
  linear: (t) => t,
  in: (t) => t * t * t,
  out: (t) => 1 - Math.pow(1 - t, 3),
  inOut: (t) => (t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2),
  outBack: (t) => 1 + (c1 + 1) * Math.pow(t - 1, 3) + c1 * Math.pow(t - 1, 2),
  outBackBig: (t) => {
    const c = 3.2;
    return 1 + (c + 1) * Math.pow(t - 1, 3) + c * Math.pow(t - 1, 2);
  },
  inBack: (t) => (c1 + 1) * t * t * t - c1 * t * t,
  outElastic: (t) =>
    t === 0 || t === 1 ? t : Math.pow(2, -10 * t) * Math.sin((t * 10 - 0.75) * ((2 * Math.PI) / 3)) + 1,
  hold: () => 0,
  expoOut: (t) => (t >= 1 ? 1 : 1 - Math.pow(2, -10 * t)),
  expoIn: (t) => (t <= 0 ? 0 : Math.pow(2, 10 * t - 10)),
};

export const clamp = (v: number, a = 0, b = 1) => Math.min(b, Math.max(a, v));
export const lerp = (a: number, b: number, t: number) => a + (b - a) * t;
/** 0→1 progress of f through [a, b] with easing. */
export const prog = (f: number, a: number, b: number, ease: EaseName = "linear") =>
  EASES[ease](clamp((f - a) / Math.max(1e-6, b - a)));

type Val = number | string | boolean | readonly unknown[] | { [k: string]: unknown } | undefined;

const mix = (a: Val, b: Val, t: number): Val => {
  if (typeof a === "number" && typeof b === "number") return lerp(a, b, t);
  if (Array.isArray(a) && Array.isArray(b)) return a.map((x, i) => mix(x as Val, b[i] as Val, t));
  if (a && b && typeof a === "object" && typeof b === "object" && !Array.isArray(a)) {
    const out: Record<string, unknown> = {};
    const ka = a as Record<string, unknown>;
    const kb = b as Record<string, unknown>;
    for (const k of new Set([...Object.keys(ka), ...Object.keys(kb)])) {
      out[k] = k in ka && k in kb ? mix(ka[k] as Val, kb[k] as Val, t) : (kb[k] ?? ka[k]);
    }
    return out;
  }
  // discrete values switch at the halfway point of the transition
  return t < 0.5 ? a : b;
};

const merge = (base: Val, over: Val): Val => {
  if (over === undefined) return base;
  if (base && over && typeof base === "object" && typeof over === "object" && !Array.isArray(base) && !Array.isArray(over)) {
    const out: Record<string, unknown> = { ...(base as Record<string, unknown>) };
    for (const [k, v] of Object.entries(over as Record<string, unknown>)) out[k] = merge(out[k] as Val, v as Val);
    return out;
  }
  return over;
};

/** A keyframe: value `v` is reached at frame `f`, travelling from the previous key with `ease`. */
export type Key<T> = { f: number; v: T; ease?: EaseName };

/**
 * Sample a sparse keyframe track. Object keys are cumulative: each key only needs the fields that
 * change, the rest carry over from earlier keys.
 */
export function sample<T>(keys: readonly Key<T>[], f: number): T {
  if (keys.length === 0) throw new Error("empty track");
  // resolve cumulative values
  const full: { f: number; v: Val; ease: EaseName }[] = [];
  let acc: Val = undefined;
  for (const k of keys) {
    acc = merge(acc, k.v as Val);
    full.push({ f: k.f, v: acc, ease: k.ease ?? "inOut" });
  }
  if (f <= full[0].f) return full[0].v as T;
  for (let i = 1; i < full.length; i++) {
    if (f <= full[i].f) {
      const a = full[i - 1];
      const b = full[i];
      const t = EASES[b.ease](clamp((f - a.f) / Math.max(1e-6, b.f - a.f)));
      return mix(a.v, b.v, t) as T;
    }
  }
  return full[full.length - 1].v as T;
}

/** Deterministic hash noise in [-1, 1]. */
export const hash = (n: number) => {
  const s = Math.sin(n * 127.1 + 311.7) * 43758.5453;
  return (s - Math.floor(s)) * 2 - 1;
};
/** Smooth 1D value noise in [-1, 1]. */
export const noise1 = (x: number, seed = 0) => {
  const i = Math.floor(x);
  const fr = x - i;
  const u = fr * fr * (3 - 2 * fr);
  return lerp(hash(i + seed * 101.3), hash(i + 1 + seed * 101.3), u);
};

/** Decaying camera/impact shake: amplitude envelope starting at frame `at`. */
export const shake = (f: number, at: number, dur: number, amp: number, seed = 1, freq = 0.9) => {
  if (f < at || f > at + dur) return 0;
  const e = 1 - (f - at) / dur;
  return noise1((f - at) * freq, seed) * amp * e * e;
};

/** Split a quad into shards (fan from an impact point) for the panel-shatter transition. */
export const shardsOf = (quad: readonly Pt[], impact: Pt, cuts: number[]): Pt[][] => {
  const edges: [Pt, Pt][] = quad.map((p, i) => [p, quad[(i + 1) % quad.length]]);
  const lens = edges.map(([a, b]) => Math.hypot(b[0] - a[0], b[1] - a[1]));
  const total = lens.reduce((s, l) => s + l, 0);
  const at = (t: number): Pt => {
    let d = (((t % 1) + 1) % 1) * total;
    for (let i = 0; i < edges.length; i++) {
      if (d <= lens[i]) {
        const [a, b] = edges[i];
        const k = d / lens[i];
        return [a[0] + (b[0] - a[0]) * k, a[1] + (b[1] - a[1]) * k];
      }
      d -= lens[i];
    }
    return quad[0];
  };
  const corners: number[] = [];
  let acc = 0;
  for (const l of lens) {
    corners.push(acc / total);
    acc += l;
  }
  const sorted = [...cuts].sort((a, b) => a - b);
  return sorted.map((c0, i) => {
    const c1 = i === sorted.length - 1 ? sorted[0] + 1 : sorted[i + 1];
    const pts: Pt[] = [impact, at(c0)];
    for (const k of [...corners, ...corners.map((x) => x + 1)]) if (k > c0 && k < c1) pts.push(at(k));
    pts.push(at(c1));
    return pts;
  });
};

    const bare = new BufferGeometry();
    bare.setAttribute("position", pos.clone());
    if (g.index) bare.setIndex(g.index.clone());

    hulls.set(g, h);

export const headRadiusAt = (y: number) => {
  const p = HEAD_PROFILE;
  if (y <= p[0][1]) return 0;
  for (let i = 1; i < p.length; i++) {
    if (y <= p[i][1]) {
      const t = (y - p[i - 1][1]) / (p[i][1] - p[i - 1][1]);
      return p[i - 1][0] + (p[i][0] - p[i - 1][0]) * t;
    }
  }
  return 0;
};

/** n-point star outline (comic stars, star eyes, sparkles). */
export const starShape = (r1: number, r2: number, n = 5) => {
  const s = new Shape();
  for (let i = 0; i <= n * 2; i++) {
    const a = (i / (n * 2)) * Math.PI * 2 + Math.PI / 2;
    const r = i % 2 === 0 ? r1 : r2;
    const x = Math.cos(a) * r;
    const y = Math.sin(a) * r;
    if (i === 0) s.moveTo(x, y);
    else s.lineTo(x, y);
  }
  return s;
};

// Colour pipeline: swatch hexes are sRGB, three converts them to linear on input and the shaders convert
// back with <colorspace_fragment> on output (no tone mapping), so flat fills land on the exact palette hex.

const COMMON = /* glsl */ `
  float aastep(float th, float v) { float w = max(fwidth(v) * 0.75, 1e-4); return smoothstep(th - w, th + w, v); }
  // Screen-space Ben-Day dot field (45°), dot coverage = tone in [0,1]. Locked to the canvas pixel grid
  // so every surface in a shot prints with one coherent screen, like a real halftone plate.
  float dots(float tone, float cell) {
    vec2 p = gl_FragCoord.xy;
    const float a = 0.785398;
    vec2 q = mat2(cos(a), sin(a), -sin(a), cos(a)) * p / cell;
    vec2 g = fract(q) - 0.5;
    float d = length(g);
    float r = sqrt(clamp(tone, 0.0, 1.0)) * 0.64;
    float w = max(fwidth(d) * 0.8, 1e-4);
    return 1.0 - smoothstep(r - w, r + w, d);
  }
`;

const VERT = /* glsl */ `
  varying vec3 vN;
  varying vec3 vPos;
  varying vec2 vUv;
  varying vec3 vObj;
  void main() {
    vObj = position;
    vec4 mv = modelViewMatrix * vec4(position, 1.0);
    vPos = mv.xyz;
    vN = normalize(normalMatrix * normal);
    vUv = uv;
    gl_Position = projectionMatrix * mv;
  }
`;

const FRAG = /* glsl */ `
  uniform vec3 uColor; uniform vec3 uShade; uniform vec3 uDot; uniform vec3 uSpecColor; uniform vec3 uRimColor;
  uniform vec3 uLight;     // world-space direction TO the key light
  uniform float uCut;      // cel terminator (n·l)
  uniform float uCell;     // halftone cell size in px
  uniform float uHalftone; // halftone strength
  uniform float uPattern;  // flat all-over Ben-Day tone (walls / print fills)
  uniform float uSpec; uniform float uSpecSize;
  uniform float uRim;
  uniform float uFlat;     // 1 = unlit graphic fill
  uniform float uOpacity;
  uniform float uDotOnly;  // 1 = print only the dot screen (graphic shadows, blush)
  uniform vec4 uClip;      // object-space front ellipse cut-out: (halfW, halfH, centreY, minZ); halfW<=0 = off
  varying vec3 vN; varying vec3 vPos; varying vec2 vUv; varying vec3 vObj;
  ${COMMON}
  void main() {
    if (uClip.x > 0.0 && vObj.z > uClip.w) {
      vec2 e = vec2(vObj.x / uClip.x, (vObj.y - uClip.z) / uClip.y);
      if (dot(e, e) < 1.0) discard;
    }
    if (uDotOnly > 0.5) {
      if (dots(uPattern, uCell) < 0.5) discard;
      gl_FragColor = vec4(uDot, uOpacity);
      #include <colorspace_fragment>
      return;
    }
    vec3 N = normalize(vN);
    if (!gl_FrontFacing) N = -N;
    vec3 L = normalize((viewMatrix * vec4(uLight, 0.0)).xyz);
    vec3 V = normalize(-vPos);
    float ndl = dot(N, L);
    vec3 col = uColor;
    if (uFlat < 0.5) {
      float lit = aastep(uCut, ndl);
      col = mix(uShade, uColor, lit);
      // halftone fade across the terminator (lit side) ...
      float t1 = 1.0 - smoothstep(uCut, uCut + 0.42, ndl);
      col = mix(col, uShade, dots(t1 * 0.62, uCell) * lit * uHalftone);
      // ... and a deeper dot screen inside the core shadow
      float t2 = smoothstep(uCut - 0.05, uCut - 0.85, ndl);
      col = mix(col, uDot, dots(t2 * 0.55, uCell) * (1.0 - lit) * uHalftone);
      // hard comic specular
      vec3 H = normalize(L + V);
      col = mix(col, uSpecColor, aastep(1.0 - uSpecSize, dot(N, H)) * uSpec);
      // back rim (graphic kicker)
      float rim = aastep(0.62, 1.0 - max(dot(N, V), 0.0)) * aastep(0.0, -ndl + 0.25);
      col = mix(col, uRimColor, rim * uRim);
    }
    if (uPattern > 0.0) col = mix(col, uDot, dots(uPattern, uCell));
    gl_FragColor = vec4(col, uOpacity);
    #include <colorspace_fragment>
  }
`;

// ── Two-bone IK ────────────────────────────────────────────────────────────────────────────────
const _v = new Vector3();
const _d = new Vector3();
const _b = new Vector3();
/** Returns [joint, end] for a chain root→joint→end of lengths a,b reaching for target, bending toward pole. */
export const solveIK = (root: Vector3, target: Vector3, a: number, b: number, pole: Vector3): [Vector3, Vector3] => {
  _d.copy(target).sub(root);
  let dist = _d.length();
  if (dist < 1e-5) {
    _d.set(0, -1, 0);
    dist = 1e-5;
  }
  _d.divideScalar(dist);
  const d = Math.min(dist, a + b - 1e-4);
  const dMin = Math.abs(a - b) + 1e-3;
  const dd = Math.max(d, dMin);
  const x = (a * a - b * b + dd * dd) / (2 * dd);
  const h = Math.sqrt(Math.max(a * a - x * x, 0));
  _b.copy(pole).addScaledVector(_d, -pole.dot(_d));
  if (_b.lengthSq() < 1e-8) _b.set(0, 0, 1).addScaledVector(_d, -_d.z);
  _b.normalize();
  const joint = new Vector3().copy(root).addScaledVector(_d, x).addScaledVector(_b, h);
  const end = new Vector3().copy(root).addScaledVector(_d, dd);
  return [joint, end];
};

const UP = new Vector3(0, 1, 0);
/** Placement (position, quaternion, length) of a +Y unit segment from a to b. */
export const segment = (a: Vector3, b: Vector3) => {
  _v.copy(b).sub(a);
  const len = _v.length();
  const q = new Quaternion().setFromUnitVectors(UP, len > 1e-6 ? _v.clone().divideScalar(len) : UP);
  return { pos: [a.x, a.y, a.z] as V3, q, len };
};

/**
 * Register a story's own elements (made for one script) next to the library items. Call it at the
 * top level of `src/pv-stories/<id>/elements.tsx`; the names become usable as set items and as
 * `item` props in that story.
 */
export const registerItems = (items: Record<string, { desc: string; render: (p: ItemProps) => React.ReactNode; wall?: boolean }>) => {
  for (const [k, v] of Object.entries(items)) LIB[k] = v;
};

/** One placed item (animated fields sampled at the shot frame). */
export const PlacedItem: React.FC<{ spec: ItemSpec; f: number }> = ({ spec, f }) => {
  const s = spec.keys ? { ...spec, ...sample<Partial<ItemSpec>>(spec.keys, f) } : spec;
  const item = LIB[s.type];
  if (!item) throw new Error(`PANELVERSE: unknown item "${s.type}" — use a library item or registerItems() it in the story's elements.tsx`);
  const rot: V3 = typeof s.rot === "number" ? [0, s.rot, 0] : s.rot ?? [0, 0, 0];
  const scale: V3 = typeof s.scale === "number" ? [s.scale, s.scale, s.scale] : s.scale ?? [1, 1, 1];
  return (
    <group position={s.pos} rotation={rot} scale={scale}>
      {item.render({ sw: s.sw, sw2: s.sw2, text: s.text, sub: s.sub, on: s.on, n: s.n, t: s.t ?? f / 60, f })}
    </group>
  );
};
