// PANELVERSE sound kit — every effect is synthesised here from oscillators + filtered noise.
// Deterministic (seeded PRNG), dependency-free, original (no samples, no licensing questions).
//   node scripts/pv-sfx.mjs            → public/sfx/pv-*.wav  (48 kHz, mono, 16-bit; skips existing files)
//   node scripts/pv-sfx.mjs --force    → rebuild all
import fs from "node:fs";
import path from "node:path";

const SR = 48000;
const OUT = path.resolve("public/sfx");
fs.mkdirSync(OUT, { recursive: true });

// ── primitives ───────────────────────────────────────────────────────────────────────────────
const rng = (seed) => () => {
  seed |= 0;
  seed = (seed + 0x6d2b79f5) | 0;
  let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
  t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
  return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
};
const buf = (sec) => new Float32Array(Math.max(1, Math.round(sec * SR)));
const white = (n, seed) => {
  const r = rng(seed);
  const b = new Float32Array(n);
  for (let i = 0; i < n; i++) b[i] = r() * 2 - 1;
  return b;
};
const brown = (n, seed) => {
  const w = white(n, seed);
  let acc = 0;
  for (let i = 0; i < n; i++) {
    acc = (acc + w[i] * 0.02) * 0.998;
    w[i] = acc * 3.5;
  }
  return w;
};
/** RBJ biquad; `f` may be a number or a function of sample index (sweeps). */
const biquad = (x, type, f, q = 0.707) => {
  const y = new Float32Array(x.length);
  let x1 = 0, x2 = 0, y1 = 0, y2 = 0;
  for (let i = 0; i < x.length; i++) {
    const fc = Math.min(SR * 0.45, Math.max(20, typeof f === "function" ? f(i) : f));
    const w = (2 * Math.PI * fc) / SR;
    const c = Math.cos(w), s = Math.sin(w), a = s / (2 * q);
    let b0, b1, b2;
    if (type === "lp") { b0 = (1 - c) / 2; b1 = 1 - c; b2 = (1 - c) / 2; }
    else if (type === "hp") { b0 = (1 + c) / 2; b1 = -(1 + c); b2 = (1 + c) / 2; }
    else { b0 = a; b1 = 0; b2 = -a; } // band-pass (constant 0 dB peak)
    const a0 = 1 + a, a1 = -2 * c, a2 = 1 - a;
    const out = (b0 * x[i] + b1 * x1 + b2 * x2 - a1 * y1 - a2 * y2) / a0;
    x2 = x1; x1 = x[i]; y2 = y1; y1 = out;
    y[i] = out;
  }
  return y;
};
/** sine with a frequency function f(t) (Hz) — phase-accumulated so glides are clean */
const osc = (sec, f, shape = "sin") => {
  const b = buf(sec);
  let ph = 0;
  for (let i = 0; i < b.length; i++) {
    const t = i / SR;
    ph += (2 * Math.PI * (typeof f === "function" ? f(t) : f)) / SR;
    b[i] = shape === "sq" ? Math.tanh(Math.sin(ph) * 4) : shape === "tri" ? (2 / Math.PI) * Math.asin(Math.sin(ph)) : Math.sin(ph);
  }
  return b;
};
const env = (b, fn) => {
  for (let i = 0; i < b.length; i++) b[i] *= fn(i / SR, i / b.length);
  return b;
};
const adsr = (a, d) => (t) => (t < a ? t / a : Math.exp(-(t - a) / d));
const bell = (t, len) => Math.sin(Math.PI * Math.min(1, Math.max(0, t / len)));
const gain = (b, g) => b.map((v) => v * g);
const mixInto = (dst, src, at = 0, g = 1) => {
  const o = Math.round(at * SR);
  for (let i = 0; i < src.length && i + o < dst.length; i++) if (i + o >= 0) dst[i + o] += src[i] * g;
  return dst;
};
const sat = (b, drive) => b.map((v) => Math.tanh(v * drive) / Math.tanh(drive));
const fadeOut = (b, sec = 0.01) => {
  const n = Math.round(sec * SR);
  for (let i = 0; i < n && i < b.length; i++) b[b.length - 1 - i] *= i / n;
  return b;
};
const normalize = (b, peak = 0.89) => {
  let m = 0;
  for (const v of b) m = Math.max(m, Math.abs(v));
  return m > 0 ? gain(b, peak / m) : b;
};
const writeWav = (name, b) => {
  b = fadeOut(normalize(b));
  const data = Buffer.alloc(b.length * 2);
  for (let i = 0; i < b.length; i++) data.writeInt16LE(Math.round(Math.max(-1, Math.min(1, b[i])) * 32767), i * 2);

// ── word clock: anchor every event to the spoken words ────────────────────────────────────────
export type Word = readonly [string, number, number, ...unknown[]];
const norm = (s: string) => s.toLowerCase().normalize("NFKD").replace(/[^\p{L}\p{N}]+/gu, "");
export type Clock = ReturnType<typeof clock>;
/**
 * `const W = clock(WORDS, { offset: 0.3 })` →
 *   W.at("oke") / W.at("dia", 2)  frame where the word (its 2nd occurrence) starts
 *   W.end("tamat")                frame where it ends
 *   W.s(2.5)                      frame of a VO second · W.voEnd frame where the VO ends
 */
export const clock = (words: readonly Word[], opt: { offset?: number } = {}) => {
  const offset = opt.offset ?? 0;
  const list = words.map(([w, a, b], i) => ({ i, w, n: norm(w), a: a + offset, b: b + offset }));
  const find = (word: string, nth = 1) => {
    const n = norm(word);
    let hits = list.filter((x) => x.n === n);
    if (!hits.length) hits = list.filter((x) => x.n.startsWith(n) || (n.length > 3 && n.startsWith(x.n)));
    const hit = hits[nth - 1];
    if (!hit) {
      const near = list.filter((x) => x.n.slice(0, 2) === n.slice(0, 2)).map((x) => x.w);
      throw new Error(`word "${word}"${nth > 1 ? ` (#${nth})` : ""} not in the VO transcript${near.length ? ` — did you mean: ${[...new Set(near)].join(", ")}` : ""}`);
    }
    return hit;
  };
  return {
    words: list,
    offset,
    at: (word: string, nth = 1) => Math.round(find(word, nth).a * FPS),
    end: (word: string, nth = 1) => Math.round(find(word, nth).b * FPS),
    s: (sec: number) => Math.round((sec + offset) * FPS),
    voEnd: Math.round(((list[list.length - 1]?.b ?? 0) + 0) * FPS),
    /** speech spans (frames) merged across short pauses — drives SFX ducking */
    spans: (mergeGap = 6): [number, number][] => {
      const out: [number, number][] = [];
      for (const x of list) {
        const s0 = Math.floor(x.a * FPS), e0 = Math.ceil(x.b * FPS);
        const last = out[out.length - 1];
        if (last && s0 - last[1] <= mergeGap) last[1] = Math.max(last[1], e0);
        else out.push([s0, e0]);
      }
      return out;
    },
  };
};

    const e = t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2;
    const fov = fov0 + (fov1 - fov0) * e;
    const d = k / Math.tan((fov * Math.PI) / 360);
    return { f: Math.round(f0 + (f1 - f0) * t), v: { pos: [target[0] + u[0] * d, target[1] + u[1] * d, target[2] + u[2] * d] as V3, target, fov, roll }, ease: "linear" as EaseName };
  });
};
/** Orbit around a target (arc move) from angle a0 to a1 (radians, around Y), at height h and distance d. */
export const orbitKeys = (f0: number, f1: number, target: V3, d: number, h: number, a0: number, a1: number, fov = 34, steps = 6): CamKey[] =>
  Array.from({ length: steps + 1 }, (_, i) => {
    const t = i / steps;
    const e = t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2;
    const a = a0 + (a1 - a0) * e;
    return { f: Math.round(f0 + (f1 - f0) * t), v: { pos: [target[0] + Math.sin(a) * d, target[1] + h, target[2] + Math.cos(a) * d] as V3, target, fov, roll: 0 }, ease: "linear" as EaseName };
  });
/** Alternating small offsets (typing, trembling, fist shake…) as keyframes. */
export const jitterKeys = <T extends Record<string, unknown>>(f0: number, f1: number, every: number, make: (i: number) => T): Key<T>[] => {
  const out: Key<T>[] = [];
  for (let f = f0, i = 0; f <= f1; f += every, i++) out.push({ f, v: make(i), ease: "inOut" });
  return out;
};

// ── panels: build a panel's life from transition segments ─────────────────────────────────────
export type Seg = { keys: PanelKey[]; moves: { type: SegmentType; at: number }[]; contentCenter?: Pt; shatter?: PanelCue["shatter"] };
const offscreen = (dir: "left" | "right" | "top" | "bottom"): Pt => (dir === "left" ? [-1200, 0] : dir === "right" ? [1200, 0] : dir === "top" ? [0, -2000] : [0, 2000]);
const WIDE: Quad = pairWide([[-260, -40], [1340, -40], [1340, 1960], [-260, 1960]], [[-460, -40], [2380, -40], [2380, 1120], [-460, 1120]]);
export const enter = {
  /** appears instantly */
  cut: (at: number, quad: Quad, border = 8): Seg => ({ keys: [{ f: at, v: { pts: quad, border, scale: 1, rot: 0, move: [0, 0], iris: 0 } }], moves: [{ type: "cut", at }] }),
  /** slams onto the page (zoom-in overshoot, or flying in from a side) */
  slam: (at: number, quad: Quad, from: "zoom" | "left" | "right" | "top" | "bottom" = "zoom", dur = 10, border = 9): Seg => {
    const m: Pt = from === "zoom" ? [0, -50] : [offscreen(from)[0] * 0.08, offscreen(from)[1] * 0.04];
    return {
      keys: [
        { f: at, v: { pts: quad, border, scale: from === "zoom" ? 1.4 : 1.12, rot: (from === "left" ? -7 : 7), move: m, iris: 0 } },
        { f: at + dur, v: { scale: 1, rot: 0, move: [0, 0] }, ease: "outBackBig" },
      ],
      moves: [{ type: "slam", at: at + Math.round(dur * 0.35) }],
    };
  },
  /** slides in from off-screen */
  slide: (at: number, quad: Quad, from: "left" | "right" | "top" | "bottom" = "right", dur = 14, border = 9): Seg => ({
    keys: [
      { f: at, v: { pts: quad, border, scale: 1, rot: from === "left" ? -4 : 4, move: offscreen(from), iris: 0 } },
      { f: at + dur, v: { move: [0, 0], rot: 0 }, ease: "expoOut" },
    ],
    moves: [{ type: "slide", at }],
  }),
  /** opens as a circle from its centre (iris in) */
  iris: (at: number, quad: Quad, dur = 16, r = 1250, border = 10): Seg => ({
    keys: [
      { f: at, v: { pts: quad, border, iris: 1, scale: 1, rot: 0, move: [0, 0] } },
      { f: at + dur, v: { iris: r }, ease: "inOut" },
      { f: at + dur + 1, v: { iris: 0 }, ease: "hold" },
    ],
    moves: [{ type: "iris", at }],
  }),
  /** slashes in diagonally from a side: a half-screen split for `hold` frames, then takes the whole frame */
  slash: (at: number, from: "left" | "right" = "right", hold = 6, border = 12): Seg => {
    const s = from === "right" ? 1 : -1;
    const Q = (x0: number, x1: number, sk: number, w: number, h: number): Quad =>
      s > 0 ? [[x0, -40], [x1, -40], [x1 - 0, h + 40], [x0 - sk, h + 40]] : [[w - x1, -40], [w - x0, -40], [w - x0 + sk, h + 40], [w - x1, h + 40]];
    // the same slash on the wide page: x stretched to 1920, the cut keeps its lean
    const P = (x0: number, x1: number, sk: number): Quad => pairWide(Q(x0, x1, sk, 1080, 1920), Q((x0 * 16) / 9, (x1 * 16) / 9, sk, 1920, 1080));
    return {
      keys: [
        { f: at, v: { pts: P(1150, 1750, 200), border, scale: 1, rot: 0, move: [0, 0], iris: 0 } },
        { f: at + 8, v: { pts: P(600, 1340, 220) }, ease: "expoOut" },
        { f: at + 8 + hold, v: { pts: P(600, 1340, 220) } },
        { f: at + 15 + hold, v: { pts: WIDE, border: 0 }, ease: "expoOut" },
      ],
      moves: [{ type: "slash", at }],
      contentCenter: [540, 960],
    };
  },
};
export const move = {
  /** grows to full screen (or to `to`) — the camera inside should push in at the same time */
  expand: (at: number, dur = 20, to: Quad = FULL): Seg => ({ keys: [{ f: at, v: {} }, { f: at + dur, v: { pts: to, border: to === FULL ? 0 : 9, rot: 0, scale: 1 }, ease: "inOut" }], moves: [{ type: "expand", at }] }),
  /** shrinks into a printed panel (freeze frames, endings) */
  collapse: (at: number, to: Quad, dur = 14, rot = -1, border = 11): Seg => ({ keys: [{ f: at, v: {} }, { f: at + dur, v: { pts: to, border, rot }, ease: "outBack" }], moves: [{ type: "collapse", at }] }),
  /** falls off the page */
  fall: (at: number, dur = 22, dir: "down" | "left" | "right" = "down", rot = 14): Seg => ({
    keys: [{ f: at, v: {} }, { f: at + dur, v: { move: dir === "down" ? [rot * 9, 1500] : dir === "left" ? [-1300, 300] : [1300, 300], rot }, ease: "in" }],
    moves: [{ type: "fall", at }],
  }),
  /** slow living drift of a panel on the page */
  drift: (at: number, dur: number, m: Pt = [0, -14], rot = 0.8): Seg => ({ keys: [{ f: at, v: {} }, { f: at + dur, v: { move: m, rot }, ease: "inOut" }], moves: [{ type: "drift", at }] }),
  /** punch the whole panel bigger (zoom) */
  zoom: (at: number, dur = 10, scale = 1.12): Seg => ({ keys: [{ f: at, v: {} }, { f: at + dur, v: { scale }, ease: "outBack" }], moves: [{ type: "zoom", at }] }),
  /** closes as a circle (iris out) to radius r (0 = gone) */
  irisOut: (at: number, dur = 16, r = 0, center?: Pt): Seg => ({ keys: [{ f: at, v: { iris: 1300 } }, { f: at + dur, v: { iris: Math.max(1, r) }, ease: "inOut" }], moves: [{ type: "iris", at }], contentCenter: center }),
  /** breaks into flying shards (the shot freezes at `at`) */
  shatter: (at: number, dur = 30, impact: Pt = [560, 980], cuts = [0.06, 0.29, 0.5, 0.77]): Seg => ({ keys: [], moves: [{ type: "shatter", at }], shatter: { at, dur, impact, cuts } }),
};
/** Assemble a panel cue from segments: `panel("rage", 349, 507, enter.slam(349, P1), move.expand(428), …)` */
export const panel = (shot: string, from: number, to: number, ...segs: Seg[]): PanelCue => {
  const keys = segs.flatMap((s) => s.keys).sort((a, b) => a.f - b.f);
  return {
    shot,
    from,
    to,
    geom: keys.length ? keys : [{ f: from, v: { pts: FULL, border: 0 } }],
    contentCenter: segs.find((s) => s.contentCenter)?.contentCenter,
    shatter: segs.find((s) => s.shatter)?.shatter,
    moves: segs.flatMap((s) => s.moves),
    shadow: true,
  };
};

// ── sound ─────────────────────────────────────────────────────────────────────────────────────
export const sfx = (name: string, at: number, volume: number, o: { duck?: number; until?: number } = {}): SfxCue => ({ at, src: `sfx/pv-${name}.wav`, volume, ...o });
const MOVE_SOUND: Partial<Record<SegmentType, [string, number]>> = {
  slam: ["slam", 0.34], slide: ["whooshShort", 0.3], iris: ["whoosh", 0.3], expand: ["whooshShort", 0.36], collapse: ["whooshShort", 0.28],
  fall: ["whoosh", 0.22], slash: ["swish", 0.5], shatter: ["bigShatter", 0.55], zoom: ["pop", 0.2], flip: ["tick", 0.25],
};
/**
 * Automatic sound for a story: every transition and comic FX gets its effect (placed on the event
 * frame), plus the story's own `sfx`. Loud hits should sit in VO pauses — the ducking handles the rest.
 */
export const soundsFor = (st: StoryDef): SfxCue[] => {
  const out: SfxCue[] = [];
  if (st.autoSfx !== false) {
    for (const p of st.panels) for (const m of p.moves ?? []) {
      const s = MOVE_SOUND[m.type];
      if (s) out.push(sfx(s[0], Math.max(0, m.at), s[1], m.type === "shatter" ? {} : { duck: 0.8 }));
    }
    for (const c of st.fx) {
      if (c.kind === "impact") out.push(sfx("punchDry", c.at + 1, 0.3, { duck: 0.4 }));
      else if (c.kind === "caption") out.push(sfx("tick", c.at + 2, 0.24));
      else if (c.kind === "burst") out.push(sfx("boom", c.at, 0.45, { duck: 0.6 }));
      else if (c.kind === "flood") [0, 0.25, 0.5].forEach((k, i) => out.push(sfx("splat", Math.round(c.at + c.dur * k), 0.45 - i * 0.05, { until: c.to })));
      else if (c.kind === "sfxText") out.push(sfx("hit", c.at, 0.3, { duck: 0.6 }));
      else if (c.kind === "impactFrame") out.push(sfx("punch", c.at, 0.34, { duck: 0.6 }));
      else if (c.kind === "stamp") out.push(sfx("stamp", c.at + 2, 0.45));
      else if (c.kind === "shock") out.push(sfx("freeze", c.from, 0.3, { duck: 0.8 }));
      else if (c.kind === "halftoneWipe") out.push(sfx("cloud", c.at, 0.25, { until: c.to }));
    }
  }
  out.push(...(st.sfx ?? []));
  out.sort((a, b) => a.at - b.at);
  // the same sound twice within 6 frames reads as a glitch: keep the first
  return out.filter((c, i) => !out.slice(0, i).some((p) => p.src === c.src && c.at - p.at < 6));
};
export const duckGain = (d: Ducking, f: number) => {
  let g = 1;
  for (const [s, e] of d.spans) {
    let k = 0;
    if (f >= s && f <= e) k = 1;
    else if (f < s && f > s - d.attack) k = 1 - (s - f) / d.attack;
    else if (f > e && f < e + d.release) k = 1 - (f - e) / d.release;
    g = Math.min(g, 1 - (1 - d.level) * k);
  }
  return g;
};

// ── the wide page (16:9, `--wide`): the same story re-laid for 1920×1080 ───────────────────────
// Stories are written once, for the portrait page. `wideStory` turns every panel into its landscape
// arrangement (with the zoom that keeps its subject framed), moves lettering and FX to the same spot of the
// picture they sat on, and scales the page camera. A cue's `wide: { x, y, size… }` overrides its placement.
const SX = 16 / 9, SY = 9 / 16;
const wideAt = (p: Pt): Pt => [p[0] * SX, p[1] * SY];
const within = (v: number, a: number, b: number) => Math.min(Math.max(v, a), Math.max(a, b));
const widePanel = (p: PanelCue): PanelCue => {
  let tall: Quad = FULL, wide: Quad = FULL_WIDE;
  const geom = p.geom.map((k, i) => {
    const v = { ...k.v };
    if (v.pts || i === 0) {
      tall = v.pts ?? tall;
      wide = wideQuad(tall);
      v.pts = wide;
      v.fit = wideFit(tall, wide);
    }
    if (v.move) {
      // the same share of the frame; a move that throws the panel out of the portrait frame also clears the wide one
      const m: Pt = [v.move[0] * SX, v.move[1] * SY], a = boundsOf(tall), b = boundsOf(wide);
      const out = a.x0 + v.move[0] >= 1080 || a.x1 + v.move[0] <= 0 || a.y0 + v.move[1] >= 1920 || a.y1 + v.move[1] <= 0;
      const need = Math.min(m[0] > 0 ? (1960 - b.x0) / m[0] : m[0] < 0 ? (-40 - b.x1) / m[0] : Infinity, m[1] > 0 ? (1120 - b.y0) / m[1] : m[1] < 0 ? (-40 - b.y1) / m[1] : Infinity);
      const t = out && Number.isFinite(need) ? Math.max(1, need) : 1;
      v.move = [m[0] * t, m[1] * t];
    }
    return { ...k, v };
  });
  return { ...p, geom, contentCenter: p.contentCenter && wideAt(p.contentCenter), shatter: p.shatter && { ...p.shatter, impact: wideAt(p.shatter.impact) } };
};
/** the panel under a portrait page point at frame t, next to its landscape twin */
const panelAt = (st: StoryDef, wide: PanelCue[], t: number, x: number, y: number) => {
  for (let i = st.panels.length - 1; i >= 0; i--) {
    const p = st.panels[i];
    if (t < p.from || t > p.to) continue;
    const g = samplePanel(p.geom, t), a = boundsOf(g.pts);
    if (!inQuad(g.pts, x - g.move[0], y - g.move[1])) continue;
    const u = (x - g.move[0] - a.x0) / (a.x1 - a.x0), v = (y - g.move[1] - a.y0) / (a.y1 - a.y0);
    const gw = samplePanel(wide[i].geom, t), b = boundsOf(gw.pts), c = p.contentCenter ?? centerOf(g.pts), cw = wide[i].contentCenter ?? centerOf(gw.pts);
    return { u, v, a, b, size: sizeOf(g.pts), wsize: sizeOf(gw.pts), k: gw.fit, m: gw.move, tm: g.move, dx: x - g.move[0] - c[0], dy: y - g.move[1] - c[1], cw };
  }
  return null;
};
/**
 * Where page-level lettering goes: the same spot of the panel it was laid on. `box` = its extent around (x, y)
 * on the portrait page [left, top, right, bottom]; the result places a w × h box of the wide page (`room` = the
 * width it may take).
 */
const wideSpot = (st: StoryDef, wide: PanelCue[], t: number, x: number, y: number, box: [number, number, number, number] = [0, 0, 0, 0]) => {
  const s = panelAt(st, wide, t, x, y);
  if (!s) return { room: 1824, at: (): Pt => widePt([x, y]) };
  const [tw, th] = s.size, [ww, wh] = s.wsize;
  if (tw > 700 && th > 1000) {
    // one picture fills the page: spread across the wide frame; in height the edge that faces the middle of the
    // picture keeps its place IN the picture, so lettering above / below a face stays clear of it at the new zoom
    const edge = (h: number) => (s.dy + box[3] < 0 ? (s.dy + box[3]) * s.k - (h * box[3]) / (box[3] - box[1] || 1) : s.dy + box[1] > 0 ? (s.dy + box[1]) * s.k - (h * box[1]) / (box[3] - box[1] || 1) : s.dy * s.k);
    return { room: 1824, at: (_w: number, h: number): Pt => [s.b.x0 + s.u * (s.b.x1 - s.b.x0) + s.m[0], s.cw[1] + edge(h) + s.m[1]] };
  }
  // a panel of a page: the same relative spot; when a band became a column, "beside the subject" becomes "above /
  // below it". Lettering that sat inside its panel stays inside it.
  const flip = tw > th !== ww > wh, inside = x + box[0] >= s.a.x0 + s.tm[0] && x + box[2] <= s.a.x1 + s.tm[0] && y + box[1] >= s.a.y0 + s.tm[1] && y + box[3] <= s.a.y1 + s.tm[1];
  const px = s.b.x0 + (flip ? s.v : s.u) * (s.b.x1 - s.b.x0), py = s.b.y0 + (flip ? s.u : s.v) * (s.b.y1 - s.b.y0);
  return {
    room: ww * 0.94,
    at: (w: number, h: number): Pt => {
      // the box hangs from (x, y) the way it did on the portrait page (centred, or from its top-left corner)
      const ax = box[0] < 0 ? w / 2 : 0, ay = box[1] < 0 ? h / 2 : 0, e = 14;
      return inside ? [within(px, s.b.x0 + e + ax, s.b.x1 - e - w + ax) + s.m[0], within(py, s.b.y0 + e + ay, s.b.y1 - e - h + ay) + s.m[1]] : [px + s.m[0], py + s.m[1]];
    },
  };
};
/** where a focus device (speed lines, shock lines) goes: it follows the 3D picture under it, with its zoom */
const wideFocus = (st: StoryDef, wide: PanelCue[], t: number, x: number, y: number): { p: Pt; k: number } => {
  const s = panelAt(st, wide, t, x, y);
  return s ? { p: [s.cw[0] + s.dx * s.k + s.m[0], s.cw[1] + s.dy * s.k + s.m[1]], k: within(s.k, 0.6, 1) } : { p: widePt([x, y]), k: 0.71 };
};
/** lettering keeps its weight on the wide page — a touch smaller, because the frame is only 1080 high */
const WIDE_TEXT = 0.85;
/** The story on the 16:9 page (1920×1080). Pure: the portrait story is not changed. */
export const wideStory = (st: StoryDef): StoryDef => {
  if (st.wide) return st;
  const panels = st.panels.map(widePanel);
  const lettered = new Map<string, Pt>(); // a burst behind lettering goes where its lettering went
  const place = (c: FxCue): FxCue => {
    const o = c.wide ?? {};
    switch (c.kind) {
      case "impact":
      case "sfxText":
      case "stamp":
      case "caption": {
        const box = c.kind === "caption", pad = box ? 80 : 0;
        const per = c.kind === "impact" && c.font === "title" ? TEXT_W.title : TEXT_W[c.kind];
        const size0 = c.size ?? (box ? 84 : 110), w0 = c.text.length * size0 * per + pad;
        // a caption box hangs from its top-left corner, the other lettering is centred on its point
        const spot = wideSpot(st, panels, c.at + Math.min(12, (c.to - c.at) / 2), c.x, c.y, box ? [0, 0, w0, size0 + 50] : [-w0 / 2, -size0 / 2, w0 / 2, size0 / 2]);
        const size = Math.round(Math.min(size0 * WIDE_TEXT, (spot.room - pad) / (c.text.length * per)));
        const w = c.text.length * size * per + pad, h = box ? size + 50 : size * 1.2;
        const p = spot.at(w, box ? h : size);
        const at = box ? { x: within(p[0], 48, 1872 - w), y: within(p[1], 44, 1036 - h) } : { x: within(p[0], 48 + w / 2, 1872 - w / 2), y: within(p[1], 24 + h / 2, 1056 - h / 2) };
        const out = { ...c, x: o.x ?? Math.round(at.x), y: o.y ?? Math.round(at.y), size: o.size ?? size, ...(o.rot !== undefined && c.kind !== "caption" && { rot: o.rot }) };
        lettered.set(`${c.x},${c.y}`, [out.x, out.y]);
        return out;
      }
      case "burst": {
        const p = lettered.get(`${c.x},${c.y}`) ?? wideSpot(st, panels, c.at + Math.min(12, c.dur / 2), c.x, c.y).at(0, 0);
        return { ...c, x: o.x ?? Math.round(p[0]), y: o.y ?? Math.round(p[1]), r: o.r ?? Math.round(c.r * WIDE_TEXT) };
      }
      case "speed": {
        const { p, k } = wideFocus(st, panels, c.from + Math.min(12, (c.to - c.from) / 2), c.cx ?? 540, c.cy ?? 960);
        return { ...c, cx: o.cx ?? Math.round(p[0]), cy: o.cy ?? Math.round(p[1]), inner: o.inner ?? Math.round((c.inner ?? 520) * k) };
      }
      case "shock": {
        const { p, k } = wideFocus(st, panels, c.from + Math.min(12, (c.to - c.from) / 2), c.x, c.y);

const blob = (cx: number, cy: number, r: number, seed: number, lobes = 11) => {

const R = BODY.headR;
/** z of the head surface at head-space (x, y) */
const surfZ = (x: number, y: number) => {
  const r = headRadiusAt(y / R) * R;
  return Math.sqrt(Math.max(r * r - x * x, 0)) * 0.92;
};
const faceYaw = (x: number, y: number) => Math.atan2(x, Math.max(surfZ(x, y), 0.05)) * 0.85;
const v3 = (a: V3) => new Vector3(a[0], a[1], a[2]);

const Seg: React.FC<{ a: Vector3; b: Vector3; r0: number; r1: number; sw: SwatchKey; ink?: number; double?: boolean }> = ({ a, b, r0, r1, sw, ink = 3.5, double }) => {

const handQuat = (elbow: Vector3, wrist: Vector3, twist: number) => {
  const y = wrist.clone().sub(elbow).normalize();
  let x = new Vector3().crossVectors(y, new Vector3(0, 0, 1));
  if (x.lengthSq() < 1e-4) x = new Vector3(1, 0, 0);
  x.normalize();
  const z = new Vector3().crossVectors(x, y).normalize();
  const q = new Quaternion().setFromRotationMatrix(new Matrix4().makeBasis(x, y, z));
  return q.multiply(new Quaternion().setFromAxisAngle(new Vector3(0, 1, 0), twist));
};

    />
  </group>

const Eye: React.FC<{ side: 1 | -1; e: Expr; open: number; lashes: boolean }> = ({ side, e, open, lashes }) => {
  const f = useShotFrame();
  const k = e.eyeSize;
  const x = side * 0.148;
  const y = 0.0 + (k - 1) * 0.02;
  const z = surfZ(x, y) - 0.032;
  const yaw = faceYaw(x, y) * 1.12;
  if (e.eyeMode === "happy") {
    return (
      <group position={[x, y - 0.01, z + 0.03]} rotation={[0, yaw, 0]}>
        <Toon geo={G.torusArc(0.07, 0.017, Math.PI)} sw="ink" mat={{ flat: true }} outline={0} scale={[k, k * 0.95, 1]} />
      </group>
    );
  }
  if (e.eyeMode === "heart" || e.eyeMode === "star") {
    const beat = 1 + 0.08 * Math.max(0, Math.sin(f * 0.3));
    return (
      <group position={[x, y, z + 0.035]} rotation={[0, yaw, e.eyeMode === "star" ? side * 0.15 : 0]} scale={beat * k}>
        {e.eyeMode === "heart" ? (
          <Toon geo={G.heart()} sw="red" mat={{ spec: 1, specSize: 0.05 }} outline={3} scale={[0.085, 0.085, 0.04]} />
        ) : (
          <Toon geo={G.extrude("starEye", () => starShape(1, 0.48), 0.2, 0.05)} sw="yellow" mat={{ spec: 1, specSize: 0.05 }} outline={3} scale={[0.1, 0.1, 0.12]} />
        )}
      </group>
    );
  }
  if (e.eyeMode === "shut" || open < 0.12) {
    return (
      <group position={[x, y - 0.02, z + 0.035]} rotation={[0, yaw, Math.PI / 2 + side * 0.08]}>
        <Toon geo={G.capsule(0.016, 0.11)} sw="ink" mat={{ flat: true }} outline={0} scale={[1, k, 1]} />
      </group>
    );
  }
  const ex = 0.088 * k;
  const ey = 0.108 * k * open;
  const px = e.look[0] * 0.034 * k;
  const py = e.look[1] * 0.04 * k * Math.min(open, 1);
  const pr = 0.047 * e.pupil * k;
  return (
    <group position={[x, y, z]} rotation={[0, yaw, 0]}>
      <Toon geo={G.sphere()} sw="white" mat={{ cut: -0.25, halftone: 0.6 }} outline={3.2} scale={[ex, ey, 0.042]} />
      <Toon geo={G.sphere()} sw="ink" mat={{ flat: true }} outline={0} position={[px, py, 0.028]} scale={[pr, Math.min(pr * 1.18, ey * 0.86), 0.022]} />
      <Toon geo={G.sphereLo()} sw="white" mat={{ flat: true }} outline={0} position={[px + pr * 0.42, py + pr * 0.5, 0.043]} scale={0.013 * k * Math.max(0.6, e.pupil)} />
      {lashes ? (
        <>
          <Toon geo={G.capsule(0.011, 0.04)} sw="ink" mat={{ flat: true }} outline={0} position={[side * ex * 0.95, ey * 0.62, 0.02]} rotation={[0, 0, -side * 0.95]} />
          <Toon geo={G.capsule(0.01, 0.03)} sw="ink" mat={{ flat: true }} outline={0} position={[side * ex * 0.7, ey * 0.9, 0.025]} rotation={[0, 0, -side * 0.55]} />
        </>
      ) : null}
    </group>
  );
};

const wobblePts = (n = 9): [number, number, number][] =>
  Array.from({ length: n }, (_, i) => {

    return [(t - 0.5) * 0.15, Math.sin(t * Math.PI * 3) * 0.012, 0];

const Mouth: React.FC<{ e: Expr; f: number }> = ({ e, f }) => {
  const y = -0.205;
  const z = surfZ(0, y) - 0.004;
  const w = e.mouthW;
  const o = e.mouthOpen;
  let body: React.ReactNode = null;
  switch (e.mouth) {
    case "smile":
      body = <Toon geo={G.torusArc(0.07, 0.015, Math.PI)} sw="ink" mat={{ flat: true }} outline={0} rotation={[0, 0, Math.PI]} scale={[w, w * 0.85, 1]} position={[0, 0.04 * w, 0]} />;
      break;
    case "smirk":
      body = <Toon geo={G.torusArc(0.06, 0.015, Math.PI * 0.62)} sw="ink" mat={{ flat: true }} outline={0} rotation={[0, 0, Math.PI + 0.55]} scale={[w, w * 0.8, 1]} position={[0.03, 0.03, 0]} />;
      break;
    case "frown":
      body = <Toon geo={G.torusArc(0.06, 0.015, Math.PI)} sw="ink" mat={{ flat: true }} outline={0} scale={[w, w * 0.6, 1]} position={[0, -0.02, 0]} />;
      break;
    case "flat":
      body = <Toon geo={G.capsule(0.014, 0.1)} sw="ink" mat={{ flat: true }} outline={0} rotation={[0, 0, Math.PI / 2]} scale={[1, w, 1]} />;
      break;
    case "wobble": {
      const step = Math.floor(f / 2);
      body = <Toon geo={G.tube("wobble", wobblePts(), 0.013)} sw="ink" mat={{ flat: true }} outline={0} scale={[w, 1 + 0.35 * Math.sin(step * 1.7), 1]} />;
      break;
    }
    case "chew": {
      const ch = Math.sin(f * 0.55);
      body = <Toon geo={G.torusArc(0.045, 0.014, Math.PI)} sw="ink" mat={{ flat: true }} outline={0} rotation={[0, 0, Math.PI + ch * 0.12]} position={[0.015 * ch, 0.02, 0]} scale={[w, 0.7 + 0.25 * ch, 1]} />;
      break;
    }
    default: {
      const round = e.mouth === "o";
      const sx = (round ? 0.05 : 0.078) * w;
      const sy = 0.012 + (round ? 0.062 : 0.058) * o;
      body = (
        <group>
          <Toon geo={G.sphere()} sw="mouth" mat={{ flat: true }} outline={3} scale={[sx, sy, 0.03]} />
          <Toon geo={G.sphere()} sw="tongue" mat={{ flat: true }} outline={0} position={[0, -sy * 0.45, 0.012]} scale={[sx * 0.62, sy * 0.42, 0.022]} />
          {e.mouth === "shout" || e.mouth === "grin" ? (
            <Toon geo={G.roundedBox(1, 1, 1, 0.3)} sw="white" mat={{ flat: true }} outline={0} position={[0, sy * 0.62, 0.02]} scale={[sx * 1.35, sy * 0.32, 0.012]} />
          ) : null}
        </group>
      );
    }
  }
  return (
    <group position={[0, y, z]} rotation={[0.34, 0, 0]}>
      {body}
    </group>
  );
};

const Drop: React.FC<{ p: V3; f: number; phase: number; s: number; fall?: number; period?: number }> = ({ p, f, phase, s, fall = 0.12, period = 70 }) => {
  const cyc = ((f + phase) % period) / period;
  const sc = s * (cyc < 0.15 ? cyc / 0.15 : cyc > 0.85 ? (1 - cyc) / 0.15 : 1);
  if (sc <= 0.01) return null;
  return (
    <group position={[p[0], p[1] - cyc * fall, p[2]]} scale={sc}>
      <Toon geo={G.sphere()} sw="sky" mat={{ spec: 1, specSize: 0.12, cut: 0.0 }} outline={2.5} scale={[0.034, 0.036, 0.03]} />
      <Toon geo={G.cone(20)} sw="sky" mat={{ cut: 0.0 }} outline={2.5} position={[0, 0.042, 0]} scale={[0.031, 0.06, 0.027]} />
    </group>
  );
};

const hijabRim = () =>
  memoGeo("hijabRim", () => {
    const pts: Vector3[] = [];
    for (let i = 0; i < 48; i++) {
      const a = (i / 48) * Math.PI * 2;
      const ux = HOLE[0] * Math.cos(a);
      const uy = HOLE[2] + HOLE[1] * Math.sin(a);
      const r = headRadiusAt(uy);
      const uz = Math.sqrt(Math.max(r * r - ux * ux, 0.0225));
      pts.push(new Vector3(ux * HIJAB_SCALE[0] + HIJAB_OFF[0], uy * HIJAB_SCALE[1] + HIJAB_OFF[1], uz * HIJAB_SCALE[2] + HIJAB_OFF[2]));
    }
    return new TubeGeometry(new CatmullRomCurve3(pts, true, "centripetal"), 96, 0.014, 8, true);
  });

export type ComicCharacterProps = {
  look: Look;
  pose: Pose;
  expr: Expr;
  heldL?: React.ReactNode;
  heldR?: React.ReactNode;
  holdRotL?: V3;
  holdRotR?: V3;
  holdOffL?: V3;
  holdOffR?: V3;
  shadow?: boolean;
  /** things parented to the head centre (thought bubbles, steam, icons…) */
  headExtras?: React.ReactNode;
  /** things placed in torso space (a phone held in both hands, a bowl on the lap…) */
  torsoExtras?: React.ReactNode;
  /** head + neck only (expression sheets, inserts) */
  bust?: boolean;
};

export const ComicCharacter: React.FC<ComicCharacterProps> = ({
  look: L,
  pose,
  expr,
  heldL,
  heldR,
  holdRotL = [0, 0, 0],
  holdRotR = [0, 0, 0],
  holdOffL = [0, 0, 0],
  holdOffR = [0, 0, 0],
  shadow = true,
  headExtras,
  torsoExtras,
  bust = false,
}) => {
  const f = useShotFrame();
  const seed = Math.abs(Math.round(hash(L.id.length * 13.7 + L.id.charCodeAt(0)) * 50));
  let open = expr.eyeOpen;
  if (expr.blink) {
    const period = 150 + Math.round(hash(seed) * 30);
    const t = (f + seed * 37) % period;
    if (t < 7) open *= Math.abs(t - 3.5) / 3.5;
  }
  const breathe = Math.sin((f + seed * 20) / 38) * 0.012;
  const tr = expr.tremble;
  const trem: V3 = [noise1(f * 0.9, seed) * 0.012 * tr, noise1(f * 1.1, seed + 3) * 0.008 * tr, 0];
  const B = BUILD[L.build];
  const sq = pose.squash;
  const pelvis = new Vector3(pose.pelvis[0], BODY.pelvisH + pose.pelvis[1], pose.pelvis[2]);

  const legs = ([1, -1] as const).map((s) => {
    const hip = pelvis.clone().add(new Vector3(s * BODY.hipX, BODY.hipY, 0));
    const [knee, ankle] = solveIK(hip, v3(s === 1 ? pose.footL : pose.footR), BODY.thigh, BODY.shin, v3(s === 1 ? pose.kneeL : pose.kneeR));
    return { s, hip, knee, ankle };
  });
  const arms = ([1, -1] as const).map((s) => {
    const sh = new Vector3(s * BODY.shoulderX * B.torso[0], BODY.shoulderY + pose.shrug * 0.06, 0);
    const [elbow, wrist] = solveIK(sh, v3(s === 1 ? pose.handL : pose.handR), BODY.upperArm, BODY.foreArm, v3(s === 1 ? pose.elbowL : pose.elbowR));
    return { s, sh, elbow, wrist };
  });
  const top = L.topSw;
  const skin = L.skin;
  const long = LONG_SLEEVES[L.top];
  const torsoGeo = G.lathe(`torso_${L.top}`, PROFILES[L.top]);
  const legSw = (part: "thigh" | "shin"): SwatchKey => (L.bottom === "pants" ? L.bottomSw : L.bottom === "shorts" ? (part === "thigh" ? L.bottomSw : skin) : skin);
  const sc = B.scale;
  const blush = Math.max(expr.blush, L.blush);
  const spineX = pose.spine[0] + B.spine;
  return (
    <group position={pose.pos} rotation={[0, pose.rotY, 0]} scale={[sc / Math.sqrt(sq), sc * sq, sc / Math.sqrt(sq)]}>
      {shadow && !bust ? (
        <Toon geo={G.circle()} sw="ink" mat={{ dotOnly: true, pattern: 0.42 }} outline={0} position={[0, 0.006, 0.02]} rotation={[-Math.PI / 2, 0, 0]} scale={[0.55, 0.42, 1]} />
      ) : null}

const spikyShape = (r1: number, r2: number, n: number, seed: number) => {
  const s = new Shape();
  for (let i = 0; i < n * 2; i++) {
    const a = (i / (n * 2)) * Math.PI * 2;
    const r = i % 2 === 0 ? r1 * (0.85 + 0.25 * (hash(seed + i) + 1) * 0.5) : r2;
    const x = Math.cos(a) * r;
    const y = Math.sin(a) * r;
    if (i === 0) s.moveTo(x, y);
    else s.lineTo(x, y);
  }
  s.closePath();
  return s;
};
