// CERDIK camera solver — picks a camera for each shot automatically. For a shot kind and its subjects it tries a fan
// of angles and distances around them and scores each: are the subjects' heads and bodies visible (rays tested
// against the world's colliders, the other people and big props), do they fit the narrow 9:16 frame, does the
// camera see their faces, is there something behind them (not empty sky), is it different from the previous shot.
// Pure TypeScript (no three).

import type { Box, PCam, ShotKind, V3, WorldMeta } from "./types.ts";

export type Subj = { id: string; p: V3; face: number; s: number; kind: "person" | "prop"; size?: V3 };
export type Occ = { c: V3; r: number; id: string };

const D = Math.PI / 180;
const add = (a: V3, b: V3, k = 1): V3 => [a[0] + b[0] * k, a[1] + b[1] * k, a[2] + b[2] * k];
const sub = (a: V3, b: V3): V3 => [a[0] - b[0], a[1] - b[1], a[2] - b[2]];
const len = (a: V3) => Math.hypot(a[0], a[1], a[2]);
const nrm = (a: V3): V3 => { const l = len(a) || 1; return [a[0] / l, a[1] / l, a[2] / l]; };
const dot = (a: V3, b: V3) => a[0] * b[0] + a[1] * b[1] + a[2] * b[2];
const wrapA = (a: number) => Math.atan2(Math.sin(a), Math.cos(a));

/** Head centre / chest of a person, centre of a prop. */
export const headOf = (s: Subj): V3 => (s.kind === "person" ? add(s.p, [0, 1.25 * s.s, 0]) : add(s.p, [0, (s.size?.[1] ?? 0.3) * 0.55, 0]));
export const chestOf = (s: Subj): V3 => (s.kind === "person" ? add(s.p, [0, 0.82 * s.s, 0]) : add(s.p, [0, (s.size?.[1] ?? 0.3) * 0.4, 0]));
export const feetOf = (s: Subj): V3 => add(s.p, [0, 0.08, 0]);

/** Ray (o + d·t, t ∈ [0, 1]) against an AABB: entry t or Infinity. */
const rayBox = (o: V3, d: V3, b: Box) => {
  let t0 = 0, t1 = 1;
  for (let i = 0; i < 3; i++) {
    if (Math.abs(d[i]) < 1e-9) { if (o[i] < b.min[i] || o[i] > b.max[i]) return Infinity; continue; }
    let a = (b.min[i] - o[i]) / d[i], c = (b.max[i] - o[i]) / d[i];
    if (a > c) [a, c] = [c, a];
    t0 = Math.max(t0, a); t1 = Math.min(t1, c);
    if (t0 > t1) return Infinity;
  }
  return t0;
};
const raySphere = (o: V3, d: V3, c: V3, r: number) => {
  const m = sub(o, c), b = dot(m, d), cc = dot(m, m) - r * r, a = dot(d, d);
  const disc = b * b - a * cc;
  if (disc < 0) return Infinity;
  const t = (-b - Math.sqrt(disc)) / a;
  return t >= 0 && t <= 1 ? t : cc < 0 ? 0 : Infinity;
};
const inside = (p: V3, b: Box, pad = 0) => p[0] > b.min[0] - pad && p[0] < b.max[0] + pad && p[1] > b.min[1] - pad && p[1] < b.max[1] + pad && p[2] > b.min[2] - pad && p[2] < b.max[2] + pad;

/** Fraction of the way from a to b that is clear (1 = nothing in between). */
const clear = (a: V3, b: V3, W: WorldMeta, occ: Occ[], skip: Set<string>) => {
  const d = sub(b, a);
  let t = 1;
  for (const c of W.colliders) t = Math.min(t, rayBox(a, d, c));
  for (const o of occ) if (!skip.has(o.id)) t = Math.min(t, raySphere(a, d, o.c, o.r));
  return t;
};

export type SolveOpts = {
  aspect: number;
  prevAz?: number;
  prevId?: string;
  side?: "left" | "right";
  move?: string;
  /** person whose shoulder frames an "ots" shot */
  over?: Subj;
  /** a direction the subject must be seen from (cutaway: the open side of a vessel), world yaw */
  fromYaw?: number;
};

/** Camera for one shot. `subs[0]` is the main subject; `occ` = everything else that can block the view. */
export const solveShot = (kind: ShotKind, subs: Subj[], W: WorldMeta, occ: Occ[], o: SolveOpts): Solve => {
  if (kind === "establish" || kind === "aerial") {
    const v = W.views[kind] ?? W.views.establish ?? Object.values(W.views)[0];
    return { a: v.a, b: v.b, vis: 1, blur: 0, box: 34, az: 0 };
  }
  if (kind === "ots" && o.over) return solveOts(subs[0], o.over, W, occ, o);
  const K = KIND[kind] ?? KIND.medium;
  const main = subs[0];
  const skip = new Set(subs.map((s) => s.id));
  const hfov = (f: number) => Math.atan(Math.tan((f * D) / 2) * o.aspect);
  // group centre / extent (wide shots frame everyone)
  const aim: V3 = subs.length > 1 && (kind === "wide" || kind === "medium")
    ? (() => { const c = subs.map(K.aim).reduce((x, y) => add(x, y), [0, 0, 0] as V3); return [c[0] / subs.length, c[1] / subs.length, c[2] / subs.length]; })()
    : K.aim(main);
  const baseFace = main.face;
  let best: (Solve & { score: number }) | null = null;
  const dists = [1, 0.82, 1.2, 0.68];
  const els = kind === "cutaway" ? [K.el, K.el + 5] : kind === "close" || kind === "reaction" ? [K.el, K.el + 9, K.el + 18] : [K.el, K.el + 9, K.el + 18, K.el - 5];
  for (const dk of dists)
    for (const el of els)
      for (const off of K.offs) {
        const A = baseFace + off * D;
        // distance: vertical extent, and the lateral spread of the group across the view
        const fov = K.fov;
        let d = K.ext(main) / (2 * Math.tan((fov * D) / 2));
        if (subs.length > 1) {
          const right: V3 = [Math.cos(A), 0, -Math.sin(A)];
          const xs = subs.map((s) => dot(sub(s.p, aim), right));
          const spread = Math.max(...xs) - Math.min(...xs) + 0.7;
          d = Math.max(d, spread / 2 / Math.tan(hfov(fov)));
          const ys = subs.map((s) => headOf(s)[1]);
          d = Math.max(d, (Math.max(...ys) + 0.25) / (2 * Math.tan((fov * D) / 2)) * 1.25);
        }
        d *= dk;
        const dir: V3 = [Math.cos(el * D) * Math.sin(A), Math.sin(el * D), Math.cos(el * D) * Math.cos(A)];
        const pos = add(aim, dir, d);
        if (kind === "low") pos[1] = Math.max(0.3, Math.min(pos[1], 0.55));
        if (kind === "ground") pos[1] = 0.32;
        // validity
        if (pos[1] < 0.2) continue;
        if (pos[0] < W.bounds.min[0] || pos[0] > W.bounds.max[0] || pos[2] < W.bounds.min[2] || pos[2] > W.bounds.max[2]) continue;
        if (W.colliders.some((c) => inside(pos, c, 0.2))) continue;
        if (occ.some((q) => !skip.has(q.id) && len(sub(pos, q.c)) < q.r + 0.3)) continue;
        // never right on top of a person in the shot (a head filling the frame)
        const minD = kind === "close" || kind === "reaction" ? 0.55 : kind === "insert" || kind === "cutaway" || kind === "top" || kind === "ground" ? 0.45 : kind === "wide" ? 2.0 : 1.4;
        if (subs.some((q) => q.kind === "person" && len(sub(pos, headOf(q))) < minD && q !== main)) continue;
        // visibility of the subjects (main counts most)
        const pts = subs.flatMap((s, i) => (s.kind === "person" ? [[headOf(s), i], [chestOf(s), i]] : [[K.aim(s), i], [add(s.p, [0, (s.size?.[1] ?? 0.3) * 0.85, 0]), i]]) as [V3, number][]);
        let vMain = 0, nMain = 0, vAll = 0;
        for (const [q, i] of pts) {
          const c = clear(pos, q, W, occ, new Set([subs[i].id])) >= 0.985 ? 1 : 0;
          if (i === 0) { vMain += c; nMain++; }
          vAll += c;
        }
        vMain /= nMain || 1;
        vAll /= pts.length || 1;
        // foreground clutter at the frame edges (something big right in front of the lens)
        const fwd = nrm(sub(aim, pos)), rgt = nrm([fwd[2], 0, -fwd[0]]);
        let clutter = 0;
        for (const k of [-0.55, 0.55]) {
          const edge = add(aim, rgt, Math.tan(hfov(fov)) * d * k);
          const tt = clear(pos, edge, W, occ, skip);
          if (tt < 0.55) clutter += 1;
        }
        // depth behind the subject: a ray past it should travel a while before meeting the world (not a wall
        // right behind, not empty sky)
        const far = add(aim, fwd, 45);
        const hitT = clear(aim, far, W, [], skip);
        const back = hitT < 1 ? Math.min(1, (hitT * 45) / 9) : 0.3;
        // faces: the people face the camera (all of them in a group, the main one counts double)
        const people = subs.filter((q) => q.kind === "person");
        const faceOf = (q: Subj) => Math.cos(wrapA(Math.atan2(pos[0] - q.p[0], pos[2] - q.p[2]) - q.face));
        const front = people.length ? (people.reduce((acc, q, i) => acc + faceOf(q) * (i === 0 ? 2 : 1), 0) / (people.length + 1)) : 1;
        let score = 4 * vMain + 1.6 * vAll + (kind === "insert" || kind === "top" || kind === "cutaway" ? 0 : 1.2 * front) - 0.7 * clutter + 0.6 * back - (kind === "wide" || kind === "medium" || kind === "close" ? 1.3 : 0.3) * Math.max(0, 1 - dk) - 0.15 * Math.max(0, dk - 1) - 0.012 * Math.abs(el - K.el);
        if (o.prevAz !== undefined && o.prevId === main.id && Math.abs(wrapA(A - o.prevAz)) < 18 * D) score -= 0.9;
        if (o.side) {
          // screen side of the camera relative to the subject's facing: left = camera on the subject's right
          const wantSign = o.side === "left" ? 1 : -1;
          if (Math.sign(off) === wantSign) score += 0.4;
        }
        if (o.fromYaw !== undefined) score += 2 * Math.cos(wrapA(A - o.fromYaw));
        if (!best || score > best.score) {
          const a: PCam = { pos, target: aim, fov };
          best = { a, b: a, vis: vMain, blur: K.blur, box: Math.max(3, Math.min(30, d * 1.2 + 2)), az: A, score };
        }
      }
  if (!best) {
    // nothing valid: straight in front, close
    const A = baseFace, d = K.ext(main) / (2 * Math.tan((K.fov * D) / 2)) * 0.7;
    const pos = add(aim, [Math.sin(A) * d, 0.3, Math.cos(A) * d]);
    return { a: { pos, target: aim, fov: K.fov }, b: { pos, target: aim, fov: K.fov }, vis: 0, blur: K.blur, box: 6, az: A, note: "no clear camera position found" };
  }
  best.b = moved(best.a, o.move ?? (kind === "insert" || kind === "cutaway" ? "push" : "push"), W);
  return best;
};

/** Over-the-shoulder: camera behind `over`'s shoulder looking at `subj`. */
const solveOts = (subj: Subj, over: Subj, W: WorldMeta, occ: Occ[], o: SolveOpts): Solve => {
  const target = add(headOf(subj), [0, -0.12 * subj.s, 0]);
  const from = headOf(over);
  const back = nrm([from[0] - target[0], 0, from[2] - target[2]]);
  const side: V3 = [back[2], 0, -back[0]];
  let best: Solve & { score: number } | null = null;
  for (const sd of [1, -1])
    for (const k of [0.75, 1.05, 0.55]) {
      const pos = add(add(add(from, back, k), side, 0.42 * sd), [0, 0.16, 0]);
      if (W.colliders.some((c) => inside(pos, c, 0.15))) continue;
      const v = clear(pos, target, W, occ, new Set([subj.id]));
      const score = (v >= 0.985 ? 4 : 0) + (o.side ? (o.side === "left" ? -sd : sd) * 0.3 : 0) - Math.abs(k - 0.75);
      if (!best || score > best.score) best = { a: { pos, target, fov: 40 }, b: { pos, target, fov: 40 }, vis: v >= 0.985 ? 1 : 0, blur: 0.8, box: 6, az: Math.atan2(pos[0] - target[0], pos[2] - target[2]), score };
    }
  if (!best) return solveShot("medium", [subj], W, occ, { ...o, over: undefined });
  best.b = moved(best.a, o.move ?? "push", W);
  return best;
};

/** The end key of a camera move. */
export const moved = (a: PCam, move: string, W: WorldMeta): PCam => {
  const v = sub(a.target, a.pos), d = len(v);
  let b: PCam = a;
  if (move === "push") b = { ...a, pos: add(a.pos, v, 0.09), fov: a.fov - 1 };
  else if (move === "pull") b = { ...a, pos: add(a.pos, v, -0.12), fov: a.fov + 1 };
  else if (move === "orbit") {
    const r = sub(a.pos, a.target), c = Math.cos(9 * D), s = Math.sin(9 * D);
    b = { ...a, pos: add(a.target, [r[0] * c + r[2] * s, r[1], -r[0] * s + r[2] * c]) };
  } else if (move === "rise") b = { ...a, pos: add(a.pos, [0, d * 0.18, 0]), target: add(a.target, [0, d * 0.05, 0]) };
  else if (move === "track") { const rgt = nrm([v[2], 0, -v[0]]); b = { ...a, pos: add(a.pos, rgt, d * 0.07), target: add(a.target, rgt, d * 0.07) }; }
  if (W.colliders.some((c) => inside(b.pos, c, 0.1))) return a;
  return b;
};

/** Problems with a final look, in plain words (empty = fine). */
export const lookProblems = (look: Record<string, unknown>): string[] => {
  const p: string[] = [];
  const known = ["build", "costume", "hair", "face", "hat", "necklace", "towel"];
  for (const k of Object.keys(look)) if (!known.includes(k)) p.push(`unknown look field "${k}" (have: ${known.join(", ")})`);
  const hex = (v: unknown, where: string) => { if (typeof v !== "string" || !HEX.test(v)) p.push(`${where} must be a colour like "#3a7bd5" (got ${JSON.stringify(v)})`); };
  const c = look.costume as Record<string, unknown> | undefined;
  if (!isObj(c)) p.push(`costume missing`);
  else for (const [k, v] of Object.entries(c)) {
    if (!COSTUME_KEYS.includes(k)) p.push(`unknown costume field "${k}" (have: ${COSTUME_KEYS.join(", ")})`);
    else if (k === "long") { if (typeof v !== "boolean") p.push(`costume.long must be true or false`); }
    else hex(v, `costume.${k}`);
  }
  const h = look.hair as Record<string, unknown> | undefined;
  if (h !== undefined) {
    if (!isObj(h)) p.push(`hair must be { "color": "#…", "style": … }`);
    else { if (!HAIR.includes(String(h.style))) p.push(`hair.style "${h.style}" unknown (have: ${HAIR.join(", ")})`); hex(h.color, "hair.color"); }
  }
  const f = look.face as Record<string, unknown> | undefined;
  if (f !== undefined) {
    if (!isObj(f)) p.push(`face must be an object`);
    else for (const [k, v] of Object.entries(f)) {
      if (k === "brows") { if (!BROWS.includes(String(v))) p.push(`face.brows "${v}" unknown (have: ${BROWS.join(", ")})`); }
      else if (k === "ink") hex(v, "face.ink");
      else if (["mustache", "beard", "glasses", "lashes", "blush"].includes(k)) { if (typeof v !== "boolean") p.push(`face.${k} must be true or false`); }
      else p.push(`unknown face field "${k}" (have: brows, mustache, beard, glasses, lashes, blush, ink)`);
    }
  }
  const t = look.hat as Record<string, unknown> | undefined;
  if (t !== undefined) {
    if (!isObj(t)) p.push(`hat must be { "kind": … }`);
    else {
      if (!HATS.includes(String(t.kind))) p.push(`hat.kind "${t.kind}" unknown (have: ${HATS.join(", ")})`);
      if (t.color !== undefined) hex(t.color, "hat.color");
      if (t.color2 !== undefined) hex(t.color2, "hat.color2");
    }
  }
  const b = look.build as Record<string, unknown> | undefined;
  if (b !== undefined) {
    if (!isObj(b)) p.push(`build must be an object`);
    else for (const [k, v] of Object.entries(b)) {
      const r = RANGE[k];
      if (!r) p.push(`unknown build field "${k}" (have: s, wide, deep, head)`);
      else if (typeof v !== "number" || v < r[0] || v > r[1]) p.push(`build.${k} must be a number ${r[0]}–${r[1]} (got ${JSON.stringify(v)})`);
    }
  }
  if (look.necklace !== undefined) hex(look.necklace, "necklace");
  if (look.towel !== undefined) hex(look.towel, "towel");
  return p;
};

export const paceWords = (text: string, t0: number, t1: number): Word[] => {
  const raw = text.split(/\s+/).filter(Boolean);
  const weights = raw.map((w) => 0.55 + w.replace(/[,.?!"“”]/g, "").length * 0.085 + (/[,.?!]["”]?$/.test(w) ? 0.5 : 0));
  const tot = weights.reduce((x, y) => x + y, 0);

  return raw.map((w, i) => { const t = t0 + ((t1 - t0) * acc) / tot; acc += weights[i]; return { w, t }; });
};
/** Sentence timing at a natural narrator pace (the genre talks fast), for previews before the voice-over exists. */
export const paceLines = (texts: string[], wps = 2.9, gap = 0.28, start = 0.25): Timed[] => {

// CERDIK faces — the 2D cartoon face drawn onto the round head.
//
// The face is a decal: a spherical patch slightly outside the head (±75° around the front) whose texture is drawn on
// a canvas. Features are designed in "face-plane" coordinates (x right as you look at the face, y up, both in head
// radii) and projected onto the sphere, so they look exactly as drawn when seen from the front and wrap naturally
// when the head turns. Expressions are discrete (like a 2D cartoon swapping faces); every combination is drawn once
// and cached.

import { CanvasTexture, LinearFilter, SRGBColorSpace } from "three";

/** Patch half-extent in longitude / latitude (radians). The decal geometry uses the same numbers. */
export const PATCH = { lon: (75 * Math.PI) / 180, lat: (75 * Math.PI) / 180 };

export type Expr =
  | "neutral" | "smug" | "happy" | "laugh" | "angry" | "surprised" | "worried" | "proud" | "sly" | "scared";

import type { FaceSpec, HairSpec } from "./look.ts";
export type { FaceSpec, HairSpec };

export type FaceState = {
  expr: Expr;
  /** mouth open 0..1 (lip flap while talking), quantised to 4 steps */
  open?: number;
  /** eyelids 0 open … 1 closed, quantised */
  blink?: number;
  /** pupils looking (x right, y up) in head radii, quantised */
  look?: [number, number];
};

type P2 = [number, number];

const projector = (W: number, H: number) => (p: P2): P2 => {
  const [x, y] = p;
  const r2 = Math.min(0.995, x * x + y * y);
  const z = Math.sqrt(1 - r2);
  const lon = Math.atan2(x, z), lat = Math.asin(Math.max(-1, Math.min(1, y)));
  return [((lon + PATCH.lon) / (2 * PATCH.lon)) * W, ((PATCH.lat - lat) / (2 * PATCH.lat)) * H];
};

// ------------------------------------------------------------------ shape helpers (face-plane coordinates)
const ellipse = (cx: number, cy: number, rx: number, ry: number, rot = 0, n = 48, a0 = 0, a1 = Math.PI * 2): P2[] => {
  const pts: P2[] = [];
  const c = Math.cos(rot), s = Math.sin(rot);
  for (let i = 0; i <= n; i++) {
    const a = a0 + ((a1 - a0) * i) / n;
    const x = Math.cos(a) * rx, y = Math.sin(a) * ry;
    pts.push([cx + x * c - y * s, cy + x * s + y * c]);
  }
  return pts;
};
/** Thick bar from a to b with rounded caps, as a polygon. */
export const bar = (a: P2, b: P2, w0: number, w1 = w0, n = 10): P2[] => {
  const dx = b[0] - a[0], dy = b[1] - a[1];
  const L = Math.hypot(dx, dy) || 1;
  const nx = -dy / L, ny = dx / L;
  const pts: P2[] = [];
  const ang = Math.atan2(dy, dx);
  for (let i = 0; i <= n; i++) {
    const t = ang - Math.PI / 2 - (Math.PI * i) / n;
    pts.push([a[0] + Math.cos(t) * w0 / 2, a[1] + Math.sin(t) * w0 / 2]);
  }
  for (let i = 0; i <= n; i++) {
    const t = ang + Math.PI / 2 - (Math.PI * i) / n;
    pts.push([b[0] + Math.cos(t) * w1 / 2, b[1] + Math.sin(t) * w1 / 2]);
  }
  void nx; void ny;
  return pts;
};
/** Quadratic curve sampled. */
const quad = (a: P2, c: P2, b: P2, n = 24): P2[] => {
  const pts: P2[] = [];
  for (let i = 0; i <= n; i++) {
    const t = i / n, u = 1 - t;
    pts.push([u * u * a[0] + 2 * u * t * c[0] + t * t * b[0], u * u * a[1] + 2 * u * t * c[1] + t * t * b[1]]);
  }
  return pts;
};

const drawEye = (C: Ctx, side: -1 | 1, kind: EyeKind, blink: number, look: P2) => {
  const cx = side * EX + look[0], cy = EY + look[1];
  const closed = blink >= 0.99;
  if (closed || kind === "closedUp") {
    // ^ shape (laughing / happy squeeze)
    stroke(C, quad([cx - 0.085, cy - 0.025], [cx, cy + 0.075], [cx + 0.085, cy - 0.025]), 0.042, INK);
    return;
  }
  if (kind === "closedDown") {
    stroke(C, quad([cx - 0.085, cy + 0.02], [cx, cy - 0.06], [cx + 0.085, cy + 0.02]), 0.042, INK);
    return;
  }
  const rx = kind === "wide" ? 0.085 : 0.068, ry = kind === "wide" ? 0.1 : 0.085;
  if (kind === "wide") fill(C, ellipse(side * EX, EY, 0.13, 0.15), "#ffffff");
  const dot = ellipse(cx, cy, rx, ry);
  // lids: cut the top (half-lidded smug / sly / blinking)
  const lid = Math.max(kind === "half" ? 0.48 : kind === "sly" ? 0.4 : 0, blink);
  if (lid <= 0.01) {
    fill(C, dot, INK);
    if (kind === "wide") fill(C, ellipse(cx + 0.025, cy + 0.03, 0.02, 0.024), "#ffffff");
    return;
  }
  const top = cy + ry - lid * 2 * ry;
  const slope = kind === "sly" ? side * 0.25 : 0;
  const pts = dot.filter((p) => p[1] <= top + slope * (p[0] - cx));
  if (pts.length > 2) fill(C, pts, INK);
  stroke(C, [[cx - rx * 1.25, top - slope * rx * 1.25], [cx + rx * 1.25, top + slope * rx * 1.25]], 0.035, INK);
};

/** Full-sphere texture (u around from the back, v from the top): white skin with the hair painted on. */
export const hairTexture = (h: HairSpec): CanvasTexture => {
  const key = JSON.stringify(h);
  const hit = hairCache.get(key);
  if (hit) return hit;
  const W = 1024, Hh = 512;
  const cv = document.createElement("canvas");
  cv.width = W;
  cv.height = Hh;
  const g = cv.getContext("2d")!;
  g.fillStyle = "#ffffff";
  g.fillRect(0, 0, W, Hh);
  // three.js sphere: u = 0.25 is the front (+z); v = 0 the top. lat (deg) → y px; lon (deg, 0 front, + toward +x) → x px.
  const X = (lonDeg: number) => (((0.25 + lonDeg / 360) % 1) + 1) % 1 * W;
  const Y = (latDeg: number) => ((90 - latDeg) / 180) * Hh;
  g.fillStyle = h.color;
  const band = (edge: (lon: number) => number) => {
    // fill from the top of the head down to edge(lon) (latitude) all the way round
    g.beginPath();
    g.moveTo(0, 0);
    for (let i = 0; i <= 360; i += 2) {
      const lon = i - 90; // u = 0 ↔ lon −90
      g.lineTo((i / 360) * W, Y(edge(lon)));
    }
    g.lineTo(W, 0);
    g.closePath();
    g.fill();
  };
  const wrapLon = (lon: number) => ((((lon + 180) % 360) + 360) % 360) - 180;
  if (h.style === "tuft" || h.style === "cap" || h.style === "side" || h.style === "bun" || h.style === "neat") {
    band((lon) => {
      const a = Math.abs(wrapLon(lon));
      // hairline high on the forehead, low at the back; little jagged fringe at the front
      const base = a < 60 ? 46 - (60 - a) * 0.05 : a < 120 ? 46 - (a - 60) * 0.55 : 13 - (a - 120) * 0.25;
      const jag = h.style === "neat" || h.style === "bun" ? 0 : a < 70 ? Math.abs(Math.sin((lon * Math.PI) / 9)) * -7 : 0;
      return h.style === "side" ? base - 6 + jag : base + jag;
    });
  }
  if (h.style === "bald-ring") {
    g.fillRect(0, Y(10), W, Y(-12) - Y(10));
    g.fillStyle = "#ffffff";
    g.fillRect(X(-70), Y(12), X(70) - X(-70), Y(-14) - Y(12));
  }
  void X;
  const t = new CanvasTexture(cv);
  t.colorSpace = SRGBColorSpace;
  t.anisotropy = 4;
  hairCache.set(key, t);
  return t;
};

// ------------------------------------------------------------------ river stone (lumpy, flat-topped)
export const stoneGeo = (() => {
  const cache = new Map<number, BufferGeometry>();
  return (seed: number) => {
    const hit = cache.get(seed);
    if (hit) return hit;
    const ico = new IcosahedronGeometry(1, 3);
    ico.deleteAttribute("normal");
    ico.deleteAttribute("uv");
    const g = mergeVertices(ico, 1e-4);
    const p = g.getAttribute("position") as BufferAttribute;
    const uv: number[] = [];
    for (let i = 0; i < p.count; i++) {
      const x = p.getX(i), y = p.getY(i), z = p.getZ(i);
      const n = 1 + 0.12 * (fbm(x * 0.3 + 0.5, z * 0.3 + 0.5, 3, 3, seed) - 0.5) + 0.05 * Math.sin(x * 7 + seed) * Math.cos(z * 6);
      // flatten: squashed top, rounded belly
      const yy = y > 0 ? y * 0.42 : y * 0.55;
      p.setXYZ(i, x * n, Math.max(yy, -0.3) * n, z * n * (0.82 + 0.1 * hash2(seed, 2)));
      uv.push(x * 0.5 + 0.5, z * 0.5 + 0.5);
    }
    g.setAttribute("uv", new Float32BufferAttribute(uv, 2));
    g.computeVertexNormals();
    cache.set(seed, g);
    return g;
  };
})();

  const SR = 0.33, base = footing === "stone" ? stoneTop(SR) + 0.0 : -0.02;
  const n = xs.length * zs.length;
  const items: React.ReactNode[] = [];
  let k = 0;
  zs.forEach((z, zi) => xs.forEach((x, xi) => {
    const order = (zi * xs.length + xi) / n;
    if (order > built + 1e-6) return;
    const seed = xi * 7 + zi * 3;

  }
  return <group>{items}</group>;

// CERDIK rain for any world: slanted streaks falling through a box around the shot and ripples where they land.
// Drops stop on whatever covers them (awnings, roofs, tables — the world's colliders), so people under a shelter stay
// dry. Everything is a pure function of t; the streaks live on FX_LAYER (no AO / DOF on them).
import React, { useLayoutEffect, useMemo, useRef } from "react";
import { AdditiveBlending, BoxGeometry, Color, InstancedMesh, MeshBasicMaterial, Object3D, RingGeometry } from "three";
import { FX_LAYER } from "./Stage.tsx";

type Box = { min: [number, number, number]; max: [number, number, number] };
const hash = (i: number, k: number) => { const x = Math.sin(i * 127.1 + k * 311.7) * 43758.5453; return x - Math.floor(x); };

const HALF = 22, TOP = 12, SLANT = 0.09;

/** Where a drop at (x, z) lands: the highest collider top above it, else the ground. */
const landAt = (x: number, z: number, ground: (x: number, z: number) => number, cover: Box[]) => {
  let y = ground(x, z);
  for (const b of cover) if (x >= b.min[0] && x <= b.max[0] && z >= b.min[2] && z <= b.max[2] && b.max[1] > y) y = b.max[1];
  return y;
};

/** `eye`: the camera position(s) of the shot — drops close to the lens thin out and fade so they never smear the frame. */
export const Rain: React.FC<{ t: number; center: [number, number]; eye: [number, number, number][]; ground: (x: number, z: number) => number; cover: Box[]; count?: number; ripples?: number }> = ({ t, center, eye, ground, cover, count = 8000, ripples = 220 }) => {
  const streaks = useRef<InstancedMesh>(null);
  const rings = useRef<InstancedMesh>(null);
  const G = useMemo(() => ({
    geo: new BoxGeometry(0.006, 0.46, 0.006),
    mat: new MeshBasicMaterial({ color: "#e2e9f2", transparent: true, opacity: 0.3, depthWrite: false }),
    ring: new RingGeometry(0.8, 1, 24).rotateX(-Math.PI / 2),
    ringMat: new MeshBasicMaterial({ color: "#ffffff", transparent: true, depthWrite: false, blending: AdditiveBlending, opacity: 0.55 }),
  }), []);
  // drops: x, z, floor, phase, speed — fixed per shot centre
  const drops = useMemo(() => {
    const top = ground(center[0], center[1]) + TOP;
    return Array.from({ length: count }, (_, i) => {
      const x = center[0] + (hash(i, 1) * 2 - 1) * HALF, z = center[1] + (hash(i, 2) * 2 - 1) * HALF;
      return { x, z, floor: landAt(x, z, ground, cover), top, ph: hash(i, 3), v: 8.5 + hash(i, 4) * 2.5, len: 0.75 + hash(i, 5) * 0.5 };
    });
  }, [center[0], center[1], ground, cover, count]);
  // ripples only on open ground near the centre (not on roofs)
  const spots = useMemo(() => {
    const out: { x: number; z: number; y: number; ph: number; per: number; r: number }[] = [];
    for (let i = 0; out.length < ripples && i < ripples * 4; i++) {
      const a = hash(i, 7) * Math.PI * 2, d = Math.sqrt(hash(i, 8)) * 11;
      const x = center[0] + Math.cos(a) * d, z = center[1] + Math.sin(a) * d;
      const g = ground(x, z), y = landAt(x, z, ground, cover);
      if (y > g + 0.01) continue;
      out.push({ x, z, y: g + 0.012, ph: hash(i, 9), per: 0.55 + hash(i, 10) * 0.5, r: 0.06 + hash(i, 11) * 0.07 });
    }
    return out;
  }, [center[0], center[1], ground, cover, ripples]);

  useLayoutEffect(() => {
    const o = new Object3D();
    const im = streaks.current!;
    im.layers.set(FX_LAYER);
    for (let i = 0; i < drops.length; i++) {
      const d = drops[i], h = d.top - d.floor;
      if (h < 0.6) { o.scale.set(0, 0, 0); o.updateMatrix(); im.setMatrixAt(i, o.matrix); continue; }
      const k = (d.ph + (t * d.v) / h) % 1, y = d.top - k * h;
      const grow = Math.min(1, (y - d.floor) / 0.4, (d.top - y) / 1.0); // shrink into the landing, grow in at the top
      const px = d.x + (d.top - y) * SLANT, py = y + 0.23;
      let near = 99;
      for (const e of eye) near = Math.min(near, Math.hypot(px - e[0], py - e[1], d.z - e[2]));
      const keep = Math.min(1, Math.max(0, (near - 1.4) / 1.6)); // gone within 1.4 m of the lens, full from 3 m
      const thin = Math.min(1, Math.max(0.3, near / 7)); // a hair-thin streak up close, not a white bar
      o.position.set(px, py, d.z);
      o.rotation.set(0, 0, SLANT);
      o.scale.set(thin * keep, Math.max(0.001, grow * d.len * keep), thin * keep);
      o.updateMatrix();
      im.setMatrixAt(i, o.matrix);
    }
    im.instanceMatrix.needsUpdate = true;
    const rm = rings.current!;
    rm.layers.set(FX_LAYER);
    const c = new Color();
    for (let i = 0; i < spots.length; i++) {
      const s = spots[i], k = ((t / s.per) + s.ph) % 1;
      const r = s.r * (0.25 + 0.75 * k);
      o.position.set(s.x, s.y, s.z);
      o.rotation.set(0, 0, 0);
      o.scale.set(r, 1, r);
      o.updateMatrix();
      rm.setMatrixAt(i, o.matrix);
      const f = Math.sin(Math.PI * k) * (1 - k) * 1.4; // additive: black = invisible, so the fade never pops
      rm.setColorAt(i, c.setRGB(f, f, f));
    }
    rm.instanceMatrix.needsUpdate = true;
    if (rm.instanceColor) rm.instanceColor.needsUpdate = true;
  });
  return (
    <group>
      <instancedMesh ref={streaks} args={[G.geo, G.mat, drops.length]} frustumCulled={false} />
      <instancedMesh ref={rings} args={[G.ring, G.ringMat, Math.max(1, spots.length)]} frustumCulled={false} />
    </group>
  );
};

const torsoProfile = (n = 28) => {

    const y = -1 + (2 * i) / n;

    const x = p.getX(i) * half[0], y = p.getY(i) * half[1], z = p.getZ(i) * half[2];

    skirtHull.translate(0, -0.165 * s, 0);
    return { ...t, decal, headHull, head, shoe, neck, lace, bun, bunHull, skirt, skirtHull };
  }, [j.torsoHalf[0], j.torsoHalf[1], j.torsoHalf[2], R, thick]); // eslint-disable-line react-hooks/exhaustive-deps
  useEffect(() => () => Object.values(G).forEach((g) => g.dispose()), [G]);

  const mats = useMemo(() => {
    const body = new MeshStandardMaterial({ map: costumeTexture(look.costume), roughness: 0.82 });
    const hairMap = hairTexture(look.hair ?? { color: "#2a1d16", style: "none" });
    // the genre's heads read paper-white: a little self-light keeps them white in shade too
    // matte (no specular): a glossy highlight turns the ink grey on the sunny side
    const head = new MeshLambertMaterial({ map: hairMap, emissive: "#ffffff", emissiveMap: hairMap, emissiveIntensity: 0.16 });
    const decal = new MeshLambertMaterial({ transparent: false, alphaTest: 0.5, polygonOffset: true, polygonOffsetFactor: -2, polygonOffsetUnits: -2 });
    const skirt = new MeshStandardMaterial({ color: look.costume.base, roughness: 0.85, side: 2 });
    return { body, head, decal, skirt };
  }, [look.costume, look.hair]);
  useEffect(() => () => Object.values(mats).forEach((m) => m.dispose()), [mats]);

  mats.decal.map = faceTexture(look.face ?? {}, face, faceRes);
  mats.decal.needsUpdate = true;
  const fl = Math.max(0, Math.min(1, flush));
  mats.head.color.copy(new Color("#ffffff").lerp(new Color(flushColor), fl));
  mats.head.emissive.copy(new Color("#ffffff").lerp(new Color(flushColor), fl));

  const headM = basis(j.head, j.headRight, j.headUp, j.headFwd);
  // torso: lathe axis = torso up
  const torsoM = basis(j.torso, j.right, j.up, j.fwd);
  const lr = DIM.limbR * s;

  const foot = (ankle: V3, side: 1 | -1) => {
    const yaw = (pose.splay ?? 0.2) * side;
    const f: V3 = [Math.sin(yaw), 0, Math.cos(yaw)];
    const rt = cross([0, 1, 0], f);
    return basis(add(add(ankle, f, 0.035 * s), [0, 1, 0], -0.012 * s), rt, [0, 1, 0], f, [0.03 * s, 0.022 * s, 0.062 * s]);
  };
  const hand = (el: V3, wr: V3) => {
    const d = nrm([wr[0] - el[0], wr[1] - el[1], wr[2] - el[2]]);
    const side = nrm(cross(d, [0, 1, 0.001]));
    const up = nrm(cross(side, d));
    return basis(add(wr, d, DIM.handR * s * 0.55), side, d, up, [DIM.handR * s * 0.95, DIM.handR * s * 1.15, DIM.handR * s * 0.75]);
  };

  return (
    <group position={pos} rotation={[0, rotY, 0]}>
      {/* torso + outline */}
      <M g={G.body} m={torsoM} material={mats.body} />
      <M g={G.hull} m={torsoM} material={mLine} shadow={false} />
      {/* neck stick */}
      <M g={G.neck} m={stick(add(j.neck, j.up, -0.03 * s), add(j.head, j.headUp, -R * 0.7), 0.011 * s)} material={mInk} />
      {/* head */}
      <M g={G.head} m={headM} material={mats.head} />
      <M g={G.decal} m={headM} material={mats.decal} shadow={false} />
      <M g={G.headHull} m={headM} material={mLine} shadow={false} />
      {look.hat && look.hat.kind !== "none" ? <Headgear hat={look.hat} j={j} thick={thick} /> : null}
      {look.hair?.style === "bun" && (!look.hat || look.hat.kind === "none") ? (
        <group>
          <M g={G.bun} m={basis(add(add(j.head, j.headFwd, -R * 0.78), j.headUp, R * 0.55), j.headRight, j.headUp, j.headFwd)} material={flat(look.hair.color, 0.7)} />
          <M g={G.bunHull} m={basis(add(add(j.head, j.headFwd, -R * 0.78), j.headUp, R * 0.55), j.headRight, j.headUp, j.headFwd)} material={mLine} shadow={false} />
        </group>
      ) : null}
      {look.costume.long ? (
        <group>
          <M g={G.skirt} m={basis(add(j.pelvis, j.up, -0.05 * s), [1, 0, 0], [0, 1, 0], [0, 0, 1])} material={mats.skirt} />
          <M g={G.skirtHull} m={basis(add(j.pelvis, j.up, -0.05 * s), [1, 0, 0], [0, 1, 0], [0, 0, 1])} material={mLine} shadow={false} />
        </group>
      ) : null}
      {look.necklace ? <M g={G.lace} m={basis(add(j.neck, j.up, -0.012 * s), j.right, j.fwd, nrm(cross(j.right, j.fwd)), [1, 1, 1])} material={flat(look.necklace, 0.35)} /> : null}
      {/* arms */}
      {([[j.shL, j.elbowL, j.wristL], [j.shR, j.elbowR, j.wristR]] as V3[][]).map(([sh, el, wr], i) => (
        <group key={"a" + i}>
          <M g={UNIT.cyl} m={stick(sh, el, lr)} material={mInk} />
          <M g={UNIT.sph} m={basis(el, [1, 0, 0], [0, 1, 0], [0, 0, 1], [lr, lr, lr])} material={mInk} />
          <M g={UNIT.cyl} m={stick(el, wr, lr)} material={mInk} />
          <M g={UNIT.sph} m={hand(el, wr)} material={mInk} />
        </group>
      ))}
      {/* legs */}
      {([[j.hipL, j.kneeL, j.ankleL, 1], [j.hipR, j.kneeR, j.ankleR, -1]] as [V3, V3, V3, 1 | -1][]).map(([hp, kn, an, sd], i) => (
        <group key={"l" + i}>
          <M g={UNIT.cyl} m={stick(add(hp, j.up, 0.03 * s), kn, lr * 1.12)} material={mInk} />
          <M g={UNIT.sph} m={basis(kn, [1, 0, 0], [0, 1, 0], [0, 0, 1], [lr * 1.12, lr * 1.12, lr * 1.12])} material={mInk} />
          <M g={UNIT.cyl} m={stick(kn, an, lr * 1.12)} material={mInk} />
          <M g={G.shoe} m={foot(an, sd)} material={mInk} />
        </group>
      ))}
      {look.towel ? <M g={UNIT.sph} m={basis(add(add(j.shR, j.up, 0.01 * s), j.right, 0.03 * s), j.right, j.up, j.fwd, [0.05 * s, 0.02 * s, 0.1 * s])} material={flat(look.towel, 0.9)} /> : null}
    </group>
  );
};

/** Deterministic blink: closed for ~0.12 s every few seconds (seeded per character). */
export const blinkAt = (t: number, seed = 1) => {
  const period = 2.7 + (seed % 5) * 0.37;
  const ph = (t + seed * 1.37) % period;
  return ph < 0.06 ? 0.5 : ph < 0.13 ? 1 : ph < 0.18 ? 0.5 : 0;
};

/** Lip flap from word timings: open while a word is spoken, pulsing per syllable. */
export const mouthAt = (t: number, words: { t0: number; t1: number; w: string }[]) => {
  for (const w of words) {
    if (t < w.t0 - 0.02 || t > w.t1 + 0.02) continue;
    const syl = Math.max(1, (w.w.match(/[aiueo]/gi) ?? []).length);
    const k = (t - w.t0) / Math.max(0.05, w.t1 - w.t0);
    return 0.25 + 0.75 * Math.abs(Math.sin(Math.PI * k * syl));
  }
  return 0;
};
