==== 9586 project/scripts/lib/cerdik-sound.mjs
// CERDIK sound from a Plan — original synth, no samples.
// Music: a curious, playful storytelling bed (pizzicato bass, marimba melody in a "mystery-major" mode, light shaker
// and woodblock), 104 bpm. It thins out in the twist (night / hot) and lands with a resolving chord at the end.
// SFX follow the plan: a soft air move on each cut, a whoosh on dissolves, a bright "ding" when something glows,
// sparkles, a sizzle under steam, a cool shimmer for "chill", a dull thud for dust, a heat drone in hot shots.
// Mix: −14 LUFS / −1.5 dBTP; the music ducks under the voice.
import fs from "node:fs";
import path from "node:path";
import { execFileSync, spawnSync } from "node:child_process";

export const buildSound = (plan, { voice, outDir, mixFile, music = true }) => {
  const DUR = plan.duration, SR = 48000, N = Math.ceil(DUR * SR);
  const mk = () => [new Float32Array(N), new Float32Array(N)];
  let seed = 4242;
  const rnd = () => (seed = (seed * 16807) % 2147483647) / 2147483647;
  const add = (buf, t, pan, len, fn) => {
    const i0 = Math.max(0, Math.round(t * SR)), n = Math.min(N - i0, Math.round(len * SR));
    const gl = Math.cos((pan + 1) * Math.PI / 4), gr = Math.sin((pan + 1) * Math.PI / 4);
    for (let i = 0; i < n; i++) { const v = fn(i / SR, i); buf[0][i0 + i] += v * gl; buf[1][i0 + i] += v * gr; }
  };
  const hz = (m) => 440 * Math.pow(2, (m - 69) / 12);
  const shotAt = (t) => plan.shots.find((s) => t >= s.t0 && t < s.t1) ?? plan.shots[plan.shots.length - 1];
  const timeAt = (t) => shotAt(t).time;
  const rainAt = (t) => shotAt(t).weather === "rain";

  // ---------------------------------------------------------------- music
  const mus = mk();
  if (music) {
    const beat = 60 / 104, bar = beat * 4;
    // F major with a raised 4th (Lydian) → curious; chords I – II – vi – IV
    const CH = [[53, 57, 60, 64], [55, 59, 62, 66], [50, 57, 60, 65], [46, 53, 58, 62]];
    const MEL = [72, 74, 76, 79, 81, 83, 84];
    const fade = (t) => Math.min(1, Math.max(0, (DUR - 0.3 - t) / 1.6)) * Math.min(1, t / 0.6);
    const bars = Math.ceil(DUR / bar);
    for (let b = 0; b < bars; b++) {
      const c = CH[b % 4], t0 = b * bar;
      const mood = timeAt(t0 + 0.1);
      const thin = mood === "hot" || mood === "night" || rainAt(t0 + 0.1) ? 0.55 : 1;
      // pizzicato bass on 1 and 3 (+ a pickup on 4&)
      for (const [bt, m] of [[0, c[0] - 12], [2, c[0] - 5], [3.5, c[0] - 10]]) {
        const f = hz(m);
        add(mus, t0 + bt * beat, -0.1, 0.5, (t) => (Math.sin(2 * Math.PI * f * t) + 0.35 * Math.sin(4 * Math.PI * f * t)) * 0.085 * Math.min(1, t / 0.004) * Math.exp(-t * 9) * fade(t0 + t));
      }
      // marimba: a light arpeggio, sparser in the twist
      for (let e = 0; e < 8; e++) {
        if (rnd() > 0.78 * thin + 0.1) continue;
        const pool = MEL.filter((p) => c.some((x) => (p - x) % 12 === 0) || rnd() < 0.25);
        const m = pool[Math.floor(rnd() * pool.length)] ?? 72;
        const f = hz(m), vel = (e % 4 === 0 ? 0.06 : 0.042) * (0.75 + rnd() * 0.4), pan = (rnd() - 0.5) * 0.8;
        add(mus, t0 + e * beat / 2 + (rnd() - 0.5) * 0.01, pan, 0.8, (t) => (Math.sin(2 * Math.PI * f * t) * Math.exp(-t * 8) + 0.35 * Math.sin(2 * Math.PI * f * 4 * t) * Math.exp(-t * 30)) * vel * Math.min(1, t / 0.002) * fade(t0 + t));
      }
      // soft pad under the whole bar
      for (const [k, m] of c.entries()) add(mus, t0, (k - 1.5) * 0.2, bar + 0.4, (t) => Math.sin(2 * Math.PI * hz(m) * t) * 0.009 * Math.min(1, t / 0.8) * Math.min(1, (bar + 0.4 - t) / 0.6) * fade(t0 + t));
      // shaker on the off-beats, woodblock on 2 and 4
      for (let e = 1; e < 8; e += 2) { let lp = 0; add(mus, t0 + e * beat / 2, 0.3, 0.06, (t) => { const n = rnd() * 2 - 1; lp += (n - lp) * 0.7; return (n - lp) * 0.016 * thin * Math.exp(-t * 70) * fade(t0 + t); }); }
      for (const bt of [1, 3]) add(mus, t0 + bt * beat, -0.3, 0.12, (t) => Math.sin(2 * Math.PI * 1150 * t) * Math.exp(-t * 55) * 0.03 * thin * fade(t0 + t));
    }
    // landing chord at the end
    const endT = Math.max(0, DUR - 2.2);
    for (const m of [53, 60, 65, 69, 72]) add(mus, endT, 0, 2.2, (t) => Math.sin(2 * Math.PI * hz(m) * t) * 0.03 * Math.min(1, t / 0.02) * Math.exp(-t * 1.4));
  }

  // ---------------------------------------------------------------- SFX
  const sfx = mk();
  const noise = (t, len, gain, lpK, hpK, env, pan = 0) => { let lp = 0, hp = 0; add(sfx, t, pan, len, (x) => { const n = rnd() * 2 - 1; lp += (n - lp) * lpK; hp += (lp - hp) * hpK; return (lp - hp) * gain * env(x); }); };
  const air = (t) => noise(t - 0.12, 0.3, 0.09, 0.05, 0.002, (x) => Math.sin(Math.PI * Math.min(1, x / 0.3)) ** 2);
  const whoosh = (t) => noise(t - 0.15, 0.8, 0.16, 0.08, 0.01, (x) => Math.sin(Math.PI * Math.min(1, x / 0.8)) ** 2, 0.2);
  const ding = (t, m = 88) => { for (const [k, mm] of [m, m + 7, m + 12].entries()) add(sfx, t + k * 0.035, 0.15, 1.1, (x) => Math.sin(2 * Math.PI * hz(mm) * x) * (0.035 - k * 0.008) * Math.min(1, x / 0.003) * Math.exp(-x * 4.5)); };
  const twinkle = (t) => { for (let k = 0; k < 4; k++) { const m = 93 + Math.floor(rnd() * 7); add(sfx, t + k * 0.09 + rnd() * 0.04, (rnd() - 0.5) * 0.8, 0.3, (x) => Math.sin(2 * Math.PI * hz(m) * x) * 0.014 * Math.exp(-x * 14)); } };
  const thud = (t) => add(sfx, t, 0, 0.4, (x) => Math.sin(2 * Math.PI * 70 * x * (1 + Math.exp(-x * 40))) * Math.exp(-x * 14) * 0.16 * Math.min(1, x / 0.003));
  const events = [];
  plan.shots.forEach((S, i) => {
    if (i) { if (S.dissolve) whoosh(S.t0); else air(S.t0); }
    for (const f of S.fx) {
      if (f.t0 < S.t0 - 0.01 || f.t0 >= S.t1) continue; // each effect sounds once, in the shot it starts in
      if (f.kind === "glow") { ding(f.t0); events.push(`${f.t0.toFixed(2)} ding`); }
      if (f.kind === "sparkle") { twinkle(f.t0 + 0.05); events.push(`${f.t0.toFixed(2)} sparkle`); }
      if (f.kind === "chill") { ding(f.t0, 91); twinkle(f.t0 + 0.2); events.push(`${f.t0.toFixed(2)} chill`); }
      if (f.kind === "dust") { thud(f.t0); events.push(`${f.t0.toFixed(2)} thud`); }
      if (f.kind === "steam") { const len = Math.min(f.t1, S.t1) - f.t0; noise(f.t0, len, 0.035, 0.6, 0.15, (x) => Math.min(1, x / 0.4) * Math.min(1, (len - x) / 0.3)); events.push(`${f.t0.toFixed(2)} sizzle`); }
      if (f.kind === "vapor") { const len = Math.min(f.t1, S.t1) - f.t0; noise(f.t0, len, 0.02, 0.3, 0.05, (x) => Math.min(1, x / 0.6) * Math.min(1, (len - x) / 0.4), -0.3); }
    }
    if (S.time === "hot") { const len = S.t1 - S.t0; add(sfx, S.t0, 0, len, (x) => (Math.sin(2 * Math.PI * 55 * x) + 0.5 * Math.sin(2 * Math.PI * 82.5 * x)) * 0.012 * Math.min(1, x / 0.3) * Math.min(1, (len - x) / 0.3)); }
  });

  // rain: a steady hiss (two filtered noise layers, stereo) that fades in / out over 0.6 s where shots are rainy
  const wet = new Float32Array(N);
  { let on = 0; const step = 1 / (0.6 * SR); for (let i = 0; i < N; i++) { const want = rainAt(i / SR) ? 1 : 0; on += Math.max(-step, Math.min(step, want - on)); wet[i] = on; } }
  if (wet.some((v) => v > 0)) {
    events.push("rain bed");
    for (const c of [0, 1]) {
      let lp = 0, hp = 0, lp2 = 0, hp2 = 0;
      for (let i = 0; i < N; i++) {
        if (!wet[i]) { lp = hp = lp2 = hp2 = 0; continue; }
        const n = rnd() * 2 - 1;
        lp += (n - lp) * 0.35; hp += (lp - hp) * 0.08; // bright hiss
        lp2 += (n - lp2) * 0.04; hp2 += (lp2 - hp2) * 0.01; // low body
        const drop = rnd() < 0.0009 ? (rnd() * 2 - 1) * 0.5 : 0; // the odd close drop
        sfx[c][i] += ((lp - hp) * 0.032 + (lp2 - hp2) * 0.045 + drop * 0.03) * wet[i];
      }
    }
  }

  // ---------------------------------------------------------------- write + mix
  const wav = (file, buf) => {
    const data = Buffer.alloc(N * 4);
    for (let i = 0; i < N; i++) for (let c = 0; c < 2; c++) data.writeInt16LE(Math.round(Math.max(-1, Math.min(1, buf[c][i])) * 32767), i * 4 + c * 2);
    const h = Buffer.alloc(44);
    h.write("RIFF", 0); h.writeUInt32LE(36 + data.length, 4); h.write("WAVEfmt ", 8); h.writeUInt32LE(16, 16); h.writeUInt16LE(1, 20); h.writeUInt16LE(2, 22);
    h.writeUInt32LE(SR, 24); h.writeUInt32LE(SR * 4, 28); h.writeUInt16LE(4, 32); h.writeUInt16LE(16, 34); h.write("data", 36); h.writeUInt32LE(data.length, 40);
    fs.writeFileSync(file, Buffer.concat([h, data]));
  };
  fs.mkdirSync(outDir, { recursive: true });
  fs.mkdirSync(path.dirname(mixFile), { recursive: true });
  const mf = path.join(outDir, "music.wav"), sf = path.join(outDir, "sfx.wav");
  wav(mf, mus);
  wav(sf, sfx);
  const tail = `apad=whole_dur=${DUR},atrim=0:${DUR},loudnorm=I=-14:TP=-1.5:LRA=11,aresample=48000[out]`;
  const filter = voice
    ? ["[0:a]highpass=f=70,acompressor=threshold=-20dB:ratio=2.5:attack=8:release=140:makeup=3,asplit=2[v][vk]", "[1:a]volume=1.1[m]", "[m][vk]sidechaincompress=threshold=0.03:ratio=6:attack=25:release=420[md]", "[2:a]volume=0.9[s]", `[v][md][s]amix=inputs=3:normalize=0:duration=longest,${tail}`].join(";")
    : ["[0:a]volume=1.5[m]", "[1:a]volume=1.0[s]", `[m][s]amix=inputs=2:normalize=0:duration=longest,${tail}`].join(";");
  execFileSync("ffmpeg", ["-y", "-v", "error", ...(voice ? ["-i", voice] : []), "-i", mf, "-i", sf, "-filter_complex", filter, "-map", "[out]", "-ac", "2", mixFile]);
  const m = spawnSync("ffmpeg", ["-hide_banner", "-i", mixFile, "-af", "loudnorm=print_format=summary", "-f", "null", "-"], { encoding: "utf8" }).stderr;
  return { lufs: (m.match(/Input Integrated:\s+(-?[\d.]+)/) ?? [])[1], peak: (m.match(/Input True Peak:\s+(-?[\d.]+)/) ?? [])[1], events: events.length };
};

==== 3814 project/scripts/stills.mjs
// Visual QA: bundle once, render chosen frames of a composition, tile them into a contact sheet.
//   node scripts/stills.mjs <outDir> <frame,frame,...> <compositionId> [scale]
// Frames may be given in seconds with an "s" suffix (e.g. 4.5s), using the composition's fps.

import path from "node:path";
import fs from "node:fs";
import { execFileSync } from "node:child_process";
import { bundle } from "@remotion/bundler";
import { openBrowser, renderStill, selectComposition } from "@remotion/renderer";
import { glMode, isWebGLError, switchToSoftware, tolerateClosedTargets } from "./gl.mjs";

tolerateClosedTargets();

const [outDir, list, id, scaleArg = "1", propsFile] = process.argv.slice(2);
const rawProps = propsFile ? JSON.parse(fs.readFileSync(propsFile, "utf8")) : {};
// a props file is either { plan, … } or a bare plan
const inputProps = propsFile ? (rawProps.plan ? rawProps : { plan: rawProps }) : {};
if (!outDir || !list || !id) {
  console.error("usage: node scripts/stills.mjs <outDir> <frame,frame,...|4.5s,...> <compositionId> [scale]");
  process.exit(1);
}
fs.mkdirSync(outDir, { recursive: true });

const serveUrl = await bundle({ entryPoint: path.resolve("src/index.ts") });
// bundles are ~50 MB each; never leave them in the temp dir, even when a render throws
const cleanup = () => fs.rmSync(serveUrl, { recursive: true, force: true });
process.on("exit", cleanup);
const composition = await selectComposition({ serveUrl, id, inputProps });
const frames = list.split(",").map((t) => (t.endsWith("s") ? Math.round(parseFloat(t) * composition.fps) : parseInt(t, 10)));
const files = [];
const renderAll = async (gl) => {
  // one browser for all stills: WebGL contexts are expensive to create and tear down
  const puppeteerInstance = await openBrowser("chrome", { chromiumOptions: { gl } });
  try {
    files.length = 0;
    for (const frame of frames) {
      const output = path.join(outDir, `f${String(frame).padStart(4, "0")}.png`);
      await renderStill({ composition, serveUrl, output, frame, inputProps, scale: Number(scaleArg), chromiumOptions: { gl }, timeoutInMilliseconds: 180000, puppeteerInstance });
      files.push(output);
      process.stdout.write(`${frame} `);
    }
  } finally {
    // after a WebGL crash the tab may already be gone; closing must not hide the original error
    await puppeteerInstance.close({ silent: true }).catch(() => {});
  }
};
const gl = glMode();
try {
  await renderAll(gl);
} catch (e) {
  if (gl === "swangle" || !isWebGLError(e)) throw e;
  await renderAll(switchToSoftware());
}
console.log();
cleanup();

// contact sheet (needs ffmpeg; skipped if missing)
try {
  const w = Math.round(composition.width * Number(scaleArg)), h = Math.round(composition.height * Number(scaleArg));
  const cols = Math.min(4, files.length), rows = Math.ceil(files.length / cols), n = cols * rows;
  if (n > 1) {
    const inputs = files.flatMap((f) => ["-i", f]);
    const pads = Array.from({ length: n - files.length }, () => ["-f", "lavfi", "-i", `color=c=black:s=${w}x${h}`]).flat();
    const filter =
      Array.from({ length: n }, (_, i) => `[${i}:v]scale=${w}:${h},setsar=1[v${i}]`).join(";") + ";" +
      Array.from({ length: n }, (_, i) => `[v${i}]`).join("") +
      `xstack=inputs=${n}:layout=` + Array.from({ length: n }, (_, i) => `${(i % cols) * w}_${Math.floor(i / cols) * h}`).join("|") +
      `,scale='min(2160,iw)':-2`;
    execFileSync("ffmpeg", ["-v", "error", "-y", ...inputs, ...pads, "-filter_complex", filter, "-frames:v", "1", path.join(outDir, "_sheet.jpg")]);
    console.log(`sheet: ${path.join(outDir, "_sheet.jpg")}`);
  }
} catch {
  console.log("(contact sheet skipped: ffmpeg not available)");
}
// a render that failed on the GPU path leaves its timeout timer behind; do not wait for it
process.exit(0);

==== 9921 project/src/cerdik/actions.ts
// CERDIK body actions — names an author writes in story.json ("do": "laugh") → a pose at a time. Pure (node + browser).
// Every action is a function of (seconds since it started, absolute time, options) returning a rig Pose in the
// person's local frame (faces +z, left = +x). Changes between actions blend over ~0.25 s in the renderer.

import { DIM, mixPose, POSES, type Pose, type V3 } from "../kits/cerdik/rig.ts";
import type { PropDef } from "./props/lib.ts";

const c01 = (x: number) => (x < 0 ? 0 : x > 1 ? 1 : x);
const sm = (x: number) => { const t = c01(x); return t * t * (3 - 2 * t); };
const wave = (t: number, hz: number, ph = 0) => 0.5 - 0.5 * Math.cos(2 * Math.PI * hz * t + ph); // 0…1, smooth
const A = DIM.ankle;
const PY = 0.655;

type Opt = { to?: V3; prop?: PropDef; seed: number };

const breathe = (p: Pose, t: number, seed: number, amt = 1): Pose => {
  const b = Math.sin(t * 2.1 + seed) * 0.006 * amt, sw = Math.sin(t * 0.9 + seed * 2) * 0.012 * amt;
  return { ...p, pelvis: [p.pelvis[0], p.pelvis[1] + b, p.pelvis[2]], pitch: p.pitch + sw * 0.4, turn: (p.turn ?? 0) + sw, handL: [p.handL[0], p.handL[1] + b, p.handL[2]], handR: [p.handR[0], p.handR[1] + b, p.handR[2]] };
};
const lift = (p: Pose, dy: number): Pose => ({ ...p, pelvis: [p.pelvis[0], p.pelvis[1] + dy, p.pelvis[2]], handL: [p.handL[0], p.handL[1] + dy, p.handL[2]], handR: [p.handR[0], p.handR[1] + dy, p.handR[2]] });

/** Hands on a held prop's grips (the prop sits at its `holdAt` in front of the person). */
export const holdHands = (def: PropDef): { L: V3; R: V3; at: V3 } => {
  const at = def.holdAt ?? [0, 0.62, 0.3];
  const g = def.grips ?? { L: [0.15, 0.1, 0], R: [-0.15, 0.1, 0] };
  return { L: [at[0] + g.L[0], at[1] + g.L[1], at[2] + g.L[2]], R: [at[0] + g.R[0], at[1] + g.R[1], at[2] + g.R[2]], at };
};

/** A prop held in one hand (both grips at the same point). */
export const oneHanded = (def: PropDef) => !!def.grips && def.grips.L.every((v, i) => Math.abs(v - def.grips!.R[i]) < 1e-6);

export const ACTIONS: Record<string, { about: string; pose: (s: number, t: number, o: Opt) => Pose }> = {
  stand: { about: "stand relaxed", pose: (_s, t, o) => breathe(POSES.stand(), t, o.seed) },
  akimbo: { about: "hands on hips, chin up (proud, confident)", pose: (_s, t, o) => breathe(POSES.akimbo(), t, o.seed) },
  crossed: { about: "arms folded (calm, unbothered, stubborn)", pose: (_s, t, o) => breathe(POSES.crossed(), t, o.seed) },
  peer: { about: "lean in, hands behind the back (inspecting)", pose: (s, t, o) => breathe(mixPose(POSES.stand(), POSES.peer(), sm(s / 0.8)), t, o.seed, 0.5) },
  laugh: { about: "belly laugh, bouncing", pose: (s, _t, o) => { const b = wave(s, 2.6, o.seed) * sm(s / 0.2); const p = POSES.laugh(); return { ...lift(p, 0.012 * b), pitch: p.pitch - 0.04 * b, nod: (p.nod ?? 0) - 0.07 * b }; } },
  mock: { about: "point at something (`to`) and laugh at it", pose: (s, _t, o) => {
    const pt = POSES.point(o.to ?? [0.2, 1.0, 2]); const b = wave(s, 2.4, o.seed);
    return { ...lift(pt, 0.01 * b), pitch: -0.08 - 0.03 * b, nod: -0.12 - 0.05 * b, handL: [0.12, 0.7 + 0.01 * b, 0.155], elbowOut: 0.6 };
  } },
  point: { about: "point at something (`to`: a person, a prop or a spot)", pose: (s, t, o) => { const pt = POSES.point(o.to ?? [0.2, 1.0, 2]); return breathe(mixPose(POSES.stand(), pt, sm(s / 0.3)), t, o.seed, 0.5); } },
  present: { about: "one arm out, palm up (showing, explaining, selling)", pose: (_s, t, o) => breathe(POSES.present(), t, o.seed) },
  talk: { about: "talking with the hands", pose: (s, t, o) => {
    const a = wave(s, 0.9, o.seed), b = wave(s, 0.7, o.seed + 2);
    return breathe({ ...POSES.stand(), nod: -0.04, handL: [0.24 + 0.04 * a, 0.78 + 0.12 * a, 0.2 + 0.05 * a], handR: [-0.2, 0.55 + 0.1 * b, 0.12], elbowOut: 0.55 }, t, o.seed);
  } },
  shock: { about: "shocked: hands to the cheeks, leaning back", pose: (s, _t, o) => {
    const k = sm(s / 0.25), j = Math.sin(s * 30) * 0.004 * (1 - sm(s / 0.6));
    return mixPose(POSES.stand(), { pelvis: [0, PY + j, -0.03], pitch: -0.12, nod: -0.05, handL: [0.15, 1.18, 0.12], handR: [-0.15, 1.18, 0.12], footL: [0.09, A, 0.04], footR: [-0.09, A, -0.04], elbowOut: 0.9 }, k);

const liveActor = (a: PActor, t: number, LIB: Record<string, PropDef>): Live => {
  let ki = 0;
  for (let i = 0; i < a.keys.length; i++) if (t >= a.keys[i].t - 1e-4) ki = i;
  // the action / expression / flush in force: the latest key that set each one
  let doI = 0, expr = "neutral", flush = 0;
  for (let i = 0; i <= ki; i++) { if (a.keys[i].do) doI = i; if (a.keys[i].expr) expr = a.keys[i].expr; if (a.keys[i].flush >= 0) flush = a.keys[i].flush; }
  const K = a.keys[doI];
  const prop = a.hold ? LIB[propKindOf(a.hold)] : undefined;
  const opt = { to: K.to, prop, seed: a.seed };
  let pose = actionPose(K.do, t - K.t, t, opt);
  // blend from the previous action over 0.25 s
  if (doI > 0) {
    const prevDo = a.keys.slice(0, doI).reverse().find((k) => k.do);
    if (prevDo) {
      const b = smoother((t - K.t) / 0.25);
      if (b < 1) pose = mixPose(actionPose(prevDo.do, t - prevDo.t, t, { to: prevDo.to, prop, seed: a.seed }), pose, b);
    }
  }
  // walking: move along the path, legs cycling, facing the way; turn to the final facing at the end
  let pos = a.p1, face = a.face1;
  if (a.walk) {
    const [w0, w1] = a.walk;
    const k = c01((t - w0) / Math.max(0.1, w1 - w0));
    const e = smoother(k);
    pos = mix3(a.p0, a.p1, e);
    const walking = t > w0 && t < w1 + 0.05;
    const inW = c01((t - w0) / 0.2) * c01((w1 + 0.15 - t) / 0.2);
    if (walking) pose = mixPose(pose, walkPose((t - w0) * 1.9), inW);
    const turn = smoother((t - w1) / 0.35);
    face = t < w1 ? a.face0 : a.face0 + wrapA(a.face1 - a.face0) * turn;
    if (t < w0) { pos = a.p0; face = a.face0; }
  }
  return { pose, pos, face, expr: expr as Expr, flush, open: mouthAt(t, a.talk) };

const doubleSided = (m: Material) => { let d = dbl.get(m); if (!d) { d = m.clone(); d.side = DoubleSide; dbl.set(m, d); } return d; };

// ------------------------------------------------------------------ geometry
/** Lathe; when cut, the sector ±CUT_HALF around local angle `dir` (0 = +z) is left open. */
const lathe = (profile: [number, number][], cut: boolean, dir = 0) =>
  new LatheGeometry(profile.map(([r, y]) => new Vector2(Math.max(1e-4, r), y)), 48, cut ? dir + CUT_HALF : 0, cut ? Math.PI * 2 - 2 * CUT_HALF : Math.PI * 2);

/** Flat section face between an outer and an inner profile, placed on the cut plane at angle phi. */
const sectionFace = (outer: [number, number][], inner: [number, number][], phi: number) => {
  const sh = new Shape();
  const pts = [...outer, ...inner.slice().reverse()];
  pts.forEach(([r, y], i) => (i ? sh.lineTo(r, y) : sh.moveTo(r, y)));
  const g = new ShapeGeometry(sh);
  // shape is in (x = r, y); rotate so +x points along the lathe direction phi (x = sin φ, z = cos φ)
  g.rotateY(phi - Math.PI / 2);
  return g;
};
const clipProfile = (p: [number, number][], ymax: number): [number, number][] => {
  const out: [number, number][] = [];
  for (let i = 0; i < p.length; i++) {
    const [r, y] = p[i];
    if (y <= ymax) out.push([r, y]);
    else {
      const [r0, y0] = p[i - 1] ?? [r, y];
      const k = (ymax - y0) / Math.max(1e-6, y - y0);
      out.push([r0 + (r - r0) * k, ymax]);
      break;
    }
  }
  out.push([0, out[out.length - 1][1]]);
  return out;
};

const partGeo = (p: Part, cut: boolean, dir = 0): BufferGeometry => {
  switch (p.shape) {
    case "box": { const g = new BoxGeometry(p.size[0], p.size[1], p.size[2]); g.translate(0, p.size[1] / 2, 0); return g; }
    case "cyl": { const g = new CylinderGeometry(p.r2 ?? p.r, p.r, p.h, p.seg ?? 28, 1); g.translate(0, p.h / 2, 0); return g; }
    case "cone": { const g = new ConeGeometry(p.r, p.h, p.seg ?? 28, 1); g.translate(0, p.h / 2, 0); return g; }
    case "sphere": return new SphereGeometry(p.r, 28, 18);
    case "torus": return new TorusGeometry(p.r, p.tube, 10, 36);
    case "lathe": return lathe(p.profile, cut && !!p.cut, dir);
  }
};


const partMatrix = (p: Part) => {
  const pos = (p as { pos?: V3 }).pos ?? [0, 0, 0];
  const rot = (p as { rot?: V3 }).rot ?? [0, 0, 0];
  const sc = p.shape === "sphere" ? p.scale ?? [1, 1, 1] : [1, 1, 1];
  return new Matrix4().compose(new Vector3(...pos), new Quaternion().setFromEuler(new Euler(rot[0] * D2R, rot[1] * D2R, rot[2] * D2R)), new Vector3(sc[0], sc[1], sc[2]));
};

type Built = { g: BufferGeometry; m: Matrix4; mat: Material; center: Vector3 }[];
const builtCache = new Map<string, Built>();
const buildProp = (key: string, def: PropDef, cut: boolean, fill?: { color: string; level: number }, dir = 0): Built => {
  const k = key + (cut ? `|cut${dir.toFixed(2)}` : "") + (fill ? `|${fill.color}|${fill.level}` : "");
  const hit = builtCache.get(k);
  if (hit) return hit;
  const out: Built = [];
  def.parts.forEach((p, i) => {
    // cut open: small parts sitting in the opening (a spout, a handle) are left out so the section reads
    if (cut && p.shape !== "lathe") {
      const q = (p as { pos?: V3 }).pos ?? [0, 0, 0];
      if (Math.hypot(q[0], q[2]) > 0.02 && Math.abs(Math.atan2(Math.sin(Math.atan2(q[0], q[2]) - dir), Math.cos(Math.atan2(q[0], q[2]) - dir))) < CUT_HALF + 0.5) return;
    }
    const g = partGeo(p, cut, dir);
    g.computeBoundingBox();
    const center = g.boundingBox!.getCenter(new Vector3());
    const m = partMatrix(p);
    // a vessel shown cut open: its inside must render too
    out.push({ g, m, mat: cut && p.shape === "lathe" && p.cut ? doubleSided(propMat(p.mat, p.color)) : propMat(p.mat, p.color), center });
    // cut vessel: wall section faces + liquid
    if (cut && p.shape === "lathe" && p.cut && def.vessel?.part === i) {
      const inner = def.vessel.inner;
      for (const phi of [dir + CUT_HALF, dir - CUT_HALF]) {
        const f = sectionFace(p.profile, inner, phi);
        out.push({ g: f, m, mat: propMat(p.mat === "glass" ? "ice" : "clayDark", p.mat === "glass" ? "#bfe6f7" : undefined), center: new Vector3() });
      }
    }
  });
  // liquid inside a vessel (always drawn; through glass it shows, in clay only when cut)
  const v = def.vessel;
  if (v && (fill || v.level) && (cut || def.parts[v.part].mat === "glass")) {
    const level = fill?.level ?? v.level ?? 0.7;
    const ymax = Math.max(...v.inner.map((q) => q[1])) * level;
    const prof = clipProfile(v.inner.map(([r, y]) => [Math.max(0, r - 0.002), y]), ymax);
    const p = def.parts[v.part];
    const m = partMatrix(p);
    const water = cut ? propMat("liquid", fill?.color) : propMat("water", fill?.color ?? v.liquid);
    out.push({ g: lathe(prof, cut, dir), m, mat: water, center: new Vector3() });
    if (cut) for (const phi of [dir + CUT_HALF, dir - CUT_HALF]) out.push({ g: sectionFace(prof.slice(0, -1), [[0, prof[prof.length - 1][1]], [0, prof[0][1]]], phi), m, mat: water, center: new Vector3() });
  }
  builtCache.set(k, out);
  return out;
};

  def: PropDef;
  pos?: V3;
  yaw?: number;
  scale?: number;

// Shared far scenery for CERDIK worlds: green hills, blue mountains, a ring of tree masses on the horizon.
import React, { useMemo } from "react";
import { BufferAttribute, IcosahedronGeometry, MeshStandardMaterial } from "three";
import { mergeVertices } from "three/examples/jsm/utils/BufferGeometryUtils.js";
import { TreeLine } from "../bedah/props.tsx";

type V3 = [number, number, number];

export const Hills: React.FC<{ hills?: [V3, V3, boolean][]; green?: string; blue?: string }> = ({ hills, green = "#4f8a3c", blue = "#5d7f8e" }) => {
  const geo = useMemo(() => {
    const ico = new IcosahedronGeometry(1, 4);
    ico.deleteAttribute("normal");
    ico.deleteAttribute("uv");
    const g = mergeVertices(ico, 1e-4);
    const p = g.getAttribute("position") as BufferAttribute;
    for (let i = 0; i < p.count; i++) {
      const x = p.getX(i), y = p.getY(i), z = p.getZ(i);
      const n = 1 + 0.08 * Math.sin(x * 4.1 + z * 2.3) + 0.05 * Math.sin(z * 7.7 - x * 3.1);
      p.setXYZ(i, x * n, Math.max(y, -0.2) * n, z * n);
    }
    g.computeVertexNormals();
    return g;
  }, []);
  const mH = useMemo(() => new MeshStandardMaterial({ color: green, roughness: 1 }), [green]);
  const mM = useMemo(() => new MeshStandardMaterial({ color: blue, roughness: 1 }), [blue]);
  const H: [V3, V3, boolean][] = hills ?? [
    [[-150, -6, -260], [130, 60, 90], false], [[120, -8, -300], [160, 75, 110], false], [[-20, -10, -380], [220, 120, 120], true],
    [[230, -8, -180], [120, 55, 90], false], [[-260, -8, -160], [130, 60, 90], false], [[-320, -10, 120], [160, 70, 110], false],
    [[300, -10, 160], [170, 80, 120], false], [[40, -12, 360], [240, 100, 120], true], [[-140, -6, 260], [130, 55, 90], false],
  ];
  return <group>{H.map(([p, s, far], i) => <mesh key={i} geometry={geo} material={far ? mM : mH} position={p} scale={s} />)}</group>;
};

import { CanvasTexture, MeshStandardMaterial as MSM, RepeatWrapping, SRGBColorSpace } from "three";
const tileCache = new Map<string, MSM>();
/** Floor tiles: square tiles with grout lines (ceramic, stone), `size` metres per tile on metre-UV geometry. */
export const tileFloor = (base: string, grout: string, size = 0.5, roughness = 0.5) => {
  const key = base + grout + size + roughness;
  const hit = tileCache.get(key);
  if (hit) return hit;
  const cv = document.createElement("canvas");
  cv.width = cv.height = 256;
  const g = cv.getContext("2d")!;
  g.fillStyle = grout;
  g.fillRect(0, 0, 256, 256);
  for (let i = 0; i < 4; i++) for (let j = 0; j < 4; j++) {
    const v = ((i * 7 + j * 3) % 5) / 40;
    g.fillStyle = base;
    g.globalAlpha = 1;
    g.fillRect(i * 64 + 2, j * 64 + 2, 60, 60);
    g.fillStyle = `rgba(0,0,0,${v})`;
    g.fillRect(i * 64 + 2, j * 64 + 2, 60, 60);
  }
  const t = new CanvasTexture(cv);
  t.wrapS = t.wrapT = RepeatWrapping;
  t.repeat.set(1 / (size * 4), 1 / (size * 4));
  t.colorSpace = SRGBColorSpace;
  const m = new MSM({ map: t, roughness });
  tileCache.set(key, m);
  return m;
};

// CERDIK effects: the glowing highlight the genre puts on "the thing that matters" (a rail gap, a stone under a post),
// dust puffs on impacts, steam puffs on angry heads. Glows live on the FX layer (no AO / DOF) and are bright enough
// to cross the Stage's high bloom threshold, so only they bloom.
import React, { useMemo } from "react";
import { AdditiveBlending, BackSide, CylinderGeometry, MeshBasicMaterial, SphereGeometry, TorusGeometry, type BufferGeometry } from "three";
import { FX_LAYER } from "./Stage.tsx";
import type { V3 } from "../bedah/cam.ts";

const c01 = (x: number) => (x < 0 ? 0 : x > 1 ? 1 : x);

/** Outline glow around any geometry (inverted hull, additive). `k` 0..1 = how lit; pulses gently once lit. */
export const GlowHull: React.FC<{ geometry: BufferGeometry; pos: V3; rot?: V3; scale: V3; grow?: number; k: number; t: number; color?: string }> = ({ geometry, pos, rot = [0, 0, 0], scale, grow = 1.09, k, t, color = "#ffc94a" }) => {
  const m = useMemo(() => new MeshBasicMaterial({ color, side: BackSide, transparent: true, blending: AdditiveBlending, depthWrite: false, toneMapped: false }), [color]);
  if (k <= 0) return null;
  const pulse = 1 + 0.12 * Math.sin(t * 7.5);
  m.opacity = c01(k);
  m.color.set(color).multiplyScalar(6 * pulse);
  return (
    <mesh geometry={geometry} material={m} position={pos} rotation={rot} scale={[scale[0] * grow, scale[1] * grow, scale[2] * grow]} renderOrder={5} onUpdate={(o) => o.layers.set(FX_LAYER)} />
  );
};

/** Glowing ring lying on a surface (around a post foot). */
export const GlowRing: React.FC<{ pos: V3; r: number; k: number; t: number; color?: string }> = ({ pos, r, k, t, color = "#ffd76a" }) => {
  const g = useMemo(() => new TorusGeometry(1, 0.06, 10, 48), []);
  const m = useMemo(() => new MeshBasicMaterial({ color, transparent: true, blending: AdditiveBlending, depthWrite: false, toneMapped: false }), [color]);
  if (k <= 0) return null;
  m.opacity = c01(k);
  m.color.set(color).multiplyScalar(7 * (1 + 0.1 * Math.sin(t * 7.5)));
  const s = r * (0.8 + 0.2 * c01(k * 1.5));
  return <mesh geometry={g} material={m} position={pos} rotation={[-Math.PI / 2, 0, 0]} scale={[s, s, s]} renderOrder={6} onUpdate={(o) => o.layers.set(FX_LAYER)} />;
};

/** Short-lived dust puff (a few soft spheres growing and fading) starting at time t0. */
export const Dust: React.FC<{ pos: V3; t: number; t0: number; size?: number; color?: string }> = ({ pos, t, t0, size = 0.25, color = "#e3cba6" }) => {
  const g = useMemo(() => new SphereGeometry(1, 16, 10), []);
  const mats = useMemo(() => Array.from({ length: 6 }, () => new MeshBasicMaterial({ color, transparent: true, depthWrite: false })), [color]);
  const a = t - t0;
  if (a < 0 || a > 0.9) return null;
  const k = a / 0.9;
  return (
    <group position={pos}>
      {mats.map((m, i) => {
        const ang = i * 1.047 + 0.3;
        const d = size * (0.4 + 1.6 * Math.sqrt(k));
        m.opacity = 0.32 * (1 - k) * (1 - k) * Math.min(1, a / 0.1);
        const r = size * (0.28 + 0.5 * k);
        return <mesh key={i} geometry={g} material={m} position={[Math.cos(ang) * d, size * 0.3 + k * size * 0.5 + (i % 2) * 0.05, Math.sin(ang) * d]} scale={[r, r * 0.75, r]} onUpdate={(o) => o.layers.set(FX_LAYER)} />;
      })}
    </group>
  );
};

// CERDIK rig — pure maths for the round-head stick figures (no React/three, so poses can be checked in node).
// Proportions follow the reference genre: a big round head (≈ ¼ of the height), a short costume torso and long
// thin black stick limbs. The body is posed from a few targets — pelvis, torso lean, hand and ankle targets — and
// elbows / knees come from two-bone IK, so hands land on props and feet stay on the ground.
// Local frame: the figure faces +z, up is +y, its LEFT side is +x.

export type V3 = [number, number, number];

export type Pose = {
  pelvis: V3;
  /** torso lean forward (+) / back (−), side lean to the left (+), shoulder twist to the left (+) — radians */
  pitch: number;
  roll?: number;
  twist?: number;
  /** head relative to the torso: nod down (+), turn left (+), tilt left (+) */
  nod?: number;
  turn?: number;
  tilt?: number;
  handL: V3;
  handR: V3;
  footL: V3;
  footR: V3;
  /** foot yaw outward */
  splay?: number;
  /** elbow pole: 0 = down/back, 1 = out to the side; knee pole sideways */
  elbowOut?: number;
  kneeOut?: number;
};

export type Build = {
  /** overall scale (1 = 1.47 m) */
  s?: number;
  /** torso half-width / half-depth multipliers (a round belly = 1.3, 1.35) */
  wide?: number;
  deep?: number;
  /** head radius multiplier */
  head?: number;
};

export const DIM = {
  hipW: 0.06, thigh: 0.335, shin: 0.315, ankle: 0.032,
  torsoUp: 0.37, shoulderUp: 0.3, shoulderW: 0.135, upper: 0.215, fore: 0.205,
  neck: 0.035, headR: 0.19,
  torsoW: 0.145, torsoD: 0.105, torsoBelow: 0.07,
  limbR: 0.0085, handR: 0.03,
};

const add = (a: V3, b: V3): V3 => [a[0] + b[0], a[1] + b[1], a[2] + b[2]];
const sub = (a: V3, b: V3): V3 => [a[0] - b[0], a[1] - b[1], a[2] - b[2]];
const mul = (a: V3, s: number): V3 => [a[0] * s, a[1] * s, a[2] * s];
const dot = (a: V3, b: V3) => a[0] * b[0] + a[1] * b[1] + a[2] * b[2];
const cross = (a: V3, b: V3): V3 => [a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0]];
const len = (a: V3) => Math.hypot(a[0], a[1], a[2]);
const norm = (a: V3): V3 => {
  const l = len(a) || 1;
  return [a[0] / l, a[1] / l, a[2] / l];
};

/** Two-bone IK: joint between root and target (lengths a, b), bent toward `pole`. Returns [mid, end]. */
export const ik2 = (root: V3, target: V3, a: number, b: number, pole: V3): [V3, V3] => {
  let d = sub(target, root);
  let L = len(d);
  const max = (a + b) * 0.999, min = Math.abs(a - b) + 1e-3;
  if (L < 1e-6) { d = [0, -1, 0]; L = 1e-6; }
  const dir = mul(d, 1 / L);
  const Lc = Math.min(max, Math.max(min, L));
  const x = (a * a - b * b + Lc * Lc) / (2 * Lc);
  const h = Math.sqrt(Math.max(0, a * a - x * x));
  let bend = sub(pole, mul(dir, dot(pole, dir)));
  if (len(bend) < 1e-5) bend = cross(dir, [1, 0, 0]);
  bend = norm(bend);
  const mid = add(add(root, mul(dir, x)), mul(bend, h));
  return [mid, add(root, mul(dir, Lc))];
};

export type Joints = {
  pelvis: V3; chest: V3; neck: V3; head: V3; up: V3; right: V3; fwd: V3; headUp: V3; headFwd: V3; headRight: V3;
  hipL: V3; hipR: V3; kneeL: V3; kneeR: V3; ankleL: V3; ankleR: V3;
  shL: V3; shR: V3; elbowL: V3; elbowR: V3; wristL: V3; wristR: V3;
  /** torso centre + half extents (for the body mesh and contact checks) */
  torso: V3; torsoHalf: V3; headR: number;
};

const rotX = (v: V3, a: number): V3 => [v[0], v[1] * Math.cos(a) - v[2] * Math.sin(a), v[1] * Math.sin(a) + v[2] * Math.cos(a)];
const rotY = (v: V3, a: number): V3 => [v[0] * Math.cos(a) + v[2] * Math.sin(a), v[1], -v[0] * Math.sin(a) + v[2] * Math.cos(a)];
const rotZ = (v: V3, a: number): V3 => [v[0] * Math.cos(a) - v[1] * Math.sin(a), v[0] * Math.sin(a) + v[1] * Math.cos(a), v[2]];

/** Solve a pose into joint positions (figure-local metres, already scaled by the build). */
export const solve = (p: Pose, b: Build = {}): Joints => {
  const s = b.s ?? 1, W = b.wide ?? 1, Dp = b.deep ?? 1, HR = (b.head ?? 1) * DIM.headR * s;
  const D = DIM;
  // poses are authored for a unit body (s = 1): scale every target with the build
  if (s !== 1) {
    const S = (v: V3): V3 => [v[0] * s, v[1] * s, v[2] * s];
    p = { ...p, pelvis: S(p.pelvis), handL: S(p.handL), handR: S(p.handR), footL: S(p.footL), footR: S(p.footR) };
  }
  const tw = p.twist ?? 0, rl = p.roll ?? 0;
  const tf = (v: V3) => rotZ(rotX(rotY(v, tw), p.pitch), -rl);
  const up = tf([0, 1, 0]), right = tf([1, 0, 0]), fwd = tf([0, 0, 1]);
  const pelvis = p.pelvis;
  const chest = add(pelvis, mul(up, D.shoulderUp * s));
  const neck = add(pelvis, mul(up, D.torsoUp * s));
  const hf = (v: V3) => tf(rotX(rotZ(rotY(v, p.turn ?? 0), -(p.tilt ?? 0)), p.nod ?? 0));
  const headUp = hf([0, 1, 0]), headFwd = hf([0, 0, 1]), headRight = hf([1, 0, 0]);
  const head = add(neck, mul(headUp, D.neck * s + HR * 0.93));
  const hipL = add(pelvis, mul(right, D.hipW * s * Math.max(1, W * 0.9)));
  const hipR = add(pelvis, mul(right, -D.hipW * s * Math.max(1, W * 0.9)));
  const ko = p.kneeOut ?? 0.12;
  const [kneeL, ankleL] = ik2(hipL, p.footL, D.thigh * s, D.shin * s, [ko, 0.2, 1]);
  const [kneeR, ankleR] = ik2(hipR, p.footR, D.thigh * s, D.shin * s, [-ko, 0.2, 1]);
  const shL = add(chest, mul(right, D.shoulderW * s * W)), shR = add(chest, mul(right, -D.shoulderW * s * W));
  const eo = p.elbowOut ?? 0.35;
  const poleL = add(add(mul(right, eo * 1.6), mul(up, -1)), mul(fwd, -0.55));
  const poleR = add(add(mul(right, -eo * 1.6), mul(up, -1)), mul(fwd, -0.55));
  const [elbowL, wristL] = ik2(shL, p.handL, D.upper * s, D.fore * s, poleL);
  const [elbowR, wristR] = ik2(shR, p.handR, D.upper * s, D.fore * s, poleR);
  const torsoHalf: V3 = [D.torsoW * s * W, ((D.torsoUp + D.torsoBelow) * s) / 2, D.torsoD * s * Dp];
  const torso = add(pelvis, mul(up, (D.torsoUp - D.torsoBelow) * s / 2));
  return { pelvis, chest, neck, head, up, right, fwd, headUp, headFwd, headRight, hipL, hipR, kneeL, kneeR, ankleL, ankleR, shL, shR, elbowL, elbowR, wristL, wristR, torso, torsoHalf, headR: HR };
};

const mixN = (a: number | undefined, b: number | undefined, t: number, d = 0) => (a ?? d) + ((b ?? d) - (a ?? d)) * t;
const mixV = (a: V3, b: V3, t: number): V3 => [a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t, a[2] + (b[2] - a[2]) * t];
/** Blend two poses (targets blend, IK re-solves, so limbs never stretch or pop). */
export const mixPose = (a: Pose, b: Pose, t: number): Pose => {
  if (t <= 0) return a;
  if (t >= 1) return b;
  return {
    pelvis: mixV(a.pelvis, b.pelvis, t), pitch: mixN(a.pitch, b.pitch, t), roll: mixN(a.roll, b.roll, t), twist: mixN(a.twist, b.twist, t),
    nod: mixN(a.nod, b.nod, t), turn: mixN(a.turn, b.turn, t), tilt: mixN(a.tilt, b.tilt, t),
    handL: mixV(a.handL, b.handL, t), handR: mixV(a.handR, b.handR, t), footL: mixV(a.footL, b.footL, t), footR: mixV(a.footR, b.footR, t),
    splay: mixN(a.splay, b.splay, t, 0.2), elbowOut: mixN(a.elbowOut, b.elbowOut, t, 0.35), kneeOut: mixN(a.kneeOut, b.kneeOut, t, 0.12),
  };
};
