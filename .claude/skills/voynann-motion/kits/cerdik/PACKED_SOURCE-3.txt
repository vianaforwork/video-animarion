// World registry (pure data for the director). Pictures: src/cerdik/worlds/scenes.tsx.
// Adding a world: write <id>.meta.ts (spots, colliders, bounds, views) + <Id>.tsx (the picture), register both.
import type { WorldMeta } from "../types.ts";
import { DESA } from "./desa.meta.ts";
import { PASAR } from "./pasar.meta.ts";
import { PANTAI } from "./pantai.meta.ts";
import { RUANG } from "./ruang.meta.ts";
import { ISTANA } from "./istana.meta.ts";

export const WORLDS: Record<string, WorldMeta> = { desa: DESA, pasar: PASAR, pantai: PANTAI, ruang: RUANG, istana: ISTANA };

/** Ground height under (x, z) in a world (raised streets, platforms, a pier). */
export const floorAt = (w: WorldMeta, x: number, z: number) => {
  for (const f of w.floor ?? []) {
    const b = f.box;
    if (!b || (x >= b.min[0] && x <= b.max[0] && z >= b.min[2] && z <= b.max[2])) return f.y;
  }
  return 0.012;
};

// CERDIK captions: the genre's word-at-a-time caption — one (sometimes two short) words, white heavy sans with a dark
// outline, centred just below the middle of the frame, swapped exactly on the spoken word. No boxes, no colour pops.
import React from "react";
import { F_CAPTION } from "../bedah/fonts.ts";

export type CapWord = { w: string; t0: number; t1: number };
export type Chunk = { text: string; t0: number; t1: number };

const JOIN_NEXT = new Set(["di", "ke", "si", "yang", "dan", "sang", "para", "tapi"]);
const JOIN_PREV = new Set(["pun", "itu", "ini", "lah", "kah"]);

/** Group words into display chunks (1–2 words, ≤ maxChars). Times are in the same clock as the words. */
export const chunkWords = (words: CapWord[], maxChars = 13, gap = 0.55): Chunk[] => {
  const out: CapWord[][] = [];
  for (let i = 0; i < words.length; i++) {
    const w = words[i];
    const bare = w.w.toLowerCase().replace(/[^a-z-]/g, "");
    const last = out[out.length - 1];
    const lastText = last ? last.map((x) => x.w).join(" ") : "";
    const lastBare = last ? last[last.length - 1].w.toLowerCase().replace(/[^a-z-]/g, "") : "";
    const lastEnds = last ? /[,.!?]$/.test(last[last.length - 1].w) : true;
    const fits = last && last.length < 2 && lastText.length + 1 + w.w.length <= maxChars && !lastEnds;
    if (fits && (JOIN_NEXT.has(lastBare) || JOIN_PREV.has(bare))) last.push(w);
    else out.push([w]);
  }
  return out.map((c, i) => {
    const next = out[i + 1];
    const end = c[c.length - 1].t1;
    return { text: c.map((x) => x.w).join(" "), t0: c[0].t0, t1: next ? Math.min(next[0].t0, end + gap) : end + gap };
  });
};

export const WordCaption: React.FC<{ t: number; chunks: Chunk[]; y?: number; size?: number }> = ({ t, chunks, y = 1170, size = 84 }) => {
  const c = chunks.find((k) => t >= k.t0 - 0.02 && t < k.t1);
  if (!c) return null;
  const age = t - c.t0;
  const pop = age < 0.05 ? 0.94 + (age / 0.05) * 0.06 : 1;
  const text = c.text.replace(/["“”]/g, "");
  return (
    <div style={{ position: "absolute", left: 0, right: 0, top: y, textAlign: "center", pointerEvents: "none" }}>
      <span
        style={{
          display: "inline-block",
          transform: `scale(${pop})`,
          fontFamily: F_CAPTION,
          fontWeight: 700,
          fontSize: size,
          letterSpacing: -0.5,
          lineHeight: 1,
          color: "#ffffff",
          WebkitTextStroke: `${Math.round(size * 0.13)}px #111111`,
          paintOrder: "stroke fill",
          textShadow: "0 4px 10px rgba(0,0,0,0.35)",
          whiteSpace: "nowrap",
        }}
      >
        {text}
      </span>
    </div>
  );
};

import React from "react";
import { useVideoConfig } from "remotion";
import { ThreeCanvas } from "@remotion/three";
import { QaProbe } from "../../core/qa/scene-qa.tsx";
import { PCFShadowMap } from "three";
import { Stage, type Look } from "./Stage.tsx";
import type { Shot } from "../bedah/cam.ts";

/** One CERDIK shot: a Remotion-synchronised canvas with the BEDAH Stage (sun, sky, post chain) around the set. */
export const ShotCanvas: React.FC<{ shot: Shot; look: Look; frame: number; t: number; children: React.ReactNode }> = ({ shot, look, frame, t, children }) => {
  const { width, height } = useVideoConfig();
  return (
    <ThreeCanvas
      width={width}
      height={height}
      shadows={{ type: PCFShadowMap }}
      dpr={1}
      gl={{ antialias: false, preserveDrawingBuffer: true, powerPreference: "high-performance" }}
      camera={{ fov: shot.fov, position: shot.pos, near: 0.08, far: 1500 }}
    >
      <Stage shot={shot} look={look} frame={frame} t={t}>
        {children}
      </Stage>
      <QaProbe />
    </ThreeCanvas>
  );
};

==== 2097 project/src/scenes/lookdev/WorldLook.tsx
// World lookdev: frame 0/1 = the world's establish move (a, b), 2/3 = aerial (a, b), 4… = a medium view of each spot.
import React, { useEffect, useState } from "react";
import { AbsoluteFill, continueRender, delayRender, getInputProps, useCurrentFrame } from "remotion";
import { ShotCanvas } from "../../kits/cerdik/ShotCanvas.tsx";
import { fontsReady } from "../../kits/bedah/fonts.ts";
import { LOOKS } from "../../kits/cerdik/looks.ts";
import { WORLDS } from "../../cerdik/worlds/index.ts";
import { SCENES } from "../../cerdik/worlds/scenes.tsx";
import type { TimeOfDay } from "../../cerdik/types.ts";

export const WorldLook: React.FC = () => {
  const f = useCurrentFrame();
  const raw = getInputProps() as { world?: string; time?: TimeOfDay; plan?: { world?: string; time?: TimeOfDay } };
  const ip = raw.plan ?? raw;
  const id = ip.world ?? "pasar", time = ip.time ?? "day";
  const W = WORLDS[id], Scene = SCENES[id];
  const [h] = useState(() => delayRender("fonts"));
  useEffect(() => { fontsReady().then(() => continueRender(h)); }, [h]);
  const spots = Object.entries(W.spots);
  let cam;
  if (f < 4) { const v = W.views[f < 2 ? "establish" : "aerial"]; cam = f % 2 ? v.b : v.a; }
  else {
    const [, s] = spots[(f - 4) % spots.length];
    const d = 3.2, fx = Math.sin(s.face), fz = Math.cos(s.face);
    const sx = -Math.cos(s.face), sz = Math.sin(s.face);
    cam = { pos: [s.pos[0] + fx * d + sx * 1.2, 1.6, s.pos[2] + fz * d + sz * 1.2] as [number, number, number], target: [s.pos[0], 1.0, s.pos[2]] as [number, number, number], fov: 44 };
  }
  return (
    <AbsoluteFill style={{ background: "#000" }}>
      <ShotCanvas shot={{ ...cam, shadowAt: cam.target, shadowBox: f < 4 ? 30 : 8 }} look={LOOKS[time]} frame={f} t={f / 30}>
        <Scene t={f / 30} time={time} avoid={[]} />
      </ShotCanvas>
      <div style={{ position: "absolute", left: 12, top: 10, color: "#fff", fontFamily: "sans-serif", fontSize: 18, textShadow: "0 1px 3px #000" }}>{f < 4 ? (f < 2 ? "establish" : "aerial") : spots[(f - 4) % spots.length][0]}</div>
    </AbsoluteFill>
  );
};
