// TIPIKAL type (all SIL OFL, licences next to the files in public/tipikal):
//   Baloo 2 (variable 400–800) — type names, stickers, bubbles
//   Barlow Condensed ExtraBold — the road-sign header, plates, shop signs
import { useEffect, useState } from "react";
import { continueRender, delayRender, staticFile } from "remotion";

export const F_ROUND = "TipikalBaloo";
export const F_SIGN = "TipikalBarlow";

let ready: Promise<void> | null = null;
export const fontsReady = (): Promise<void> => {
  if (ready) return ready;
  if (typeof document === "undefined") return (ready = Promise.resolve());
  const faces = [
    new FontFace(F_ROUND, `url(${staticFile("tipikal/Baloo2-Var.ttf")}) format("truetype")`, { weight: "400 800" }),
    new FontFace(F_SIGN, `url(${staticFile("tipikal/BarlowCondensed-ExtraBold.ttf")}) format("truetype")`, { weight: "800" }),
  ];
  ready = Promise.all(
    faces.map(async (f) => {
      await f.load();
      (document.fonts as unknown as { add(f: FontFace): void }).add(f);
    }),
  ).then(() => undefined);
  return ready;
};

/** Holds the render until the fonts are in (canvas textures draw text with them). */
export const useFontsReady = () => {
  const [ok, setOk] = useState(false);
  const [handle] = useState(() => delayRender("tipikal fonts"));
  useEffect(() => {
    fontsReady().then(
      () => { setOk(true); continueRender(handle); },
      () => { setOk(true); continueRender(handle); },
    );
  }, [handle]);
  return ok;
};

// TIPIKAL maths: vectors, easing, C1 keyframe tracks and camera keys (pure, no React).
// Track sampler forked from src/core/motion/track.ts (Catmull-Rom tangents, monotone clamp, hold keys).
import { PerspectiveCamera, Vector3 } from "three";

export type V3 = [number, number, number];
export type V2 = [number, number];

export const clamp01 = (x: number) => (x < 0 ? 0 : x > 1 ? 1 : x);
export const lin = (s: number, a: number, b: number) => clamp01((s - a) / (b - a));
export const sm = (t: number) => {
  const x = clamp01(t);
  return x * x * (3 - 2 * x);
};
export const smoother = (t: number) => {
  const x = clamp01(t);
  return x * x * x * (x * (x * 6 - 15) + 10);
};
/** ease-out with a small overshoot (labels landing) */
export const backOut = (t: number, k = 1.6) => {
  const x = clamp01(t) - 1;
  return 1 + (k + 1) * x * x * x + k * x * x;
};
export const easeOut = (t: number) => 1 - Math.pow(1 - clamp01(t), 3);
export const easeIn = (t: number) => Math.pow(clamp01(t), 3);
export const mix = (a: number, b: number, t: number) => a + (b - a) * t;
export const mix3 = (a: V3, b: V3, t: number): V3 => [mix(a[0], b[0], t), mix(a[1], b[1], t), mix(a[2], b[2], t)];
export const add = (a: V3, b: V3, s = 1): V3 => [a[0] + b[0] * s, a[1] + b[1] * s, a[2] + b[2] * s];
export const sub = (a: V3, b: V3): V3 => [a[0] - b[0], a[1] - b[1], a[2] - b[2]];
export const mul = (a: V3, s: number): V3 => [a[0] * s, a[1] * s, a[2] * s];
export const dot = (a: V3, b: V3) => a[0] * b[0] + a[1] * b[1] + a[2] * b[2];
export const cross = (a: V3, b: V3): V3 => [a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0]];
export const len = (a: V3) => Math.hypot(a[0], a[1], a[2]);
export const nrm = (a: V3): V3 => {
  const l = len(a) || 1;
  return [a[0] / l, a[1] / l, a[2] / l];
};
export const rotY = (v: V3, a: number): V3 => [v[0] * Math.cos(a) + v[2] * Math.sin(a), v[1], -v[0] * Math.sin(a) + v[2] * Math.cos(a)];
export const wrapA = (a: number) => Math.atan2(Math.sin(a), Math.cos(a));

/** deterministic hash 0..1 */
export const hash = (n: number) => {
  const x = Math.sin(n * 127.1 + 311.7) * 43758.5453;
  return x - Math.floor(x);
};

// ------------------------------------------------------------------ keyframe tracks
export type Key = { f: number; v: number[]; hold?: boolean };
const hermite = (p0: number, p1: number, m0: number, m1: number, t: number, h: number) => {
  const t2 = t * t, t3 = t2 * t;
  return (2 * t3 - 3 * t2 + 1) * p0 + (t3 - 2 * t2 + t) * h * m0 + (-2 * t3 + 3 * t2) * p1 + (t3 - t2) * h * m1;
};
const tangent = (keys: Key[], i: number, c: number) => {
  const k = keys[i];
  if (k.hold || i === 0 || i === keys.length - 1) return 0;
  const a = keys[i - 1], b = keys[i + 1];
  const m = (b.v[c] - a.v[c]) / (b.f - a.f);
  const d0 = (k.v[c] - a.v[c]) / (k.f - a.f);
  const d1 = (b.v[c] - k.v[c]) / (b.f - k.f);
  if (d0 * d1 <= 0) return 0;
  const lim = 3 * Math.min(Math.abs(d0), Math.abs(d1));
  return Math.sign(m) * Math.min(Math.abs(m), lim);
};
export const sample = (keys: Key[], f: number): number[] => {
  if (f <= keys[0].f) return keys[0].v.slice();
  const last = keys[keys.length - 1];
  if (f >= last.f) return last.v.slice();
  let i = 0;
  while (i < keys.length - 2 && f >= keys[i + 1].f) i++;
  const a = keys[i], b = keys[i + 1];
  const h = b.f - a.f;
  const t = (f - a.f) / h;
  return a.v.map((_, c) => hermite(a.v[c], b.v[c], tangent(keys, i, c), tangent(keys, i + 1, c), t, h));
};

// ------------------------------------------------------------------ camera
export type Shot = { pos: V3; target: V3; fov: number; roll?: number; shadowAt?: V3; shadowBox?: number };
export type CamKey = { t: number; pos: V3; target: V3; fov?: number; roll?: number; hold?: boolean };
export const camAt = (keys: CamKey[], s: number): Shot => {
  const k: Key[] = keys.map((c) => ({ f: c.t, v: [...c.pos, ...c.target, c.fov ?? 40, c.roll ?? 0], hold: c.hold }));
  const v = sample(k, s);
  return { pos: [v[0], v[1], v[2]], target: [v[3], v[4], v[5]], fov: v[6], roll: v[7] };
};

const tmpCam = new PerspectiveCamera();
/** World point → overlay space (0..1, y down). */
export const project = (shot: Shot, aspect: number, p: V3) => {
  tmpCam.fov = shot.fov;
  tmpCam.aspect = aspect;
  tmpCam.near = 0.08;
  tmpCam.far = 1500;
  tmpCam.position.set(...shot.pos);
  tmpCam.up.set(Math.sin(shot.roll ?? 0), Math.cos(shot.roll ?? 0), 0);
  tmpCam.lookAt(...shot.target);
  tmpCam.updateProjectionMatrix();
  tmpCam.updateMatrixWorld();
  const v = new Vector3(...p).project(tmpCam);
  return { x: (v.x + 1) / 2, y: (1 - v.y) / 2, vis: v.z < 1 && v.z > -1 };
};

// TIPIKAL materials: flat cel-shaded toon colours (3 light steps, no gradients), low-poly set materials, ink hulls.
import React from "react";
import {
  BackSide, BufferGeometry, DataTexture, Matrix4, MeshBasicMaterial, MeshStandardMaterial, MeshToonMaterial, NearestFilter,
  RedFormat, Vector3, type Material,
} from "three";
import { mergeVertices } from "three/examples/jsm/utils/BufferGeometryUtils.js";
import type { V3 } from "./math.ts";

export const INK = "#1b1512";

const mkRamp = (v: number[]) => {
  const d = new Uint8Array(v);
  const t = new DataTexture(d, 3, 1, RedFormat);
  t.minFilter = NearestFilter;
  t.magFilter = NearestFilter;
  t.generateMipmaps = false;
  t.needsUpdate = true;
  return t;
};
const ramp = mkRamp([120, 200, 255]);
const softRamp = mkRamp([185, 228, 255]);
export const SOFT_RAMP = softRamp;

const toonCache = new Map<string, MeshToonMaterial>();
/** cel-shaded colour for characters and vehicles (soft = skin: a gentle shadow step so faces stay readable) */
export const toon = (c: string, soft = false) => {
  const k = c + (soft ? "s" : "");
  let m = toonCache.get(k);
  if (!m) toonCache.set(k, (m = new MeshToonMaterial({ color: c, gradientMap: soft ? softRamp : ramp })));
  return m;
};
const flatCache = new Map<string, MeshStandardMaterial>();
/** low-poly set colour (faceted, matte) */
export const flat = (c: string, rough = 0.85, faceted = true) => {
  const k = `${c}|${rough}|${faceted}`;
  let m = flatCache.get(k);
  if (!m) flatCache.set(k, (m = new MeshStandardMaterial({ color: c, roughness: rough, metalness: 0, flatShading: faceted })));
  return m;
};
const glowCache = new Map<string, MeshBasicMaterial>();
/** lamps that are ON: unlit flat colour (no bloom, no glow) */
export const lamp = (c: string) => {
  let m = glowCache.get(c);
  if (!m) glowCache.set(c, (m = new MeshBasicMaterial({ color: c, toneMapped: false })));
  return m;
};
export const inkHull = new MeshBasicMaterial({ color: INK, side: BackSide });

/** Outline hull for any closed geometry: welded copy pushed out along smooth vertex normals. */
export const hullOf = (g: BufferGeometry, thick: number) => {
  const c = g.clone();
  c.deleteAttribute("normal");
  if (c.getAttribute("uv")) c.deleteAttribute("uv");
  if (c.getAttribute("uv1")) c.deleteAttribute("uv1");
  const w = mergeVertices(c, 1e-4);
  w.computeVertexNormals();
  const p = w.getAttribute("position"), n = w.getAttribute("normal");
  for (let i = 0; i < p.count; i++) p.setXYZ(i, p.getX(i) + n.getX(i) * thick, p.getY(i) + n.getY(i) * thick, p.getZ(i) + n.getZ(i) * thick);
  p.needsUpdate = true;
  w.computeBoundingSphere();
  c.dispose();
  return w;
};

// ------------------------------------------------------------------ matrix helpers
/** unit +Y cylinder (base at 0, top at 1) from a to b with radius r */
export const stick = (a: V3, b: V3, r: number) => {
  const A = new Vector3(...a), B = new Vector3(...b);
  const d = B.clone().sub(A);
  const L = d.length() || 1e-6;
  const m = new Matrix4();
  const dir = d.clone().normalize();
  // basis with y = dir
  const x = Math.abs(dir.y) < 0.95 ? new Vector3(0, 1, 0).cross(dir).normalize() : new Vector3(1, 0, 0).cross(dir).normalize();
  const z = x.clone().cross(dir).normalize();
  m.makeBasis(x.multiplyScalar(r), dir.multiplyScalar(L), z.multiplyScalar(r));
  m.setPosition(A);
  return m;
};
export const basis = (p: V3, x: V3, y: V3, z: V3, s: V3 = [1, 1, 1]) => {
  const m = new Matrix4().makeBasis(new Vector3(...x), new Vector3(...y), new Vector3(...z));
  m.scale(new Vector3(...s));
  m.setPosition(...p);
  return m;
};
export const at = (p: V3, s: V3 = [1, 1, 1]) => basis(p, [1, 0, 0], [0, 1, 0], [0, 0, 1], s);

// TIPIKAL prop renderer: builds a PropDef's parts (box / cyl / cone / sphere / torus / lathe) with cel-shaded flat
// colours and a thin ink outline, like the people. Geometries are cached by their parameters.
import React from "react";
import { BoxGeometry, BufferGeometry, CylinderGeometry, Euler, LatheGeometry, Matrix4, MeshStandardMaterial, Quaternion, SphereGeometry, TorusGeometry, Vector2, Vector3 } from "three";
import { RoundedBoxGeometry } from "three/examples/jsm/geometries/RoundedBoxGeometry.js";
import { flat, inkHull, lamp, M, toon } from "../../kits/tipikal/mats.ts";
import type { Part, PropDef } from "./lib.ts";

const geoCache = new Map<string, BufferGeometry>();
const geo = (p: Part, grow = 0): BufferGeometry => {
  const key = JSON.stringify({ ...p, pos: 0, rot: 0, color: 0, mat: 0, grow });
  let g = geoCache.get(key);
  if (g) return g;
  switch (p.shape) {
    case "box": g = p.round ? new RoundedBoxGeometry(p.size[0] + grow * 2, p.size[1] + grow * 2, p.size[2] + grow * 2, 2, Math.min(...p.size) * p.round) : new BoxGeometry(p.size[0] + grow * 2, p.size[1] + grow * 2, p.size[2] + grow * 2); g.translate(0, p.size[1] / 2, 0); break;
    case "cyl": g = new CylinderGeometry((p.r2 ?? p.r) + grow, p.r + grow, p.h + grow * 2, p.seg ?? 18, 1); g.translate(0, p.h / 2, 0); break;
    case "cone": g = new CylinderGeometry(0, p.r + grow, p.h + grow, p.seg ?? 18, 1); g.translate(0, p.h / 2, 0); break;
    case "sphere": g = new SphereGeometry(p.r + grow, 18, 12); break;
    case "torus": g = new TorusGeometry(p.r, p.tube + grow, 8, 24, ((p.arc ?? 360) * Math.PI) / 180); break;
    case "lathe": g = new LatheGeometry(p.profile.map(([r, y]) => new Vector2(r + grow, y)), 24); break;
  }
  geoCache.set(key, g!);
  return g!;
};
const shiny = new Map<string, MeshStandardMaterial>();
const matOf = (p: Part) => {
  const m = p.mat ?? "toon";
  if (m === "flat") return flat(p.color, 0.85, false);
  if (m === "screen") return lamp(p.color);
  if (m === "shiny") { let s = shiny.get(p.color); if (!s) shiny.set(p.color, (s = new MeshStandardMaterial({ color: p.color, roughness: 0.35, metalness: 0.4 }))); return s; }
  return toon(p.color);
};
const D = Math.PI / 180;
const partMatrix = (p: Part) => {
  const pos = p.pos ?? [0, 0, 0];
  const rot = "rot" in p && p.rot ? p.rot : [0, 0, 0];
  const sc = p.shape === "sphere" && p.scale ? p.scale : [1, 1, 1];
  return new Matrix4().compose(new Vector3(...pos), new Quaternion().setFromEuler(new Euler(rot[0] * D, rot[1] * D, rot[2] * D)), new Vector3(...sc));
};

/** A prop at the origin of its group (bottom centre); the caller positions / rotates the group. */
export const PropModel: React.FC<{ def: PropDef }> = ({ def }) => (
  <group>
    {def.parts.map((p, i) => {
      const m = partMatrix(p);
      const ink = !def.noInk && (p.mat ?? "toon") === "toon" && p.shape !== "torus";
      return (
        <group key={i}>
          <M g={geo(p)} m={m} material={matOf(p)} />
          {ink ? <M g={geo(p, 0.006)} m={m} material={inkHull} shadow={false} /> : null}
        </group>
      );
    })}
  </group>
);

export const floorAt = (S: SetMeta, x: number, z: number) => {
  let y = 0;
  for (const f of S.floor ?? []) if (x >= f.box.min[0] && x <= f.box.max[0] && z >= f.box.min[2] && z <= f.box.max[2]) y = Math.max(y, f.y);

  }
  return y;
};

// Text fitting for the pops (sticker, bubble, tag, sfx, stamp) — pure TypeScript, shared by the renderer (Overlay /
// Film: box sizes, font sizes, keeping a pop inside the frame) and the compiler (a plain-language warning when a text
// is too long to stay big). Widths are estimates of the two OFL display fonts per character class, a little generous
// so a pop never spills out of its badge.

type FontKey = "round" | "sign";
// advance widths in em: [uppercase, lowercase, digit, space, narrow punctuation, other]
const EM: Record<FontKey, { up: number; low: number; dig: number; sp: number; nar: number; oth: number; wide: number; thin: number }> = {
  round: { up: 0.66, low: 0.56, dig: 0.6, sp: 0.26, nar: 0.3, oth: 0.62, wide: 0.92, thin: 0.32 },
  sign: { up: 0.5, low: 0.45, dig: 0.5, sp: 0.2, nar: 0.24, oth: 0.52, wide: 0.72, thin: 0.25 },
};
const charW = (c: string, f: FontKey) => {
  const e = EM[f];
  if (c === " ") return e.sp;
  if (".,:;'!|".includes(c)) return e.nar;
  if ("MWmw%@".includes(c)) return e.wide;
  if ("IJiljrt1()[]".includes(c)) return e.thin;
  if (/[0-9]/.test(c)) return e.dig;
  if (/[A-Z]/.test(c)) return e.up;
  if (/[a-z]/.test(c)) return e.low;
  return e.oth;
};
/** estimated width (px) of a single line at font size fs (px), with letter spacing ls (px) */
export const textW = (text: string, fs: number, font: FontKey = "round", ls = 0) => [...text].reduce((w, c) => w + charW(c, font) * fs + ls, 0);

export type PopFit = { w: number; h: number; fs: number; nominal: number; pill?: boolean };
const MARGIN = 28;

/** Box size (layout px) and font size of a pop with this text in a frame `LW` wide. */
export const popFit = (kind: string, text: string, LW = 1080): PopFit => {
  const maxW = LW - 2 * MARGIN;
  const fit = (nominal: number, pad: number, font: FontKey, ls: number, h: number, cap = maxW): PopFit => {
    const w0 = textW(text, nominal, font, ls);
    const fs = w0 + pad > cap ? nominal * Math.max(0.2, (cap - pad) / w0) : nominal;
    return { w: Math.min(cap, textW(text, fs, font, ls) + pad), h, fs, nominal };
  };
  switch (kind) {
    case "bubble": return fit(text.length > 18 ? 51 : 64, 64 + 14, "round", 0, 100);
    case "tag": return fit(58, 60 + 12, "round", 0, 84);
    case "sfx": return fit(124, 0.24 * 124, "round", 2, 150);
    case "stamp": {
      // rotated −12°: the box's footprint grows by its height × sin 12°
      const f = fit(92, 56 + 36, "sign", 4, 120, (maxW - 120 * 0.21) / 0.98);
      return { ...f, w: f.w * 0.98 + 120 * 0.21 };
    }
    default: {
      // sticker: a round badge, or a pill when the text is wider than the badge (e.g. "100.000")
      const size = 132;
      const nominal = text.length > 2 ? size * 0.42 : text.length > 1 ? size * 0.56 : size * 0.7;
      const w0 = textW(text, nominal);
      if (w0 <= size * 0.78) return { w: size, h: size, fs: nominal, nominal };
      const cap = Math.min(maxW, 620), pad = size * 0.5;
      const fs = w0 + pad > cap ? nominal * Math.max(0.2, (cap - pad) / w0) : nominal;
      return { w: Math.min(cap, textW(text, fs) + pad), h: size, fs, nominal, pill: true };
    }
  }
};

/** letters that still fit at full size (for the warning text) */
export const popMaxLetters = (kind: string, LW = 1080) => {
  let n = 4;
  while (n < 80 && popFit(kind, "A".repeat(n + 1), LW).fs >= popFit(kind, "A".repeat(n + 1), LW).nominal * 0.999) n++;
  return n;
};

/** Centre x for a box `w` wide anchored near `x`, kept fully inside the frame. */
export const clampBox = (x: number, w: number, LW: number) => Math.max(MARGIN + w / 2, Math.min(LW - MARGIN - w / 2, x));
export const POP_MARGIN = MARGIN;
