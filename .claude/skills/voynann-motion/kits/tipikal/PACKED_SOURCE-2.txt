const renderCards = (groups, h, outDir, names) => {
  fs.mkdirSync(outDir, { recursive: true });
  const tmp = path.join(outDir, "_strips"), pf = path.join(outDir, "_card-props.json");
  fs.rmSync(tmp, { recursive: true, force: true });
  fs.writeFileSync(pf, JSON.stringify({ strips: groups.flat(), h }));
  sh("npx", ["remotion", "render", "src/index.ts", "TipikalCard", tmp, "--sequence", "--image-format=jpeg", "--jpeg-quality=92", `--props=${pf}`, "--concurrency=2", "--log=error"]);
  const files = fs.readdirSync(tmp).filter((f) => /^element-\d+\.jpe?g$/.test(f)).sort((a, b) => parseInt(a.slice(8)) - parseInt(b.slice(8))).map((f) => path.join(tmp, f));
  let k = 0;
  const made = groups.map((g, gi) => {
    const part = files.slice(k, (k += g.length)), to = path.join(outDir, names[gi]);
    if (part.length === 1) fs.copyFileSync(part[0], to);
    else execFileSync("ffmpeg", ["-v", "error", "-y", ...part.flatMap((f) => ["-i", f]), "-filter_complex", part.map((_, i) => `[${i}:v]`).join("") + `vstack=inputs=${part.length}`, "-q:v", "3", to]);
    return to;
  });
  fs.rmSync(tmp, { recursive: true, force: true });
  fs.rmSync(pf, { force: true });
  return made;
};

  const cache = path.join(out, "timing.json");
  const stamp = `${fs.statSync(vo).mtimeMs}|${lead}|${texts.join("|")}`;
  const c = fs.existsSync(cache) ? JSON.parse(fs.readFileSync(cache, "utf8")) : null;
  if (c?.stamp === stamp && fs.existsSync(c.voice)) { lines = c.lines; voice = c.voice; console.log(`voice: ${vo} (timings cached)`); }
  else {
    const r = syncVo(vo, texts, path.join(out, "vo"), lead);
    lines = r.lines; voice = r.voice;
    // this genre leaves long pauses between the type lines (the vignette plays): a line whose first words whisper
    // placed inside a pause starts where the voice comes back
    const sd = spawnSync("ffmpeg", ["-hide_banner", "-i", vo, "-af", "silencedetect=noise=-38dB:d=0.35", "-f", "null", "-"], { encoding: "utf8" }).stderr;
    const ss = [...sd.matchAll(/silence_start: ([\d.]+)/g)].map((m) => +m[1] + lead), se = [...sd.matchAll(/silence_end: ([\d.]+)/g)].map((m) => +m[1] + lead);
    for (const L of lines) {
      const last = L.words[L.words.length - 1].t;
      const k = ss.findIndex((a, n) => se[n] !== undefined && se[n] - a >= 0.6 && a >= L.t0 - 0.15 && se[n] <= last + 0.05);
      if (k < 0) continue;
      let j = 0;
      L.words = L.words.map((w) => (w.t < se[k] ? { ...w, t: se[k] + 0.12 * j++ } : w));
      for (let n = 1; n < L.words.length; n++) if (L.words[n].t < L.words[n - 1].t + 0.05) L.words[n] = { ...L.words[n], t: L.words[n - 1].t + 0.05 };
      L.t0 = se[k];
      L.t1 = Math.max(L.t1, L.words[L.words.length - 1].t + 0.2);
    }
    fs.writeFileSync(cache, JSON.stringify({ stamp, lines, voice }));
    console.log(r.method === "whisper" ? `voice: ${vo} — whisper matched all but ${r.missed} word(s)` : `voice: ${vo} — whisper.cpp not found: lines laid onto the speech runs`);
    if (r.missed > texts.join(" ").split(/\s+/).length * 0.15) console.log(`! many words not matched: is the recording these lines?\n  heard: ${r.heard}`);
  }

// Technique adapted from the CERDIK face decal (face-plane coordinates projected on a sphere patch, discrete
// expressions cached), redrawn for this look: white eyes with big pupils and lids, a nose, chunky brows and a wide
// meme mouth. Face-plane coords: x right as you look at the face, y up, both in head radii.
import { CanvasTexture, LinearFilter, SRGBColorSpace } from "three";
import type { FaceSpec } from "./spec.ts";

export const PATCH = { lon: (72 * Math.PI) / 180, lat: (72 * Math.PI) / 180 };

export type Expr =
  | "neutral" | "happy" | "grin" | "smug" | "shock" | "huh" | "panic" | "relief" | "innocent" | "focus" | "angry" | "proud"
  | "laugh" | "sleep" | "sad" | "cry" | "bored" | "annoyed" | "scared" | "wow" | "chew";
export const EXPRS: Expr[] = ["neutral", "happy", "grin", "smug", "shock", "huh", "panic", "relief", "innocent", "focus", "angry", "proud", "laugh", "sleep", "sad", "cry", "bored", "annoyed", "scared", "wow", "chew"];

export type FaceState = {
  expr: Expr;
  /** eyelids 0 open … 1 closed (blink) */
  blink?: number;
  /** pupils looking (x right as seen, y up) in head radii */
  look?: [number, number];
  /** talking: mouth open (0 / 1) */
  open?: number;
};

type P2 = [number, number];
const projector = (W: number, H: number) => (p: P2): P2 => {
  const [x, y] = p;
  const r2 = Math.min(0.995, x * x + y * y);
  const z = Math.sqrt(1 - r2);
  const lon = Math.atan2(x, z), lat = Math.asin(Math.max(-1, Math.min(1, y)));
  return [((lon + PATCH.lon) / (2 * PATCH.lon)) * W, ((PATCH.lat - lat) / (2 * PATCH.lat)) * H];
};

const ellipse = (cx: number, cy: number, rx: number, ry: number, rot = 0, n = 56): P2[] => {
  const pts: P2[] = [];
  const c = Math.cos(rot), s = Math.sin(rot);
  for (let i = 0; i <= n; i++) {
    const a = (Math.PI * 2 * i) / n;
    const x = Math.cos(a) * rx, y = Math.sin(a) * ry;
    pts.push([cx + x * c - y * s, cy + x * s + y * c]);
  }
  return pts;
};
const quad = (a: P2, c: P2, b: P2, n = 24): P2[] => {
  const pts: P2[] = [];
  for (let i = 0; i <= n; i++) {
    const t = i / n, u = 1 - t;
    pts.push([u * u * a[0] + 2 * u * t * c[0] + t * t * b[0], u * u * a[1] + 2 * u * t * c[1] + t * t * b[1]]);
  }
  return pts;
};
const lens = (upper: P2[], lower: P2[]): P2[] => [...upper, ...lower.slice().reverse()];
/** thick tapered stroke along a polyline as a filled polygon */
const ribbon = (pts: P2[], w0: number, w1: number): P2[] => {
  const L: P2[] = [], R: P2[] = [];
  for (let i = 0; i < pts.length; i++) {
    const a = pts[Math.max(0, i - 1)], b = pts[Math.min(pts.length - 1, i + 1)];
    const dx = b[0] - a[0], dy = b[1] - a[1];
    const l = Math.hypot(dx, dy) || 1;
    const t = i / (pts.length - 1);
    const w = (w0 + (w1 - w0) * t) * (0.55 + 0.45 * Math.sin(Math.PI * Math.min(1, t * 1.15 + 0.08)));
    L.push([pts[i][0] - (dy / l) * w / 2, pts[i][1] + (dx / l) * w / 2]);
    R.push([pts[i][0] + (dy / l) * w / 2, pts[i][1] - (dx / l) * w / 2]);
  }
  return [...L, ...R.reverse()];
};

type Ctx = { g: CanvasRenderingContext2D; P: (p: P2) => P2; k: number };
const path = (C: Ctx, pts: P2[]) => {
  const { g, P } = C;
  g.beginPath();
  pts.forEach((p, i) => { const q = P(p); if (i) g.lineTo(q[0], q[1]); else g.moveTo(q[0], q[1]); });
  g.closePath();
};
const fill = (C: Ctx, pts: P2[], color: string) => { path(C, pts); C.g.fillStyle = color; C.g.fill(); };
const stroke = (C: Ctx, pts: P2[], w: number, color: string, closed = false) => {
  const { g, P, k } = C;
  g.beginPath();
  pts.forEach((p, i) => { const q = P(p); if (i) g.lineTo(q[0], q[1]); else g.moveTo(q[0], q[1]); });
  if (closed) g.closePath();
  g.lineWidth = w * k;
  g.lineCap = "round";
  g.lineJoin = "round";
  g.strokeStyle = color;
  g.stroke();
};

const drawEye = (C: Ctx, side: -1 | 1, e: Eye, blink: number, look: P2, skinShade: string, spec: FaceSpec) => {
  const sz = e.size ?? 1;
  const cx = side * EX, cy = EY;
  const rx = 0.135 * sz, ry = 0.165 * sz;
  const open = e.closed ? 0 : e.open * (1 - blink);
  if (open < 0.08) {
    const up = e.closed === "up";
    const pts = up
      ? quad([cx - rx, cy - 0.03], [cx, cy + 0.11], [cx + rx, cy - 0.03])
      : quad([cx - rx, cy + 0.02], [cx, cy - 0.07], [cx + rx, cy + 0.02]);
    stroke(C, pts, 0.045, INK);
    return;
  }
  const top = cy - ry + 2 * ry * open;
  const low = cy - ry + 2 * ry * (e.low ?? 0) * 0.5;
  const slope = (e.slope ?? 0) * -side;
  const vis: P2[] = ellipse(cx, cy, rx, ry).map(([x, y]) => {
    const lim = top + slope * (x - cx);
    return [x, Math.max(low, Math.min(lim, y))];
  });
  fill(C, vis, "#ffffff");
  C.g.save();
  path(C, vis);
  C.g.clip();
  const pr = 0.085 * (e.pupil ?? 1) * sz;
  const px = cx + look[0] * 0.55, py = cy + look[1] * 0.55 - 0.01;
  fill(C, ellipse(px, py, pr, pr * 1.12), spec.pupil ?? "#2a1a12");
  fill(C, ellipse(px, py, pr * 0.52, pr * 0.58), "#0b0706");
  fill(C, ellipse(px + pr * 0.38, py + pr * 0.42, pr * 0.3, pr * 0.3), "#ffffff");
  C.g.restore();
  stroke(C, vis, 0.03, INK, true);
  // upper lid line, heavier
  const lidPts: P2[] = [];
  for (let i = 0; i <= 20; i++) {
    const x = cx - rx * 1.05 + (2.1 * rx * i) / 20;
    const yEll = cy + ry * Math.sqrt(Math.max(0, 1 - ((x - cx) / rx) ** 2));
    lidPts.push([x, Math.min(yEll, top + slope * (x - cx))]);

  }
  void skinShade;
};

const cache = new Map<string, CanvasTexture>();

export const faceTexture = (spec: FaceSpec, st: FaceState, skin: string, size = 1024): CanvasTexture => {
  const blink = q(st.blink ?? 0, 2);
  const look: P2 = [Math.round((st.look?.[0] ?? 0) * 20) / 20, Math.round((st.look?.[1] ?? 0) * 20) / 20];
  const open = (st.open ?? 0) > 0.5 ? 1 : 0;
  const key = `${JSON.stringify(spec)}|${st.expr}|${blink}|${look}|${open}|${skin}|${size}`;
  const hit = cache.get(key);
  if (hit) return hit;
  const cv = document.createElement("canvas");
  cv.width = size;
  cv.height = size;
  const g = cv.getContext("2d")!;
  g.clearRect(0, 0, size, size);

  const FS = 1.24, CY = -0.02;
  const k = (size / (2 * PATCH.lon)) * FS;
  const P0 = projector(size, size);
  const C: Ctx = { g, P: (p) => P0([p[0] * FS, (p[1] - CY) * FS + CY]), k };

  const shade = "rgba(90,40,20,0.35)";

    const up = quad([-0.34, -0.42], [0, -0.6], [0.34, -0.42]);
    const dn = quad([-0.34, -0.42], [0, -0.9], [0.34, -0.42]);
    fill(C, lens(up, dn), "#2a1d16");
  }
  if (spec.blush || R.blush) {
    fill(C, ellipse(-0.47, -0.16, 0.11, 0.06), "rgba(235,95,95,0.5)");
    fill(C, ellipse(0.47, -0.16, 0.11, 0.06), "rgba(235,95,95,0.5)");
  }
  drawEye(C, -1, R.eye, blink, look, shade, spec);
  drawEye(C, 1, R.eyeR ?? R.eye, blink, look, shade, spec);
  if (spec.glasses) {
    for (const sd of [-1, 1] as const) stroke(C, ellipse(sd * EX, EY, 0.18, 0.17), 0.032, INK, true);
    stroke(C, quad([-EX + 0.18, EY + 0.03], [0, EY + 0.07], [EX - 0.18, EY + 0.03]), 0.03, INK);

// Generic matic scooter — dimensions and contact anchors (pure maths, shared by the renderer and the poses).
// Local frame: faces +z, up +y, LEFT side = +x, origin on the ground midway between the wheels. No brand marks.
import { add, cross, dot, mul, nrm, sub, type V3 } from "./math.ts";

/** The scooter is modelled at real size and drawn at K (a slightly toy-like scale that suits the chunky cast:
 *  the riders' feet reach the road when they stop). Every anchor below is already scaled. */
export const K = 0.88;
const k3 = (v: V3): V3 => [v[0] * K, v[1] * K, v[2] * K];
export const MOTOR = {
  wheelR: 0.235 * K,
  rearZ: -0.6 * K,
  frontZ: 0.64 * K,
  /** seat surface height under the rider / pillions */
  seatTop: 0.815 * K,
  riderZ: -0.2 * K,
  pillionZ: [-0.5 * K, -0.7 * K] as number[],
  floorTop: 0.345 * K,
  floorZ: [-0.06 * K, 0.37 * K] as [number, number],
  /** steering axis: top (head) → front hub */
  axisTop: k3([0, 1.02, 0.45]),
  axisBot: k3([0, 0.235, 0.64]),
  grip: k3([0.3, 1.045, 0.42]),
  gripR: 0.021 * K,
  rackTop: 0.845 * K,
  rackZ: -0.98 * K,
  /** half width of the body at the seat / floor */
  halfW: 0.23 * K,
};

/** Rodrigues rotation of point p about the axis through a (direction k, unit) by angle th. */
export const rotAxis = (p: V3, a: V3, k: V3, th: number): V3 => {
  const v = sub(p, a);
  const c = Math.cos(th), s = Math.sin(th);
  const r = add(add(mul(v, c), cross(k, v), s), k, dot(k, v) * (1 - c));
  return add(a, r);
};
export const steerAxisDir = (): V3 => nrm(sub(MOTOR.axisBot, MOTOR.axisTop));

/** Grip centre (motor-local) for the left (+1) / right (−1) hand with the bars turned by `steer` (rad, + = left). */
export const gripAt = (side: 1 | -1, steer = 0): V3 => {
  const g: V3 = [MOTOR.grip[0] * side, MOTOR.grip[1], MOTOR.grip[2]];
  // rotating about an axis pointing DOWN-forward: + angle turns the bars to the right; flip so + = left turn
  return rotAxis(g, MOTOR.axisTop, steerAxisDir(), -steer);
};

/** Wrist target that puts the palm on top of a grip (fingers wrapping it), given the shoulder position. */
export const wristOnGrip = (grip: V3, shoulder: V3, s = 1): V3 => {
  const d = nrm(sub(grip, shoulder));
  return add(add(grip, [0, 0.047 * s, 0]), d, -0.045 * s);
};

// TIPIKAL camera solver — picks a camera for every shot automatically. For a shot kind and its subjects it tries a fan
// of angles, heights and distances and scores each: are the subjects' heads and bodies visible (rays tested against
// the set's colliders, the other people and vehicles), do faces sit BELOW the header + type card (the label-safe band
// of the frame), does the camera see faces, is there something behind them, is it a different angle from the last
// shot. Pure TypeScript (no three).

import type { Box, PCam, SetMeta, ShotKind, V3 } from "./types.ts";

export type Subj = { id: string; p: V3; face: number; s: number; kind: "person" | "prop" | "vehicle"; head: number; size?: V3 };
export type Occ = { c: V3; r: number; id: string };

const D = Math.PI / 180;
const add = (a: V3, b: V3, k = 1): V3 => [a[0] + b[0] * k, a[1] + b[1] * k, a[2] + b[2] * k];
const sub = (a: V3, b: V3): V3 => [a[0] - b[0], a[1] - b[1], a[2] - b[2]];
const len = (a: V3) => Math.hypot(a[0], a[1], a[2]);
const nrm = (a: V3): V3 => { const l = len(a) || 1; return [a[0] / l, a[1] / l, a[2] / l]; };
const dot = (a: V3, b: V3) => a[0] * b[0] + a[1] * b[1] + a[2] * b[2];
const wrapA = (a: number) => Math.atan2(Math.sin(a), Math.cos(a));

/** Head / chest of a person, centre of a prop or vehicle (world). */
export const headOf = (s: Subj): V3 => (s.kind === "person" ? add(s.p, [0, s.head, 0]) : add(s.p, [0, (s.size?.[1] ?? 0.3) * 0.6, 0]));
export const chestOf = (s: Subj): V3 => (s.kind === "person" ? add(s.p, [0, s.head - 0.5 * s.s, 0]) : add(s.p, [0, (s.size?.[1] ?? 0.3) * 0.4, 0]));

const rayBox = (o: V3, d: V3, b: Box) => {
  let t0 = 0, t1 = 1;
  for (let i = 0; i < 3; i++) {
    if (Math.abs(d[i]) < 1e-9) { if (o[i] < b.min[i] || o[i] > b.max[i]) return Infinity; continue; }
    let a = (b.min[i] - o[i]) / d[i], c = (b.max[i] - o[i]) / d[i];
    if (a > c) [a, c] = [c, a];
    t0 = Math.max(t0, a); t1 = Math.min(t1, c);
    if (t0 > t1) return Infinity;
  }
  return t0;
};
const raySphere = (o: V3, d: V3, c: V3, r: number) => {
  const m = sub(o, c), b = dot(m, d), cc = dot(m, m) - r * r, a = dot(d, d);
  const disc = b * b - a * cc;
  if (disc < 0) return Infinity;
  const t = (-b - Math.sqrt(disc)) / a;
  return t >= 0 && t <= 1 ? t : cc < 0 ? 0 : Infinity;
};
const inside = (p: V3, b: Box, pad = 0) => p[0] > b.min[0] - pad && p[0] < b.max[0] + pad && p[1] > b.min[1] - pad && p[1] < b.max[1] + pad && p[2] > b.min[2] - pad && p[2] < b.max[2] + pad;
/** Fraction of the way from a to b that is clear (1 = nothing in between). */
export const clearRay = (a: V3, b: V3, W: SetMeta, occ: Occ[], skip: Set<string>) => {
  const d = sub(b, a);
  let t = 1;
  for (const c of W.colliders) t = Math.min(t, rayBox(a, d, c));
  for (const o of occ) if (!skip.has(o.id)) t = Math.min(t, raySphere(a, d, o.c, o.r));
  return t;
};

export type Solve = { a: PCam; b: PCam; vis: number; box: number; az: number; note?: string };

/** V = vertical span at the subject (m), headY = where the main head sits in the frame (0 top … 1 bottom). The top
 *  ~28 % of the frame belongs to the header + type card, so heads sit at 0.4 – 0.55. */
type Kind = { fov: number; el: number; offs: number[]; V: (s: Subj) => number; headY: number };
const pV = (s: Subj, person: number, prop = 2.4) => (s.kind === "person" ? person * Math.max(0.75, s.s) : Math.max(0.45, Math.max(...(s.size ?? [0.3, 0.3, 0.3])) * prop));
const KIND: Record<string, Kind> = {

  reaction: { fov: 34, el: 3, offs: [20, -20, 34, -34, 48, -48, 8, -8], V: (s) => pV(s, 1.1, 1.5), headY: 0.56 },
  medium: { fov: 40, el: 8, offs: [34, -34, 50, -50, 20, -20, 64, -64, 0], V: (s) => pV(s, 1.95, 2.0), headY: 0.46 },
  low: { fov: 44, el: -10, offs: [22, -22, 36, -36, 10, -10], V: (s) => pV(s, 2.3, 2.2), headY: 0.46 },
  wide: { fov: 44, el: 13, offs: [0, 24, -24, 42, -42, 66, -66, 95, -95], V: (s) => pV(s, 3.6, 3.2), headY: 0.48 },
  establish: { fov: 46, el: 28, offs: [30, -30, 50, -50, 10, -10, 75, -75], V: (s) => pV(s, 5.5, 5.0), headY: 0.5 },
  top: { fov: 44, el: 60, offs: [0, 35, -35, 90, -90, 180], V: (s) => pV(s, 3.4, 3.0), headY: 0.5 },
  insert: { fov: 38, el: 30, offs: [0, 30, -30, 55, -55, 80, -80, 180], V: (s) => pV(s, 1.0, 2.4), headY: 0.55 },
  follow: { fov: 42, el: 11, offs: [155, -155, 135, -135, 175, 120, -120, 100, -100], V: (s) => pV(s, 2.4, 2.0), headY: 0.46 },
};

export type SolveOpts = {
  aspect: number;
  prevAz?: number;
  prevId?: string;
  side?: "left" | "right";
  move?: string;
  over?: Subj;
  /** keep heads above this frame height (the outro card sits below) */
  maxHeadY?: number;

/** Camera for one shot. `subs[0]` is the main subject; `occ` = everything else that can block the view. */
export const solveShot = (kind: ShotKind, subs: Subj[], W: SetMeta, occ: Occ[], o: SolveOpts): Solve => {
  if (kind === "ots" && o.over) return solveOts(subs[0], o.over, W, occ, o);
  const K = KIND[kind] ?? KIND.medium;
  const main = subs[0];
  const skip = new Set(subs.map((s) => s.id));
  const hfov = (f: number) => Math.atan(Math.tan((f * D) / 2) * o.aspect);
  const V0 = K.V(main);
  // the aim point: the main head at headY of the frame (groups: their centre, heads still below the labels)
  const aimFor = (s: Subj): V3 => {
    if (s.kind !== "person") return add(s.p, [0, (s.size?.[1] ?? 0.3) * 0.5, 0]);
    const h = headOf(s);
    return [h[0], h[1] - (0.5 - K.headY) * V0, h[2]];
  };
  const group = subs.length > 1 && kind !== "close" && kind !== "reaction" && kind !== "insert";
  const groupAim: V3 = (() => { const c = subs.map(aimFor).reduce((x, y) => add(x, y), [0, 0, 0] as V3); const k = 1 / subs.length; const top = Math.max(...subs.map((s) => headOf(s)[1])); return [c[0] * k, Math.min(c[1] * k, top - (0.5 - K.headY) * V0), c[2] * k]; })();
  // a group is framed on its centre; when it cannot all fit, the main subject alone (the others may leave the frame)
  const aims: V3[] = group ? [groupAim, aimFor(main)] : [aimFor(main)];
  const baseFace = main.face;
  // the subjects can hide each other too (a rider in front of the kids behind her)
  const occAll: Occ[] = [...occ, ...subs.filter((q) => q.kind === "person").flatMap((q) => [{ c: headOf(q), r: 0.2 * q.s, id: q.id }, { c: chestOf(q), r: 0.19 * q.s, id: q.id }])];
  const tanV = Math.tan((K.fov * D) / 2), tanH = tanV * o.aspect;
  /** where a point lands in the frame (0..1, y down) */
  const screen = (pos: V3, aimP: V3, P: V3) => {
    const f = nrm(sub(aimP, pos)), r = nrm([f[2], 0, -f[0]]), u: V3 = [r[1] * f[2] - r[2] * f[1], r[2] * f[0] - r[0] * f[2], r[0] * f[1] - r[1] * f[0]];
    const v = sub(P, pos), z = dot(v, f);
    if (z < 0.05) return { x: 9, y: 9 };
    return { x: 0.5 - (0.5 * dot(v, r)) / (z * tanH), y: 0.5 - (0.5 * dot(v, u)) / (z * tanV) };
  };
  let best: (Solve & { score: number }) | null = null;
  const dists = [1, 0.85, 1.2, 0.7, 1.45];
  const els = kind === "close" || kind === "reaction" ? [K.el, K.el + 8, K.el + 16, K.el - 4] : kind === "establish" || kind === "top" ? [K.el, K.el + 10, K.el - 10] : [K.el, K.el + 9, K.el + 18, K.el - 5];
  for (const aim of aims)
  for (const dk of dists)
    for (const el of els)
      for (const off of K.offs) {
        const A = baseFace + off * D;
        let d = V0 / (2 * Math.tan((K.fov * D) / 2));
        if (subs.length > 1 && aim === groupAim) {
          const right: V3 = [Math.cos(A), 0, -Math.sin(A)];
          const xs = subs.map((s) => dot(sub(s.p, aim), right));
          const ext = subs.map((s) => (s.kind === "vehicle" ? Math.max(...(s.size ?? [1, 1, 1])) * 0.5 : 0.35 * s.s));
          const spread = Math.max(...xs.map((x, i) => x + ext[i])) - Math.min(...xs.map((x, i) => x - ext[i])) + 0.3;
          d = Math.max(d, spread / 2 / Math.tan(hfov(K.fov) * 0.92));
          const tops = subs.map((s) => headOf(s)[1] + 0.25 * s.s), bots = subs.map((s) => s.p[1]);
          const vspan = (Math.max(...tops) - Math.min(...bots)) / (1 - (0.5 - K.headY) * 2 - 0.12);
          if (group && kind !== "medium") d = Math.max(d, vspan / (2 * Math.tan((K.fov * D) / 2)));
        }
        d *= dk;
        const dir: V3 = [Math.cos(el * D) * Math.sin(A), Math.sin(el * D), Math.cos(el * D) * Math.cos(A)];
        const pos = add(aim, dir, d);
        if (kind === "low") pos[1] = Math.max(0.3, Math.min(pos[1], main.p[1] + 0.6));
        if (pos[1] < 0.2 || pos[1] < W.bounds.min[1] || pos[1] > W.bounds.max[1]) continue;
        if (pos[0] < W.bounds.min[0] || pos[0] > W.bounds.max[0] || pos[2] < W.bounds.min[2] || pos[2] > W.bounds.max[2]) continue;
        if (W.colliders.some((c) => inside(pos, c, 0.22))) continue;
        if (occ.some((q) => !skip.has(q.id) && len(sub(pos, q.c)) < q.r + 0.55)) continue;
        const minD = kind === "close" || kind === "reaction" ? 0.55 : kind === "insert" || kind === "top" ? 0.45 : kind === "wide" || kind === "establish" ? 1.8 : 1.2;
        if (subs.some((q) => len(sub(pos, headOf(q))) < minD * (q === main ? 0.7 : 1))) continue;
        // visibility of the subjects (main counts most)
        // a face counts as seen only when its centre AND both sides are clear (a thin post can hide half a face while
        // the ray to its centre slips past)
        const headPts = (q: Subj): V3[] => {
          const h = headOf(q), side = nrm([pos[2] - h[2], 0, -(pos[0] - h[0])]), r = 0.13 * q.s;
          return [h, add(h, side, r), add(h, side, -r)];
        };
        const pts = subs.flatMap((s, i) => (s.kind === "person" ? [[headPts(s), i], [[chestOf(s)], i]] : [[[add(s.p, [0, (s.size?.[1] ?? 0.3) * 0.5, 0])], i], [[add(s.p, [0, (s.size?.[1] ?? 0.3) * 0.9, 0])], i]]) as [V3[], number][]);
        let vMain = 0, nMain = 0, vAll = 0, hiddenFaces = 0;
        for (const [qs, i] of pts) {
          const c = qs.every((q) => clearRay(pos, q, W, occAll, new Set([subs[i].id])) >= 0.985) ? 1 : 0;
          if (i === 0) { vMain += c; nMain++; }
          else if (qs.length === 3 && !c) hiddenFaces++;
          vAll += c;
        }
        vMain /= nMain || 1;
        vAll /= pts.length || 1;
        // something big right in front of the lens at the frame edges
        const fwd = nrm(sub(aim, pos)), rgt = nrm([fwd[2], 0, -fwd[0]]);
        let clutter = 0;
        for (const k of [-0.55, 0.55]) {
          const edge = add(aim, rgt, Math.tan(hfov(K.fov)) * d * k);
          if (clearRay(pos, edge, W, occ, skip) < 0.55) clutter += 1;
        }
        // depth behind the subject (not a wall right behind, not empty sky)
        const far = add(aim, fwd, 40);
        const hitT = clearRay(aim, far, W, [], skip);
        const back = hitT < 1 ? Math.min(1, (hitT * 40) / 7) : 0.35;
        // faces toward the camera (all people in the shot; the main one counts double)
        const people = subs.filter((q) => q.kind === "person");
        const faceOf = (q: Subj) => Math.cos(wrapA(Math.atan2(pos[0] - q.p[0], pos[2] - q.p[2]) - q.face));
        const front = people.length ? people.reduce((acc, q, i) => acc + faceOf(q) * (i === 0 ? 2 : 1), 0) / (people.length + 1) : 1;
        const wantFront = kind === "insert" || kind === "top" || kind === "follow" ? 0 : kind === "establish" ? 0.6 : 1.3;
        // in frame: heads inside the picture and below the labels (main counts most)
        let frameP = 0;
        subs.forEach((q, i) => {
          const h = screen(pos, aim, kind === "insert" || q.kind !== "person" ? add(q.p, [0, (q.size?.[1] ?? 0.3) * 0.5, 0]) : headOf(q));
          const mx = i === 0 ? 0.14 : 0.08;
          const okX = h.x > mx && h.x < 1 - mx, okY = h.y > (q.kind === "person" ? 0.31 : 0.2) && h.y < (o.maxHeadY ?? 0.86);
          if (!okX || !okY) frameP += i === 0 ? 3 : 1.2 / Math.max(1, subs.length - 1);
          if (o.maxHeadY && h.y > o.maxHeadY) frameP += 2.5;
        });
        let score = 4 * vMain + 1.6 * vAll - 1.4 * hiddenFaces - frameP + wantFront * front - 0.7 * clutter + 0.6 * back - 0.9 * Math.max(0, 1 - dk) - 0.25 * Math.max(0, dk - 1) - 0.01 * Math.abs(el - K.el);
        if (o.prevAz !== undefined && o.prevId === main.id && Math.abs(wrapA(A - o.prevAz)) < 18 * D) score -= 0.9;
        if (o.side) { const want = o.side === "left" ? 1 : -1; if (Math.sign(off) === want) score += 0.45; }
        if (!best || score > best.score) {
          const a: PCam = { pos, target: aim, fov: K.fov };
          best = { a, b: a, vis: vMain, box: Math.max(4, Math.min(30, d * 1.3 + 3)), az: A, score };
        }
      }
  if (!best) {
    const A = baseFace, d = (V0 / (2 * Math.tan((K.fov * D) / 2))) * 0.7;
    const aim = aims[aims.length - 1];
    const pos = add(aim, [Math.sin(A) * d, 0.3, Math.cos(A) * d]);
    return { a: { pos, target: aim, fov: K.fov }, b: { pos, target: aim, fov: K.fov }, vis: 0, box: 8, az: A, note: "no clear camera position found" };
  }
  best.b = moved(best.a, o.move ?? (kind === "establish" || kind === "wide" ? "push" : "push"), W);
  return best;
};

/** Over-the-shoulder: camera behind `over`'s shoulder looking at `subj`. */
const solveOts = (subj: Subj, over: Subj, W: SetMeta, occ: Occ[], o: SolveOpts): Solve => {
  const target = add(headOf(subj), [0, -0.1 * subj.s, 0]);
  const from = headOf(over);
  const back = nrm([from[0] - target[0], 0, from[2] - target[2]]);
  const side: V3 = [back[2], 0, -back[0]];
  let best: (Solve & { score: number }) | null = null;
  for (const sd of [1, -1])
    for (const k of [0.75, 1.05, 0.55]) {
      const pos = add(add(add(from, back, k), side, 0.42 * sd), [0, 0.12, 0]);
      if (W.colliders.some((c) => inside(pos, c, 0.15))) continue;
      const v = clearRay(pos, target, W, occ, new Set([subj.id]));
      const score = (v >= 0.985 ? 4 : 0) + (o.side ? (o.side === "left" ? -sd : sd) * 0.3 : 0) - Math.abs(k - 0.75);
      // aim a little above the head so it sits below the labels
      const aim: V3 = add(target, [0, 0.08, 0]);
      if (!best || score > best.score) best = { a: { pos, target: aim, fov: 40 }, b: { pos, target: aim, fov: 40 }, vis: v >= 0.985 ? 1 : 0, box: 6, az: Math.atan2(pos[0] - target[0], pos[2] - target[2]), score };
    }
  if (!best) return solveShot("medium", [subj], W, occ, { ...o, over: undefined });
  best.b = moved(best.a, o.move ?? "push", W);
  return best;
};

/** The end key of a camera move (small: the genre cuts often, the camera only breathes). */
export const moved = (a: PCam, move: string, W: SetMeta): PCam => {
  const v = sub(a.target, a.pos), d = len(v);
  let b: PCam = a;
  if (move === "push") b = { ...a, pos: add(a.pos, v, 0.07), fov: a.fov - 0.8 };
  else if (move === "pull") b = { ...a, pos: add(a.pos, v, -0.1), fov: a.fov + 1 };
  else if (move === "orbit") {
    const r = sub(a.pos, a.target), c = Math.cos(8 * D), s = Math.sin(8 * D);
    b = { ...a, pos: add(a.target, [r[0] * c + r[2] * s, r[1], -r[0] * s + r[2] * c]) };
  } else if (move === "rise") b = { ...a, pos: add(a.pos, [0, d * 0.15, 0]), target: add(a.target, [0, d * 0.04, 0]) };
  else if (move === "track") { const rgt = nrm([v[2], 0, -v[0]]); b = { ...a, pos: add(a.pos, rgt, d * 0.06), target: add(a.target, rgt, d * 0.06) }; }
  if (W.colliders.some((c) => inside(b.pos, c, 0.1)) || b.pos[0] < W.bounds.min[0] || b.pos[0] > W.bounds.max[0] || b.pos[2] < W.bounds.min[2] || b.pos[2] > W.bounds.max[2]) return a;
  return b;
};

export const lookProblems = (look: Record<string, unknown>): string[] => {
  const p: string[] = [];
  const known = ["name", "build", "skin", "hair", "hat", "top", "bottom", "feet", "face", "bag", "towel"];
  for (const k of Object.keys(look)) if (!known.includes(k)) p.push(`unknown look field "${k}" (have: ${known.filter((x) => x !== "name").join(", ")})`);
  const hex = (v: unknown, where: string) => { if (typeof v !== "string" || !HEX.test(v)) p.push(`${where} must be a colour like "#2f6fd0" (got ${JSON.stringify(v)})`); };
  hex(look.skin, "skin");
  const part = (k: "top" | "bottom" | "feet", extra: Record<string, "hex" | "bool"> = {}) => {
    const v = look[k] as Record<string, unknown> | undefined;
    if (!isObj(v)) { p.push(`${k} missing (needs { "kind", "color" })`); return; }
    if (!SETS[k].includes(String(v.kind))) p.push(`${k}.kind "${v.kind}" unknown (have: ${SETS[k].join(", ")})`);
    hex(v.color, `${k}.color`);
    for (const [f, val] of Object.entries(v)) {
      if (f === "kind" || f === "color") continue;
      if (!extra[f]) p.push(`unknown ${k} field "${f}" (have: kind, color${Object.keys(extra).length ? ", " + Object.keys(extra).join(", ") : ""})`);
      else if (extra[f] === "bool") { if (typeof val !== "boolean") p.push(`${k}.${f} must be true or false`); }
      else hex(val, `${k}.${f}`);
    }
  };
  part("top", { accent: "hex", collar: "hex", tie: "hex", vest: "hex", apron: "hex", lanyard: "hex", band: "hex", short: "bool" });
  part("bottom");
  part("feet");
  const h = look.hair as Record<string, unknown> | undefined;
  if (h !== undefined) {
    if (!isObj(h)) p.push(`hair must be { "color": "#…", "style": … }`);
    else { if (!SETS.hair.includes(String(h.style))) p.push(`hair.style "${h.style}" unknown (have: ${SETS.hair.join(", ")})`); hex(h.color, "hair.color"); }
  }
  const t = look.hat as Record<string, unknown> | undefined;
  if (t !== undefined) {
    if (!isObj(t)) p.push(`hat must be { "kind": … }`);
    else {
      if (!SETS.hat.includes(String(t.kind))) p.push(`hat.kind "${t.kind}" unknown (have: ${SETS.hat.join(", ")})`);
      if (t.color !== undefined) hex(t.color, "hat.color");
      if (t.color2 !== undefined) hex(t.color2, "hat.color2");
    }
  }
  const f = look.face as Record<string, unknown> | undefined;
  if (f !== undefined) {
    if (!isObj(f)) p.push(`face must be an object`);
    else for (const [k, v] of Object.entries(f)) {
      if (k === "brows") { if (!["thin", "thick", "bushy"].includes(String(v))) p.push(`face.brows "${v}" unknown (have: thin, thick, bushy)`); }
      else if (k === "pupil") hex(v, "face.pupil");
      else if (["mustache", "beard", "glasses", "lashes", "blush", "old"].includes(k)) { if (typeof v !== "boolean") p.push(`face.${k} must be true or false`); }
      else p.push(`unknown face field "${k}" (have: brows, mustache, beard, glasses, lashes, blush, old)`);
    }
  }
  const b = look.build as Record<string, unknown> | undefined;
  if (b !== undefined) {
    if (!isObj(b)) p.push(`build must be an object`);
    else for (const [k, v] of Object.entries(b)) {
      const r = RANGE[k];
      if (!r) p.push(`unknown build field "${k}" (have: s, wide, head)`);

    }
  }
  if (look.bag !== undefined) hex(look.bag, "bag");
  if (look.towel !== undefined) hex(look.towel, "towel");
  return p;
};

/** Render "*word*" spans in the theme's accent colour. */
const rich = (s: string, accent: string) => s.split(/(\*[^*]+\*)/g).map((p, i) => (p.startsWith("*") ? <span key={i} style={{ color: accent }}>{p.slice(1, -1)}</span> : <span key={i}>{p}</span>));

export type Layout = { w: number; h: number; wide: boolean };
const headerTop = (L: Layout) => (L.wide ? 34 : 168);
const cardTop = (L: Layout) => (L.wide ? 150 : 318);

/** Header card (drops in over the first ~0.4 s, then stays). */
export const Header: React.FC<{ t: number; small: string; big: string; theme: LabelTheme; L: Layout }> = ({ t, small, big, theme, L }) => {
  const T = THEMES[theme];
  const k = backOut(t / 0.42, 1.4);
  const text = `${small ? small + " " : ""}${big}`;
  const maxW = L.wide ? 1300 : 840;
  const size = Math.round(Math.min(70, maxW / Math.max(6, text.length * 0.47)));
  const sign = theme === "jalan" || theme === "kantor" || theme === "warung";
  return (
    <div style={{ position: "absolute", left: 0, right: 0, top: headerTop(L), display: "flex", justifyContent: "center", transform: `translateY(${-300 * (1 - k)}px)` }}>
      <div style={{ position: "relative", background: T.hBg, borderRadius: theme === "kelas" ? 10 : 20, padding: theme === "kelas" ? 12 : 8, boxShadow: `0 7px 0 ${INK}`, border: `${theme === "kelas" ? 12 : 5}px solid ${T.hBorder}` }}>
        {theme === "warung" ? (
          <div style={{ position: "absolute", left: -5, right: -5, top: -26, height: 26, display: "flex", overflow: "hidden", borderRadius: "10px 10px 0 0", border: `5px solid ${INK}`, borderBottom: "none" }}>
            {Array.from({ length: 14 }, (_, i) => <div key={i} style={{ flex: 1, background: i % 2 ? "#ffffff" : "#c8352e" }} />)}
          </div>
        ) : null}
        {theme === "kantor" ? <div style={{ position: "absolute", left: "50%", top: -34, width: 70, height: 30, marginLeft: -35, background: "#9cc3e6", border: `5px solid ${INK}`, borderRadius: "8px 8px 0 0" }} /> : null}
        <div style={{ border: T.hInner && sign ? `5px solid ${T.hInner}` : theme === "kondangan" ? `4px double ${T.hInner}` : theme === "rumah" ? `4px dashed ${T.hInner}` : "none", borderRadius: 13, padding: theme === "kondangan" ? "10px 30px 8px 22px" : "8px 30px 6px 20px", display: "flex", alignItems: "center", gap: 18 }}>
          <Picto kind={T.picto} size={Math.round(size * 1.2)} color={T.hFg} />
          <div style={{ color: T.hFg, fontFamily: theme === "kelas" || theme === "kondangan" || theme === "rumah" ?         </div>

/** Type card: enters at s = 0 (slide + settle ~0.3 s), leaves over the last 0.18 s before `dur`. */
export const TypeCard: React.FC<{ s: number; dur: number; n: number; text: string; theme: LabelTheme; L: Layout }> = ({ s, dur, n, text, theme, L }) => {
  if (s < 0 || s > dur) return null;
  const T = THEMES[theme];
  const kin = backOut(s / 0.3, 1.7), kout = easeIn((s - (dur - 0.18)) / 0.18);
  const x = -1250 * (1 - kin) + 1250 * kout;
  const rot = -4 * (1 - clamp01(s / 0.3)) + 2 * kout;
  const pop = backOut((s - 0.12) / 0.26, 2.2);
  const W = L.wide ? 1180 : 900;
  const fs = text.replace(/\*/g, "").length > 34 ? 56 : 62;
  const lined = theme === "kelas";
  return (
    <div style={{ position: "absolute", left: 0, right: 0, top: cardTop(L), display: "flex", justifyContent: "center", transform: `translateX(${x}px) rotate(${rot}deg)` }}>
      <div style={{ position: "relative", width: W, paddingLeft: 92 }}>
        <div style={{
          background: T.cardBg, overflow: "hidden",
          border: `6px solid ${INK}`, borderRadius: theme === "kantor" ? 14 : 28, boxShadow: `10px 10px 0 ${INK}`,
          padding: "18px 30px 14px 100px", fontFamily: F_ROUND, fontWeight: 800, fontSize: fs, lineHeight: 1.0, color: T.cardFg, letterSpacing: -0.5, position: "relative",
          outline: theme === "kondangan" ? `4px solid ${T.numBg}` : undefined, outlineOffset: theme === "kondangan" ? -14 : undefined,
        }}>
          {lined ? Array.from({ length: 6 }, (_, i) => <div key={i} style={{ position: "absolute", left: 0, right: 0, top: 14 + i * 43, height: 3, background: "#b9d6ee" }} />) : null}
          {lined ? <div style={{ position: "absolute", left: 82, top: 0, bottom: 0, width: 4, background: "#e2563d" }} /> : null}
          {lined ? <span style={{ position: "relative" }}>{rich(text, T.accent)}</span> : null}
          {lined ? null : rich(text, T.accent)}
        </div>
        {T.numShape === "tab" ? (
          <div style={{ position: "absolute", left: 20, top: "50%", width: 130, height: 150, marginTop: -75, background: T.numBg, border: `6px solid ${INK}`, borderRadius: 18, boxSizing: "border-box", display: "flex", alignItems: "center", justifyContent: "center", transform: `scale(${pop})`, fontFamily: F_SIGN, fontWeight: 800, fontSize: 90, color: T.numFg, paddingTop: 6 }}>{n}</div>
        ) : T.numShape === "grade" ? (
          <div style={{ position: "absolute", left: 4, top: "50%", width: 156, height: 150, marginTop: -75, display: "flex", alignItems: "center", justifyContent: "center", transform: `scale(${pop}) rotate(-8deg)` }}>
            <svg width="156" height="150" viewBox="0 0 156 150" style={{ position: "absolute", left: 0, top: 0 }}><ellipse cx="78" cy="75" rx="66" ry="60" fill="#ffffff" stroke={INK} strokeWidth="6" /><ellipse cx="78" cy="75" rx="56" ry="50" fill="none" stroke={T.numRing} strokeWidth="9" /></svg>
            <span style={{ position: "relative", fontFamily: F_ROUND, fontWeight: 800, fontSize: 92, color: T.numFg, lineHeight: 1, paddingTop: 8 }}>{n}</span>
          </div>
        ) : (
          <div style={{ position: "absolute", left: 0, top: "50%", width: 160, height: 160, marginTop: -80, borderRadius: "50%", background: T.numBg, border: `19px solid ${T.numRing}`, outline: `6px solid ${INK}`, boxSizing: "border-box", display: "flex", alignItems: "center", justifyContent: "center", transform: `scale(${pop})`, fontFamily: F_SIGN, fontWeight: 800, fontSize: 86, color: T.numFg, lineHeight: 1, paddingTop: 6 }}>{n}</div>
        )}
      </div>
    </div>
  );
};

/** Closing card ("Kamu tipe yang mana?") — slams in, holds. */
export const OutroCard: React.FC<{ s: number; text: string; theme: LabelTheme; L: Layout }> = ({ s, text, theme, L }) => {
  if (s < 0) return null;
  const T = THEMES[theme];

    <div style={{ position: "absolute", left: 0, right: 0, top: L.wide ? 790 : 1390, display: "flex", justifyContent: "center", transform: `scale(${k}) rotate(${-3 * (1 - clamp01(s / 0.34))}deg)` }}>
      <div style={{ background: T.hBg, color: T.hFg, border: `7px solid ${INK}`, borderRadius: 26, boxShadow: `10px 10px 0 ${INK}`, padding: "20px 44px 14px", fontFamily: F_ROUND, fontWeight: 800, fontSize: text.length > 24 ? 60 : 72, lineHeight: 1.0, textAlign: "center", whiteSpace: text.length > 30 ? "normal" : "nowrap", maxWidth: L.wide ? 1300 : 960 }}>

      </div>

/** "?" / "!" sticker that pops at s = 0 and gently rocks; (x, y) = the point it hangs from. */
/** Round sticker; a text wider than the badge (e.g. "100.000") makes it a pill, and the type shrinks only if needed. */
export const Sticker: React.FC<{ s: number; x: number; y: number; text: string; fill?: string; size?: number; fit?: PopFit }> = ({ s, x, y, text, fill = "#ffffff", size = 132, fit }) => {
  if (s < 0) return null;
  const k = backOut(s / 0.24, 2.4);
  const rock = Math.sin(s * 2 * Math.PI * 0.9) * (fit?.pill ? 3 : 5) * clamp01(s / 0.3);
  const F = fit ?? popFit("sticker", text);
  const fs = F.fs, w = F.pill ? F.w : size;
  return (
    <div style={{ position: "absolute", left: x - w / 2, top: y - size, width: w, height: size, transform: `scale(${k}) rotate(${rock}deg)`, transformOrigin: "50% 100%" }}>
      <div style={{ width: "100%", height: "100%", borderRadius: size / 2, whiteSpace: "nowrap", background: fill, border: `7px solid ${INK}`, boxSizing: "border-box", boxShadow: `6px 6px 0 ${INK}`, display: "flex", alignItems: "center", justifyContent: "center", fontFamily: F_ROUND, fontWeight: 800, fontSize: fs, color: text === "♥" ? "#e94b6a" : INK, lineHeight: 1, paddingTop: size * 0.08 }}>
        {text}
      </div>
    </div>
  );
};
/** Speech bubble with a tail toward (x, y). */

/** Comic lettering (CIIIT!, TIIIN!) — slams in, holds, fades out over the last 0.15 s of `dur`. */
export const Sfx: React.FC<{ s: number; dur: number; x: number; y: number; text: string; color?: string; rot?: number; fit?: PopFit }> = ({ s, dur, x, y, text, color = "#ffd23a", rot = -8, fit }) => {
  if (s < 0 || s > dur) return null;
  const size = (fit ?? popFit("sfx", text)).fs;
  const k = backOut(s / 0.2, 2.2);
  const o = 1 - clamp01((s - (dur - 0.15)) / 0.15);
  return (
    <div style={{ position: "absolute", left: x, top: y, transform: `translate(-50%, -50%) rotate(${rot}deg) scale(${k})`, opacity: o }}>
      <span style={{ fontFamily: F_ROUND, fontWeight: 800, fontSize: size, color, WebkitTextStroke: `${Math.round(size * 0.1)}px ${INK}`, paintOrder: "stroke fill", letterSpacing: 2, whiteSpace: "nowrap", textShadow: `7px 7px 0 ${INK}` }}>{text}</span>
    </div>
  );
};

// TIPIKAL prop library — every object is plain data: a list of simple parts (box, cyl, cone, sphere, torus, lathe)
// with a material and a flat colour. Prop.tsx builds the meshes (cel-shaded with an ink outline, like the people);
// the director reads sizes (cameras) and the hold spec (hands). Authors add objects in stories/<id>/props.json with
// the same format. Units: metres. A part's `pos` is its BOTTOM centre (sphere / torus: centre). No brands, no logos.
//
// hold: how a hand carries it
//   hand "R" | "L" | "both";  mode "hand" = follows the hand (phone, book, fan, broom: prop +y along the fingers,
//   +z out of the palm), "upright" = stays upright (cup, plate, box, jug); at = the prop point that sits on the palm
//   (for "both": the point between the two palms).

export type V3 = [number, number, number];
export type Mat = "toon" | "flat" | "shiny" | "screen";
export type Part =
  | { shape: "box"; size: V3; pos?: V3; rot?: V3; color: string; mat?: Mat; round?: number }
  | { shape: "cyl"; r: number; r2?: number; h: number; pos?: V3; rot?: V3; color: string; mat?: Mat; seg?: number }
  | { shape: "cone"; r: number; h: number; pos?: V3; rot?: V3; color: string; mat?: Mat; seg?: number }
  | { shape: "sphere"; r: number; pos?: V3; scale?: V3; color: string; mat?: Mat }
  | { shape: "torus"; r: number; tube: number; pos?: V3; rot?: V3; color: string; mat?: Mat; arc?: number }
  | { shape: "lathe"; profile: [number, number][]; pos?: V3; rot?: V3; color: string; mat?: Mat };

export type Hold = { hand: "R" | "L" | "both"; mode: "hand" | "upright"; at: V3; /** "hand" mode: turn the prop (degrees, x y z) from how it rests to how the hand holds it */ rot?: V3 };
export type PropDef = {
  about: string;
  parts: Part[];
  /** bounding size (w, h, d) — cameras frame it, occluders use it */
  size: V3;
  hold?: Hold;
  /** no ink outline (thin / flat things) */
  noInk?: boolean;
};

/** Plain-language problems in a prop definition (props.json). */
export const propProblems = (id: string, d: PropDef): string[] => {
  const p: string[] = [];
  if (!d || typeof d !== "object") return [`prop "${id}": must be an object { about, size, parts }`];
  if (!Array.isArray(d.parts) || !d.parts.length) p.push(`prop "${id}": "parts" must be a list of shapes`);
  if (!Array.isArray(d.size) || d.size.length !== 3) p.push(`prop "${id}": "size" must be [w, h, d] in metres`);
  const shapes = ["box", "cyl", "cone", "sphere", "torus", "lathe"];
  (d.parts ?? []).forEach((pt, i) => {
    if (!shapes.includes((pt as { shape: string }).shape)) p.push(`prop "${id}" part ${i + 1}: shape must be one of ${shapes.join(", ")}`);
    if (typeof (pt as { color?: string }).color !== "string" || !/^#[0-9a-fA-F]{6}$/.test((pt as { color: string }).color)) p.push(`prop "${id}" part ${i + 1}: "color" must be like "#d63b30"`);
  });
  if (d.hold && (!["R", "L", "both"].includes(d.hold.hand) || !["hand", "upright"].includes(d.hold.mode) || !Array.isArray(d.hold.at))) p.push(`prop "${id}": hold = { "hand": "R"|"L"|"both", "mode": "hand"|"upright", "at": [x, y, z] }`);
  return p;
};

const texCache = new Map<string, CanvasTexture>();
export const canvasTex = (key: string, w: number, h: number, draw: (g: CanvasRenderingContext2D) => void, repeat?: [number, number]) => {
  const hit = texCache.get(key);
  if (hit) return hit;
  const cv = document.createElement("canvas");
  cv.width = w;
  cv.height = h;
  draw(cv.getContext("2d")!);
  const t = new CanvasTexture(cv);
  t.colorSpace = SRGBColorSpace;
  t.anisotropy = 8;
  t.minFilter = LinearMipmapLinearFilter;
  if (repeat) {
    t.wrapS = t.wrapT = RepeatWrapping;
    t.repeat.set(repeat[0], repeat[1]);
  }
  texCache.set(key, t);
  return t;
};

/** Catenary cable between two points (a few straight segments). */
const Cable: React.FC<{ a: V3; b: V3; sag: number }> = ({ a, b, sag }) => {
  const pts: V3[] = [];
  for (let i = 0; i <= 8; i++) {
    const t = i / 8;
    pts.push([a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t - sag * 4 * t * (1 - t), a[2] + (b[2] - a[2]) * t]);
  }
  return <group>{pts.slice(0, -1).map((p, i) => <M key={i} g={GEO.cyl} m={stick(p, pts[i + 1], 0.018)} material={flat("#222326", 0.8, false)} shadow={false} />)}</group>;
};

  const edge = L.roadHalf + 0.2 + L.walk;

  const segs: [number, number][] = [];
  // curbs/sidewalks are split around the cross street
  if (cross && cross.x > mid - len / 2 && cross.x < mid + len / 2) {
    segs.push([mid - len / 2, cross.x - cross.half]);
    segs.push([cross.x + cross.half, mid + len / 2]);
  } else segs.push([mid - len / 2, mid + len / 2]);
  const out: React.ReactNode[] = [];

  }
  for (const [a, b] of segs) {

  return <group>{out}</group>;

/** A wall along x (at z) or along z (at x) from a to b, height h, thickness 0.2, with window openings [from, to, sill, top]. */
export const Wall: React.FC<{ along: "x" | "z"; at: number; a: number; b: number; h: number; c: string; holes?: [number, number, number, number][]; base?: string; shadow?: boolean }> = ({ along, at: w, a, b, h, c, holes = [], base, shadow = false }) => {
  const out: React.ReactNode[] = [];
  const seg = (u0: number, u1: number, y0: number, y1: number, k: string) => {
    if (u1 - u0 < 0.01 || y1 - y0 < 0.01) return;
    const p: V3 = along === "x" ? [(u0 + u1) / 2, (y0 + y1) / 2, w] : [w, (y0 + y1) / 2, (u0 + u1) / 2];
    const s: V3 = along === "x" ? [u1 - u0, y1 - y0, 0.2] : [0.2, y1 - y0, u1 - u0];
    out.push(<M key={k} g={GEO.box} m={at(p, s)} material={flat(c, 0.92, false)} shadow={shadow} />);
  };
  const hs = [...holes].sort((x, y) => x[0] - y[0]);
  let u = a;
  hs.forEach(([h0, h1, sill, top], i) => {
    seg(u, h0, 0, h, "w" + i);
    seg(h0, h1, 0, sill, "s" + i);
    seg(h0, h1, top, h, "t" + i);
    u = h1;
  });
  seg(u, b, 0, h, "end");
  if (base) {
    const p: V3 = along === "x" ? [(a + b) / 2, 0.06, w] : [w, 0.06, (a + b) / 2];
    const s: V3 = along === "x" ? [b - a, 0.12, 0.24] : [0.24, 0.12, b - a];
    out.push(<M key="base" g={GEO.box} m={at(p, s)} material={flat(base, 0.8, false)} shadow={false} />);
  }
  return <group>{out}</group>;
};

const Sepeda: React.FC<{ c: string; c2: string; steer: number; spin: number; children?: React.ReactNode }> = ({ c, c2, steer, spin, children }) => {
  const d = VEHICLES.sepeda;
  const fr = toon(c), dark = toon("#2c2d31");
  const P = d.pedals!, th = spin / 0.32;
  const crank = (a: number, sd: 1 | -1): V3 => [sd * P.x, P.c[1] + Math.sin(a) * P.r, P.c[2] + Math.cos(a) * P.r];
  const piv = d.axisTop!;
  const Wheel = ({ z }: { z: number }) => (
    <group position={[0, 0.3, z]} rotation={[spin / 0.3, 0, 0]}>
      <mesh geometry={G.sTire} material={toon("#26272c")} castShadow />

      <M g={G.cyl} m={stick([-0.03, 0, 0], [0.03, 0, 0], 0.025)} material={toon("#9ea3a9")} />

  );
  const T = (a: V3, b: V3, r = 0.022, k = "") => <M key={k} g={G.cyl} m={stick(a, b, r)} material={fr} />;
  return (
    <group>

      ); })}
      <Wheel z={-0.5} />
      <group position={piv} quaternion={steerQ(d, steer)}>
        <group position={[-piv[0], -piv[1], -piv[2]]}>
          {T([0, 0.86, 0.38], [0, 0.3, 0.5], 0.02, "fork")}
          {T([0, 0.86, 0.38], [0, 0.99, 0.34], 0.018, "stem")}

export type VehicleProps = { kind: string; color?: string; color2?: string; plate?: string; steer?: number; spin?: number; lean?: number; lamps?: Lamps; scale?: number; rack?: boolean; children?: React.ReactNode };
/** Any vehicle of the library by kind. */
export const Vehicle: React.FC<VehicleProps> = ({ kind, color, color2, plate = "", steer = 0, spin = 0, lean = 0, lamps = {}, scale = 1, rack = false, children }) => {
  const d = VEHICLES[kind];
  const c = color ?? d.color, c2 = color2 ?? d.color2;
  const sp = spin / scale;
  let inner: React.ReactNode;
  if (d.kind === "matic") inner = <Motor spec={{ body: c, accent: c2, seat: "#25262b", plate, rack }} steer={steer} spin={sp} lamps={lamps} />;
  else if (d.kind === "bebek") inner = <group><Bebek c={c} c2={c2} plate={plate} steer={steer} spin={sp} lamps={lamps} />{rack ? B([0, d.rack![1] - 0.012, d.rack![2]], [0.3, 0.024, 0.28], toon("#3b3d42"), "rk") : null}</group>;
  else if (d.kind === "sepeda") inner = <Sepeda c={c} c2={c2} steer={steer} spin={sp} />;
  else if (d.kind === "mobil") inner = <Mobil c={c} c2={c2} plate={plate} steer={steer} spin={sp} lamps={lamps} />;
  else inner = <Angkot spec={{ body: c, stripe: c2 }} spin={sp} brake={lamps.brake} />;
  // riders sit in the leaning group but outside the scaled model (a small bike does not shrink its rider)
  return (
    <group rotation={[0, 0, d.kind === "mobil" || d.kind === "angkot" ? 0 : -lean]}>
      <group scale={scale}>{inner}</group>
      {children}
    </group>
  );
};
