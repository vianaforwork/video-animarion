// TIPIKAL sound from a Plan — original synth, no samples, no meme clips.
// Music: a bouncy sitcom-sketch groove (plucky bass, offbeat chord stabs, toy-piano hook, light hats + soft clap),
// 118 bpm, C major I–vi–IV–V. It stops dead under a freeze (record scratch) and comes back on the next type card;
// it lands on a "ta-da" under the outro card.
// SFX follow the plan: header drop, a slam per type card, pops / boings / stings for the stickers, comic sound words
// (brake squeal, two-tone horn, crash, thud, slap, whoosh, splash, chomp), beat cues, a putter bed under moving motors,
// a slide-whistle + thud for falls. Mix: −14 LUFS, true peak ≤ −1.5 dBTP; the music ducks under a voice-over.
import fs from "node:fs";
import path from "node:path";
import { execFileSync, spawnSync } from "node:child_process";
import { segAt } from "../../src/tipikal/track.ts";

export const buildSound = (plan, { voice, outDir, mixFile, music = true }) => {
  const DUR = plan.duration, SR = 48000, N = Math.ceil(DUR * SR);
  const mk = () => [new Float32Array(N), new Float32Array(N)];
  let seed = 7777;
  const rnd = () => (seed = (seed * 16807) % 2147483647) / 2147483647;
  const add = (buf, t, pan, len, fn) => {
    const i0 = Math.max(0, Math.round(t * SR)), n = Math.min(N - i0, Math.round(len * SR));
    const gl = Math.cos(((pan + 1) * Math.PI) / 4), gr = Math.sin(((pan + 1) * Math.PI) / 4);
    for (let i = 0; i < n; i++) { const v = fn(i / SR, i); buf[0][i0 + i] += v * gl; buf[1][i0 + i] += v * gr; }
  };
  const hz = (m) => 440 * Math.pow(2, (m - 69) / 12);
  const env = (t, a, d) => Math.min(1, t / a) * Math.exp(-t * d);

  // freezes and the outro silence the groove
  const freezes = plan.shots.filter((s) => s.freeze !== undefined && s.type !== -2).map((s) => [s.freeze, s.t1]);
  const outroT = plan.outro ? plan.outro.t0 : DUR + 1;
  const gate = (t) => { for (const [a, b] of freezes) if (t >= a - 0.02 && t < b) return 0; return t >= outroT ? 0 : 1; };

  // ---------------------------------------------------------------- music
  const mus = mk();
  if (music) {
    const beat = 60 / 118, bar = beat * 4;
    const CH = [[48, 52, 55], [45, 48, 52], [41, 45, 48], [43, 47, 50]];
    const HOOK = [[0, 76], [0.75, 79], [1.5, 81], [2, 79], [2.75, 76], [3.5, 74]];
    const start = plan.intro ? 0.15 : 0;
    const bars = Math.ceil(DUR / bar) + 1;
    for (let b = 0; b < bars; b++) {
      const c = CH[b % 4], t0 = start + b * bar;
      // plucky bass: root on 1, octave bounce on 2&, fifth on 3, approach on 4&
      for (const [bt, m] of [[0, c[0] - 12], [1.5, c[0]], [2, c[0] - 5], [3.5, c[0] - 10]]) {
        const tt = t0 + bt * beat; if (!gate(tt)) continue;
        const f = hz(m);
        add(mus, tt, -0.05, 0.4, (t) => { const ph = (f * t) % 1; return (ph < 0.5 ? 1 : -1) * 0.04 * env(t, 0.003, 11) + Math.sin(2 * Math.PI * f * t) * 0.07 * env(t, 0.003, 7); });
      }
      // offbeat chord stabs (ukulele-ish triangle plucks)
      for (let e = 1; e < 8; e += 2) {
        const tt = t0 + (e * beat) / 2; if (!gate(tt)) continue;
        for (const [k, m] of c.entries()) { const f = hz(m + 12); add(mus, tt + k * 0.006, 0.25, 0.22, (t) => { const ph = (f * t) % 1; return (4 * Math.abs(ph - 0.5) - 1) * 0.02 * env(t, 0.002, 18); }); }
      }
      // toy-piano hook every other bar, a lighter answer in between
      const hook = b % 2 === 0 ? HOOK : HOOK.filter((_, i) => i % 2 === 0).map(([bt, m]) => [bt + 0.5, m - 3]);
      for (const [bt, m] of hook) {
        const tt = t0 + bt * beat; if (!gate(tt)) continue;
        const mm = c.some((x) => (m - x) % 12 === 0) ? m : m - 1;
        const f = hz(mm);
        add(mus, tt, -0.3 + rnd() * 0.2, 0.6, (t) => (Math.sin(2 * Math.PI * f * t) * Math.exp(-t * 7) + 0.3 * Math.sin(2 * Math.PI * f * 4.02 * t) * Math.exp(-t * 26)) * 0.045 * Math.min(1, t / 0.002));
      }
      // hats on 8ths, soft clap on 2 and 4
      for (let e = 0; e < 8; e++) { const tt = t0 + (e * beat) / 2; if (!gate(tt)) continue; let hp = 0; add(mus, tt, 0.35, 0.05, (t) => { const n = rnd() * 2 - 1; const o = n - hp; hp = n; return o * (e % 2 ? 0.012 : 0.007) * Math.exp(-t * 90); }); }
      for (const bt of [1, 3]) { const tt = t0 + bt * beat; if (!gate(tt)) continue; let lp = 0; add(mus, tt, -0.1, 0.16, (t) => { const n = rnd() * 2 - 1; lp += (n - lp) * 0.4; return lp * 0.05 * Math.exp(-t * 32); }); }
    }
    // ta-da under the outro card
    if (plan.outro) for (const [k, m] of [[0, 67], [0.14, 72]]) add(mus, plan.outro.t0 + 0.12 + k, 0, 1.6, (t) => { const f = hz(m); const ph = (f * t) % 1; return ((2 * ph - 1) * 0.5 + Math.sin(2 * Math.PI * f * t)) * 0.045 * Math.min(1, t / 0.01) * Math.exp(-t * (k ? 1.6 : 5)); });
    if (plan.outro) for (const m of [48, 55, 60, 64]) add(mus, plan.outro.t0 + 0.26, 0, 1.8, (t) => Math.sin(2 * Math.PI * hz(m) * t) * 0.022 * Math.min(1, t / 0.02) * Math.exp(-t * 1.3));
  }

  // ---------------------------------------------------------------- SFX
  const sfx = mk();
  const events = [];
  const cue = (t, kind) => { if (t < 0 || t > DUR) return; events.push(`${t.toFixed(2)} ${kind}`); SFX[kind]?.(t); };
  const noise = (t, len, gain, lpK, hpK, envf, pan = 0) => { let lp = 0, hp = 0; add(sfx, t, pan, len, (x) => { const n = rnd() * 2 - 1; lp += (n - lp) * lpK; hp += (lp - hp) * hpK; return (lp - hp) * gain * envf(x); }); };
  const tone = (t, len, f0, f1, gain, decay, pan = 0, shape = "sine") => { let ph = 0; add(sfx, t, pan, len, (x) => { const f = f0 + (f1 - f0) * Math.min(1, x / len); ph += f / SR; const p = ph % 1; const w = shape === "square" ? (p < 0.5 ? 1 : -1) : shape === "tri" ? 4 * Math.abs(p - 0.5) - 1 : Math.sin(2 * Math.PI * p); return w * gain * Math.min(1, x / 0.004) * Math.exp(-x * decay); }); };
  const SFX = {
    whoosh: (t) => noise(t - 0.12, 0.36, 0.16, 0.12, 0.02, (x) => Math.sin(Math.PI * Math.min(1, x / 0.36)) ** 2, 0.2),
    slam: (t) => { SFX.whoosh(t - 0.05); tone(t + 0.06, 0.3, 120, 60, 0.22, 16); noise(t + 0.06, 0.08, 0.12, 0.5, 0.05, (x) => Math.exp(-x * 60)); },
    drop: (t) => { SFX.whoosh(t); tone(t + 0.28, 0.4, 90, 45, 0.2, 9); },
    pop: (t) => tone(t, 0.12, 900, 420, 0.12, 30, 0.1),
    boing: (t) => { let ph = 0; add(sfx, t, 0.15, 0.55, (x) => { const f = 220 + 140 * Math.sin(2 * Math.PI * 9 * x) * Math.exp(-x * 4) + 90 * Math.exp(-x * 10); ph += f / SR; return Math.sin(2 * Math.PI * ph) * 0.11 * Math.min(1, x / 0.003) * Math.exp(-x * 4.5); }); },
    sting: (t) => { tone(t, 0.18, hz(79), hz(79), 0.07, 12, 0.1, "tri"); tone(t + 0.09, 0.32, hz(84), hz(84), 0.08, 8, 0.1, "tri"); },
    tick: (t) => { for (let k = 0; k < 3; k++) tone(t + k * 0.26, 0.04, 2600, 2400, 0.06, 80, -0.2); },
    ding: (t) => { for (const [k, m] of [[0, 88], [1, 95]]) tone(t + k * 0.05, 0.9, hz(m), hz(m), 0.04, 5, 0.15); },
    snore: (t) => noise(t, 1.2, 0.07, 0.04, 0.01, (x) => (0.5 - 0.5 * Math.cos(2 * Math.PI * x / 1.2)) * (1 + 0.6 * Math.sin(2 * Math.PI * 28 * x)), -0.2),
    stamp: (t) => { tone(t, 0.25, 110, 70, 0.22, 18); noise(t, 0.06, 0.18, 0.6, 0.1, (x) => Math.exp(-x * 70)); },
    brake: (t) => { tone(t, 0.75, 2900, 2300, 0.05, 2.2, 0.1, "tri"); noise(t, 0.75, 0.06, 0.6, 0.3, (x) => Math.min(1, x / 0.05) * Math.exp(-x * 2.5)); },
    horn: (t) => { for (const [k, f] of [[0, 415], [0.22, 415]]) { tone(t + k, 0.18, f, f, 0.05, 1.5, -0.1, "square"); tone(t + k, 0.18, f * 1.26, f * 1.26, 0.035, 1.5, -0.1, "square"); } },
    crash: (t) => { noise(t, 0.6, 0.25, 0.5, 0.02, (x) => Math.exp(-x * 7)); tone(t, 0.4, 85, 40, 0.25, 9); for (const f of [1800, 2650, 3900]) tone(t + 0.02, 0.7, f, f * 0.98, 0.02, 6, 0.3); },
    thud: (t) => tone(t, 0.35, 95, 48, 0.24, 13),
    slap: (t) => noise(t, 0.09, 0.3, 0.8, 0.3, (x) => Math.exp(-x * 55)),
    slide: (t) => tone(t, 0.55, 1500, 300, 0.07, 1.5, 0, "sine"),
    scratch: (t) => { let ph = 0; add(sfx, t, 0, 0.42, (x) => { const sp = Math.sin(2 * Math.PI * 4.5 * x); ph += (900 + 700 * sp) / SR; return ((rnd() * 2 - 1) * 0.5 + Math.sin(2 * Math.PI * ph)) * 0.08 * Math.min(1, x / 0.01) * Math.min(1, (0.42 - x) / 0.05); }); },
    splash: (t) => { noise(t, 0.6, 0.18, 0.4, 0.05, (x) => Math.exp(-x * 5)); for (let k = 0; k < 4; k++) tone(t + 0.08 + k * 0.07 + rnd() * 0.04, 0.1, 1200 + rnd() * 800, 600, 0.04, 30, rnd() - 0.5); },
    chomp: (t) => { for (const k of [0, 0.16]) noise(t + k, 0.08, 0.2, 0.7, 0.2, (x) => Math.exp(-x * 45)); },
    cheer: (t) => { [72, 76, 79, 84].forEach((m, k) => tone(t + k * 0.06, 0.4, hz(m), hz(m), 0.045, 6, (k - 1.5) * 0.3, "tri")); },
    engine: (t) => tone(t, 0.6, 70, 110, 0.07, 3, 0, "square"),
  };

  // header drop + one slam per type card
  cue(0.05, "drop");
  for (const ty of plan.types) cue(ty.t0 + 0.02, ty.n === 1 && plan.intro ? "slam" : "slam");
  // marks, beat cues, freezes, falls
  for (const S of plan.shots) {
    for (const m of S.marks) if (m.sound) cue(m.t0, m.sound);
    for (const s of S.sounds ?? []) cue(s.t, s.kind);
    if (S.freeze !== undefined && S.type !== -2) cue(S.freeze - 0.05, "scratch");
    for (const a of S.actors) for (const k of a.keys) if (k.do === "fall" && k.t >= S.t0 - 0.01 && k.t < S.t1) { cue(k.t, "slide"); cue(k.t + 0.34, "thud"); }
  }
  // putter bed under moving scooters / underbones (one continuous cue)
  const motorTracks = Object.values(plan.tracks).filter((tr) => plan.shots.some((S) => S.props.some((p) => p.track === tr.id && ["motor", "bebek", "mobil", "angkot"].includes(p.kind))));
  if (motorTracks.length) {
    const BL = 240, speeds = new Float32Array(Math.ceil(N / BL) + 1);
    let used = false;
    for (let b = 0; b < speeds.length; b++) {
      const t = (b * BL) / SR;
      const S = plan.shots.find((s) => t >= s.t0 && t < s.t1);
      if (!S || (S.freeze !== undefined && t >= S.freeze) || t >= outroT) continue;
      for (const tr of motorTracks) if (S.props.some((p) => p.track === tr.id)) speeds[b] = Math.max(speeds[b], segAt(tr, t).speed);
    }
    let ph = 0, lvl = 0;
    for (let i = 0; i < N; i++) {
      const b = i / BL, b0 = Math.floor(b), sp = speeds[b0] + (speeds[Math.min(speeds.length - 1, b0 + 1)] - speeds[b0]) * (b - b0);
      const want = sp < 0.15 ? 0 : Math.min(1, sp / 2.5);
      lvl += (want - lvl) * 0.0008;
      if (lvl < 0.002) continue;
      used = true;
      ph += (32 + sp * 7) / SR;
      const v = (((ph % 1) < 0.18 ? 1 : 0) - 0.18) * 0.05 * lvl;
      sfx[0][i] += v; sfx[1][i] += v * 0.9;
    }
    if (used) events.push("motor bed");
  }

  // ---------------------------------------------------------------- write + mix
  const wav = (file, buf) => {
    const data = Buffer.alloc(N * 4);
    for (let i = 0; i < N; i++) for (let c = 0; c < 2; c++) data.writeInt16LE(Math.round(Math.max(-1, Math.min(1, buf[c][i])) * 32767), i * 4 + c * 2);
    const h = Buffer.alloc(44);
    h.write("RIFF", 0); h.writeUInt32LE(36 + data.length, 4); h.write("WAVEfmt ", 8); h.writeUInt32LE(16, 16); h.writeUInt16LE(1, 20); h.writeUInt16LE(2, 22);
    h.writeUInt32LE(SR, 24); h.writeUInt32LE(SR * 4, 28); h.writeUInt16LE(4, 32); h.writeUInt16LE(16, 34); h.write("data", 36); h.writeUInt32LE(data.length, 40);
    fs.writeFileSync(file, Buffer.concat([h, data]));
  };
  fs.mkdirSync(outDir, { recursive: true });
  fs.mkdirSync(path.dirname(mixFile), { recursive: true });
  const mf = path.join(outDir, "music.wav"), sf = path.join(outDir, "sfx.wav");
  wav(mf, mus);
  wav(sf, sfx);
  const tail = `apad=whole_dur=${DUR},atrim=0:${DUR},loudnorm=I=-14:TP=-1.5:LRA=11,aresample=48000[out]`;
  const filter = voice
    ? ["[0:a]highpass=f=70,acompressor=threshold=-20dB:ratio=2.5:attack=8:release=140:makeup=3,asplit=2[v][vk]", "[1:a]volume=1.0[m]", "[m][vk]sidechaincompress=threshold=0.03:ratio=6:attack=25:release=420[md]", "[2:a]volume=0.9[s]", `[v][md][s]amix=inputs=3:normalize=0:duration=longest,${tail}`].join(";")
    : ["[0:a]volume=1.4[m]", "[1:a]volume=1.0[s]", `[m][s]amix=inputs=2:normalize=0:duration=longest,${tail}`].join(";");
  execFileSync("ffmpeg", ["-y", "-v", "error", ...(voice ? ["-i", voice] : []), "-i", mf, "-i", sf, "-filter_complex", filter, "-map", "[out]", "-ac", "2", mixFile]);
  const m = spawnSync("ffmpeg", ["-hide_banner", "-i", mixFile, "-af", "loudnorm=print_format=summary", "-f", "null", "-"], { encoding: "utf8" }).stderr;
  return { lufs: (m.match(/Input Integrated:\s+(-?[\d.]+)/) ?? [])[1], peak: (m.match(/Input True Peak:\s+(-?[\d.]+)/) ?? [])[1], events: events.length };
};

import React, { useEffect, useMemo } from "react";
import {
  type Material, CanvasTexture, CylinderGeometry, DoubleSide, ExtrudeGeometry, MeshToonMaterial, Quaternion, Shape, SphereGeometry, SRGBColorSpace, Vector3,

type P = [number, number];
/** side profile (z, y) extruded across x, centred, with rounded edges */
const profile = (pts: P[], width: number, bevel = 0.02) => {
  const sh = new Shape();
  pts.forEach(([z, y], i) => (i ? sh.lineTo(z, y) : sh.moveTo(z, y)));
  sh.closePath();
  const depth = Math.max(0.01, width - 2 * bevel);
  const g = new ExtrudeGeometry(sh, { depth, bevelEnabled: true, bevelThickness: bevel, bevelSize: bevel * 0.8, bevelSegments: 2, curveSegments: 4 });
  g.rotateY(-Math.PI / 2);
  g.translate(depth / 2, 0, 0);
  g.computeVertexNormals();
  return g;
};

// raw (unscaled) model dimensions; the whole body is drawn inside a group scaled by K

const plateCache = new Map<string, CanvasTexture>();

  cv.height = 96;

  t.anisotropy = 4;

/** Scooter (children are carried with it: riders, cargo). Lean rolls the whole group about the ground line. */
export const Motor: React.FC<MotorProps> = ({ spec, steer = 0, spin = 0, lean = 0, lamps = {}, children }) => {
  const body = toon(spec.body), trim = toon(spec.trim ?? "#2c2d31"), seat = toon(spec.seat), accent = toon(spec.accent ?? spec.body);

  const plateMat = useMemo(() => new MeshToonMaterial({ map: plateTex(spec.plate ?? "") }), [spec.plate]);
  useEffect(() => () => plateMat.dispose(), [plateMat]);
  const fenderMat = useMemo(() => new MeshToonMaterial({ color: spec.body, side: DoubleSide }), [spec.body]);
  useEffect(() => () => fenderMat.dispose(), [fenderMat]);
  const k = steerAxisDir();

/** A hand as numbers: curl of each finger (index, middle, ring, pinky) [first, second joint] radians, spread, and the
 *  thumb direction as weights of (along the hand, out of the palm, across the palm). Shapes blend by mixing these. */
export type Fingers = { f: [number, number][]; spread: number; thumb: V3 };
const TH = { wrap: [0.75, 0.55, 0.15] as V3, up: [0, 0, 1] as V3, flat: [0.8, 0, 0.45] as V3, open: [0.55, 0, 0.8] as V3 };
export const fingersOf = (shape: HandShape | undefined): Fingers => {
  const all = (a: number, b: number) => [[a, b], [a, b], [a, b], [a, b]] as [number, number][];
  switch (shape) {
    case "grip": return { f: all(1.25, 1.2), spread: 0, thumb: TH.wrap };
    case "fist": return { f: all(1.55, 1.6), spread: 0, thumb: TH.wrap };
    case "thumb": return { f: all(1.55, 1.6), spread: 0, thumb: TH.up };
    case "wave": return { f: all(0.05, 0.05), spread: 0.18, thumb: TH.open };
    case "flat": return { f: all(0.02, 0.02), spread: 0.02, thumb: TH.flat };
    case "point": return { f: [[0.05, 0.05], [1.55, 1.6], [1.55, 1.6], [1.5, 1.6]], spread: 0, thumb: TH.wrap };
    case "peace": return { f: [[0.05, 0.05], [0.05, 0.05], [1.55, 1.6], [1.5, 1.6]], spread: 0.22, thumb: TH.wrap };
    default: return { f: all(0.25, 0.3), spread: 0.06, thumb: TH.open };
  }
};
const mixF = (a: Fingers, b: Fingers, t: number): Fingers => ({
  f: a.f.map((x, i) => [x[0] + (b.f[i][0] - x[0]) * t, x[1] + (b.f[i][1] - x[1]) * t] as [number, number]),
  spread: a.spread + (b.spread - a.spread) * t,
  thumb: [a.thumb[0] + (b.thumb[0] - a.thumb[0]) * t, a.thumb[1] + (b.thumb[1] - a.thumb[1]) * t, a.thumb[2] + (b.thumb[2] - a.thumb[2]) * t],

export const DIM = {

/** Two-bone IK: joint between root and target (lengths a, b), bent toward `pole`. Returns [mid, end]. */
export const ik2 = (root: V3, target: V3, a: number, b: number, pole: V3): [V3, V3] => {
  let d = sub(target, root);
  let L = len(d);
  const max = (a + b) * 0.999, min = Math.abs(a - b) + 1e-3;
  if (L < 1e-6) { d = [0, -1, 0]; L = 1e-6; }
  const dir = mul(d, 1 / L);
  const Lc = Math.min(max, Math.max(min, L));
  const x = (a * a - b * b + Lc * Lc) / (2 * Lc);
  const h = Math.sqrt(Math.max(0, a * a - x * x));
  let bend = sub(pole, mul(dir, dot(pole, dir)));
  if (len(bend) < 1e-5) bend = cross(dir, [1, 0, 0]);
  bend = nrm(bend);
  const mid = add(add(root, mul(dir, x)), mul(bend, h));
  return [mid, add(root, mul(dir, Lc))];
};

export type Joints = {
  pelvis: V3; chest: V3; neck: V3; head: V3; up: V3; right: V3; fwd: V3; headUp: V3; headFwd: V3; headRight: V3;
  hipL: V3; hipR: V3; kneeL: V3; kneeR: V3; ankleL: V3; ankleR: V3;
  shL: V3; shR: V3; elbowL: V3; elbowR: V3; wristL: V3; wristR: V3;
  headR: V3; s: number; W: number;
};

const rX = (v: V3, a: number): V3 => [v[0], v[1] * Math.cos(a) - v[2] * Math.sin(a), v[1] * Math.sin(a) + v[2] * Math.cos(a)];
const rY = (v: V3, a: number): V3 => [v[0] * Math.cos(a) + v[2] * Math.sin(a), v[1], -v[0] * Math.sin(a) + v[2] * Math.cos(a)];
const rZ = (v: V3, a: number): V3 => [v[0] * Math.cos(a) - v[1] * Math.sin(a), v[0] * Math.sin(a) + v[1] * Math.cos(a), v[2]];

/** Solve a pose (targets in figure-local metres for THIS build — poses are authored at the build's size). */
export const solve = (p: Pose, b: Build = {}): Joints => {
  const s = b.s ?? 1, W = b.wide ?? 1, HK = (b.head ?? 1) * (s < 0.8 ? 1.25 : 1);
  const D = DIM;
  const tw = p.twist ?? 0, rl = p.roll ?? 0;
  const tf = (v: V3) => rZ(rX(rY(v, tw), p.pitch), -rl);
  const up = tf([0, 1, 0]), right = tf([1, 0, 0]), fwd = tf([0, 0, 1]);
  const pelvis = p.pelvis;
  const chest = add(pelvis, mul(up, D.shoulderUp * s));
  const neck = add(pelvis, mul(up, D.torsoUp * s));
  const hf = (v: V3) => tf(rX(rZ(rY(v, p.turn ?? 0), -(p.tilt ?? 0)), p.nod ?? 0));
  const headUp = hf([0, 1, 0]), headFwd = hf([0, 0, 1]), headRight = hf([1, 0, 0]);
  const headR: V3 = [D.headR[0] * s * HK, D.headR[1] * s * HK, D.headR[2] * s * HK];
  const head = add(neck, mul(headUp, D.neck * s + headR[1] * 0.9));
  const hipL = add(pelvis, mul(right, D.hipW * s * W));
  const hipR = add(pelvis, mul(right, -D.hipW * s * W));
  const ko = p.kneeOut ?? 0.15;
  const [kneeL, ankleL] = ik2(hipL, p.footL, D.thigh * s, D.shin * s, [ko, 0.15, 1]);
  const [kneeR, ankleR] = ik2(hipR, p.footR, D.thigh * s, D.shin * s, [-ko, 0.15, 1]);
  const shL = add(chest, mul(right, D.shoulderW * s * W)), shR = add(chest, mul(right, -D.shoulderW * s * W));
  const eo = p.elbowOut ?? 0.45;
  const poleL = add(add(mul(right, eo * 1.6), mul(up, -1)), mul(fwd, -0.5));
  const poleR = add(add(mul(right, -eo * 1.6), mul(up, -1)), mul(fwd, -0.5));
  const [elbowL, wristL] = ik2(shL, p.handL, D.upper * s, D.fore * s, poleL);
  const [elbowR, wristR] = ik2(shR, p.handR, D.upper * s, D.fore * s, poleR);
  return { pelvis, chest, neck, head, up, right, fwd, headUp, headFwd, headRight, hipL, hipR, kneeL, kneeR, ankleL, ankleR, shL, shR, elbowL, elbowR, wristL, wristR, headR, s, W };
};

const mixN = (a: number | undefined, b: number | undefined, t: number, d = 0) => (a ?? d) + ((b ?? d) - (a ?? d)) * t;
const mixV = (a: V3, b: V3, t: number): V3 => [a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t, a[2] + (b[2] - a[2]) * t];
/** the palm hint a hand uses when the pose gives none: palms in toward the body, in the torso's frame (as Person) */
const palmDefault = (p: Pose, side: 1 | -1): V3 => {
  const tw = p.twist ?? 0, rl = p.roll ?? 0;
  return rZ(rX(rY([-side, 0, 0], tw), p.pitch), -rl);
};
const mixPalm = (a: Pose, b: Pose, side: 1 | -1, t: number): V3 | undefined => {
  const pa = side === 1 ? a.palmL : a.palmR, pb = side === 1 ? b.palmL : b.palmR;
  if (!pa && !pb) return undefined;
  return mixV(pa ?? palmDefault(a, side), pb ?? palmDefault(b, side), t);
};
/** Blend two poses (targets blend and IK re-solves, so limbs never stretch or pop). */
export const mixPose = (a: Pose, b: Pose, t: number): Pose => {
  if (t <= 0) return a;
  if (t >= 1) return b;
  return {
    pelvis: mixV(a.pelvis, b.pelvis, t), pitch: mixN(a.pitch, b.pitch, t), roll: mixN(a.roll, b.roll, t), twist: mixN(a.twist, b.twist, t),
    nod: mixN(a.nod, b.nod, t), turn: mixN(a.turn, b.turn, t), tilt: mixN(a.tilt, b.tilt, t),
    handL: mixV(a.handL, b.handL, t), handR: mixV(a.handR, b.handR, t), footL: mixV(a.footL, b.footL, t), footR: mixV(a.footR, b.footR, t),
    palmL: mixPalm(a, b, 1, t), palmR: mixPalm(a, b, -1, t),
    shapeL: t < 0.5 ? a.shapeL : b.shapeL, shapeR: t < 0.5 ? a.shapeR : b.shapeR,
    fingL: mixF(a.fingL ?? fingersOf(a.shapeL), b.fingL ?? fingersOf(b.shapeL), t), fingR: mixF(a.fingR ?? fingersOf(a.shapeR), b.fingR ?? fingersOf(b.shapeR), t),
    splay: mixN(a.splay, b.splay, t, 0.2), toe: mixN(a.toe, b.toe, t, 0), elbowOut: mixN(a.elbowOut, b.elbowOut, t, 0.45), kneeOut: mixN(a.kneeOut, b.kneeOut, t, 0.15),
    fall: mixN(a.fall, b.fall, t, 0),
  };
};

export const DustPuffs: React.FC<{ t: number; t0: number; t1: number; at: (t: number) => V3; rate?: number; life?: number; seed?: number }> = ({ t, t0, t1, at: origin, rate = 14, life = 0.55, seed = 1 }) => {

const gable = (() => {

// TIPIKAL motion tracks (pure): where a person / vehicle is at time t. A track is a list of segments: a placement
// (one point) or a move along a smooth path (Catmull-Rom through the points, arc-length timed: eased start / stop,
// or constant speed when `pass`). Used by the director (cameras) and the renderer (positions, heading, wheel spin,
// walk phase, lean).
import type { Seg, Track, V3 } from "./types.ts";

type Cache = { pts: V3[]; cum: number[]; L: number };
const cache = new WeakMap<Seg, Cache>();
const cr = (p0: V3, p1: V3, p2: V3, p3: V3, t: number): V3 => {
  const t2 = t * t, t3 = t2 * t;
  const f = (a: number, b: number, c: number, d: number) => 0.5 * (2 * b + (-a + c) * t + (2 * a - 5 * b + 4 * c - d) * t2 + (-a + 3 * b - 3 * c + d) * t3);
  return [f(p0[0], p1[0], p2[0], p3[0]), f(p0[1], p1[1], p2[1], p3[1]), f(p0[2], p1[2], p2[2], p3[2])];
};
const sampled = (s: Seg): Cache => {
  const hit = cache.get(s);
  if (hit) return hit;
  const P = s.pts;
  const pts: V3[] = [P[0]];
  for (let i = 0; i < P.length - 1; i++) {
    const p0 = P[Math.max(0, i - 1)], p1 = P[i], p2 = P[i + 1], p3 = P[Math.min(P.length - 1, i + 2)];
    const n = P.length === 2 ? 1 : 24;
    for (let k = 1; k <= n; k++) pts.push(n === 1 ? p2 : cr(p0, p1, p2, p3, k / n));
  }
  const cum = [0];
  for (let i = 1; i < pts.length; i++) cum.push(cum[i - 1] + Math.hypot(pts[i][0] - pts[i - 1][0], pts[i][2] - pts[i - 1][2]));
  const c = { pts, cum, L: cum[cum.length - 1] };
  cache.set(s, c);
  return c;
};
const at = (c: Cache, d: number): V3 => {
  if (d <= 0) return c.pts[0];
  if (d >= c.L) return c.pts[c.pts.length - 1];
  let lo = 0, hi = c.cum.length - 1;
  while (hi - lo > 1) { const m = (lo + hi) >> 1; if (c.cum[m] <= d) lo = m; else hi = m; }
  const u = (d - c.cum[lo]) / Math.max(1e-9, c.cum[hi] - c.cum[lo]);
  const a = c.pts[lo], b = c.pts[hi];
  return [a[0] + (b[0] - a[0]) * u, a[1] + (b[1] - a[1]) * u, a[2] + (b[2] - a[2]) * u];
};
const smoother = (x: number) => { const t = Math.min(1, Math.max(0, x)); return t * t * t * (t * (t * 6 - 15) + 10); };
const ease = (s: Seg, t: number) => { const u = (t - s.t0) / Math.max(1e-6, s.t1 - s.t0); return s.pass ? Math.min(1, Math.max(0, u)) : smoother(u); };
const headingOf = (c: Cache, d: number) => {
  const a = at(c, Math.max(0, d - 0.08)), b = at(c, Math.min(c.L, d + 0.08));
  return Math.atan2(b[0] - a[0], b[2] - a[2]);
};

export type TrackState = { pos: V3; face: number; speed: number; dist: number; moving: boolean; walk: boolean; run: boolean; arrived: number };

/** State of a track at time t. `arrived` = seconds since the last move ended (for turning to the final facing). */
export const segAt = (tr: Track, t: number): TrackState => {
  let k = -1;
  for (let i = 0; i < tr.segs.length; i++) if (tr.segs[i].t0 <= t + 1e-6) k = i;
  let dist = 0;
  for (let i = 0; i < Math.max(0, k); i++) { const s = tr.segs[i]; if (s.pts.length > 1 && s.t1 > s.t0) dist += sampled(s).L * ease(s, Math.min(t, s.t1)); }
  if (k < 0) { const s0 = tr.segs[0]; return { pos: s0 ? s0.pts[0] : [0, 0, 0], face: tr.face0, speed: 0, dist: 0, moving: false, walk: false, run: false, arrived: 99 }; }
  const s = tr.segs[k];
  if (s.pts.length < 2 || s.t1 <= s.t0) return { pos: s.pts[0], face: s.face, speed: 0, dist, moving: false, walk: false, run: false, arrived: t - s.t0 };
  const c = sampled(s);
  const e = ease(s, t), d = c.L * e;
  dist += d;
  const pos = at(c, d);
  const dt = 0.02, e2 = ease(s, t + dt), e1 = ease(s, t - dt);
  const speed = (c.L * (e2 - e1)) / (2 * dt);
  const moving = t < s.t1;
  let head = headingOf(c, Math.min(d, c.L - 0.01));
  // turn smoothly from the facing before the move into the path direction (no snap when setting off)
  const prev = tr.segs[k - 1];
  const before = !prev ? tr.face0 : prev.pts.length > 1 ? headingOf(sampled(prev), sampled(prev).L - 0.01) : prev.face;
  const turnIn = smoother((t - s.t0) / 0.35);
  if (turnIn < 1) head = before + Math.atan2(Math.sin(head - before), Math.cos(head - before)) * turnIn;
  return { pos, face: head, speed: moving ? speed : 0, dist, moving, walk: !!s.walk, run: !!s.run, arrived: moving ? -1 : t - s.t1 };
};

/** Heading the walker had when the move ended (to turn smoothly to the final facing). */
export const endHeading = (tr: Track, t: number) => {
  let k = -1;
  for (let i = 0; i < tr.segs.length; i++) if (tr.segs[i].t0 <= t + 1e-6) k = i;
  const s = tr.segs[k];
  if (!s || s.pts.length < 2) return s?.face ?? tr.face0;
  const c = sampled(s);
  return headingOf(c, c.L - 0.01);
};
/** The final facing of the current segment. */
export const segFace = (tr: Track, t: number) => {
  let k = -1;
  for (let i = 0; i < tr.segs.length; i++) if (tr.segs[i].t0 <= t + 1e-6) k = i;
  return k < 0 ? tr.face0 : tr.segs[k].face;
};

/** Rodrigues rotation of p about the axis through a (unit direction k) by angle th. */
const rotAxis = (p: V3, a: V3, k: V3, th: number): V3 => {
  const v: V3 = [p[0] - a[0], p[1] - a[1], p[2] - a[2]];
  const c = Math.cos(th), s = Math.sin(th);
  const kxv: V3 = [k[1] * v[2] - k[2] * v[1], k[2] * v[0] - k[0] * v[2], k[0] * v[1] - k[1] * v[0]];
  const kv = k[0] * v[0] + k[1] * v[1] + k[2] * v[2];
  return [a[0] + v[0] * c + kxv[0] * s + k[0] * kv * (1 - c), a[1] + v[1] * c + kxv[1] * s + k[1] * kv * (1 - c), a[2] + v[2] * c + kxv[2] * s + k[2] * kv * (1 - c)];
};
export const steerAxis = (d: VehDef): V3 => {
  const a = d.axisTop!, b = d.axisBot!;
  const v: V3 = [b[0] - a[0], b[1] - a[1], b[2] - a[2]];
  const l = Math.hypot(...v) || 1;
  return [v[0] / l, v[1] / l, v[2] / l];
};
/** Grip centre (vehicle-local) of the left (+1) / right (−1) hand with the bars turned by `steer` (rad, + = left). */
export const gripOf = (d: VehDef, side: 1 | -1, steer = 0, scale = 1): V3 => {
  const g: V3 = [d.grip![0] * side, d.grip![1], d.grip![2]];
  const p = rotAxis(g, d.axisTop!, steerAxis(d), -steer);
  return [p[0] * scale, p[1] * scale, p[2] * scale];
};

/** Head height above the ground of a person sitting on a seat (camera framing). */
export const seatHead = (d: VehDef, seat: number, s = 1, scale = 1) => {
  const S = d.seats[seat] ?? d.seats[0];
  if (!S) return 1.5 * s;
  if (S.kind === "floor") return S.top * scale + 1.5 * s;
  return S.top * scale + 0.115 * s + 0.72 * s;
};
