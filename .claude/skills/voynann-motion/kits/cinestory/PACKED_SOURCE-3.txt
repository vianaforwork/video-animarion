// CINÉSTORY character: one procedural rig, many people. A CharSpec (the character bible as data) sets
// skin, hair style, face shape, wardrobe and proportions; the rig is posed per frame from a Pose
// (pose.ts). Head = one sculpted sphere (brow ridge, sockets, nose, cheekbones, lips, chin), eyes with
// lids + lash lines; arms FK or 2-bone IK toward a target in chest space (door handle, key, phone).

/** Tapered limb along −Y from 0 to −len with round caps. */
const limb = (len: number, r0: number, r1: number, mid = 0) =>
  cached(`limb${len},${r0},${r1},${mid}`, () => {
    const pts: [number, number][] = [];
    const n = 8;
    for (let i = 0; i <= n; i++) {
      const a = (i / n) * (Math.PI / 2);
      pts.push([Math.sin(a) * r1, -len - Math.cos(a) * r1]);
    }
    for (let i = 1; i < 10; i++) {
      const k = i / 10;
      const bulge = Math.sin(k * Math.PI) * mid;
      pts.push([r1 + (r0 - r1) * k + bulge, -len + len * k]);
    }
    for (let i = 0; i <= n; i++) {
      const a = (i / n) * (Math.PI / 2);
      pts.push([Math.cos(a) * r0, Math.sin(a) * r0]);
    }
    return lathe(`limbL${len},${r0},${r1},${mid}`, pts, 28);
  });

const g2 = (x: number, y: number, sx: number, sy: number) => Math.exp(-(x * x) / (sx * sx) - (y * y) / (sy * sy));

/** Sculpt a unit-sphere point into the head surface (+Z = face), in head units. Shape knobs from the spec. */
const sculpt = (x: number, y: number, z: number, f: CharSpec["face"]): [number, number, number] => {
  const front = Math.max(0, z);
  const ax = Math.abs(x);
  // displacement along the normal (unit sphere: normal = position)
  let d = 0;
  d += 0.035 * (0.6 + f.brow) * g2(ax - 0.3, y - 0.27, 0.32, 0.075) * Math.pow(front, 2); // brow ridge
  d -= 0.06 * g2(ax - 0.375, y - 0.07, 0.17, 0.12) * front; // eye sockets
  const ridge = g2(x, 0, 0.075, 1) * (y < 0.18 && y > -0.33 ? Math.pow((0.18 - y) / 0.51, 1.3) : 0);
  d += (0.2 + 0.08 * f.nose) * ridge * Math.pow(front, 4); // nose bridge → tip
  d += 0.1 * g2(x, y + 0.27, 0.075, 0.065) * front; // tip
  d += 0.06 * g2(ax - 0.12, y + 0.3, 0.06, 0.05) * front; // nostril wings
  d -= 0.025 * g2(ax - 0.2, y + 0.38, 0.05, 0.06) * front; // nasolabial
  d += 0.035 * (0.5 + f.cheek) * g2(ax - 0.55, y + 0.05, 0.2, 0.16) * front; // cheekbones
  d += 0.022 * g2(x, y + 0.5, 0.17, 0.035) * front; // upper lip
  d += 0.026 * g2(x, y + 0.6, 0.15, 0.04) * front; // lower lip
  d -= 0.018 * g2(x, y + 0.69, 0.18, 0.03) * front; // under-lip groove
  d += (0.03 + 0.04 * f.chin) * g2(x, y + 0.82, 0.2, 0.08) * front; // chin
  x *= 1 + d; y *= 1 + d; z *= 1 + d;
  // jaw taper (narrower lower face), forward chin, fuller lower cheeks
  const low = Math.max(0, -y);
  x *= 1 - (0.24 - 0.12 * f.jaw) * Math.pow(low, 1.5);
  z += 0.14 * low * Math.max(0, z) * Math.max(0, z);
  y *= 1 + 0.12 * low;
  return [x * 0.102, y * 0.118, z * 0.112];
};

/** Sculpted head on a unit sphere, then scaled to head size. */
const headGeo = (f: CharSpec["face"]) =>
  cached(`head${f.jaw},${f.cheek},${f.nose},${f.brow},${f.chin}`, () => {
    const g = new SphereGeometry(1, 96, 72);
    const a = g.attributes.position;
    for (let i = 0; i < a.count; i++) a.setXYZ(i, ...sculpt(a.getX(i), a.getY(i), a.getZ(i), f));
    g.computeVertexNormals();
    return g;
  });

/** A shell that follows the sculpted head: `lift(ux, uy, uz)` (relative, on the unit sphere) > 0 stands off the
 *  skin, < 0 dives under it. The skin hides the dived part, so edges are smooth curves (no cut triangles). */
const skinShell = (key: string, f: CharSpec["face"], lift: (x: number, y: number, z: number) => number) =>
  cached(`shell${key}${f.jaw},${f.cheek},${f.nose},${f.brow},${f.chin}`, () => {
    const g = new SphereGeometry(1, 96, 72);
    const a = g.attributes.position;
    for (let i = 0; i < a.count; i++) {
      const ux = a.getX(i), uy = a.getY(i), uz = a.getZ(i);
      const [x, y, z] = sculpt(ux, uy, uz, f);
      const k = 1 + lift(ux, uy, uz);
      a.setXYZ(i, x * k, y * k, z * k);
    }
    g.computeVertexNormals();
    return g;
  });
const sstep = (a: number, b: number, x: number) => { const t = Math.min(1, Math.max(0, (x - a) / (b - a))); return t * t * (3 - 2 * t); };

/** Hijab: lifted head surface with an oval face opening; extra volume behind and wrapping under the chin. */
const hijabGeo = (f: CharSpec["face"]) =>
  skinShell("hijab", f, (x, y, z) => {
    const o = (x / 0.6) ** 2 + ((y + 0.04) / 0.8) ** 2;
    const open = sstep(1.12, 0.9, o) * sstep(0.0, 0.25, z);
    const lift = 0.09 + 0.06 * Math.max(0, -z) + 0.1 * Math.max(0, y - 0.6) + 0.3 * Math.max(0, -y - 0.6) * (1 - 0.6 * Math.max(0, z));
    return lift * (1 - open) - 0.06 * open;
  });

/** Beard: jaw, chin and cheeks below the cheekbones, a gap for the mouth; dives under the skin elsewhere. */
const beardGeo = (f: CharSpec["face"]) =>
  skinShell("beard", f, (x, y, z) => {
    const jaw = sstep(-0.4, -0.58, y) * sstep(-0.35, 0.05, z);
    const mouth = Math.exp(-(x * x) / (0.2 * 0.2) - ((y + 0.55) ** 2) / (0.09 * 0.09));
    const m = jaw * (1 - 0.95 * mouth);
    return 0.05 * m - 0.04 * (1 - m);
  });

const UA = 0.28, FA = 0.25;
const _q = new Quaternion();
const ikArm = (S: Vector3, T: Vector3, pole: Vector3) => {
  const d0 = new Vector3().subVectors(T, S);
  const d = Math.min(Math.max(d0.length(), Math.abs(UA - FA) + 1e-3), UA + FA - 1e-3);
  const u = d0.normalize();
  const cosA = (UA * UA + d * d - FA * FA) / (2 * UA * d);
  const sinA = Math.sqrt(Math.max(0, 1 - cosA * cosA));
  const w = pole.clone().addScaledVector(u, -pole.dot(u));
  if (w.lengthSq() < 1e-8) w.set(0, 0, -1).addScaledVector(u, -u.z);
  w.normalize();
  const upper = u.clone().multiplyScalar(cosA).addScaledVector(w, sinA).normalize();
  const E = S.clone().addScaledVector(upper, UA);
  const fore = new Vector3().subVectors(S.clone().addScaledVector(u, d), E).normalize();
  const cosB = Math.min(1, Math.max(-1, upper.dot(fore)));
  const bend = Math.acos(cosB);
  const Y = upper.clone().negate();
  let Z = fore.clone().addScaledVector(upper, -fore.dot(upper));
  if (Z.lengthSq() < 1e-8) Z = w.clone().negate();
  Z.normalize();
  const X = new Vector3().crossVectors(Y, Z).normalize();
  Z.crossVectors(X, Y).normalize();
  const m = new Matrix4().makeBasis(X, Y, Z);
  return { q: new Quaternion().setFromRotationMatrix(m), bend };
};

const ArmSide: React.FC<{ s: number; a: Arm; hand: Hand; c: CharSpec; t: number; tremble: number; phone?: Material; hold?: Hold | null }> = ({ s, a, hand, c, t, tremble, phone, hold }) => {
  const S = new Vector3(s * 0.19, 0.2, -0.01);
  const cl = clothes(c);
  let q: Quaternion;
  let el = a.el;
  const fk = new Quaternion().setFromEuler(new Euler(-a.sw, s * a.tw, s * a.ab, "XZY"));
  if (a.ik && a.ik.w > 0) {
    const T = new Vector3(a.ik.t[0] * s, a.ik.t[1], a.ik.t[2]);
    const pole = new Vector3(a.ik.pole[0] * s, a.ik.pole[1], a.ik.pole[2]);
    const r = ikArm(S, T, pole);
    q = fk.clone().slerp(r.q, a.ik.w);
    el = el + (r.bend - el) * a.ik.w;
  } else q = fk;
  const jit = tremble * 0.06;
  const wr = new Euler(a.wr[0] + jit * Math.sin(t * 37), s * a.wr[1], s * a.wr[2] + jit * Math.sin(t * 29 + 1), "XYZ");
  const skin = skinMat(c.skin);
  const short = c.top === "tee";
  return (
    <group position={S} quaternion={q}>
      {short ? (
        <>
          <mesh geometry={limb(0.14, 0.064, 0.058)} material={cl.sleeve} castShadow receiveShadow />
          <mesh geometry={limb(0.28, 0.044, 0.037)} material={skin} castShadow receiveShadow />
        </>
      ) : (
        <mesh geometry={limb(0.28, 0.064, 0.054, 0.007)} material={cl.sleeve} castShadow receiveShadow />
      )}
      <group position={[0, -UA, 0]} rotation={[-el, 0, 0]}>
        {short ? (
          <mesh geometry={limb(0.25, 0.037, 0.03, 0.004)} material={skin} castShadow receiveShadow />

          <>
            <mesh geometry={limb(0.225, 0.054, 0.047, 0.005)} material={cl.sleeve} castShadow receiveShadow />
            <mesh geometry={cyl(0.042, 0.04, 0.045, 20)} material={cl.sleeve} position={[0, -0.222, 0]} castShadow />

/** Hair clumps on the cap surface: local y = surface normal, local z = sweep direction. */
const hairClumps = (style: CharSpec["hairStyle"]) => {
  const out: { p: [number, number, number]; q: Quaternion; s: [number, number, number] }[] = [];
  const C = new Vector3(0, 0.016, -0.004);
  const r = mulberry32(style === "crop" ? 5 : 9);
  const put = (az: number, el: number, rad: number, sweep: number, sc: [number, number, number]) => {
    const n = new Vector3(Math.sin(az) * Math.cos(el), Math.sin(el), Math.cos(az) * Math.cos(el));
    const pos = C.clone().addScaledVector(n, rad);
    const q = new Quaternion().setFromUnitVectors(new Vector3(0, 1, 0), n);
    q.multiply(new Quaternion().setFromAxisAngle(new Vector3(0, 1, 0), sweep));
    out.push({ p: [pos.x, pos.y, pos.z], q, s: sc });
  };
  if (style === "crop") {
    // soft fringe swept to his right, short textured top, tidy sides
    for (let i = 0; i < 12; i++) put(-0.72 + i * 0.13 + (r() - 0.5) * 0.04, 0.56 + 0.05 * r(), 0.116, 2.7 + 0.08 * i + (r() - 0.5) * 0.3, [0.42, 0.2, 1.25 + 0.3 * r()]);
    for (let i = 0; i < 30; i++) put(r() * Math.PI * 2, 0.75 + 0.6 * r(), 0.119, Math.PI + (r() - 0.5) * 1.2, [0.45 + 0.15 * r(), 0.2, 1.2 + 0.3 * r()]);

const ridged = (x: number, y: number, z: number) => {
  const wx = x + 0.6 * noise3(x * 0.7 + 3, y * 0.7, z * 0.7);
  const wz = z + 0.6 * noise3(x * 0.7, y * 0.7 + 5, z * 0.7);
  const a = 1 - Math.abs(2 * noise3(wx, y, wz) - 1);
  const b = 1 - Math.abs(2 * noise3(wx * 2.1 + 9, y * 2.1, wz * 2.1) - 1);
  return Math.pow(a, 1.6) * 0.72 + b * 0.28;
};

export const brainGeo = () =>
  cached("brainShell", () => {
    let g: BufferGeometry = new IcosahedronGeometry(1, 6);
    g.deleteAttribute("normal");
    g.deleteAttribute("uv");
    g = mergeVertices(g);
    const a = g.attributes.position;
    const fold = new Float32Array(a.count);
    for (let i = 0; i < a.count; i++) {
      const v = new Vector3(a.getX(i), a.getY(i), a.getZ(i));
      const p = new Vector3(v.x * 0.8, v.y * 0.68, v.z * 1.0);
      // longitudinal fissure between the hemispheres
      const fis = Math.exp(-((v.x / 0.09) ** 2)) * Math.min(1, Math.max(0, (v.y + 0.25) * 1.6));
      p.y -= 0.1 * fis;
      p.x *= 1 + 0.04 * fis;
      // flatter base, temporal lobes, narrower occipital
      if (p.y < -0.2) p.y = -0.2 + (p.y + 0.2) * 0.55;
      const temp = Math.exp(-(((Math.abs(v.x) - 0.85) / 0.25) ** 2) - ((v.y + 0.35) / 0.3) ** 2 - ((v.z - 0.15) / 0.45) ** 2);
      p.x += Math.sign(v.x) * 0.06 * temp;
      p.y -= 0.05 * temp;
      if (v.z < -0.3) p.x *= 1 - 0.1 * (-0.3 - v.z);
      const f = ridged(v.x * 4.2, v.y * 4.2, v.z * 4.2);
      fold[i] = f;
      const n = p.clone().normalize();
      p.addScaledVector(n, (f - 0.55) * 0.055 * (1 - 0.7 * fis));
      a.setXYZ(i, p.x, p.y, p.z);
    }
    g.setAttribute("fold", new Float32BufferAttribute(fold, 1));
    g.computeVertexNormals();
    return g;
  });

export type Net = { pts: Vector3[]; edges: [number, number][] };
export const makeNet = (seed: number, n: number, accept: (p: Vector3) => boolean, rng: () => Vector3, k = 2): Net => {
  const r = mulberry32(seed);
  const pts: Vector3[] = [];
  let guard = 0;
  while (pts.length < n && guard++ < n * 50) {
    const p = rng();
    if (accept(p)) pts.push(p);
  }
  const edges: [number, number][] = [];
  for (let i = 0; i < pts.length; i++) {
    const d = pts.map((q, j) => [q.distanceToSquared(pts[i]), j] as const).sort((a, b) => a[0] - b[0]);
    for (let m = 1; m <= k; m++) if (d[m] && (r() < 0.9 || m === 1)) edges.push([i, d[m][1]]);
  }
  return { pts, edges };
};

const neuronGeo = (seed: number) =>
  cached(`neuron${seed}`, () => {
    const r = mulberry32(seed);
    const tubes: BufferGeometry[] = [];
    const branch = (o: Vector3, dir: Vector3, len: number, rad: number, depth: number) => {
      const pts = [o.clone()];
      let p = o.clone(), d = dir.clone();
      for (let i = 0; i < 5; i++) {
        d.add(new Vector3(r() - 0.5, r() - 0.5, r() - 0.5).multiplyScalar(0.7)).normalize();
        p = p.clone().addScaledVector(d, len / 5);
        pts.push(p);
      }
      const curve = new CatmullRomCurve3(pts);
      const tg = new TubeGeometry(curve, 24, rad, 6, false);
      // taper
      const a = tg.attributes.position;
      for (let i = 0; i < a.count; i++) {
        const seg = Math.floor(i / 7) / 24;
        const c = curve.getPoint(Math.min(1, seg));
        const v = new Vector3(a.getX(i), a.getY(i), a.getZ(i)).sub(c).multiplyScalar(1 - 0.7 * seg).add(c);
        a.setXYZ(i, v.x, v.y, v.z);
      }
      tubes.push(tg);
      if (depth > 0) for (let k = 0; k < 2; k++) branch(p, d.clone().add(new Vector3(r() - 0.5, r() - 0.5, r() - 0.5)).normalize(), len * 0.62, rad * 0.6, depth - 1);
    };
    for (let k = 0; k < 6; k++) {
      const d = new Vector3(r() - 0.5, r() - 0.5, r() - 0.5).normalize();
      branch(d.clone().multiplyScalar(0.12), d, 0.55 + r() * 0.3, 0.03, 2);
    }
    branch(new Vector3(0, -0.1, 0), new Vector3(0.2, -1, 0.3).normalize(), 1.6, 0.035, 0);
    const parts = tubes.map((g) => { g.deleteAttribute("uv"); return g.index ? g.toNonIndexed() : g; });
    const merged = new BufferGeometry();
    let total = 0;
    for (const p of parts) total += p.attributes.position.count;
    const pos = new Float32Array(total * 3), nor = new Float32Array(total * 3);
    let off = 0;
    for (const p of parts) {
      pos.set(p.attributes.position.array as Float32Array, off * 3);
      nor.set(p.attributes.normal.array as Float32Array, off * 3);
      off += p.attributes.position.count;
    }
    merged.setAttribute("position", new Float32BufferAttribute(pos, 3));
    merged.setAttribute("normal", new Float32BufferAttribute(nor, 3));
    return merged;
  });

// The rig applies the shot for this frame before R3F advances, then the composer renders: scene (MSAA) → ground-truth AO → depth of field → restrained
// bloom → Khronos PBR Neutral tone mapping → grade (warmth, vignette, fine grain).

import React, { useLayoutEffect, useMemo } from "react";
import { useFrame, useThree } from "@react-three/fiber";
import { type Camera, HalfFloatType, NeutralToneMapping, PerspectiveCamera, Vector2, WebGLRenderTarget } from "three";
import { EffectComposer } from "three/examples/jsm/postprocessing/EffectComposer.js";
import { RenderPass } from "three/examples/jsm/postprocessing/RenderPass.js";
import { GTAOPass } from "three/examples/jsm/postprocessing/GTAOPass.js";
import { BokehPass } from "three/examples/jsm/postprocessing/BokehPass.js";
import { UnrealBloomPass } from "three/examples/jsm/postprocessing/UnrealBloomPass.js";
import { OutputPass } from "three/examples/jsm/postprocessing/OutputPass.js";
import { ShaderPass } from "three/examples/jsm/postprocessing/ShaderPass.js";

/**
 * Render layer for additive / volumetric effects (light shafts, glows, halos, beams). Put such
 * meshes on this layer ONLY (`mesh.layers.set(FX_LAYER)`): the camera still renders them, but the
 * depth-based passes (GTAO, depth of field) skip them. Large transparent volumes on the default
 * layer corrupt those passes and can silently blank the whole frame.
 */
export const FX_LAYER = 1;

/** Run a pass with the FX layer hidden from the camera (depth/normal passes must not see effects). */
export const withoutFx = <P extends Pass>(pass: P, camera: Camera): P => {
  const render = pass.render.bind(pass);
  pass.render = (renderer, writeBuffer, readBuffer, deltaTime, maskActive) => {
    camera.layers.disable(FX_LAYER);
    try {
      render(renderer, writeBuffer, readBuffer, deltaTime, maskActive);
    } finally {
      camera.layers.enable(FX_LAYER);
    }
  };
  return pass;
};

export type Shot = {
  pos: [number, number, number];
  target: [number, number, number];
  fov: number;
  roll?: number;
  /** Focus distance (world units) and lens aperture for the bokeh pass. */
  focus: number;
  aperture: number;
  maxBlur: number;
  exposure: number;
};

export const CameraRig: React.FC<{ shot: Shot; near?: number; far?: number }> = ({ shot, near = 1, far = 900 }) => {
  const { camera, gl } = useThree();
  useLayoutEffect(() => {
    const cam = camera as PerspectiveCamera;
    cam.position.set(...shot.pos);
    cam.up.set(Math.sin(shot.roll ?? 0), Math.cos(shot.roll ?? 0), 0);
    cam.lookAt(...shot.target);
    cam.fov = shot.fov;
    cam.near = near;
    cam.far = far;
    cam.updateProjectionMatrix();
    cam.updateMatrixWorld();
    cam.layers.enable(FX_LAYER);
    gl.toneMapping = NeutralToneMapping;
    gl.toneMappingExposure = shot.exposure;
  }, [shot, camera, gl, near, far]);
  return null;
};

/**
 * Replaces non-finite pixels (NaN/Inf from a faulty custom shader) right after the scene render.
 * Without it, bloom spreads a single NaN pixel across the whole frame and the frame renders blank
 * with no error — found in testing with a custom light-shaft shader.
 */
const NanGuardShader = {
  uniforms: { tDiffuse: { value: null } },
  vertexShader: /* glsl */ `varying vec2 vUv; void main(){ vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }`,
  fragmentShader: /* glsl */ `uniform sampler2D tDiffuse; varying vec2 vUv;
    void main(){ vec4 c = texture2D(tDiffuse, vUv);
      if (any(isnan(c)) || any(isinf(c))) c = vec4(0.0, 0.0, 0.0, 1.0);
      gl_FragColor = clamp(c, 0.0, 64.0); }`,
};

const GradeShader = {
  uniforms: {
    tDiffuse: { value: null },
    uTime: { value: 0 },
    uVignette: { value: 0.14 },
    uGrain: { value: 0.012 },
    uRes: { value: new Vector2(1080, 1920) },
  },
  vertexShader: /* glsl */ `varying vec2 vUv; void main(){ vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }`,
  fragmentShader: /* glsl */ `
    uniform sampler2D tDiffuse; uniform float uTime; uniform float uVignette; uniform float uGrain; uniform vec2 uRes;
    varying vec2 vUv;
    float hash(vec2 p){ p = fract(p * vec2(443.897, 441.423)); p += dot(p, p.yx + 19.19); return fract((p.x + p.y) * p.x); }
    void main(){
      vec4 c = texture2D(tDiffuse, vUv);
      float l = dot(c.rgb, vec3(0.2126, 0.7152, 0.0722));
      // ivory highlights, a breath of cool in the shadows
      c.rgb += smoothstep(0.6, 1.0, l) * vec3(0.002, 0.001, -0.003);
      c.rgb += (1.0 - smoothstep(0.0, 0.5, l)) * vec3(-0.004, 0.0, 0.006);
      // gentle S-curve: deeper form shadows without crushing the whites
      c.rgb = mix(c.rgb, c.rgb * c.rgb * (3.0 - 2.0 * c.rgb), 0.24);
      vec2 d = (vUv - 0.5) * vec2(1.0, 1.25);
      c.rgb *= 1.0 - uVignette * smoothstep(0.35, 0.95, length(d));
      float g = hash(vUv * uRes + uTime * 17.0) - 0.5;
      c.rgb += g * uGrain;
      gl_FragColor = c;
    }`,
};

export const Post: React.FC<{ shot: Shot; frame: number; ao?: boolean; aoRadius?: number }> = ({ shot, frame, ao = true, aoRadius = 0.28 }) => {
  const { gl, scene, camera, size } = useThree();
  const chain = useMemo(() => {
    const w = size.width, h = size.height;
    const rt = new WebGLRenderTarget(w, h, { type: HalfFloatType, samples: 4 });
    const composer = new EffectComposer(gl, rt);
    composer.setPixelRatio(1);
    composer.setSize(w, h);
    composer.addPass(new RenderPass(scene, camera));
    composer.addPass(new ShaderPass(NanGuardShader));
    if (ao) {
      const gtao = new GTAOPass(scene, camera, w, h);
      gtao.blendIntensity = 1.0;
      gtao.updateGtaoMaterial({ radius: aoRadius, distanceExponent: 1.8, thickness: 0.5, scale: 1.0, samples: 16, distanceFallOff: 1.0, screenSpaceRadius: false });
      gtao.updatePdMaterial({ lumaPhi: 10, depthPhi: 2, normalPhi: 3, radius: 5, rings: 2, samples: 16 });
      composer.addPass(withoutFx(gtao, camera));
    }
    const dof = withoutFx(new BokehPass(scene, camera, { focus: 10, aperture: 0.002, maxblur: 0.008 }), camera);
    composer.addPass(dof);
    const bloom = new UnrealBloomPass(new Vector2(w, h), 0.14, 0.45, 2.2);
    composer.addPass(bloom);
    composer.addPass(new OutputPass());
    const grade = new ShaderPass(GradeShader);
    grade.uniforms.uRes.value.set(w, h);
    composer.addPass(grade);
    return { composer, dof, grade };
  }, [gl, scene, camera, size.width, size.height, ao, aoRadius]);

  useLayoutEffect(() => {
    const u = chain.dof.uniforms as unknown as Record<string, { value: number }>;
    u.focus.value = shot.focus;
    u.aperture.value = shot.aperture;
    u.maxblur.value = shot.maxBlur;
    chain.dof.enabled = shot.maxBlur > 0 && shot.aperture > 0;
    chain.grade.uniforms.uTime.value = frame % 997;
  }, [frame, chain, shot]);

  useFrame(() => {
    chain.composer.render();
  }, 1);
  return null;
};

const armIK = (p: Pose, side: 1 | -1, t: [number, number, number], w: number, pole: [number, number, number] = [0.5, -0.7, -0.2]) => {
  const key = side === 1 ? "armL" : "armR";
  p[key] = { ...p[key], ik: { t: side === 1 ? t : [-t[0], t[1], t[2]], pole, w } };
};
/** World point → IK target for the right arm (true chest coordinates, like every armIK target), clamped to reach. */
const reachR = (p: Pose, w: V3, c: CharSpec, reach = 0.5): [number, number, number] => {
  const l = toChest(p, w, c);
  const S = [-0.19, 0.2, -0.01];
  const d = [l[0] - S[0], l[1] - S[1], l[2] - S[2]], n = Math.hypot(d[0], d[1], d[2]) || 1;
  const k = Math.min(1, reach / n);
  // true chest coordinates: armIK's mirror for the right arm and Character's side sign cancel out
  return [S[0] + d[0] * k, S[1] + d[1] * k, S[2] + d[2] * k];
};

/** A point on a surface (desk, counter) the right hand can reach: when the real point is too far, the
 *  nearest reachable point at the SAME height toward it — never a shortened reach down into the furniture. */
const onSurface = (p: Pose, w: V3, c: CharSpec, reach: number): V3 => {
  const sh = fromChest(p, [-0.19, 0.2, -0.01], c);
  const R = reach * c.scale * 0.97, dy = w[1] - sh[1];
  const dx = w[0] - sh[0], dz = w[2] - sh[2], hd = Math.hypot(dx, dz);
  if (Math.hypot(hd, dy) <= R || Math.abs(dy) >= R) return w;
  const hmax = Math.sqrt(R * R - dy * dy), k = hmax / (hd || 1);
  return [sh[0] + dx * k, w[1], sh[2] + dz * k];
};

// ── the actions ─────────────────────────────────────────────────────────────

export type ActCtx = { seg: Seg; t: number; c: CharSpec; talk: (t: number) => number };

const idle = ({ seg, t, c }: ActCtx, lookK = 1): Pose => {
  const p = base(seg.to, c);
  if (seg.to.posture === "stand") shift(p, t, seg.seed);
  p.face = FACE[seg.face];
  look(p, seg.target, c, lookK * ss(seg.t0, seg.t0 + 0.7, t));
  return p;
};

const ACT: Record<ActionName, (x: ActCtx) => Pose> = {
  stand: (x) => idle(x, 0.85),
  look: (x) => idle(x, 1),
  turn: (x) => {
    const { seg, t, c } = x;
    const k = ss(seg.t0, seg.t0 + 0.9, t);
    const p = base({ ...seg.to, yaw: seg.from.yaw + wrap(seg.to.yaw - seg.from.yaw) * k }, c);
    const step = Math.sin(k * Math.PI);
    p.legL = { ...p.legL, sw: 0.25 * step * Math.sin(k * Math.PI * 2), kn: 0.35 * step };
    p.legR = { ...p.legR, sw: -0.2 * step * Math.sin(k * Math.PI * 2), kn: 0.2 * step };
    p.face = FACE[seg.face];
    look(p, seg.target, c, ss(seg.t0, seg.t0 + 0.5, t));
    return p;
  },
  walk: (x) => move(x, false),
  run: (x) => move(x, true),
  startle: (x) => {
    const { seg, t, c } = x;
    const p = idle(x, 0);
    const u = t - seg.t0, jolt = Math.exp(-Math.max(0, u) * 4) * ss(0, 0.06, u);
    p.chest = [p.chest[0] - 0.12 * jolt, p.chest[1], p.chest[2]];
    p.spine = [p.spine[0] - 0.05 * jolt, p.spine[1], p.spine[2]];
    p.armL = { ...p.armL, ab: p.armL.ab + 0.25 * jolt + 0.05, el: p.armL.el + 0.5 * jolt };
    p.armR = { ...p.armR, ab: p.armR.ab + 0.25 * jolt + 0.05, el: p.armR.el + 0.5 * jolt };
    p.handL = { curl: 0.1, index: 0.05, thumb: 0.4 };
    p.handR = { curl: 0.1, index: 0.05, thumb: 0.4 };
    p.breath = 1.6;
    if (seg.to.posture === "stand") { const back = 0.12 * ss(0, 0.35, u); p.x -= Math.sin(p.yaw) * back; p.z -= Math.cos(p.yaw) * back; }
    look(p, seg.target, c, ss(seg.t0, seg.t0 + 0.25, t));
    return p;
  },
  freeze: (x) => {
    const { seg, t, c } = x;
    const p = base(seg.to, c);
    p.face = FACE[seg.face];
    p.breath = 1.9;
    p.tremble = 0.6;
    if (seg.to.posture === "stand") {
      p.hipY = -0.03;
      p.legL = { ...p.legL, kn: 0.15 };
      p.legR = { ...p.legR, kn: 0.12 };
      p.armL = { ...p.armL, ab: 0.15, el: 0.45 };
      p.armR = { ...p.armR, ab: 0.15, el: 0.42 };
    }
    look(p, seg.target, c, ss(seg.t0, seg.t0 + 0.4, t));
    return p;
  },
  peer: (x) => {
    const { seg, t, c } = x;
    const p = idle(x, 0);
    const k = ss(seg.t0, seg.t0 + 0.8, t);
    // leans in, head forward, squinting (hands stay down: a raised hand reads as a salute and hides the face)
    p.spine = [p.spine[0] + 0.06 * k, p.spine[1], p.spine[2]];
    p.chest = [p.chest[0] + 0.1 * k, p.chest[1], p.chest[2]];
    p.neck = [p.neck[0] + 0.08 * k, p.neck[1], p.neck[2]];
    p.armL = { ...p.armL, el: p.armL.el + 0.25 * k };
    p.armR = { ...p.armR, el: p.armR.el + 0.25 * k };
    look(p, seg.target, c, ss(seg.t0, seg.t0 + 0.5, t));
    return p;
  },
  point: (x) => {
    const { seg, t, c } = x;
    const p = idle(x, 1);
    const k = ss(seg.t0 + 0.1, seg.t0 + 0.7, t) * (1 - ss(seg.t1 - 0.5, seg.t1, t));
    if (seg.target) armIK(p, -1, reachR(p, seg.target, c, 0.53), k, [0.3, -0.8, 0.1]);
    p.handR = { curl: lerpN(0.35, 1, k), index: lerpN(0.3, 0, k), thumb: 0.5 };
    return p;
  },
  reach: (x) => {
    const { seg, t, c } = x;
    const p = idle(x, 1);
    const k = ss(seg.t0 + 0.1, seg.t0 + 0.8, t) * (1 - ss(seg.t1 - 0.6, seg.t1, t));
    if (seg.target) armIK(p, -1, reachR(p, seg.target, c, 0.55), k, [0.4, -0.8, -0.1]);
    p.chest = [p.chest[0] + 0.08 * k, p.chest[1], p.chest[2]];
    p.handR = { curl: 0.25, index: 0.15, thumb: 0.3 };
    return p;
  },
  wave: (x) => {
    const { seg, t } = x;
    const p = idle(x, 1);
    const k = ss(seg.t0, seg.t0 + 0.4, t) * (1 - ss(seg.t1 - 0.4, seg.t1, t));
    p.armR = { ...p.armR, sw: lerpN(p.armR.sw, 0.3, k), ab: lerpN(p.armR.ab, 1.3, k), el: lerpN(p.armR.el, 0.9 + 0.35 * Math.sin(t * 9), k), wr: [0, 0, 0.2 * Math.sin(t * 9)] };
    p.handR = { curl: 0.05, index: 0.0, thumb: 0.3 };
    return p;
  },
  phone: (x) => {
    const { seg, t, c } = x;
    const p = idle(x, 0);
    const k = ss(seg.t0, seg.t0 + 0.7, t) * (1 - ss(seg.t1 - 0.5, seg.t1, t));
    p.phone = 2;
    p.phoneOn = ss(seg.t0, seg.t0 + 0.3, t);
    const arm = phoneArm();
    p.armL = { ...arm, ik: { ...arm.ik!, w: k } };
    p.head = [lerpN(p.head[0], 0.45, k), p.head[1], p.head[2]];
    p.neck = [lerpN(p.neck[0], 0.12, k), p.neck[1], p.neck[2]];
    if (k < 0.5) look(p, seg.target, c, 1 - k * 2);
    return p;
  },
  binoculars: (x) => {
    const { seg, t } = x;
    const p = idle(x, 1);
    const k = ss(seg.t0 + 0.1, seg.t0 + 0.9, t) * (1 - ss(seg.t1 - 0.6, seg.t1, t));
    // both hands up to the eyes (binoculars sit on the face once raised)
    const head = p.head, aimDown = Math.max(-0.4, Math.min(0.3, head[0] + p.neck[0]));
    armIK(p, 1, [0.06, 0.42 - 0.2 * aimDown, 0.26], k, [0.7, -0.6, -0.1]);
    armIK(p, -1, [0.06, 0.42 - 0.2 * aimDown, 0.26], k, [0.7, -0.6, -0.1]);
    p.armL = { ...p.armL, wr: [0.2, 0, 0.6] };
    p.armR = { ...p.armR, wr: [0.2, 0, 0.6] };
    p.handL = { curl: 0.6, index: 0.5, thumb: 0.4 };
    p.handR = { curl: 0.6, index: 0.5, thumb: 0.4 };
    p.hold = { kind: "binoculars", side: -1, on: k };
    return p;
  },
  raise: (x) => {
    const { seg, t, c } = x;
    const p = idle(x, 1);
    const k = ss(seg.t0 + 0.1, seg.t0 + 0.8, t) * (1 - ss(seg.t1 - 0.6, seg.t1, t));
    // aim at the target when there is one (the arm points the light at it), else straight ahead
    const tgt: [number, number, number] = seg.target ? reachR(p, seg.target, c, 0.56) : seg.hold === "lantern" ? [0.1, 0.32, 0.42] : [0.0, 0.15, 0.5];
    armIK(p, -1, tgt, k, [0.5, -0.8, 0.0]);
    if (seg.hold === "flashlight") p.armR = { ...p.armR, wr: [-0.9 * k, 0, 0] };
    if (seg.target && k > 0.3) p.aim = seg.target;
    return p;
  },
  think: (x) => {
    const { seg, t } = x;
    const p = idle(x, 0.5);
    const k = ss(seg.t0 + 0.1, seg.t0 + 0.8, t) * (1 - ss(seg.t1 - 0.5, seg.t1, t));
    // the hand raised under the jaw for a moment, clear of the mouth (higher lands on the mouth / nose and reads as
    // biting nails or rubbing it; held, or every time, a hand in front of the face is all a close-up shows) — the head
    // and the eyes keep thinking
    const off = Math.min(seg.t1 - 0.5, seg.t0 + 2.2);
    const kh = seg.again ? 0 : ss(seg.t0 + 0.1, seg.t0 + 0.8, t) * (1 - ss(off, off + 0.6, t));
    if (kh > 0) armIK(p, -1, [0.0, 0.25, 0.235], kh, [0.6, -0.8, 0.2]);
    if (kh > 0) p.handR = { curl: 0.7, index: 0.4, thumb: 0.5 };
    p.head = [p.head[0] + 0.08 * k, p.head[1] + 0.12 * k, p.head[2] + 0.08 * k];
    return p;
  },
  nod: (x) => {
    const { seg, t } = x;
    const p = idle(x, 1);
    const u = t - seg.t0;
    p.head = [p.head[0] + 0.16 * Math.sin(u * 9) * Math.exp(-u * 1.1) * ss(0, 0.1, u), p.head[1], p.head[2]];
    return p;
  },
  shake: (x) => {
    const { seg, t } = x;
    const p = idle(x, 1);
    const u = t - seg.t0;
    p.head = [p.head[0], p.head[1] + 0.22 * Math.sin(u * 8) * Math.exp(-u * 0.9) * ss(0, 0.1, u), p.head[2]];
    return p;
  },
  sigh: (x) => {
    const { seg, t } = x;
    const p = idle(x, 0.6);
    const u = t - seg.t0, k = Math.sin(Math.min(1, u / 2.2) * Math.PI);
    p.chest = [p.chest[0] + 0.1 * k, p.chest[1], p.chest[2]];
    p.head = [p.head[0] + 0.18 * k, p.head[1], p.head[2]];
    p.breath = 1.4;
    return p;
  },
  talk: (x) => {
    const { seg, t, talk } = x;
    const p = idle(x, 1);
    const a = talk(t);
    p.talk = Math.min(0.6, a * 1.3);
    const g = Math.sin(t * 2.1 + seg.seed) * 0.5 + 0.5;
    p.armR = { ...p.armR, sw: p.armR.sw + 0.35 * a + 0.1 * g, el: p.armR.el + 0.7 * a + 0.3 * g, ab: p.armR.ab + 0.1 };
    p.handR = { curl: 0.2, index: 0.1, thumb: 0.3 };
    p.head = [p.head[0] + 0.05 * Math.sin(t * 5.3) * a, p.head[1] + 0.04 * Math.sin(t * 1.7), p.head[2]];
    return p;
  },
  shiver: (x) => {
    const { seg, t } = x;
    const p = idle(x, 0.6);
    const k = ss(seg.t0, seg.t0 + 0.6, t);
    armIK(p, 1, [-0.12, 0.1, 0.16], k, [0.6, -0.6, -0.2]);
    armIK(p, -1, [-0.12, 0.06, 0.18], k, [0.6, -0.6, -0.2]);
    p.chest = [p.chest[0] + 0.12 * k, p.chest[1], p.chest[2]];
    p.tremble = 0.8 * k;
    p.chest = [p.chest[0], p.chest[1] + 0.02 * Math.sin(t * 31) * k, p.chest[2]];
    return p;
  },
  sit: (x) => {
    const { seg, t, c } = x;
    // walk to the seat (if needed) is a separate segment; here: stand → sit over 1 s
    const k = ss(seg.t0, seg.t0 + 1.1, t);
    const st = base({ ...seg.to, posture: "stand" }, c), si = base(seg.to, c);
    const p = blend(st, si, k);
    p.chest = [p.chest[0] + 0.25 * Math.sin(k * Math.PI), p.chest[1], p.chest[2]];
    p.face = FACE[seg.face];
    look(p, seg.target, c, ss(seg.t0 + 0.8, seg.t0 + 1.5, t));
    return p;
  },
  standUp: (x) => {
    const { seg, t, c } = x;
    const k = ss(seg.t0, seg.t0 + 1.1, t);
    const si = base(seg.from, c), st = base(seg.to, c);
    const p = blend(si, st, k);
    p.chest = [p.chest[0] + 0.3 * Math.sin(k * Math.PI), p.chest[1], p.chest[2]];
    p.face = FACE[seg.face];
    look(p, seg.target, c, ss(seg.t0 + 0.9, seg.t0 + 1.6, t));
    return p;
  },
  write: (x) => {
    const { seg, t, c } = x;
    const p = base(seg.to, c);
    const k = ss(seg.t0, seg.t0 + 0.8, t) * (1 - ss(seg.t1 - 0.5, seg.t1, t));
    p.face = FACE[seg.face];
    // lean over the desk — further when the paper is further away — so the forearm comes over the edge
    const far = seg.target ? Math.max(0, Math.min(0.28, (Math.hypot(seg.target[0] - seg.to.x, seg.target[2] - seg.to.z) - 0.3) * 0.9)) : 0;
    p.spine = [p.spine[0] + far * 0.5 * k, p.spine[1], p.spine[2]];
    p.chest = [p.chest[0] + (0.22 + far * 0.5) * k, p.chest[1], p.chest[2]];
    p.neck = [p.neck[0] + 0.15 * k, p.neck[1], p.neck[2]];
    p.head = [p.head[0] + 0.32 * k, p.head[1], p.head[2]];
    // write on the target (papers on the desk) when given, else just in front of the chest;
    // the hand rests ON the paper (its centre ~5 cm above the surface), never inside the desk
    const base3: [number, number, number] = seg.target ? reachR(p, onSurface(p, [seg.target[0], seg.target[1] + 0.05, seg.target[2]], c, 0.6), c, 0.6) : [-0.08, -0.12, 0.36];
    const wob: [number, number, number] = [base3[0] + 0.04 * Math.sin(t * 7.3), base3[1] + 0.012 * Math.sin(t * 19), base3[2] + 0.02 * Math.sin(t * 2.3)];
    // elbow out to the side and up, as when writing at a desk
    armIK(p, -1, [wob[0], wob[1], wob[2]], k, [0.8, -0.35, -0.2]);
    armIK(p, 1, [0.14, -0.16, 0.32], 0.8 * k, [0.6, -0.8, -0.2]);
    p.hold = { kind: "pen", side: -1, on: 1 };
    p.handR = { curl: 0.7, index: 0.5, thumb: 0.6 };
    return p;
  },
  push: (x) => move(x, false),
  pickUp: (x) => reachTo(x, 0.55),
  putDown: (x) => reachTo(x, 0.55),
  give: (x) => reachTo(x, 0.42),
  receive: (x) => reachTo(x, 0.42),
  open: (x) => reachTo(x, 0.55),
  close: (x) => reachTo(x, 0.55),
  kneel: (x) => {
    const { seg, t, c } = x;
    const p = idle(x, 1);
    const k = ss(seg.t0, seg.t0 + 0.8, t) * (1 - ss(seg.t1 - 0.6, seg.t1, t));
    p.hipY = -0.42 * k;
    p.legL = { ...p.legL, sw: 1.25 * k, kn: 2.0 * k, an: -0.2 * k };
    p.legR = { ...p.legR, sw: -0.15 * k, kn: 1.75 * k, an: 0.9 * k };
    p.chest = [p.chest[0] + 0.12 * k, p.chest[1], p.chest[2]];
    if (seg.target) armIK(p, -1, reachR(p, seg.target, c, 0.55), ss(seg.t0 + 0.5, seg.t0 + 1.1, t) * (1 - ss(seg.t1 - 0.6, seg.t1, t)), [0.4, -0.8, -0.1]);
    return p;
  },
  lie: (x) => { const p = idle(x, 0.6); return p; },
  sleep: (x) => {
    const p = base(x.seg.to, x.c);
    p.face = FACE.sleep;
    p.breath = 0.7;
    return p;
  },
  wake: (x) => {
    const { seg, t, c } = x;
    const p = base(seg.to, c);
    const k = ss(seg.t0 + 0.2, seg.t0 + 1.4, t);
    p.face = { ...blend({ ...p, face: FACE.sleep }, { ...p, face: FACE[seg.face] }, k).face };
    look(p, seg.target, c, 0.6 * ss(seg.t0 + 1.0, seg.t0 + 2.0, t));
    return p;
  },
  sitUp: (x) => {
    const { seg, t, c } = x;
    const k = ss(seg.t0, seg.t0 + 1.4, t);
    const p = blend(lying(seg.from, c), bedSit(seg.to, c), k);
    p.face = FACE[seg.face];
    look(p, seg.target, c, ss(seg.t0 + 1.2, seg.t0 + 1.9, t));
    return p;
  },
};

/** reach toward the target (hold on it in the middle of the beat), used by pickUp / putDown / give / open */
function reachTo(x: ActCtx, reach: number): Pose {
  const { seg, t, c } = x;
  const p = idle(x, 1);
  const dur = Math.min(1.6, seg.t1 - seg.t0);
  const k = ss(seg.t0 + 0.05, seg.t0 + 0.55, t) * (1 - ss(seg.t0 + dur - 0.4, seg.t0 + dur, t));
  if (seg.target) {
    const low = seg.target[1] < p.y + 0.75;
    if (low) { p.spine = [p.spine[0] + 0.25 * k, p.spine[1], p.spine[2]]; p.chest = [p.chest[0] + 0.3 * k, p.chest[1], p.chest[2]]; p.hipY -= 0.08 * k; p.legL = { ...p.legL, kn: p.legL.kn + 0.4 * k }; p.legR = { ...p.legR, kn: p.legR.kn + 0.4 * k }; }
    armIK(p, -1, reachR(p, seg.target, c, reach), k, [0.4, -0.8, -0.1]);
  }
  p.handR = { curl: lerpN(0.35, 0.15, k), index: 0.1, thumb: 0.3 };
  return p;
}

function move(x: ActCtx, running: boolean): Pose {
  const { seg, t, c } = x;
  if (!seg.path || seg.moveT0 === undefined || seg.moveT1 === undefined) return idle(x);
  const t0 = seg.moveT0, t1 = seg.moveT1;
  if (t <= t1) {
    const w = running ? walkState(seg.path, t, t0, t1, 0.35, 0.45, 1.9, 1.0) : walkState(seg.path, t, t0, t1, 0.4, 0.6, 1.15, 0.85);
    // turn into the path first: the body comes round from where it faced (0.2–0.6 s by the angle) while the first
    // steps ease in — never a jump to the path direction on the first frame of the walk
    const dYaw = wrap(w.yaw - seg.from.yaw);
    const turnT = Math.min(0.6, Math.max(0.2, (Math.abs(dYaw) / Math.PI) * 0.6));
    const yaw = Math.abs(dYaw) > 0.05 ? seg.from.yaw + dYaw * ss(t0, t0 + turnT, t) : w.yaw;
    let p = base({ ...seg.from, x: w.p[0], z: w.p[1], yaw, posture: "stand" }, c);
    p = running ? runCycle(p, w.phase, Math.min(1, w.speed / 2.4)) : walkCycle(p, w.phase, w.moving);
    p.face = FACE[seg.face];
    look(p, seg.target, c, 0.7);
    return p;
  }
  // arrived: settle into the mark's facing
  const k = ss(t1, t1 + 0.6, t);
  const endYaw = Math.atan2(seg.path.at(seg.path.length).dir[0], seg.path.at(seg.path.length).dir[1]);
  const p = base({ ...seg.to, yaw: endYaw + wrap(seg.to.yaw - endYaw) * k }, c);
  shift(p, t, seg.seed);
  p.face = FACE[seg.face];
  look(p, seg.target, c, 0.8 * k);
  if (running) p.breath = 1.8;
  return p;
}

/** The pose for one segment at film time t, alive (blinks etc.). */
export const actPose = (x: ActCtx): Pose => {
  const { seg, t, c } = x;
  const p = ACT[seg.act](x);
  const r = seg.ride?.(t);
  if (r) {
    // rotate around the object's pivot, then move with it
    const ox = p.x - r.px, oz = p.z - r.pz, cs = Math.cos(r.dyaw), sn = Math.sin(r.dyaw);
    p.x = r.px + ox * cs + oz * sn + r.dx; p.z = r.pz - ox * sn + oz * cs + r.dz; p.yaw += r.dyaw;
  }

  }

const rot3 = (p: V3, r: V3 = [0, 0, 0]): V3 => {
  let [x, y, z] = p;
  const [ax, ay, az] = [r[0] * deg, r[1] * deg, r[2] * deg];
  // XYZ euler (three.js default): apply X, then Y, then Z in object space = R = Rz·Ry·Rx? three uses intrinsic XYZ → R = Rx·Ry·Rz applied to the vector
  let c = Math.cos(az), s = Math.sin(az); [x, y] = [x * c - y * s, x * s + y * c];
  c = Math.cos(ay); s = Math.sin(ay); [x, z] = [x * c + z * s, -x * s + z * c];
  c = Math.cos(ax); s = Math.sin(ax); [y, z] = [y * c - z * s, y * s + z * c];
  return [x, y, z];
};

const partBox = (p: PartSpec): Box3 => {
  const sc = p.scale ?? [1, 1, 1];
  let half: V3;
  switch (p.shape) {
    case "box": { const s = p.size ?? [0.1, 0.1, 0.1]; half = [s[0] / 2, s[1] / 2, s[2] / 2]; break; }
    case "plane": case "text": { const s = p.size ?? [0.5, 0.3]; half = [s[0] / 2, s[1] / 2, 0.005]; break; }
    case "cyl": case "cone": { const r = Math.max(p.r ?? 0.05, p.r2 ?? p.r ?? 0.05); half = [r, (p.h ?? 0.1) / 2, r]; break; }
    case "capsule": { const r = p.r ?? 0.05; half = [r, (p.h ?? 0.1) / 2 + r, r]; break; }
    case "sphere": { const r = p.r ?? 0.05; half = [r, r, r]; break; }
    case "torus": { const R = (p.R ?? 0.2) + (p.r ?? 0.02); half = [R, R, p.r ?? 0.02]; break; }
    case "lathe": { const pr = p.profile ?? [[0.1, 0], [0.1, 0.1]]; const R = Math.max(...pr.map((q) => q[0])); const y0 = Math.min(...pr.map((q) => q[1])), y1 = Math.max(...pr.map((q) => q[1])); return { min: add(p.at ?? [0, 0, 0], [-R * sc[0], y0 * sc[1], -R * sc[2]]), max: add(p.at ?? [0, 0, 0], [R * sc[0], y1 * sc[1], R * sc[2]]) }; }
    case "rod": { const a = p.from ?? [0, 0, 0], b = p.to ?? [0, 1, 0], r = p.r ?? 0.02; return { min: [Math.min(a[0], b[0]) - r, Math.min(a[1], b[1]) - r, Math.min(a[2], b[2]) - r], max: [Math.max(a[0], b[0]) + r, Math.max(a[1], b[1]) + r, Math.max(a[2], b[2]) + r] }; }
    case "pile": { const s = p.size ?? [0.4, 0.15, 0.3]; half = [s[0] / 2, s[1] / 2, s[2] / 2]; break; }
    default: half = [0.05, 0.05, 0.05];
  }
  half = [half[0] * sc[0], half[1] * sc[1], half[2] * sc[2]];
  const at = p.at ?? [0, 0, 0];
  const lo: V3 = [Infinity, Infinity, Infinity], hi: V3 = [-Infinity, -Infinity, -Infinity];
  for (const sx of [-1, 1]) for (const sy of [-1, 1]) for (const sz of [-1, 1]) {
    const q = rot3([half[0] * sx, half[1] * sy, half[2] * sz], p.rot);
    for (let k = 0; k < 3; k++) { lo[k] = Math.min(lo[k], at[k] + q[k]); hi[k] = Math.max(hi[k], at[k] + q[k]); }
  }
  return { min: lo, max: hi };
};

/** Expand repeat / mirror into concrete parts (the renderer uses the same expansion). */
export const expandParts = (parts: PartSpec[]): PartSpec[] => {
  const out: PartSpec[] = [];
  const rotY = (v: V3, a: number): V3 => [v[0] * Math.cos(a) + v[2] * Math.sin(a), v[1], -v[0] * Math.sin(a) + v[2] * Math.cos(a)];
  for (const p of parts) {
    const n = p.repeat?.n ?? 1;
    for (let i = 0; i < n; i++) {
      const step = p.repeat?.step ?? [0, 0, 0], sr = p.repeat?.stepRot ?? [0, 0, 0];
      const off: V3 = [step[0] * i, step[1] * i, step[2] * i];
      let q: PartSpec;
      if (p.repeat?.around) {
        const a = sr[1] * i * deg;
        const r0 = p.rot ?? [0, 0, 0];
        q = { ...p, repeat: undefined, at: add(rotY(p.at ?? [0, 0, 0], a), off), rot: [r0[0], r0[1] + sr[1] * i, r0[2]] };
        if (p.from && p.to) { q.from = add(rotY(p.from, a), off); q.to = add(rotY(p.to, a), off); }
      } else {
        q = { ...p, repeat: undefined, at: add(p.at ?? [0, 0, 0], off), rot: add(p.rot ?? [0, 0, 0], [sr[0] * i, sr[1] * i, sr[2] * i]) };
        if (p.from && p.to) { q.from = add(p.from, off); q.to = add(p.to, off); }
      }
      out.push(q);
      if (p.mirror) {
        const k = p.mirror === "x" ? 0 : 2;
        const m = (v: V3): V3 => { const c: V3 = [...v]; c[k] = -c[k]; return c; };
        const r = q.rot ?? [0, 0, 0];
        out.push({ ...q, at: m(q.at ?? [0, 0, 0]), from: q.from ? m(q.from) : undefined, to: q.to ? m(q.to) : undefined, rot: p.mirror === "x" ? [r[0], -r[1], -r[2]] : [-r[0], -r[1], r[2]] });
      }
    }
  }
  return out;
};

    if (Math.max(...sz) < 0.06 || sz.filter((v) => v < 0.03).length >= 2) continue; // thin rods / tiny bits never block a camera

export const buildWorld = (spec: WorldSpec, objectList: ObjectSpec[] = []): { data: WorldData; built: Built } => {
  const objects: Record<string, ObjectSpec> = Object.fromEntries(objectList.map((o) => [o.id, o]));
  const errs: string[] = [];
  const rooms = (spec.rooms ?? []).map((r) => ({ ...r, y: r.y ?? 0, height: r.height ?? 2.9 }));
  const roomById = Object.fromEntries(rooms.map((r) => [r.id, r]));
  // terrain
  const g = spec.ground;
  // under (and just around) a room the terrain sinks below its floor: never coplanar (flicker) or poking through
  const ground = g ? (x: number, z: number) => {
    const v = (g.y ?? 0) + (g.hills ?? 0) * fbm(x * 0.035 + (g.seed ?? 3), z * 0.035, 4) * 4;
    for (const r of rooms) if (x > r.rect[0] - 0.4 && x < r.rect[2] + 0.4 && z > r.rect[1] - 0.4 && z < r.rect[3] + 0.4) return Math.min(v, r.y - 0.03);
    return v;
  } : undefined;
  // paths flatten the terrain a little: compute height as is (characters follow it)
  const floorAt = (x: number, z: number) => {
    for (const r of rooms) if (x >= r.rect[0] - 0.01 && x <= r.rect[2] + 0.01 && z >= r.rect[1] - 0.01 && z <= r.rect[3] + 0.01) return r.y;
    return ground ? ground(x, z) : 0;
  };
  // openings
  const openings: Opening[] = [];
  for (const r of rooms) for (const o of r.openings ?? []) {
    const [x0, z0, x1, z1] = r.rect;
    const axis: "x" | "z" = o.wall === "north" || o.wall === "south" ? "x" : "z";
    const wallC = o.wall === "north" ? z0 : o.wall === "south" ? z1 : o.wall === "west" ? x0 : x1;
    const w = o.w ?? (o.type === "double-door" ? 1.5 : o.type === "window" ? 1.2 : o.type === "arch" ? 1.4 : 0.95);
    const h = o.h ?? (o.type === "window" ? 1.1 : 2.1);
    const sill = o.type === "window" ? o.sill ?? 0.95 : 0;
    const slug = (x: string) => x.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "");
    const kindName = o.type === "window" ? "jendela" : o.type === "arch" ? "lengkung" : "pintu";

    while (openings.some((q) => q.id === id)) id += "-2";

  }
  // objects
  const inRoomXZ = (x: number, z: number) => rooms.some((r) => x >= r.rect[0] && x <= r.rect[2] && z >= r.rect[1] && z <= r.rect[3]);

    const [oid, anchor = "top"] = ref.split(":");
    const host = insts.find((i) => i.id === oid);
    if (!host) { errs.push(`object placed "on": "${ref}" but "${oid}" is not defined before it`); return undefined; }
    const a = host.info.anchors[anchor] ?? [0, host.info.bounds[1], 0];
    return toWorld(host.pos, host.yaw, host.scale, a);
  };
  (spec.objects ?? []).forEach((u: ObjectUse, k) => {
    let info: ObjInfo;
    try { info = objectInfo(u.use, objects); } catch (e) { errs.push((e as Error).message); return; }
    const yaw = u.yaw ?? 0, scale = u.scale ?? 1;
    let pos: V3;
    if (u.on) { const s = surfaceOf(u.on); pos = s ? [u.at.length === 3 ? s[0] + (u.at as V3)[0] : s[0] + u.at[0], s[1], u.at.length === 3 ? s[2] + (u.at as V3)[2] : s[2] + (u.at as V2)[1]] : [0, 0, 0]; }
    else if (u.at.length === 3) { const a3 = u.at as V3; pos = inRoomXZ(a3[0], a3[2]) || !ground ? a3 : [a3[0], ground(a3[0], a3[2]) + a3[1], a3[2]]; }
    else pos = [u.at[0], floorAt(u.at[0], u.at[1]), u.at[1]];

    insts.push({ id, use: u.use, pos, yaw, scale, info, variant: u.variant, color: u.color, states: { ...info.states, ...(u.states ?? {}) }, box: placedBox(info, pos, yaw, scale), parts: partBoxes(u.use, objects, pos, yaw, scale) });
  });
  // paths
  const paths = spec.paths ?? [];
  const nearPath = (x: number, z: number, m: number) => paths.some((p) => { for (let i = 1; i < p.points.length; i++) { const [ax, az] = p.points[i - 1], [bx, bz] = p.points[i]; const dx = bx - ax, dz = bz - az, L = dx * dx + dz * dz || 1, k = Math.max(0, Math.min(1, ((x - ax) * dx + (z - az) * dz) / L)); if (Math.hypot(x - ax - dx * k, z - az - dz * k) < (p.width ?? 1.2) / 2 + m) return true; } return false; });

  // scatter
  const scatter = (spec.scatter ?? []).map((sc: ScatterSpec, k) => {
    let info: ObjInfo;
    try { info = objectInfo(sc.use, objects); } catch (e) { errs.push((e as Error).message); return null; }
    const r = rng(1000 + (sc.seed ?? k * 77)), items: { pos: V3; yaw: number; scale: number }[] = [];
    const [ax0, az0, ax1, az1] = sc.area, minD = sc.minDist ?? Math.max(info.bounds[0], info.bounds[2]) * 0.8;
    for (let tries = 0; items.length < sc.count && tries < sc.count * 40; tries++) {
      const x = ax0 + r() * (ax1 - ax0), z = az0 + r() * (az1 - az0);
      const av = sc.avoid ?? ["paths", "rooms", "objects"];
      // trees keep only their trunk off the path (crowns may overhang it, like a real forest track)
      if (av.includes("paths") && nearPath(x, z, sc.use.startsWith("tree") ? 1.3 : Math.max(info.bounds[0], info.bounds[2]) / 2 + 0.3)) continue;
      if (av.includes("rooms") && inRoom(x, z, 0.5)) continue;
      if (av.includes("objects") && insts.some((i) => x > i.box.min[0] - 0.6 && x < i.box.max[0] + 0.6 && z > i.box.min[2] - 0.6 && z < i.box.max[2] + 0.6)) continue;
      if (items.some((q) => Math.hypot(q.pos[0] - x, q.pos[2] - z) < minD)) continue;
      const s = (sc.scale?.[0] ?? 0.8) + r() * ((sc.scale?.[1] ?? 1.2) - (sc.scale?.[0] ?? 0.8));
      items.push({ pos: [x, floorAt(x, z) - 0.02, z], yaw: r() * Math.PI * 2, scale: s });
    }
    return { use: sc.use, info, items, variant: sc.variant };
  }).filter((x): x is NonNullable<typeof x> => !!x);
  // fixtures
  // outdoors (not inside a room) a fixture's y is its height above the ground
  const lift = (p: V3): V3 => (ground && !inRoomXZ(p[0], p[2]) ? [p[0], ground(p[0], p[2]) + p[1], p[2]] : p);
  // a lamp post given at its foot (y < 1.5) gets its head at the post height
  const head = (f: FixtureSpec): V3 => (f.type === "lamp-post" && f.at[1] < 1.5 ? [f.at[0], f.at[1] + 4.2, f.at[2]] : f.at);
  const fixtures = (spec.lights ?? []).map((f, i) => ({ ...f, at: lift(head(f)), to: f.to ? lift(f.to) : undefined, target: f.target ? lift(f.target) : undefined, id: f.id ?? `${f.type}-${i + 1}` }));
  // world bounds
  let wb: Box3 = { min: [-5, -1, -5], max: [5, 4, 5] };
  if (rooms.length) wb = rooms.map((r) => ({ min: [r.rect[0], r.y, r.rect[1]] as V3, max: [r.rect[2], r.y + r.height, r.rect[3]] as V3 })).reduce(union);
  if (g) wb = union(wb, { min: [-g.size / 2, -2, -g.size / 2], max: [g.size / 2, 20, g.size / 2] });
  // ── navigation grid: walkable floor minus solids ───────────────────────────
  const cell = 0.2, R = 0.28;
  const nx0 = Math.floor(wb.min[0]) - 1, nz0 = Math.floor(wb.min[2]) - 1;
  const nx = Math.ceil((wb.max[0] - nx0 + 1) / cell), nz = Math.ceil((wb.max[2] - nz0 + 1) / cell);
  const free = new Uint8Array(nx * nz);
  // per-part boxes that reach body height block walking (a stall's canopy or a hanging lamp does not)
  const solidsXZ = insts.filter((i) => !i.info.walkable && i.box.max[1] - i.box.min[1] > 0.12).flatMap((i) => {
    const fl0 = i.pos[1];
    const ps = (i.parts.length ? i.parts : [i.box]).filter((b) => b.min[1] < fl0 + 1.7 && b.max[1] > fl0 + 0.12);
    return ps.length ? ps : [];
  });
  // trees block with their trunk only; walkable undergrowth (ferns, grass) does not block
  const scatterSolids = scatter.filter((s) => !s.info.walkable).flatMap((s) => s.items.map((it): Box3 => {
    if (!s.use.startsWith("tree")) return placedBox(s.info, it.pos, it.yaw, it.scale);
    const r = 0.3 * it.scale;
    return { min: [it.pos[0] - r, it.pos[1], it.pos[2] - r], max: [it.pos[0] + r, it.pos[1] + 4.5 * it.scale, it.pos[2] + r] };
  }));
  const bld = (spec.buildings ?? []).map((b) => b.rect);
  for (let j = 0; j < nz; j++) for (let i = 0; i < nx; i++) {
    const x = nx0 + (i + 0.5) * cell, z = nz0 + (j + 0.5) * cell;
    let ok = false;
    if (rooms.length) {
      for (const r of rooms) if (x > r.rect[0] + R && x < r.rect[2] - R && z > r.rect[1] + R && z < r.rect[3] - R) ok = true;
      // doorways connect rooms
      for (const o of openings) if (o.type !== "window" && (o.axis === "x" ? Math.abs(z - o.wallC) < 0.5 && Math.abs(x - o.x) < o.w / 2 - 0.12 : Math.abs(x - o.wallC) < 0.5 && Math.abs(z - o.z) < o.w / 2 - 0.12)) ok = true;
    }
    if (g && !inRoom(x, z, 0.3)) ok = Math.abs(x) < g.size / 2 - 1 && Math.abs(z) < g.size / 2 - 1;
    if (ok && bld.some((b) => x > b[0] - R && x < b[2] + R && z > b[1] - R && z < b[3] + R)) ok = false;
    if (ok && solidsXZ.some((b) => x > b.min[0] - R && x < b.max[0] + R && z > b.min[2] - R && z < b.max[2] + R)) ok = false;
    if (ok && scatterSolids.some((b) => x > b.min[0] - R * 0.6 && x < b.max[0] + R * 0.6 && z > b.min[2] - R * 0.6 && z < b.max[2] + R * 0.6)) ok = false;
    free[j * nx + i] = ok ? 1 : 0;
  }
  const nav: Nav = { x0: nx0, z0: nz0, cell, nx, nz, free, floor: floorAt };
  const built: Built = { spec, rooms, openings, insts, scatter, fixtures, ground, bounds: wb, nav, objects };

  // ── places ──────────────────────────────────────────────────────────────────
  const places: Record<string, Place> = {};
  for (const [pid, ps] of Object.entries(spec.places)) {
    const pr = (ps.rooms ?? []).map((r) => roomById[r]).filter(Boolean);
    let area = ps.area;
    if (!area && pr.length) area = [Math.min(...pr.map((r) => r.rect[0])), Math.min(...pr.map((r) => r.rect[1])), Math.max(...pr.map((r) => r.rect[2])), Math.max(...pr.map((r) => r.rect[3]))];
    if (!area) area = [wb.min[0], wb.min[2], wb.max[0], wb.max[2]];
    const kind = ps.kind ?? (pr.length ? "interior" : "exterior");
    const inArea = (x: number, z: number, m = 0) => x >= area![0] - m && x <= area![2] + m && z >= area![1] - m && z <= area![3] + m;
    const fl = (x: number, z: number) => floorAt(x, z);
    // marks: explicit + automatic (object anchors, doors, room centres)
    const marks: Record<string, Mark> = {};
    for (const r of pr.length ? pr : rooms.filter((r) => inArea((r.rect[0] + r.rect[2]) / 2, (r.rect[1] + r.rect[3]) / 2))) marks[`${r.id}:center`] = { p: [(r.rect[0] + r.rect[2]) / 2, (r.rect[1] + r.rect[3]) / 2], y: r.y, doc: `middle of ${r.id}` };
    for (const o of openings) {
      if (o.type === "window" || !inArea(o.x, o.z, 1)) continue;
      const r = roomById[o.room];
      const inward = o.wall === "north" ? [0, 1] : o.wall === "south" ? [0, -1] : o.wall === "west" ? [1, 0] : [-1, 0];
      const facing = Math.atan2(-inward[0], -inward[1]);
      marks[`${o.id}:in`] = { p: [o.x + inward[0] * 0.75, o.z + inward[1] * 0.75], y: r.y, yaw: facing, doc: `inside ${o.room}, facing ${o.id}` };
      marks[`${o.id}:out`] = { p: [o.x - inward[0] * 0.75, o.z - inward[1] * 0.75], y: r.y, yaw: facing + Math.PI, doc: `outside ${o.room} at ${o.id}, facing it` };
    }
    for (const i of insts) {
      if (!inArea(i.pos[0], i.pos[2], 0.5)) continue;
      const a = i.info.anchors;
      const frontLocal: V3 = a.front ?? [0, 0, i.info.bounds[2] / 2 * (i.info.center[2] >= 0 ? 1 : 1) + 0.55];
      const fw = toWorld(i.pos, i.yaw, i.scale, frontLocal);
      marks[`${i.id}:front`] = { p: [fw[0], fw[2]], y: fl(fw[0], fw[2]), yaw: i.yaw + Math.PI, doc: `in front of ${i.id}, facing it` };
      if (a.seat) { const sw = toWorld(i.pos, i.yaw, i.scale, a.seat); marks[`${i.id}:seat`] = { p: [sw[0], sw[2]], y: fl(sw[0], sw[2]), yaw: i.yaw, seat: sw[1] - fl(sw[0], sw[2]), doc: `sitting on ${i.id}` }; }
      if (a.push) { const pw = toWorld(i.pos, i.yaw, i.scale, [a.push[0], 0, a.push[2] - 0.38]); marks[`${i.id}:push`] = { p: [pw[0], pw[2]], y: fl(pw[0], pw[2]), yaw: i.yaw, doc: `behind ${i.id}, hands on its handles` }; }
      // a bed (anchors "bed" = mattress top centre, "head" = pillow end): lie / sleep / wake / sitUp there
      if (a.bed) { const bw = toWorld(i.pos, i.yaw, i.scale, a.bed), hw = toWorld(i.pos, i.yaw, i.scale, a.head ?? [a.bed[0], a.bed[1], a.bed[2] - 0.8]); marks[`${i.id}:bed`] = { p: [bw[0], bw[2]], y: fl(bw[0], bw[2]), yaw: i.yaw, bed: { head: [hw[0], hw[2]], y: bw[1] }, doc: `lying in ${i.id}` }; }
      if (a.behind) { const bw = toWorld(i.pos, i.yaw, i.scale, a.behind); marks[`${i.id}:behind`] = { p: [bw[0], bw[2]], y: fl(bw[0], bw[2]), yaw: i.yaw, doc: `behind ${i.id} (the seller's side), facing over it` }; }
    }
    for (const [k, m] of Object.entries(ps.marks ?? {})) {
      const y = m.at.length === 3 ? (m.at as V3)[1] : fl(m.at[0], (m.at as V2)[1]);
      const p: [number, number] = m.at.length === 3 ? [(m.at as V3)[0], (m.at as V3)[2]] : [m.at[0], (m.at as V2)[1]];
      marks[k] = { p, y, yaw: m.yaw, seat: m.seat, bed: m.bed ? { head: m.bed.head, y: m.bed.y } : undefined, doc: m.doc };
    }
    // POIs: explicit + objects, openings, fixtures
    const pois: Record<string, V3> = {};
    for (const i of insts) if (inArea(i.pos[0], i.pos[2], 2)) pois[i.id] = [i.pos[0], i.box.max[1] - (i.box.max[1] - i.box.min[1]) * 0.35, i.pos[2]];
    for (const o of openings) if (inArea(o.x, o.z, 2)) pois[o.id] = [o.x, o.y + o.sill + o.h * 0.55, o.z];
    for (const f of fixtures) if (inArea(f.at[0], f.at[2], 2)) pois[f.id] = f.at;
    Object.assign(pois, ps.pois ?? {});
    // camera space and solids
    const yTop = pr.length ? Math.max(...pr.map((r) => r.y + r.height)) - 0.2 : 40;
    const yBot = pr.length ? Math.min(...pr.map((r) => r.y)) + 0.3 : (ground ? -5 : 0.3);
    const m = kind === "interior" ? -0.25 : 25;
    const solids = insts.filter((i) => !i.info.walkable && inArea(i.pos[0], i.pos[2], 8)).flatMap((i) => (i.parts.length ? i.parts : [i.box]));
    for (const b of spec.buildings ?? []) solids.push({ min: [b.rect[0], (b.y ?? 0) - 1, b.rect[1]], max: [b.rect[2], (b.y ?? 0) + b.height, b.rect[3]] });
    // seen from outside, a closed room is a building: cameras of an exterior place stay out of it
    if (kind === "exterior") for (const r of rooms) if (!pr.includes(r) && !(r.open?.length)) solids.push({ min: [r.rect[0] - 0.1, r.y - 0.5, r.rect[1] - 0.1], max: [r.rect[2] + 0.1, r.y + r.height + 0.3, r.rect[3] + 0.1] });
    for (const s of scatter) for (const it of s.items) {
      if (!inArea(it.pos[0], it.pos[2], 15) || s.info.bounds[1] <= 1.2) continue;
      // trees: the trunk blocks (crowns are above the action); other tall things: their box
      if (s.use.startsWith("tree")) { const r = 0.3 * it.scale; solids.push({ min: [it.pos[0] - r, it.pos[1], it.pos[2] - r], max: [it.pos[0] + r, it.pos[1] + 4.5 * it.scale, it.pos[2] + r] }); }
      else solids.push(placedBox(s.info, it.pos, it.yaw, it.scale));
    }
    // foliage the camera must not sit in (tree crowns, bushes) — scatter and placed built-ins
    const crownBox = (use: string, pos: V3, yaw: number, scale: number): Box3 | null => {
      const c = BUILTIN[use]?.crown;
      if (!c) return null;
      const lo: V3 = [Infinity, pos[1] + c[0][1] * scale, Infinity], hi: V3 = [-Infinity, pos[1] + c[1][1] * scale, -Infinity];
      for (const x of [c[0][0], c[1][0]]) for (const z of [c[0][2], c[1][2]]) { const w = toWorld(pos, yaw, scale, [x, 0, z]); lo[0] = Math.min(lo[0], w[0]); hi[0] = Math.max(hi[0], w[0]); lo[2] = Math.min(lo[2], w[2]); hi[2] = Math.max(hi[2], w[2]); }
      return { min: lo, max: hi };
    };
    const soft: Box3[] = [];
    for (const sc of scatter) for (const it of sc.items) if (inArea(it.pos[0], it.pos[2], 20)) { const b = crownBox(sc.use, it.pos, it.yaw, it.scale); if (b) soft.push(b); }
    for (const i of insts) if (inArea(i.pos[0], i.pos[2], 20)) { const b = crownBox(i.use, i.pos, i.yaw, i.scale); if (b) soft.push(b); }
    const cx = (area[0] + area[2]) / 2, cz = (area[1] + area[3]) / 2, rad = Math.max(area[2] - area[0], area[3] - area[1]) / 2;
    const est = ps.establish;
    const firstMark = Object.values(marks)[0];
    places[pid] = {
      id: pid, doc: ps.doc, kind, floor: fl(cx, cz), marks, pois,
      box: { min: [area[0] - m, yBot, area[1] - m], max: [area[2] + m, yTop, area[3] + m] },
      zones: kind === "interior" && pr.length ? pr.map((r) => ({ min: [r.rect[0] + 0.25, r.y + 0.3, r.rect[1] + 0.25] as V3, max: [r.rect[2] - 0.25, r.y + r.height - 0.2, r.rect[3] - 0.25] as V3 })) : undefined,
      solids, soft, ground: ground && kind === "exterior" ? ground : undefined,
      establish: est ? { center: [est[0], est[1], est[2]], radius: est[3], height: est[4], yaw: est[5] } : { center: [cx, fl(cx, cz) + 1.2, cz], radius: kind === "interior" ? Math.min(rad, 4) : Math.max(8, rad * 0.9), height: kind === "interior" ? 1.0 : Math.max(4, rad * 0.4), yaw: 0.7 },
      view: ps.view ?? Object.keys(pois)[0] ?? "",
      window: undefined,
      ambience: ps.ambience ?? (kind === "interior" ? "room" : "outdoor"), surface: (ps.surface ?? (kind === "interior" ? "tile" : "grass")) as Place["surface"], reverb: kind === "interior" ? 0.35 : 0.05,
    };
    void firstMark;
  }

  // wall pieces (full height, between openings) block the view: line-of-sight checks for cameras
  const occl: Box3[] = [];
  for (const r of rooms) {
    const [x0, z0, x1, z1] = r.rect, Tw = 0.15;
    const sides = [{ s: "north", ax: "x", c: z0 - Tw / 2, a0: x0, a1: x1 }, { s: "south", ax: "x", c: z1 + Tw / 2, a0: x0, a1: x1 }, { s: "west", ax: "z", c: x0 - Tw / 2, a0: z0, a1: z1 }, { s: "east", ax: "z", c: x1 + Tw / 2, a0: z0, a1: z1 }] as const;
    for (const w of sides) {
      if (r.open?.includes(w.s)) continue;
      const holes = openings.filter((o) => o.axis === w.ax && Math.abs(o.wallC - (w.ax === "x" ? (w.s === "north" ? z0 : z1) : (w.s === "west" ? x0 : x1))) < 0.05 && o.type !== "window").map((o) => [(w.ax === "x" ? o.x : o.z) - o.w / 2, (w.ax === "x" ? o.x : o.z) + o.w / 2]).sort((a, b) => a[0] - b[0]);
      let cur: number = w.a0;
      const piece = (a: number, b: number) => { if (b - a > 0.2) occl.push(w.ax === "x" ? { min: [a, r.y, w.c - Tw / 2], max: [b, r.y + r.height, w.c + Tw / 2] } : { min: [w.c - Tw / 2, r.y, a], max: [w.c + Tw / 2, r.y + r.height, b] }); };
      for (const [a, b] of holes) { piece(cur, a); cur = Math.max(cur, b); }
      piece(cur, w.a1);
    }
  }
  for (const P of Object.values(places)) P.occluders = occl;
  // the swing of every door leaf: cameras stay out of it (an open leaf would fill the frame)
  const swings: Box3[] = openings.filter((o) => o.type === "door" || o.type === "double-door").map((o) => {
    const into = o.wall === "north" || o.wall === "west" ? 1 : -1, sw = (o.swing ?? 1) * into;
    const lw = o.type === "double-door" ? o.w / 2 : o.w, d0 = Math.min(0, sw * lw), d1 = Math.max(0, sw * lw);
    return o.axis === "x" ? { min: [o.x - o.w / 2, o.y, o.z + d0], max: [o.x + o.w / 2, o.y + o.h, o.z + d1] } : { min: [o.x + d0, o.y, o.z - o.w / 2], max: [o.x + d1, o.y + o.h, o.z + o.w / 2] };
  });
  for (const P of Object.values(places)) P.swings = swings;

  // ── light rig per place ─────────────────────────────────────────────────────
  const light = (place: string, time: string, s: SetState, t: number): LightState => {
    const P = places[place];
    const Tl = TIMES[time] ?? TIMES.night;
    const interior = P?.kind === "interior";
    const c: V3 = P ? [(P.box.min[0] + P.box.max[0]) / 2, 1.5, (P.box.min[2] + P.box.max[2]) / 2] : [0, 1.5, 0];
    const near = fixtures.filter((f) => f.type !== "spot").map((f) => ({ f, d: Math.hypot(f.at[0] - c[0], f.at[2] - c[2]) })).sort((a, b) => a.d - b.d).slice(0, 5);
    const pts: PointL[] = near.map(({ f }) => {
      const k = FIX[f.type], lv = fixtureLevel(f, s, t) * (f.i ?? 1);
      const pos: V3 = f.type === "string" && f.to ? [(f.at[0] + f.to[0]) / 2, (f.at[1] + f.to[1]) / 2 - 0.2, (f.at[2] + f.to[2]) / 2] : f.type === "panel" || f.type === "tube" ? [f.at[0], f.at[1] - 0.5, f.at[2]] : f.at;
      return { pos, color: f.color ?? k.color, i: k.i * lv * (Tl.dark > 0.3 || interior ? 1 : 0.3), dist: k.dist };
    });
    while (pts.length < 5) pts.push(P_OFF);
    // a soft fill so night interiors never crush
    pts.push({ pos: [c[0], (P?.box.max[1] ?? 3) - 0.3, c[2]], color: interior ? "#8aa0c0" : Tl.hemiSky, i: interior ? 0.35 : 0.5, dist: interior ? 7 : 20 });
    const spotF = fixtures.find((f) => f.type === "spot");
    const spot0: SpotL = spotF ? { pos: spotF.at, target: spotF.target ?? [spotF.at[0], 0, spotF.at[2]], color: spotF.color ?? FIX.spot.color, i: FIX.spot.i * fixtureLevel(spotF, s, t) * (spotF.i ?? 1), angle: 0.6, penumbra: 0.6, dist: 20 } : S_OFF;
    const box = P ? { min: [P.box.min[0] - 2, P.box.min[1] - 1, P.box.min[2] - 2] as V3, max: [P.box.max[0] + 2, Math.min(P.box.max[1], (interior ? 4 : 18)), P.box.max[2] + 2] as V3 } : wb;
    const fog = spec.fog ?? (interior ? 0.1 : 0.3);
    return {
      bg: Tl.bg, fog: { color: Tl.fog, density: interior ? 0.006 + 0.01 * fog : 0.004 + 0.02 * fog }, env: Tl.env * (interior ? 0.5 : 1) + 0.02,
      hemi: { sky: interior ? spec.ambient ?? "#9aa8c0" : Tl.hemiSky, ground: Tl.hemiGround, i: interior ? 0.36 + Tl.hemi * 0.25 : Tl.hemi },
      moon: { dir: spec.sky?.sun ?? Tl.dir, color: Tl.key, i: Tl.keyI * (interior ? 0.8 : 1), box },
      spots: [spot0, S_OFF],
      pts: pts.slice(0, 6) as LightState["pts"],
    };
  };

  const props: Record<string, PropDoc> = {};
  for (const [k, p] of Object.entries(spec.props ?? {})) props[k] = { default: p.default, doc: p.doc, sfx: p.sfx ?? (p.door ? { up: "door-open", down: "door-close" } : p.lights ? { up: "click", down: "click" } : undefined) };
  // every door gets an open prop automatically: "<door id>" 0..1
  for (const o of openings) if ((o.type === "door" || o.type === "double-door") && !props[o.id]) props[o.id] = { default: 0, doc: `${o.id} (0 shut, 1 open)`, sfx: { up: "door-open", down: "door-close" } };
  for (const i of insts) for (const [k, v] of Object.entries(i.states)) props[`${i.id}.${k}`] = { default: v, doc: `${i.id} ${k} (0..1)` };
  // every object can appear / leave on a beat: "set": { "<id>.visible": 0 } (start hidden: "states": { "visible": 0 })
  for (const i of insts) if (!props[`${i.id}.visible`]) props[`${i.id}.visible`] = { default: 1, doc: `${i.id} on screen (1) or gone (0)` };
  for (const f of fixtures) if (f.prop && !props[f.prop]) props[f.prop] = { default: 1, doc: `lights ${f.prop} (0 off, 1 on)`, sfx: { up: "click", down: "click" } };
  if (errs.length) throw new Error(errs.join("\n"));
  const data: WorldData = { id: spec.id, doc: spec.doc, places, times: spec.times, props, light };
  return { data, built };
};

// ── navigation ───────────────────────────────────────────────────────────────

/** A* on the walk grid from a to b, then string-pulled to a few waypoints. Falls back to the straight line. */
export const navPath = (nav: Nav, a: V2, b: V2): V2[] => {
  const { x0, z0, cell, nx, nz, free } = nav;
  const ci = (x: number) => Math.max(0, Math.min(nx - 1, Math.floor((x - x0) / cell))), cj = (z: number) => Math.max(0, Math.min(nz - 1, Math.floor((z - z0) / cell)));
  const nearestFree = (i: number, j: number) => {
    if (free[j * nx + i]) return [i, j];
    for (let r = 1; r < 12; r++) for (let dj = -r; dj <= r; dj++) for (let di = -r; di <= r; di++) { const ii = i + di, jj = j + dj; if (ii >= 0 && jj >= 0 && ii < nx && jj < nz && free[jj * nx + ii]) return [ii, jj]; }
    return [i, j];
  };
  const [si, sj] = nearestFree(ci(a[0]), cj(a[1])), [ti, tj] = nearestFree(ci(b[0]), cj(b[1]));
  const N = nx * nz, g = new Float32Array(N).fill(Infinity), came = new Int32Array(N).fill(-1), open: number[] = [];

// ── the boat and its lamps (the "light at sea") ─────────────────────────────

const Boat: React.FC<{ t: number; amp: number; seaLight: number; boat: number; fog: number; cam: Cam }> = ({ t, amp, seaLight, boat, fog, cam }) => {
  const hy = waveHeight(BOAT[0], BOAT[2], t, amp) * 0.85;
  const [sx, sz] = waveSlope(BOAT[0], BOAT[2], t, amp);
  const hull = phys("hull", { color: "#2c4a6a", roughness: 0.6 }), stripe = phys("hullStripe", { color: "#e6e0d0", roughness: 0.6 }), wood = MAT.wood("#5a4030");
  // blink: the lamps dip behind the swells now and then (their reflection on the water comes from the point light)
  const blink = 0.55 + 0.45 * Math.min(1, Math.max(0, 0.5 + 0.9 * (Math.sin(t * 1.25) + 0.6 * Math.sin(t * 0.71 + 1.0))));
  const on = Math.max(seaLight * blink, boat);
  const lampM = phys(`boatLamp${Math.round(on * 10)}`, { color: "#fff8e8", emissive: "#ffe4b0", emissiveIntensity: 8 * on, roughness: 0.2 });
  const dist = Math.hypot(cam.pos[0] - SEA_LIGHT[0], cam.pos[2] - SEA_LIGHT[2]);
  const far = Math.min(1, Math.max(0, (dist - 30) / 120));
  return (
    <group>
      <group position={[BOAT[0], hy, BOAT[2]]} rotation={[sz * 0.8, BOAT_YAW, -sx * 0.8]}>
        <mesh geometry={cached("boatHull", () => new IcosahedronGeometry(1, 3))} material={hull} scale={[1.15, 0.7, 5.0]} position={[0, 0.25, 0]} castShadow />
        <mesh geometry={box(1.7, 0.06, 7.6, 0.03)} material={wood} position={[0, 0.72, 0]} castShadow receiveShadow />
        <mesh geometry={box(2.2, 0.12, 0.12, 0.02)} material={stripe} position={[0, 0.62, 3.2]} />
        <mesh geometry={box(1.3, 0.85, 1.5, 0.05)} material={phys("cabin", { color: "#c9b48a", roughness: 0.7 })} position={[0, 1.18, 2.0]} castShadow />
        <mesh geometry={box(1.55, 0.06, 1.8, 0.02)} material={phys("cabinRoof", { color: "#3a5a7a", roughness: 0.6 })} position={[0, 1.64, 2.0]} rotation={[0.08, 0, 0]} castShadow />
        <mesh geometry={cyl(0.07, 0.09, 5.2, 8)} material={wood} position={[0, 2.95, -0.6]} castShadow />
        <mesh geometry={cyl(0.04, 0.04, 3.4, 6)} material={wood} position={[0, 4.2, -0.6]} rotation={[0, 0, Math.PI / 2]} />
        {[-1.5, -0.5, 0.5, 1.5].map((x) => <mesh key={x} geometry={sphere(0.14, 12, 8)} material={lampM} position={[x, 4.05, -0.6]} />)}
        {/* outriggers (cadik) */}
        {[1, -1].map((k) => (
          <group key={k}>
            <mesh geometry={cyl(0.12, 0.12, 8.5, 8)} material={wood} position={[k * 2.6, 0.1, 0]} rotation={[Math.PI / 2, 0, 0]} />
            {[-2.2, 2.2].map((z) => <mesh key={z} geometry={cyl(0.05, 0.05, 2.7, 6)} material={wood} position={[k * 1.35, 0.55, z]} rotation={[0, 0, Math.PI / 2 - k * 0.2]} />)}
          </group>
        ))}
      </group>
      {/* the light as seen from far away: glow + halo in fog + its streak on the water */}
      <Glow pos={[SEA_LIGHT[0], SEA_LIGHT[1] + hy, SEA_LIGHT[2]]} size={(3 + 4 * fog) * (0.4 + 0.6 * far)} color="#ffe8c4" i={on * (0.55 + 0.45 * far)} />
      <Glow pos={[SEA_LIGHT[0], SEA_LIGHT[1] + hy, SEA_LIGHT[2]]} size={(6 + 7 * fog) * far} color="#c8d8ff" i={on * fog * 0.14 * far} />
    </group>
  );
};
