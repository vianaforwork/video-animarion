VOYNANN Motion - packed source files of this folder. Do not edit by hand.
The skill upload in the Claude app accepts at most 200 files, so the source files are packed here.
`node <skill folder>/new-project.mjs <project folder> --kit=<style>` unpacks them into the project.
Format: per file one line with four '=' signs, the byte length and the path, then exactly that many bytes.

==== 164 project/assets/objects/README.md
Your own objects (`<id>.json`, see the skill's references/objects.md). They override library objects with the same id and are reused by every film in this project.

==== 711 project/README.md
# CINÉSTORY 3D project

Made from the VOYNANN Motion skill, style 09 CINÉSTORY 3D (portrait 9:16 by default). Per video you write `stories/<id>/story.json`;
`node scripts/make.mjs stories/<id>/story.json --final` does the rest (plan → sound → render → QA).

- `src/director` — story.json → film plan (acting, cameras, light, grade, UI, sound cues)
- `src/cine` — character rig, light rig, post/grade, FX, motion-graphics kit, set-building kit
- `src/worlds` — `coast`, `kos`, and the `mind` studio
- `src/film` — the generic renderer · `scripts/` — transcribe, TTS, sound, render, QA

Docs: the skill folder's `references/cinestory.md`, `references/cinestory/` and `kits/cinestory/KIT.md`.

import { AdditiveBlending, CanvasTexture, Color, DoubleSide, PointsMaterial, ShaderMaterial, type Side } from "three";

const h3 = (x: number, y: number, z: number) => {
  let n = Math.sin(x * 127.1 + y * 311.7 + z * 74.7) * 43758.5453;
  return n - Math.floor(n);
};
const sm = (t: number) => t * t * (3 - 2 * t);
/** Smooth 3D value noise in 0..1. */
export const noise3 = (x: number, y: number, z: number) => {
  const xi = Math.floor(x), yi = Math.floor(y), zi = Math.floor(z);
  const xf = sm(x - xi), yf = sm(y - yi), zf = sm(z - zi);
  const l = (a: number, b: number, t: number) => a + (b - a) * t;
  const c = (dx: number, dy: number, dz: number) => h3(xi + dx, yi + dy, zi + dz);
  return l(
    l(l(c(0, 0, 0), c(1, 0, 0), xf), l(c(0, 1, 0), c(1, 1, 0), xf), yf),
    l(l(c(0, 0, 1), c(1, 0, 1), xf), l(c(0, 1, 1), c(1, 1, 1), xf), yf),
    zf,
  );
};

export const glowMat = (o: GlowOpts) =>

    transparent: true,
    depthWrite: false,
    blending: AdditiveBlending,
    side: o.side ?? DoubleSide,
    fog: false,
    uniforms: {
      uColor: { value: new Color(o.color) },
      uRim: { value: o.rim ?? 1.0 },
      uCore: { value: o.core ?? 0.15 },
      uPow: { value: o.power ?? 2.0 },
      uOpacity: { value: o.opacity ?? 1 },

// PRECISION easing vocabulary. Every curve is a pure function of normalised time, so the
// whole film stays deterministic: same frame in, same pose out.

export const clamp = (x: number, lo = 0, hi = 1) => Math.min(hi, Math.max(lo, x));
export const mix = (a: number, b: number, t: number) => a + (b - a) * t;
export const remap = (x: number, a: number, b: number) => clamp((x - a) / (b - a));

export const smoothstep = (t: number) => {
  const x = clamp(t);
  return x * x * (3 - 2 * x);
};

/** C2-continuous: zero velocity and acceleration at both ends. The house curve. */
export const smootherstep = (t: number) => {
  const x = clamp(t);
  return x * x * x * (x * (x * 6 - 15) + 10);
};

/** CSS-style cubic bezier easing (x1,y1,x2,y2), solved by Newton + bisection. */
export const bezier = (x1: number, y1: number, x2: number, y2: number) => {
  const cx = 3 * x1, bx = 3 * (x2 - x1) - cx, ax = 1 - cx - bx;
  const cy = 3 * y1, by = 3 * (y2 - y1) - cy, ay = 1 - cy - by;
  const sx = (t: number) => ((ax * t + bx) * t + cx) * t;
  const sy = (t: number) => ((ay * t + by) * t + cy) * t;
  const dx = (t: number) => (3 * ax * t + 2 * bx) * t + cx;
  return (x: number) => {
    const X = clamp(x);
    let t = X;
    for (let i = 0; i < 8; i++) {
      const e = sx(t) - X;
      const d = dx(t);
      if (Math.abs(e) < 1e-6) return sy(t);
      if (Math.abs(d) < 1e-6) break;
      t -= e / d;
    }
    let lo = 0, hi = 1;
    t = X;
    for (let i = 0; i < 30; i++) {
      const v = sx(t);
      if (Math.abs(v - X) < 1e-6) break;
      if (v < X) lo = t;
      else hi = t;
      t = (lo + hi) / 2;
    }
    return sy(t);
  };
};

// Named curves, tuned for product cinematography (long, confident tails; no bounce).
export const ease = {
  /** Default move: decisive start, very long settle. */
  glide: bezier(0.45, 0, 0.12, 1),
  /** Symmetric, weighty. Camera and large objects. */
  inOut: bezier(0.65, 0, 0.35, 1),
  /** Soft symmetric for subtle drifts. */
  sine: (t: number) => 0.5 - 0.5 * Math.cos(Math.PI * clamp(t)),
  /** Starts with energy (a magnet letting go), long deceleration. */
  release: bezier(0.2, 0.7, 0.1, 1),
  /** Gathers speed and arrives fast (magnetic pull into a dock). */
  attract: bezier(0.55, 0, 0.95, 0.55),
  /** Typographic reveal. */
  type: bezier(0.2, 0, 0, 1),
  out: bezier(0, 0, 0.2, 1),
  in: bezier(0.6, 0, 1, 1),
  smoother: smootherstep,
};

/** Eased progress of `frame` through [start, end]. */
export const seg = (frame: number, start: number, end: number, curve: (t: number) => number = ease.inOut) =>
  curve(remap(frame, start, end));

/**
 * Critically damped settle used after a mechanical stop: decays to zero with at most one
 * tiny overshoot. `t` in frames since the stop, `amp` the initial offset.
 */
export const settle = (t: number, amp: number, freq = 0.11, damping = 0.16) =>
  t <= 0 ? 0 : amp * Math.exp(-damping * t) * Math.cos(freq * t * Math.PI);

// Spoken-word cues → VO seconds, and the VO's pauses (where cuts like to land).

import type { Cue, Word } from "./story.ts";

export const norm = (w: string) => w.toLowerCase().normalize("NFKD").replace(/[^a-z0-9]/g, "");

export type Cues = {
  words: Word[];
  offset: number;
  voEnd: number;
  /** VO seconds of a cue (throws a plain-language error when the word isn't spoken) */
  at: (c: Cue, label?: string, after?: number) => number;
  /** pauses in the VO: [start, end] VO seconds, gaps ≥ 0.18 s */
  pauses: [number, number][];
  /** the best cut time (VO s) for an action that starts at `v`: a pause just before it, else v − 0.12 */
  cutFor: (v: number) => number;
};

export const makeCues = (words: Word[], offset: number): Cues => {
  const map = new Map<string, number[]>();
  words.forEach(([w, s]) => {
    const k = norm(w);
    if (!k) return;
    if (!map.has(k)) map.set(k, []);
    map.get(k)!.push(s);
  });
  const pauses: [number, number][] = [];
  for (let i = 1; i < words.length; i++) if (words[i][1] - words[i - 1][2] >= 0.18) pauses.push([words[i - 1][2], words[i][1]]);
  const voEnd = words.length ? words[words.length - 1][2] : 0;
  const toks = words.map(([w]) => norm(w));
  /** VO seconds of a cue. Without "#n" the first occurrence at/after `after` (VO s) is used, so beats
   *  written in order never need counting. A cue may be a short phrase ("di balik"). */
  const at = (c: Cue, label = "cue", after = -1) => {
    if (typeof c === "number") return c;
    const [w, n] = c.split("#");
    const parts = w.split(/\s+/).map(norm).filter(Boolean);
    if (!parts.length) throw new Error(`${label}: empty cue`);
    const hits: number[] = [];
    for (let i = 0; i + parts.length <= toks.length; i++) if (parts.every((p, k) => toks[i + k] === p)) hits.push(words[i][1]);
    if (!hits.length) {
      const k = parts[0];
      const near = [...map.keys()].filter((x) => x.startsWith(k.slice(0, 3))).slice(0, 6);
      throw new Error(`${label}: "${w}" is not spoken in the VO${near.length ? ` (similar words: ${near.join(", ")})` : ""}`);
    }
    if (n !== undefined) {
      const i = Number(n);
      if (!(i >= 0 && i < hits.length)) throw new Error(`${label}: "${w}" is spoken ${hits.length}× — "${c}" asks for occurrence #${n} (use #0…#${hits.length - 1}, or leave "#n" out to take the next one in order)`);
      return hits[i];
    }
    const next = hits.find((h) => h >= after - 0.05);
    if (next === undefined) throw new Error(`${label}: "${w}" is not spoken again after ${after.toFixed(2)} s in the VO (beats must be in VO order; spoken at ${hits.map((h) => h.toFixed(2)).join(", ")} s)`);
    return next;
  };
  const cutFor = (v: number) => {
    // the latest pause that ends just before (or at) the word: cut inside it, shortly before the word
    let best: [number, number] | null = null;
    for (const p of pauses) if (p[1] <= v + 0.05 && p[1] >= v - 0.9 && (!best || p[1] > best[1])) best = p;
    return best ? Math.max(best[0] + 0.05, best[1] - 0.15) : v - 0.12;
  };
  return { words, offset, voEnd, at, pauses, cutFor };
};

import { Vector3 } from "three";
import type { Cam } from "../cine/Post.tsx";
import type { V3 } from "../cine/Lights.tsx";

/** Project a world point to overlay pixels (W × H: 1080×1920 portrait, 1920×1080 landscape); z < 0 = behind. */
export const project = (cam: Cam, p: V3, W = 1920, H = 1080): { x: number; y: number; z: number } => {
  const pos = new Vector3(...cam.pos), f = new Vector3(...cam.target).sub(pos).normalize();
  const right = new Vector3().crossVectors(f, new Vector3(0, 1, 0)).normalize();
  let up = new Vector3().crossVectors(right, f).normalize();
  const r = cam.roll ?? 0;
  if (r) {
    const u2 = up.clone().multiplyScalar(Math.cos(r)).addScaledVector(right, Math.sin(r));
    right.crossVectors(f, u2).normalize();
    up = u2;
  }
  const d = new Vector3(...p).sub(pos);
  const z = d.dot(f), x = d.dot(right), y = d.dot(up);
  const th = Math.tan((cam.fov * Math.PI) / 360);
  return { x: W / 2 + (x / (z * th * (W / H))) * (W / 2), y: H / 2 - (y / (z * th)) * (H / 2), z };
};

// arch profiles, geodesic frames. Geometries are cached by key so a film builds each once.

import {
  BufferGeometry,
  CylinderGeometry,
  IcosahedronGeometry,
  LatheGeometry,
  Matrix4,
  Quaternion,
  Shape,
  SphereGeometry,
  TorusGeometry,
  Vector2,
  Vector3,
} from "three";
import { RoundedBoxGeometry } from "three/examples/jsm/geometries/RoundedBoxGeometry.js";
import { mergeGeometries } from "three/examples/jsm/utils/BufferGeometryUtils.js";

const cache = new Map<string, BufferGeometry>();
export const cached = <G extends BufferGeometry>(key: string, make: () => G): G => {
  let g = cache.get(key);
  if (!g) {
    g = make();
    cache.set(key, g);
  }
  return g as G;
};

export const cyl = (rt: number, rb: number, h: number, seg = 32, open = false, ts = 0, tl = Math.PI * 2) =>

export const sphere = (r: number, ws = 32, hs = 16, phiLen = Math.PI * 2, thetaLen = Math.PI) =>

export const torus = (R: number, r: number, rs = 8, ts = 48, arc = Math.PI * 2) =>

export const lathe = (key: string, pts: [number, number][], seg = 64) =>
  cached(`lathe${key}`, () => new LatheGeometry(pts.map(([x, y]) => new Vector2(x, y)), seg));

const UP = new Vector3(0, 1, 0);
/** A cylinder from p to q (world-free; positions in the caller's frame). */
export const strutGeo = (p: Vector3, q: Vector3, r: number, seg = 6) => {
  const d = new Vector3().subVectors(q, p);
  const len = d.length();
  const g = new CylinderGeometry(r, r, len, seg, 1, false);
  const m = new Matrix4().compose(
    new Vector3().addVectors(p, q).multiplyScalar(0.5),
    new Quaternion().setFromUnitVectors(UP, d.normalize()),
    new Vector3(1, 1, 1),
  );
  g.applyMatrix4(m);
  return g;
};

export const merged = (key: string, make: () => BufferGeometry[]) =>
  cached(key, () => {
    const parts = make().map((g) => (g.index ? g.toNonIndexed() : g));
    for (const p of parts) {
      for (const name of Object.keys(p.attributes)) if (name !== "position" && name !== "normal" && name !== "uv") p.deleteAttribute(name);
    }
    return mergeGeometries(parts, false)!;
  });

/** Geodesic frame over the upper hemisphere (edges of a subdivided icosahedron). */
export const geodesicFrame = (R: number, detail: number, strutR: number) =>
  merged(`geo${R},${detail},${strutR}`, () => {
    const ico = new IcosahedronGeometry(R, detail);
    const p = ico.attributes.position;
    const key = (v: Vector3) => `${v.x.toFixed(3)},${v.y.toFixed(3)},${v.z.toFixed(3)}`;
    const edges = new Map<string, [Vector3, Vector3]>();
    for (let i = 0; i < p.count; i += 3) {
      const tri = [0, 1, 2].map((k) => new Vector3(p.getX(i + k), p.getY(i + k), p.getZ(i + k)));
      for (let k = 0; k < 3; k++) {
        const a = tri[k], b = tri[(k + 1) % 3];
        if (a.y < -0.01 || b.y < -0.01) continue;
        const ka = key(a), kb = key(b);
        const id = ka < kb ? ka + "|" + kb : kb + "|" + ka;
        if (!edges.has(id)) edges.set(id, [a, b]);
      }
    }

==== 2090 project/src/kits/worldforge/path.ts
// Plan-view paths for vehicles and people: Catmull-Rom through the points, sampled by arc
// length so speed along the path is exact (wheels and footsteps can be derived from distance).

export type P2 = [number, number];

export type PlanPath = {
  length: number;
  /** Position and unit direction at arc length s (clamped). */
  at: (s: number) => { p: P2; dir: P2 };
};

const cr = (a: number, b: number, c: number, d: number, t: number) =>
  0.5 * (2 * b + (-a + c) * t + (2 * a - 5 * b + 4 * c - d) * t * t + (-a + 3 * b - 3 * c + d) * t * t * t);

export const planPath = (pts: P2[], perSeg = 40): PlanPath => {
  const n = pts.length;
  const get = (i: number) => pts[Math.max(0, Math.min(n - 1, i))];
  const samples: P2[] = [];
  for (let i = 0; i < n - 1; i++) {
    const a = get(i - 1), b = get(i), c = get(i + 1), d = get(i + 2);
    for (let k = 0; k < perSeg; k++) {
      const t = k / perSeg;
      samples.push([cr(a[0], b[0], c[0], d[0], t), cr(a[1], b[1], c[1], d[1], t)]);
    }
  }
  samples.push(pts[n - 1]);
  const cum = [0];
  for (let i = 1; i < samples.length; i++) cum.push(cum[i - 1] + Math.hypot(samples[i][0] - samples[i - 1][0], samples[i][1] - samples[i - 1][1]));
  const length = cum[cum.length - 1];
  const at = (s: number) => {
    const x = Math.max(0, Math.min(length - 1e-6, s));
    let lo = 0, hi = cum.length - 1;
    while (hi - lo > 1) {
      const m = (lo + hi) >> 1;
      if (cum[m] <= x) lo = m;
      else hi = m;
    }
    const f = (x - cum[lo]) / Math.max(1e-9, cum[hi] - cum[lo]);
    const A = samples[lo], B = samples[hi];
    const dx = B[0] - A[0], dz = B[1] - A[1];
    const l = Math.hypot(dx, dz) || 1;
    return { p: [A[0] + dx * f, A[1] + dz * f] as P2, dir: [dx / l, dz / l] as P2 };
  };
  return { length, at };
};

/** Distance travelled with a smooth start: rest until t0, accelerate over `ramp`, then speed v. */
export const travel = (t: number, t0: number, ramp: number, v: number) => {
  if (t <= t0) return 0;
  const u = t - t0;
  if (u < ramp) return (v * u * u) / (2 * ramp);
  return v * (u - ramp / 2);
};

==== 560 project/worlds/README.md
Reusable worlds (`<name>.world.json`, see the skill's references/cinestory/dynamic-worlds.md and worlds.md).

Ready-made: `rumah` (house: living room, kitchen, bedroom, back corridor, terrace), `sekolah` (classroom, veranda corridor,
