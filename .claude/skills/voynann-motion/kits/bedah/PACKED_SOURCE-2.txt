// Sound from a Plan — original synth, no samples. Music: a light marimba-and-pad bed (D major pentatonic, 92 bpm).
// SFX follow the plan: soft rain while it rains, knocks when timber lands, clay ticks when tiles land, a breath of
// fire, a quiet chime when a term is pinned, a faint air move on each cut. Mix −14 LUFS / −1.5 dBTP; the music ducks
// under the voice when there is one.
import fs from "node:fs";
import path from "node:path";
import { execFileSync, spawnSync } from "node:child_process";

export const buildSound = (plan, D, { voice, outDir, mixFile, music = true }) => {
  const DUR = plan.duration, SR = 48000, N = Math.ceil(DUR * SR);
  const mk = () => [new Float32Array(N), new Float32Array(N)];
  let seed = 12345;
  const rnd = () => (seed = (seed * 16807) % 2147483647) / 2147483647;
  const add = (buf, t, pan, len, fn) => {
    const i0 = Math.max(0, Math.round(t * SR)), n = Math.min(N - i0, Math.round(len * SR));
    const gl = Math.cos((pan + 1) * Math.PI / 4), gr = Math.sin((pan + 1) * Math.PI / 4);
    for (let i = 0; i < n; i++) { const v = fn(i / SR, i); buf[0][i0 + i] += v * gl; buf[1][i0 + i] += v * gr; }
  };
  const hz = (m) => 440 * Math.pow(2, (m - 69) / 12);

  // ---------------------------------------------------------------- music
  const mus = mk();
  if (music) {
    const beat = 60 / 92, bar = beat * 4;
    const CH = [[50, 57, 62, 66], [47, 54, 59, 62], [43, 50, 55, 59], [45, 52, 57, 61]];
    const PENTA = [62, 64, 66, 69, 71, 74, 76, 78];
    const fade = (t) => Math.min(1, Math.max(0, (DUR - 0.15 - t) / 2.2));
    for (let b = 0; b < Math.ceil(DUR / bar); b++) {
      const c = CH[b % 4], t0 = b * bar;
      for (const [k, m] of c.entries())
        for (const det of [-0.06, 0.07]) {
          const f = hz(m + 12 + det);
          add(mus, t0, det * 6, bar + 0.6, (t) => Math.sin(2 * Math.PI * f * t) * 0.016 * (k === 0 ? 0.8 : 1) * Math.min(1, t / 0.9) * Math.min(1, (bar + 0.6 - t) / 0.7) * fade(t0 + t));
        }
      for (const bt of [0, 2]) {
        const f = hz(c[0] - 12);
        add(mus, t0 + bt * beat, 0, 1.1, (t) => (Math.sin(2 * Math.PI * f * t) + 0.25 * Math.sin(4 * Math.PI * f * t)) * 0.05 * Math.min(1, t / 0.012) * Math.exp(-t * 2.6) * fade(t0 + t));
      }
      for (let e = 0; e < 8; e++) {
        const r = rnd();
        if (r < 0.26 && e % 2) continue;
        const pool = PENTA.filter((p) => c.some((x) => (p - x) % 12 === 0) || rnd() < 0.3);
        const m = pool[Math.floor(rnd() * pool.length)] ?? 69;
        const f = hz(m), vel = (e % 4 === 0 ? 0.075 : 0.05) * (0.75 + rnd() * 0.4), pan = (rnd() - 0.5) * 0.7;
        add(mus, t0 + e * beat / 2 + (rnd() - 0.5) * 0.008, pan, 0.9, (t) => (Math.sin(2 * Math.PI * f * t) * Math.exp(-t * 7) + 0.3 * Math.sin(2 * Math.PI * f * 3.93 * t) * Math.exp(-t * 26)) * vel * Math.min(1, t / 0.003) * fade(t0 + t));
      }
      for (let e = 1; e < 8; e += 2) {
        let lp = 0;
        add(mus, t0 + e * beat / 2, 0.25, 0.07, (t) => { const n = rnd() * 2 - 1; lp += (n - lp) * 0.6; return (n - lp) * 0.02 * Math.exp(-t * 60) * fade(t0 + t); });
      }
    }
  }

  // ---------------------------------------------------------------- SFX
  const sfx = mk();
  const bed = (a, d, gain, lpK, hpK, fin = 0.8, fout = 0.5) => {
    for (const pan of [-0.6, 0.6]) {
      let lp = 0, hp = 0;
      add(sfx, a, pan, d, (t) => { const n = rnd() * 2 - 1; lp += (n - lp) * lpK; hp += (lp - hp) * hpK; return (lp - hp) * gain * Math.min(1, t / fin) * Math.min(1, (d - t) / fout); });
    }
  };
  const knock = (t, f = 190, g = 0.14, pan = 0) => add(sfx, t, pan, 0.35, (x) => (Math.sin(2 * Math.PI * f * x * (1 + 0.6 * Math.exp(-x * 60))) * Math.exp(-x * 22) + (rnd() * 2 - 1) * 0.25 * Math.exp(-x * 140)) * g * Math.min(1, x / 0.002));
  const tick = (t, f = 2100, g = 0.075, pan = 0) => add(sfx, t, pan, 0.18, (x) => (Math.sin(2 * Math.PI * f * x) * Math.exp(-x * 55) + 0.5 * Math.sin(2 * Math.PI * f * 2.7 * x) * Math.exp(-x * 90)) * g * Math.min(1, x / 0.001));
  const chime = (t, m = 81) => { for (const [k, mm] of [m, m + 7].entries()) add(sfx, t + k * 0.07, 0.2, 0.9, (x) => Math.sin(2 * Math.PI * hz(mm) * x) * 0.03 * Math.min(1, x / 0.004) * Math.exp(-x * 5.5)); };
  const air = (t) => { let lp = 0; add(sfx, t - 0.14, 0, 0.36, (x) => { const n = rnd() * 2 - 1; lp += (n - lp) * 0.05; return lp * 0.11 * Math.sin(Math.PI * Math.min(1, x / 0.36)) ** 2; }); };

  const events = [];
  plan.shots.forEach((S, i) => {
    const d = S.t1 - S.t0;
    if (i) air(S.t0);
    // rain: on while this shot rains; it tails off over the start of the shot that clears it
    if (S.rain[1] > 0) { bed(S.t0 + (S.rain[0] > 0 ? 0 : 0.2), d + (plan.shots[i + 1] && plan.shots[i + 1].rain[1] === 0 ? Math.min(1.1, (plan.shots[i + 1].t1 - plan.shots[i + 1].t0) * 0.3) : 0), 0.09, 0.35, 0.02, S.rain[0] > 0 ? 0.3 : 1.2, S.rain[0] > 0 && plan.shots[i + 1]?.rain[1] > 0 ? 0.25 : 1.0); events.push(`${S.t0.toFixed(2)} rain`); }
    for (const p of S.parts) {
      if (!p.build) continue;
      const kind = D.parts[p.name]?.sound ?? "none", a = S.t0 + p.build.a, len = p.build.b - p.build.a;
      if (kind === "wood") { const n = Math.max(1, Math.min(4, Math.round(len / 0.6))); for (let k = 0; k < n; k++) knock(a + len * (0.42 + (0.55 * k) / n), 160 + k * 24 + rnd() * 20, 0.13, 0.25 - k * 0.15); }
      else if (kind === "clay") { const n = Math.max(6, Math.min(27, Math.round(len * 7))); for (let k = 0; k < n; k++) tick(a + len * (0.14 + (0.84 * k) / n), 1400 + rnd() * 900, 0.075, -0.3 + (k % 3) * 0.25); }
      else if (kind === "soft") { for (let k = 0; k < 2; k++) knock(a + len * (0.4 + k * 0.35), 110, 0.05, 0.2); }
      if (kind !== "none") events.push(`${a.toFixed(2)} ${kind} ${p.name}`);
    }
    for (const an of S.anims) {
      const ps = D.paramSound?.[an.what] ?? D.paramSound?.[`${an.what.split(".")[0].split("#")[0]}.${an.what.split(".")[1]}`];
      if (ps === "fire") bed(S.t0, d, 0.5, 0.012, 0.002, 0.4, 0.4);
      if (ps === "thud") knock(S.t0 + (an.a + an.b) / 2, 85, 0.13, 0);
      if (ps) events.push(`${S.t0.toFixed(2)} ${ps}`);
    }
    if (S.worker?.do === "cetak") for (let k = 0; k * 1.5 + 0.75 < d; k++) knock(S.t0 + 0.75 + k * 1.5, 120, 0.07, 0.3);
    S.pins.forEach((p, k) => chime(S.t0 + p.a + 0.2, 79 + ((i + k) % 3) * 2));
  });
  if (plan.shots.length > 2) chime(plan.shots[plan.shots.length - 1].t0 + 0.25, 74);

  // ---------------------------------------------------------------- write + mix
  const wav = (file, buf) => {
    const data = Buffer.alloc(N * 4);
    for (let i = 0; i < N; i++) for (let c = 0; c < 2; c++) data.writeInt16LE(Math.round(Math.max(-1, Math.min(1, buf[c][i])) * 32767), i * 4 + c * 2);
    const h = Buffer.alloc(44);
    h.write("RIFF", 0); h.writeUInt32LE(36 + data.length, 4); h.write("WAVEfmt ", 8); h.writeUInt32LE(16, 16); h.writeUInt16LE(1, 20); h.writeUInt16LE(2, 22);
    h.writeUInt32LE(SR, 24); h.writeUInt32LE(SR * 4, 28); h.writeUInt16LE(4, 32); h.writeUInt16LE(16, 34); h.write("data", 36); h.writeUInt32LE(data.length, 40);
    fs.writeFileSync(file, Buffer.concat([h, data]));
  };
  fs.mkdirSync(outDir, { recursive: true });
  fs.mkdirSync(path.dirname(mixFile), { recursive: true });
  const mf = path.join(outDir, "music.wav"), sf = path.join(outDir, "sfx.wav");
  wav(mf, mus);
  wav(sf, sfx);
  const tail = `apad=whole_dur=${DUR},atrim=0:${DUR},loudnorm=I=-14:TP=-1.5:LRA=11,aresample=48000[out]`;
  const filter = voice
    ? ["[0:a]highpass=f=70,acompressor=threshold=-20dB:ratio=2.5:attack=8:release=140:makeup=3,asplit=2[v][vk]", "[1:a]volume=1.3[m]", "[m][vk]sidechaincompress=threshold=0.03:ratio=6:attack=25:release=420[md]", "[2:a]volume=0.9[s]", `[v][md][s]amix=inputs=3:normalize=0:duration=longest,${tail}`].join(";")
    : ["[0:a]volume=1.6[m]", "[1:a]volume=1.0[s]", `[m][s]amix=inputs=2:normalize=0:duration=longest,${tail}`].join(";");
  execFileSync("ffmpeg", ["-y", "-v", "error", ...(voice ? ["-i", voice] : []), "-i", mf, "-i", sf, "-filter_complex", filter, "-map", "[out]", "-ac", "2", mixFile]);
  const m = spawnSync("ffmpeg", ["-hide_banner", "-i", mixFile, "-af", "loudnorm=print_format=summary", "-f", "null", "-"], { encoding: "utf8" }).stderr;
  return { lufs: (m.match(/Input Integrated:\s+(-?[\d.]+)/) ?? [])[1], peak: (m.match(/Input True Peak:\s+(-?[\d.]+)/) ?? [])[1], events };
};

// Voice-over → word timings for a list of sentences (whisper.cpp), plus the timed voice track.
import fs from "node:fs";
import os from "node:os";
import path from "node:path";
import { execFileSync, spawnSync } from "node:child_process";

const exists = (f) => !!f && fs.existsSync(f);
export const findWhisper = () => {
  const home = os.homedir();
  const onPath = spawnSync("sh", ["-c", "command -v whisper-cli"], { encoding: "utf8" }).stdout?.trim();
  if (process.env.NO_WHISPER) return null; // test switch: force the speech-run alignment
  const cli = [process.env.WHISPER_CLI, onPath, "/opt/homebrew/bin/whisper-cli", "/usr/local/bin/whisper-cli", path.join(home, "whisper.cpp/build/bin/whisper-cli")].find(exists);
  const dirs = [path.join(home, ".cache/whisper.cpp"), "models", "/opt/homebrew/share/whisper-cpp", path.join(home, "whisper.cpp/models")];
  const models = dirs.flatMap((d) => (fs.existsSync(d) ? fs.readdirSync(d).filter((f) => /^ggml-.*\.bin$/.test(f)).map((f) => path.join(d, f)) : []));
  const model = [process.env.WHISPER_MODEL].find(exists) ?? ["large-v3-turbo", "large-v3", "medium", "small"].map((p) => models.find((m) => m.includes(p))).find(Boolean) ?? models[0];
  return cli && model ? { cli, model } : null;
};
const norm = (w) => w.toLowerCase().normalize("NFD").replace(/[̀-ͯ]/g, "").replace(/[^a-z0-9]/g, "");

const silences = (vo) => {
  const sd = spawnSync("ffmpeg", ["-hide_banner", "-i", vo, "-af", "silencedetect=noise=-38dB:d=0.2", "-f", "null", "-"], { encoding: "utf8" }).stderr;
  const ss = [...sd.matchAll(/silence_start: ([\d.]+)/g)].map((m) => +m[1]), se = [...sd.matchAll(/silence_end: ([\d.]+)/g)].map((m) => +m[1]);
  return ss.map((a, i) => [a, se[i] ?? a + 0.3]);
};
const weight = (w) => 0.55 + w.replace(/[,.?!]/g, "").length * 0.085;

/**
 * Without whisper: the sentences are laid onto the recording's speech runs. Every sentence gets its share of the
 * spoken time (by word length); a sentence start that falls near a pause is moved to where the voice comes back
 * (sentences are normally separated by the longest pauses). Words are paced inside their sentence.
 */
const ontoRuns = (vo, texts, dur) => {
  const sil = silences(vo).filter(([a, b]) => b - a >= 0.2);
  const runs = [];
  let at = 0;
  for (const [a, b] of sil) { if (a - at > 0.05) runs.push([at, a]); at = b; }
  if (dur - at > 0.05) runs.push([at, dur]);
  const spoken = runs.reduce((n, [a, b]) => n + (b - a), 0);
  const real = (sp) => { let acc = 0; for (const [a, b] of runs) { if (sp <= acc + (b - a)) return a + (sp - acc); acc += b - a; } return runs.at(-1)[1]; };
  const W = texts.map((t) => t.split(/\s+/).filter(Boolean).map(weight));
  const tot = W.flat().reduce((x, y) => x + y, 0);
  // candidate sentence starts = run starts; pick, in order, the one nearest to each sentence's share of speech
  const starts = runs.map((r) => r[0]);
  let acc = 0, minIdx = 0;
  const t0 = texts.map((_, i) => {
    const guess = real((acc / tot) * spoken);
    acc += W[i].reduce((x, y) => x + y, 0);
    if (i === 0) return starts[0];
    let best = -1;
    // nearness to the sentence's share of speech, with a bonus for a long pause in front (sentences end on the long ones)
    const cost = (k) => Math.abs(starts[k] - guess) - 1.5 * Math.min(0.6, starts[k] - runs[k - 1][1]);
    for (let k = minIdx + 1; k < starts.length - (texts.length - 1 - i); k++) if (Math.abs(starts[k] - guess) < 1.2 && (best < 0 || cost(k) < cost(best))) best = k;
    if (best >= 0) { minIdx = best; return starts[best]; }
    return guess;
  });
  return texts.map((text, i) => {
    const a = t0[i], endRun = runs.filter((r) => r[1] <= (t0[i + 1] ?? dur) + 0.01).at(-1), b = Math.max(a + 0.3, Math.min(t0[i + 1] ?? dur, endRun ? endRun[1] : dur));
    const ws = text.split(/\s+/).filter(Boolean), sum = W[i].reduce((x, y) => x + y, 0);
    let c = 0;
    // pace the words over the speech inside [a, b] (pauses inside the sentence are skipped)
    const inside = runs.map(([x, y]) => [Math.max(x, a), Math.min(y, b)]).filter(([x, y]) => y > x);
    const span = inside.reduce((n, [x, y]) => n + (y - x), 0) || b - a;
    const at2 = (sp) => { let q = 0; for (const [x, y] of inside) { if (sp <= q + (y - x)) return x + (sp - q); q += y - x; } return b; };
    return { a, b, e0: a, words: ws.map((w, k) => { const t = inside.length ? at2((c / sum) * span) : a + ((b - a) * c) / sum; c += W[i][k]; return { w, t, e: t + 0.2 }; }) };
  });
};

/**
 * @returns {{ lines: {text, t0, t1, words: {w, t}[]}[], duration: number, voice: string, heard: string, missed: number }}
 * Times include `lead` seconds of silence before the first word; `voice` is the track to mix (lead + recording).
 */
export const syncVo = (vo, texts, out, lead = 0.35) => {
  fs.mkdirSync(out, { recursive: true });
  const dur = Number(execFileSync("ffprobe", ["-v", "error", "-show_entries", "format=duration", "-of", "csv=p=0", vo]).toString());
  const W = findWhisper();
  let lines, missed = 0, heardText = "", method = "whisper";
  if (!W) {
    lines = ontoRuns(vo, texts, dur);
    method = "runs";
  } else {
    const wav = path.join(out, "vo16k.wav");
    execFileSync("ffmpeg", ["-y", "-v", "error", "-i", vo, "-ar", "16000", "-ac", "1", wav]);
    execFileSync(W.cli, ["-m", W.model, "-f", wav, "-l", "id", "-ml", "1", "-sow", "-ojf", "-of", path.join(out, "vo_words")], { stdio: ["ignore", "ignore", "ignore"] });
    const j = JSON.parse(fs.readFileSync(path.join(out, "vo_words.json"), "utf8"));
    heardText = j.transcription.map((x) => x.text.trim()).filter(Boolean).join(" ");
    const heard = j.transcription.map((x) => ({ w: norm(x.text), a: x.offsets.from / 1000, b: x.offsets.to / 1000 })).filter((x) => x.w);
    let h = 0;
    lines = texts.map((text) => {
      const ws = text.split(/\s+/).filter(Boolean).map((w) => {
        const n = norm(w);
        for (let k = h; k < Math.min(heard.length, h + 10); k++) {
          if (heard[k].w === n || (n.length > 3 && heard[k].w.startsWith(n.slice(0, 4)))) { h = k + 1; return { w, t: heard[k].a, e: heard[k].b }; }
          for (let m = 2; m <= 3 && k + m <= heard.length; m++) {
            const joined = heard.slice(k, k + m).map((x) => x.w).join("");
            if (joined === n || (n.length > 5 && joined.startsWith(n.slice(0, 6)))) { const r = { w, t: heard[k].a, e: heard[k + m - 1].b }; h = k + m; return r; }
          }
        }
        missed++;
        return { w, t: null, e: null };
      });
      for (let i = 0; i < ws.length; i++) if (ws[i].t === null) {
        const prev = ws.slice(0, i).reverse().find((x) => x.t !== null), next = ws.slice(i + 1).find((x) => x.t !== null);
        ws[i].t = prev && next ? (prev.e + next.t) / 2 : prev ? prev.e + 0.05 : next ? next.t - 0.25 : heard[Math.min(h, heard.length - 1)]?.a ?? 0;
        ws[i].e = ws[i].t + 0.25;
      }
      return { a: ws[0].t, b: ws[ws.length - 1].e, e0: ws[0].e, words: ws };
    });
    // whisper word starts run early: a line whisper started inside a pause starts where the voice comes back
    const sil = silences(vo);
    for (const T of lines) {
      const o = sil.find(([a, b]) => T.a >= a - 0.15 && T.a < b && b - T.a < 2.5 && !(T.e0 <= a + 0.05 && T.e0 - T.a > 0.15))?.[1];
      if (o !== undefined && o > T.a) {
        const d = o - T.a;
        T.a += d;
        T.words = T.words.map((w, k) => ({ ...w, t: Math.max(w.t + d * Math.max(0, 1 - k / Math.max(3, T.words.length)), o + k * 0.05) }));
      }
    }
  }
  let last = -1;
  for (const T of lines) for (const w of T.words) { if (w.t < last + 0.04) w.t = last + 0.04; last = w.t; }
  lines.forEach((T, i) => { T.a = Math.min(T.a, T.words[0].t); T.b = Math.min(Math.max(T.b, T.words[T.words.length - 1].t + 0.2), (lines[i + 1]?.words[0].t ?? dur) - 0.02, dur); });
  const r3 = (x) => Math.round((x + lead) * 1000) / 1000;
  const voice = path.join(out, "voice.wav");
  execFileSync("ffmpeg", ["-y", "-v", "error", "-i", vo, "-af", `adelay=${Math.round(lead * 1000)}:all=1`, "-ar", "48000", "-ac", "2", voice]);
  return {
    lines: lines.map((T, i) => ({ text: texts[i], t0: r3(T.a), t1: r3(T.b), words: T.words.map((w) => ({ w: w.w, t: r3(w.t) })) })),
    duration: dur + lead, voice, missed, heard: heardText, method,
  };
};

// BEDAH director — compiles a Story into a Plan: cut times from the sentence timing, the state of the build at
// every cut, arrivals and parameter changes inside each shot, a camera per shot, worker, pins, marks, weather.
// Pure TypeScript: runs in node (scripts/make.mjs) and in the browser (Film.tsx reads the same helpers).

import { boxAround, frame, unionBox } from "./camera.ts";
import type { Box, DomainMeta, MarkGeo, PAnim, PPart, PShot, Params, Plan, ShotKind, ShotSpec, Story, Timed, V3, When, Word } from "./types.ts";

/** a cut lands just before its sentence starts */
export const LEAD = 0.16;
const c01 = (x: number) => (x < 0 ? 0 : x > 1 ? 1 : x);
const sm = (x: number) => { const t = c01(x); return t * t * (3 - 2 * t); };
const norm = (w: string) => w.toLowerCase().normalize("NFD").replace(/[̀-ͯ]/g, "").replace(/[^a-z0-9-]/g, "");

// ------------------------------------------------------------------ timing without a voice
/** Evenly paced words for a sentence (longer words and punctuation take longer). */
export const paceWords = (text: string, t0: number, t1: number): Word[] => {
  const raw = text.split(/\s+/).filter(Boolean);
  const weights = raw.map((w) => 0.55 + w.replace(/[,.?!]/g, "").length * 0.085 + (/[,.?!]$/.test(w) ? 0.5 : 0));
  const tot = weights.reduce((x, y) => x + y, 0);
  let acc = 0;
  return raw.map((w, i) => { const t = t0 + ((t1 - t0) * acc) / tot; acc += weights[i]; return { w, t }; });
};
/** Sentence timing at a natural speaking rate, for previews before the voice-over exists. */
export const paceLines = (texts: string[], wps = 2.75, gap = 0.42, start = 0.3): Timed[] => {
  let t = start;
  return texts.map((text) => {
    const n = text.split(/\s+/).length;
    const commas = (text.match(/,/g) ?? []).length, stops = (text.slice(0, -1).match(/[.?!]/g) ?? []).length;
    const d = n / wps + commas * 0.16 + stops * 0.3;
    const line = { text, t0: t, t1: t + d, words: paceWords(text, t, t + d) };
    t += d + gap;
    return line;
  });
};

// ------------------------------------------------------------------ references
/** "name", "name#2", "name:param=1,other=0.5" → key (name#id), part name, parameter overrides */
export const parseRef = (ref: string): { key: string; name: string; params: Params } => {
  const [key, rest = ""] = ref.split(":");
  const params: Params = {};
  for (const kv of rest.split(",").filter(Boolean)) { const [k, v] = kv.split("="); params[k.trim()] = Number(v); }
  return { key: key.trim(), name: key.split("#")[0].trim(), params };
};

type Live = { key: string; name: string; params: Params };
const find = (parts: { key: string; name: string }[], id: string) => parts.findIndex((p) => p.key === id) >= 0 ? parts.findIndex((p) => p.key === id) : parts.findIndex((p) => p.name === id);

/** Parameters and arrival of every part, and the globals, at `s` seconds into a shot. */
export const stateAt = (shot: PShot, s: number) => {
  const globals: Params = { ...shot.globals };
  const parts = shot.parts.map((p) => ({ key: p.key, name: p.name, params: { ...p.params }, e: p.build ? p.build.from + (1 - p.build.from) * c01((s - p.build.a) / Math.max(1e-3, p.build.b - p.build.a)) : p.e0 }));
  for (const an of shot.anims) {
    const k = sm((s - an.a) / Math.max(1e-3, an.b - an.a));
    const v = an.from + (an.to - an.from) * k;
    const [who, what] = an.what.split(".");
    if (who === "g") globals[what] = v;
    else { const i = find(parts, who); if (i >= 0) parts[i].params[what] = v; }
  }
  return { parts, globals };
};

/** World position of "part.anchor" in a resolved state (null when the part or anchor is not there). */
export const anchorAt = (D: DomainMeta, st: ReturnType<typeof stateAt>, ref: string): V3 | null => {
  const [who, name] = ref.split(".");
  const i = find(st.parts, who);
  if (i < 0) return null;
  const M = D.parts[st.parts[i].name];
  return M?.anchors?.({ ...M.params, ...st.parts[i].params }, st.globals)[name] ?? null;
};
/** Annotation geometry of "part.mark" in a resolved state. */
export const markAt = (D: DomainMeta, st: ReturnType<typeof stateAt>, ref: string): MarkGeo | null => {
  const [who, name] = ref.split(".");
  const i = find(st.parts, who);
  if (i < 0) return null;
  const M = D.parts[st.parts[i].name];
  return M?.marks?.({ ...M.params, ...st.parts[i].params }, st.globals)[name] ?? null;
};

// ------------------------------------------------------------------ compile
export const buildPlan = (story: Story, lines: Timed[], D: DomainMeta, opts: { audio?: string; aspect?: number } = {}): Plan => {
  const warnings: string[] = [];
  const warn = (i: number, m: string) => warnings.push(`shot ${i + 1}: ${m}`);
  const duration = Math.ceil((lines[lines.length - 1].t1 + 1.3) * 10) / 10;

  // ---- pass 1: which shots, when
  type Raw = { spec: ShotSpec; line0: number; line1: number; t0: number };
  const raws: Raw[] = [];
  const wordIn = (l0: number, l1: number, prefix: string, after = -1): number | null => {
    const n = norm(prefix);
    for (let l = l0; l <= l1; l++) for (const w of lines[l].words) if (w.t >= after && norm(w.w).startsWith(n)) return w.t;
    return null;
  };
  story.lines.forEach((L, i) => {
    if (L.same && raws.length) {
      const prev = raws[raws.length - 1];
      prev.line1 = i;
      for (const k of ["pin", "mark", "tag", "build", "set"] as const) if (L[k]) (prev.spec as Record<string, unknown[]>)[k] = [...((prev.spec[k] as unknown[]) ?? []), ...(L[k] as unknown[])];
      return;
    }
    raws.push({ spec: L, line0: i, line1: i, t0: raws.length ? lines[i].t0 - LEAD : 0 });
    for (const T of L.then ?? []) {
      const w = wordIn(i, i, T.at);
      if (w === null) warnings.push(`line ${i + 1}: "then" word "${T.at}" is not in the sentence`);
      raws.push({ spec: T, line0: i, line1: i, t0: (w ?? (lines[i].t0 + lines[i].t1) / 2) - LEAD });
    }
  });

  // ---- pass 2: state, arrivals, camera
  let live: Live[] = [];
  let globals: Params = { ...D.globals };
  let rain = 0;
  const put = (ref: string, i: number): Live | null => {
    const r = parseRef(ref);
    const M = D.parts[r.name];
    if (!M) { warn(i, `unknown part "${r.name}" (see --parts)`); return null; }
    for (const k of Object.keys(r.params)) if (!(k in (M.params ?? {}))) warn(i, `part "${r.name}" has no parameter "${k}"`);
    if (M.slot) live = live.filter((p) => p.key === r.key || D.parts[p.name]?.slot !== M.slot);
    const at = live.findIndex((p) => p.key === r.key);
    const item = { key: r.key, name: r.name, params: { ...(at >= 0 ? live[at].params : {}), ...r.params } };
    if (at >= 0) live[at] = item;
    else live.push(item);
    return item;
  };
  for (const ref of story.start ?? []) put(ref, -1);

  const shots: PShot[] = raws.map((R, i) => {
    const S = R.spec;
    const t0 = R.t0, t1 = raws[i + 1]?.t0 ?? duration, d = t1 - t0;
    const when = (w: When | undefined, def: number) => {
      if (w === undefined) return def;
      if (typeof w === "number") return c01(w) * d;
      const t = wordIn(R.line0, R.line1, w, t0 - 0.05);
      if (t === null) { warn(i, `word "${w}" is not spoken in this shot`); return def; }
      return Math.max(0, t - t0);
    };
    // state changes at the cut
    if (S.only) { live = []; for (const r of S.only) put(r, i); }
    for (const r of S.remove ?? []) { const k = find(live, parseRef(r).key); if (k >= 0) live.splice(k, 1); else warn(i, `remove: "${r}" is not there`); }
    for (const r of S.add ?? []) put(r, i);
    // arrivals
    const builds = (S.build ?? []).map((b) => (typeof b === "string" ? { part: b } : b));
    const built: PPart["build"][] = [];
    const builtKeys: string[] = [];
    builds.forEach((b, k) => {
      const item = put(b.part, i);
      if (!item) return;
      const M = D.parts[item.name];
      let a = when(b.at, d * (0.08 + (0.55 * k) / Math.max(1, builds.length)));
      const natural = M.build ?? 1.2;
      const len = b.over !== undefined ? b.over * d : Math.min(natural, Math.max(0.35, d - a - 0.3));
      // "at" a word: the part is landing as the word is said, so it sets off a little earlier
      if (typeof b.at === "string") a = Math.max(0, a - Math.min(0.45, len * 0.6));
      built.push({ a, b: Math.min(d - 0.05, a + len), from: b.from ?? 0 });
      builtKeys.push(item.key);
    });
    // parameter changes
    const anims: PAnim[] = [];
    const setNow = (what: string, v: number) => {
      const [who, name] = what.split(".");
      if (who === "g") { if (!(name in D.globals)) warn(i, `no global "${name}"`); globals = { ...globals, [name]: v }; return; }
      const k = find(live, who);
      if (k < 0) { warn(i, `set: part "${who}" is not there`); return; }
      if (!(name in (D.parts[live[k].name].params ?? {}))) warn(i, `part "${live[k].name}" has no parameter "${name}"`);
      live[k] = { ...live[k], params: { ...live[k].params, [name]: v } };
    };
    const getNow = (what: string) => {
      const [who, name] = what.split(".");
      if (who === "g") return globals[name] ?? 0;
      const k = find(live, who);
      return k < 0 ? 0 : live[k].params[name] ?? D.parts[live[k].name].params?.[name] ?? 0;
    };
    const after: [string, number][] = [];
    for (const st of S.set ?? []) {
      if (st.over === undefined && st.at === undefined) { setNow(st.what, st.to); continue; }
      const from = st.from ?? getNow(st.what);
      const a = when(st.at, d * 0.14), b = Math.min(d - 0.05, a + (st.over ?? 0.5) * d);
      setNow(st.what, from);
      anims.push({ what: st.what, from, to: st.to, a, b });
      after.push([st.what, st.to]);
    }
    globals = { ...globals, ...(D.derive?.(live.map((p) => p.name), globals) ?? {}) };
    const parts: PPart[] = live.map((p) => {
      const bi = builtKeys.indexOf(p.key);
      return { key: p.key, name: p.name, params: { ...D.parts[p.name].params, ...p.params }, e0: 1, ...(bi >= 0 ? { e0: built[bi]!.from, build: built[bi] } : {}) };
    });
    const shotGlobals = { ...globals };
    for (const [what, v] of after) setNow(what, v);
    globals = { ...globals, ...(D.derive?.(live.map((p) => p.name), globals) ?? {}) };

    // what the camera frames (measured on the state at the end of the shot)
    const endParams = (key: string) => { const k = find(live, key); return k < 0 ? null : { M: D.parts[live[k].name], p: { ...D.parts[live[k].name].params, ...live[k].params } }; };
    const kind: ShotKind = S.shot ?? (i === 0 || i === raws.length - 1 ? "wide" : builds.length ? "close" : "front");
    const boxOf = (ref: string): Box | null => {
      const [who, name] = ref.split(".");
      const e = endParams(who);
      if (!e) { warn(i, `focus: "${who}" is not there`); return null; }
      if (!name) return e.M.box(e.p, globals);
      const a = e.M.anchors?.(e.p, globals)[name];
      if (!a) { warn(i, `focus: part "${who}" has no anchor "${name}"`); return null; }
      return boxAround(a, kind === "detail" ? 0.6 : 1.15);
    };
    const refs = S.focus ? (Array.isArray(S.focus) ? S.focus : [S.focus]) : kind !== "wide" && kind !== "front" && builtKeys.length ? builtKeys : [];
    const boxes = refs.map(boxOf).filter((b): b is Box => !!b);
    const box = boxes.length ? unionBox(boxes) : D.subject(live.map((p) => p.name), globals);
    const side: 1 | -1 = S.side ? (S.side === "right" ? 1 : -1) : i % 2 ? 1 : -1;
    const move = S.move ?? (i === raws.length - 1 ? "pull" : "push");
    const SC = (opts.aspect ?? 9 / 16) > 1 && S.camWide ? S.camWide : S.cam;
    const cam = SC ? { a: SC.from, b: SC.to ?? SC.from, move } : { ...frame(kind, box, side, move, D.cutZ, i, opts.aspect ?? 9 / 16), move };

    // worker
    let worker: PShot["worker"];
    if (S.worker) {
      const W = typeof S.worker === "string" ? { do: S.worker } : S.worker;
      if (!D.acts.includes(W.do)) warn(i, `worker cannot "${W.do}" (acts: ${D.acts.join(", ")})`);
      const onKey = W.on ?? (refs[0] ? refs[0].split(".")[0] : builtKeys[0]);
      const e = onKey ? endParams(onKey) : null;

      else if (spot) worker = { ...spot, do: W.do, face: W.face ?? spot.face };
      else warn(i, `worker: give "at": [x, z] (no work spot known for "${onKey ?? "this shot"}")`);
    }

    const target = S.weather === "rain" ? 1 : 0;
    const shot: PShot = {

/** The roof every roof part sits on: the light bamboo roof, or (timber house) the raised truss roof. */
export const roofOf = (g: Params): RoofDims =>
  g.kayu
    ? { hw: 2.5, ridgeY: ridgeForWalls(HUT.hz, HUT.plate, 2.0, 2.0 * g.slope), run: 2.0, drop: 2.0 * g.slope }
    : { ...roofForSlope(g.slope), ...(g.sag ? { sag: g.sag } : {}) };

const angle = (g: Params) => {
  const th = Math.atan(g.slope), x = 2.62, R = 1.0;
  const C: V3 = [x, HUT.plate + (g.kayu ? roofOf(g).ridgeY - HUT.hz * g.slope - HUT.plate : 0), HUT.hz];
  const arc: V3[] = Array.from({ length: 13 }, (_, i) => { const a = (th * i) / 12; return [x, C[1] + R * Math.sin(a), C[2] - R * Math.cos(a)] as V3; });
  return { C, arc, base: [C, [x, C[1], C[2] - 1.45] as V3], label: [x, C[1] + 1.42 * Math.sin(th * 0.5), C[2] - 1.42 * Math.cos(th * 0.5)] as V3 };
};

const c01 = (x: number) => (x < 0 ? 0 : x > 1 ? 1 : x);
const smooth = (x: number) => { const t = c01(x); return t * t * (3 - 2 * t); };

// ------------------------------------------------------------------ rain
/** Slanted streaks falling through a box around `center`. `amount` 0…1 thins them in without a pop. */
export const Rain: React.FC<{ t: number; amount: number; center: V3; size?: V3; count?: number }> = ({ t, amount, center, size = [18, 11, 16], count = 800 }) => {
  const ref = useRef<InstancedMesh>(null);
  const G = useMemo(() => ({ geo: new BoxGeometry(0.007, 1.9, 0.007), m: new MeshBasicMaterial({ color: "#dfe6f0", transparent: true, opacity: 0.3, depthWrite: false, fog: false }) }), []);
  useLayoutEffect(() => {
    const im = ref.current!;
    im.layers.set(FX_LAYER);
    const o = new Object3D();
    const speed = 7.5;
    for (let i = 0; i < count; i++) {
      const fall = (hash2(i, 1, 3) * size[1] + t * speed * (0.85 + 0.3 * hash2(i, 2, 3))) % size[1];
      const y = center[1] + size[1] / 2 - fall;
      o.position.set(center[0] + (hash2(i, 3, 3) - 0.5) * size[0] + (size[1] / 2 - fall) * -0.1, y, center[2] + (hash2(i, 4, 3) - 0.5) * size[2]);
      o.rotation.set(0, 0, -0.1);
      const on = smooth((amount - hash2(i, 5, 3)) / 0.08); // streaks fade in one by one as the rain builds
      o.scale.set(on, on * (0.7 + 0.6 * hash2(i, 6, 3)), on);
      o.updateMatrix();
      im.setMatrixAt(i, o.matrix);
    }
    im.instanceMatrix.needsUpdate = true;
    G.m.opacity = 0.13 * Math.min(1, amount * 1.5);
  });
  return <instancedMesh ref={ref} args={[G.geo, G.m, count]} frustumCulled={false} />;
};

/** Drops running off an eave edge (a line from a to b), falling to the ground. */
export const EaveDrips: React.FC<{ t: number; amount: number; a: V3; b: V3; count?: number }> = ({ t, amount, a, b, count = 70 }) => {
  const ref = useRef<InstancedMesh>(null);
  const G = useMemo(() => ({ geo: new SphereGeometry(0.006, 6, 5), m: new MeshBasicMaterial({ color: "#e6edf5", transparent: true, opacity: 0.22, depthWrite: false, fog: false }) }), []);
  useLayoutEffect(() => {
    const im = ref.current!;
    im.layers.set(FX_LAYER);
    const o = new Object3D();
    const T = Math.sqrt((2 * a[1]) / 9.8);
    for (let i = 0; i < count; i++) {
      const k = (i + hash2(i, 1, 9) * 0.8) / count;
      const tau = (t * 1.0 + hash2(i, 2, 9) * T) % T;
      const on = smooth((amount - hash2(i, 3, 9)) / 0.1);
      o.position.set(a[0] + (b[0] - a[0]) * k, a[1] + (b[1] - a[1]) * k - 0.5 * 9.8 * tau * tau, a[2] + (b[2] - a[2]) * k);
      o.scale.set(on, on * (10 + tau * 26), on);
      o.updateMatrix();
      im.setMatrixAt(i, o.matrix);
    }
    im.instanceMatrix.needsUpdate = true;
  });
  return <instancedMesh ref={ref} args={[G.geo, G.m, count]} frustumCulled={false} />;
};

// ------------------------------------------------------------------ captions
export type Word = { w: string; t: number };
export type Line = { text: string; t0: number; t1: number };

/** Evenly paced words for a line (until real VO word timings replace them). */
export const pace = (text: string, t0: number, t1: number): Word[] => {
  const raw = text.split(/\s+/).filter(Boolean);
  const weights = raw.map((w) => 0.55 + w.replace(/[,.?!]/g, "").length * 0.085 + (/[,.?!]$/.test(w) ? 0.5 : 0));
  const tot = weights.reduce((a, b) => a + b, 0);
  let acc = 0;
  return raw.map((w, i) => {
    const t = t0 + ((t1 - t0) * acc) / tot;
    acc += weights[i];
    return { w, t };
  });
};

/** Split words into phrases of ≤ `max` words and ≤ `maxChars` characters, breaking after punctuation. */
export const chunk = (words: Word[], max = 3, maxChars = 21): Word[][] => {
  const chunks: Word[][] = [];
  let cur: Word[] = [];
  const chars = (c: Word[]) => c.reduce((n, w) => n + w.w.length + 1, -1);
  for (const w of words) {
    if (cur.length && (cur.length >= max || chars(cur) + 1 + w.w.length > maxChars)) { chunks.push(cur); cur = []; }
    cur.push(w);
    if (/[,.?!]$/.test(w.w)) { chunks.push(cur); cur = []; }
  }
  if (cur.length) chunks.push(cur);
  // avoid a lone orphan word when it still fits on the previous phrase
  for (let i = chunks.length - 1; i > 0; i--) {
    const prev = chunks[i - 1];
    if (chunks[i].length === 1 && !/[,.?!]$/.test(prev[prev.length - 1].w) && chars(prev) + 1 + chunks[i][0].w.length <= maxChars + 3) { prev.push(...chunks[i]); chunks.splice(i, 1); }
  }
  return chunks;
};

/** `ends[i]` = when phrase i must be gone (end of its sentence): a caption never hangs over the next shot. */
export const Captions: React.FC<{ s: number; chunks: Word[][]; ends?: number[]; y?: number; hold?: number }> = ({ s, chunks, ends, y, hold = 0.9 }) => {
  const L = React.useContext(LayoutCtx);
  const top = y ?? (L.w > L.h ? Math.round(L.h * 0.8) : 1196);
  let ci = -1;
  for (let i = 0; i < chunks.length; i++) if (s >= chunks[i][0].t - 0.02) ci = i;
  if (ci < 0) return null;
  const ch = chunks[ci];
  const last = ch[ch.length - 1].t;
  const end = ends?.[ci] ?? last + hold;
  if (s > end) return null;
  return (
    <div style={{ position: "absolute", left: 40, right: 40, top: top, textAlign: "center", fontFamily: F_CAPTION, fontWeight: 700, fontSize: 56, lineHeight: 1.2, letterSpacing: -0.4, color: "#fff", textShadow: SHADOW, whiteSpace: "nowrap", opacity: 1 - lin(s, end - 0.12, end) }}>
      {ch.map((w, i) => (
        <span key={i} style={{ opacity: lin(s, w.t - 0.03, w.t + 0.07) }}>{w.w.replace(/[,.]$/, "")}{i < ch.length - 1 ? " " : ""}</span>
      ))}

  const e = out3(lin(s, t0, t0 + dur)), o = 1 - lin(s, t1 - 0.25, t1);

  let tip = P[0], dirx = 1, diry = 0;
  for (let i = 0; i < segs.length && left > 0; i++) {
    const k = Math.min(1, left / Math.max(segs[i], 1e-3));
    const x = P[i][0] + (P[i + 1][0] - P[i][0]) * k, y = P[i][1] + (P[i + 1][1] - P[i][1]) * k;
    d.push(`L${x.toFixed(1)} ${y.toFixed(1)}`);
    tip = [x, y];
    dirx = (P[i + 1][0] - P[i][0]) / Math.max(segs[i], 1e-3);
    diry = (P[i + 1][1] - P[i][1]) / Math.max(segs[i], 1e-3);
    left -= segs[i];
  }
  const ah = width * 3.4;
  return (

      ) : null}

  );
};

/** A leaf blade along +Z from the origin, length L, width W, arching down by `droop` and folded along its midrib. */
const bladeGeo = (L: number, W: number, droop: number, fold = 0.25, seg = 8) => {
  const g = new PlaneGeometry(W, L, 2, seg);
  const p = g.getAttribute("position") as BufferAttribute;
  for (let i = 0; i < p.count; i++) {
    const x = p.getX(i), t = p.getY(i) / L + 0.5; // 0 at the base … 1 at the tip
    const y = Math.sin(t * Math.PI * 0.5) * L * 0.25 - t * t * L * droop - Math.abs(x) * fold;
    const z = t * L * (1 - 0.18 * droop * t);
    p.setXYZ(i, x, y, z);
  }
  g.computeVertexNormals();
  return g;
};

/** Thatch material with weathering: `rot` darkens the leaves and eats holes through them, `wet` soaks them. */
export const thatchMaterial = (state: ThatchState) => {
  const t = thatch();
  const m = new MeshStandardMaterial({ map: t.map.clone(), normalMap: t.normalMap.clone(), alphaTest: 0.5, alphaToCoverage: true, side: DoubleSide, roughness: 0.95 });
  m.map!.repeat.set(1 / 1.15, 1);
  m.normalMap!.repeat.set(1 / 1.15, 1);
  m.map!.needsUpdate = true;
  m.normalMap!.needsUpdate = true;
  m.onBeforeCompile = (sh) => {
    sh.uniforms.uRot = state.rot;
    sh.uniforms.uWet = state.wet;
    sh.uniforms.uNoise = { value: noiseTex() };
    sh.vertexShader = sh.vertexShader.replace("#include <common>", "#include <common>\nattribute vec2 aRoof;\nvarying vec2 vRoof;").replace("#include <begin_vertex>", "#include <begin_vertex>\nvRoof = aRoof;");
    sh.fragmentShader = sh.fragmentShader
      .replace("#include <common>", "#include <common>\nuniform float uRot; uniform float uWet; uniform sampler2D uNoise; varying vec2 vRoof;")
      .replace(
        "#include <map_fragment>",
        `#include <map_fragment>
        {
          vec3 nz = texture2D(uNoise, vRoof * vec2(0.11, 0.17)).rgb;
          float blotch = nz.r * 0.75 + nz.g * 0.25;
          float fine = texture2D(uNoise, vRoof * 0.9).b;
          // rot spreads from the damp patches outward
          float r = smoothstep(0.0, 0.5, uRot * 1.5 - blotch * 0.9 + 0.06 * fine);
          diffuseColor.rgb = mix(diffuseColor.rgb, diffuseColor.rgb * vec3(0.30, 0.27, 0.24) + vec3(0.012, 0.016, 0.01), r * 0.88);
          diffuseColor.rgb *= mix(1.0, 0.62, uWet);
          float hole = uRot * 0.47 - 0.1;
          float edge = blotch + (fine - 0.5) * 0.09;
          if (edge < hole) diffuseColor.a = 0.0;
          diffuseColor.rgb *= mix(1.0, 0.45, smoothstep(hole + 0.05, hole, edge) * step(0.001, uRot));
        }`,
      );
  };
  m.customProgramCacheKey = () => "bedah-thatch";
  return m;
};

const stripGeo = (f: ReturnType<typeof roofFrame>, s: number, hw: number, dt: number, db: number, e: number, uOff: number, lt = 0.015, lb = 0.06) => {
  const bot = dt + (db - dt) * e, lbot = lt + (lb - lt) * e;
  const a = f.pt(s, -hw, dt, lt), b = f.pt(s, hw, dt, lt), c = f.pt(s, hw, bot, lbot), d0 = f.pt(s, -hw, bot, lbot);
  const g = new BufferGeometry();
  g.setAttribute("position", new Float32BufferAttribute([...a, ...b, ...c, ...d0], 3));
  const n = f.normal(s);
  g.setAttribute("normal", new Float32BufferAttribute([...n, ...n, ...n, ...n], 3));
  g.setAttribute("uv", new Float32BufferAttribute([-hw + uOff, 1, hw + uOff, 1, hw + uOff, 1 - e, -hw + uOff, 1 - e], 2));
  g.setAttribute("aRoof", new Float32BufferAttribute([-hw, dt * s + 9, hw, dt * s + 9, hw, bot * s + 9, -hw, bot * s + 9], 2));
  g.setIndex(s > 0 ? [0, 3, 2, 0, 2, 1] : [0, 2, 3, 0, 1, 2]);
  return g;
};

/** One pressed clay pan-tile: an S-wave across (water channel + side roll), straight along its length. Origin at the head, +z = tail. */
export const tileGeometry = (() => {
  let g: BufferGeometry | null = null;
  return () => {
    if (g) return g;
    const { W, L, A, T } = TILE;
    const n = 14;
    const pos: number[] = [], nor: number[] = [], uv: number[] = [], idx: number[] = [];
    const prof = (i: number) => {
      const t = i / n, ph = 2 * Math.PI * t - 2.2;
      const x = (t - 0.5) * W, y = A * Math.sin(ph), dy = (A * 2 * Math.PI * Math.cos(ph)) / W;
      const il = 1 / Math.hypot(dy, 1);
      return { x, y, nx: -dy * il, ny: il };
    };
    const quad = (a: number, b: number, c: number, d: number) => idx.push(a, b, c, a, c, d);
    // top and bottom skins
    for (let i = 0; i <= n; i++) {
      const p = prof(i);
      for (const z of [0, L]) { pos.push(p.x, p.y, z); nor.push(p.nx, p.ny, 0); uv.push(p.x + 0.5, z); }
    }
    for (let i = 0; i <= n; i++) {
      const p = prof(i);
      for (const z of [0, L]) { pos.push(p.x - p.nx * T, p.y - p.ny * T, z); nor.push(-p.nx, -p.ny, 0); uv.push(p.x + 0.2, z + 0.4); }
    }
    const top = (i: number, k: number) => i * 2 + k, bot = (i: number, k: number) => (n + 1) * 2 + i * 2 + k;
    for (let i = 0; i < n; i++) {
      quad(top(i, 0), top(i, 1), top(i + 1, 1), top(i + 1, 0));
      quad(bot(i, 0), bot(i + 1, 0), bot(i + 1, 1), bot(i, 1));
    }
    // head / tail edges
    for (const [z, nz] of [[0, -1], [L, 1]] as const) {
      const base = pos.length / 3;
      for (let i = 0; i <= n; i++) {
        const p = prof(i);
        pos.push(p.x, p.y, z, p.x - p.nx * T, p.y - p.ny * T, z);
        nor.push(0, 0, nz, 0, 0, nz);
        uv.push(p.x, 0, p.x, T);
      }
      for (let i = 0; i < n; i++) {
        const a = base + i * 2, b = a + 1, c = a + 3, d = a + 2;
        if (nz > 0) quad(a, b, c, d);
        else quad(a, d, c, b);
      }
    }
    // side edges
    for (const [i, sx] of [[0, -1], [n, 1]] as const) {
      const p = prof(i);
      const base = pos.length / 3;
      pos.push(p.x, p.y, 0, p.x, p.y, L, p.x - p.nx * T, p.y - p.ny * T, L, p.x - p.nx * T, p.y - p.ny * T, 0);
      for (let k = 0; k < 4; k++) nor.push(sx, 0, 0);
      uv.push(0, 0, L, 0, L, T, 0, T);
      if (sx > 0) quad(base, base + 1, base + 2, base + 3);
      else quad(base, base + 3, base + 2, base + 1);
    }
    // nib: a small lug under the head that hooks behind the batten
    {
      const x0 = -0.032, x1 = 0.032, y0 = -0.034, y1 = 0.004, z0 = 0.0, z1 = 0.011;
      const c: [number, number, number][] = [[x0, y0, z0], [x1, y0, z0], [x1, y1, z0], [x0, y1, z0], [x0, y0, z1], [x1, y0, z1], [x1, y1, z1], [x0, y1, z1]];
      const faces: [number[], [number, number, number]][] = [[[1, 0, 3, 2], [0, 0, -1]], [[4, 5, 6, 7], [0, 0, 1]], [[0, 4, 7, 3], [-1, 0, 0]], [[5, 1, 2, 6], [1, 0, 0]], [[0, 1, 5, 4], [0, -1, 0]]];
      for (const [f, nn] of faces) {
        const base = pos.length / 3;
        f.forEach((k, j) => { pos.push(...c[k]); nor.push(...nn); uv.push(j % 2 ? 0.1 : 0, j > 1 ? 0.1 : 0); });
        quad(base, base + 1, base + 2, base + 3);
      }
    }
    g = new BufferGeometry();
    g.setAttribute("position", new Float32BufferAttribute(pos, 3));
    g.setAttribute("normal", new Float32BufferAttribute(nor, 3));
    g.setAttribute("uv", new Float32BufferAttribute(uv, 2));
    g.setIndex(idx);
    return g;
  };
})();

export const tileMaterial = () => mat("tileClay", clay(), { repeat: [3, 3], roughness: 0.82, bump: 0.8 });
/** Terracotta colour of tile #id (most orange-red, some darker, a few pale); `raw` 1 = unfired grey clay. */
export const tileColor = (id: number, raw = 0, out = new Color()) => {
  const r = hash2(id, 3, 11), r2 = hash2(id, 5, 11);
  out.setHSL(0.03 + r * 0.02, 0.6 + r2 * 0.12, r < 0.14 ? 0.27 + r2 * 0.05 : 0.36 + r2 * 0.09);
  if (raw > 0) out.lerp(new Color().setHSL(0.09, 0.14, 0.5 + r * 0.06), raw);
  return out;
};

const smooth = (x: number) => {
  const t = x < 0 ? 0 : x > 1 ? 1 : x;
  return t * t * (3 - 2 * t);
};
const outC = (x: number) => 1 - Math.pow(1 - (x < 0 ? 0 : x > 1 ? 1 : x), 3);

/**
 * Tiles on one slope, laid from the eave up with head-lap and side-lap. `progress(col, row, index)` → 0…1 lets a
 * film land them one by one (a tile slides down onto its batten); 1 everywhere = finished roof.
 */
export const TileField: React.FC<{ dims: RoofDims; side: 1 | -1; x0?: number; x1?: number; progress?: (col: number, row: number, i: number) => number; lift?: number }> = ({ dims, side, x0 = -dims.hw, x1 = dims.hw, progress, lift = 0.055 }) => {
  const f = useMemo(() => roofFrame(dims), [dims.hw, dims.ridgeY, dims.run, dims.drop, dims.sag]); // eslint-disable-line react-hooks/exhaustive-deps
  const cols = Math.max(1, Math.round((x1 - x0) / TILE.cover)), rows = Math.floor((f.sl + 0.04 - TILE.L) / TILE.gauge) + 1;
  const ref = useRef<InstancedMesh>(null);
  const material = tileMaterial();
  const geo = tileGeometry();
  useLayoutEffect(() => {
    const im = ref.current!;
    const c = new Color();
    for (let i = 0; i < cols * rows; i++) im.setColorAt(i, tileColor(i + (side > 0 ? 0 : 5000), 0, c));
    im.instanceColor!.needsUpdate = true;
  }, [cols, rows, side]);
  useLayoutEffect(() => {
    const im = ref.current!;
    const m = new Matrix4(), xv = new Vector3(), yv = new Vector3(), zv = new Vector3(), p = new Vector3(), n = new Vector3(...f.normal(side)), dn = new Vector3(...f.down(side));
    let i = 0;
    for (let r = 0; r < rows; r++)
      for (let c = 0; c < cols; c++, i++) {
        const e = progress ? progress(c, r, i) : 1;
        const x = side > 0 ? x0 + (c + 0.5) * TILE.cover : x1 - (c + 0.5) * TILE.cover;
        const tail = f.sl + 0.04 - r * TILE.gauge, head = tail - TILE.L;
        const ph = f.pt(side, x, head, lift), pt = f.pt(side, x, tail, lift + 0.03);
        zv.set(pt[0] - ph[0], pt[1] - ph[1], pt[2] - ph[2]).normalize();
        xv.set(side, 0, 0);
        yv.crossVectors(zv, xv).normalize();
        const k = outC(e);
        const sc = e <= 0 ? 1e-4 : smooth(e / 0.3);
        p.set(...ph).addScaledVector(n, (1 - k) * 0.42).addScaledVector(dn, -(1 - k) * 0.3);
        m.makeBasis(xv, yv, zv).scale(new Vector3(sc, sc, sc)).setPosition(p);
        im.setMatrixAt(i, m);
      }
    im.instanceMatrix.needsUpdate = true;
  });
  return <instancedMesh ref={ref} args={[geo, material, cols * rows]} castShadow receiveShadow frustumCulled={false} />;
};

/** Ridge tiles (kerpus): tapered half-round caps along the ridge, each lapping over the next. */
export const RidgeCap: React.FC<{ dims: RoofDims; x0?: number; x1?: number; progress?: (i: number, n: number) => number; lift?: number }> = ({ dims, x0 = -dims.hw, x1 = dims.hw, progress, lift = 0.05 }) => {
  const f = useMemo(() => roofFrame(dims), [dims.hw, dims.ridgeY, dims.run, dims.drop, dims.sag]); // eslint-disable-line react-hooks/exhaustive-deps
  const step = 0.3, len = 0.37;
  const n = Math.max(1, Math.round((x1 - x0) / step));
  const ref = useRef<InstancedMesh>(null);
  const geo = useMemo(() => {
    const g = new CylinderGeometry(0.112, 0.097, len, 16, 1, true, -Math.PI / 2, Math.PI);
    g.rotateZ(-Math.PI / 2);
    g.rotateX(-Math.PI / 2);
    return g;
  }, []);
  const material = mat("ridgeClay", clay(), { repeat: [3, 3], roughness: 0.82, bump: 0.8, double: true });
  useLayoutEffect(() => {
    const im = ref.current!;
    const c = new Color();
    for (let i = 0; i < n; i++) im.setColorAt(i, tileColor(i + 7000, 0, c));
    im.instanceColor!.needsUpdate = true;
  }, [n]);
  useLayoutEffect(() => {
    const im = ref.current!;
    const o = new Object3D();
    for (let i = 0; i < n; i++) {
      const e = progress ? progress(i, n) : 1;
      const x = x0 + (i + 0.5) * ((x1 - x0) / n);
      const p = f.pt(1, x, 0, 0);
      // follow the ridge line when it sags
      const slope = (f.pt(1, x + 0.15, 0, 0)[1] - f.pt(1, x - 0.15, 0, 0)[1]) / 0.3;
      o.position.set(x, p[1] + lift + (1 - outC(e)) * 0.4, 0);
      o.rotation.set(0, 0, Math.atan(slope));
      o.scale.setScalar(e <= 0 ? 1e-4 : smooth(e / 0.3));
      o.updateMatrix();
      im.setMatrixAt(i, o.matrix);
    }
    im.instanceMatrix.needsUpdate = true;
  });
  return <instancedMesh ref={ref} args={[geo, material, n]} castShadow receiveShadow frustumCulled={false} />;
};

/** Ridge pole, rafters and split-bamboo battens. With dims.sag the members bow between the gables. */
export const BambooFrame: React.FC<{ dims: RoofDims; rafters?: number; battens?: boolean; gauge?: number }> = ({ dims, rafters = 9, battens = true, gauge = 0.25 }) => {
  const f = useMemo(() => roofFrame(dims), [dims.hw, dims.ridgeY, dims.run, dims.drop, dims.sag]); // eslint-disable-line react-hooks/exhaustive-deps
  const mBam = mat("bamboo", bamboo(), { roughness: 0.55 });
  const mSplit = mat("bambooSplit", bamboo({ base: "#d6c08a", dark: "#a98f58" }), { roughness: 0.6 });
  const xs = Array.from({ length: rafters }, (_, i) => -dims.hw + 0.2 + ((2 * dims.hw - 0.4) * i) / (rafters - 1));
  const segs = dims.sag ? 6 : 1;
  const out: React.ReactNode[] = [];
  // ridge
  for (let k = 0; k < segs; k++) {
    const xa = -dims.hw + (2 * dims.hw * k) / segs, xb = -dims.hw + (2 * dims.hw * (k + 1)) / segs;
    out.push(<Culm key={`rg${k}`} a={f.pt(1, xa, 0, -0.05)} b={f.pt(1, xb, 0, -0.05)} r={0.045} material={mBam} seed={k} />);
  }
  xs.forEach((x, i) => {
    for (const s of [1, -1])
      for (let k = 0; k < segs; k++) {
        const da = 0.02 + ((f.sl - 0.04) * k) / segs, db = 0.02 + ((f.sl - 0.04) * (k + 1)) / segs;
        out.push(<Culm key={`rf${i}_${s}_${k}`} a={f.pt(s, x, da, -0.032)} b={f.pt(s, x, db, -0.032)} r={0.03} material={mBam} seed={i + (s > 0 ? 0 : 20)} />);
      }
  });
  if (battens) {
    const nb = Math.floor((f.sl - 0.1) / gauge) + 1;
    for (const s of [1, -1])
      for (let b = 0; b < nb; b++) {
        const dist = f.sl - 0.02 - b * gauge;
        for (let k = 0; k < segs; k++) {
          const xa = -dims.hw + (2 * dims.hw * k) / segs, xb = -dims.hw + (2 * dims.hw * (k + 1)) / segs;
          out.push(<Culm key={`bt${s}_${b}_${k}`} a={f.pt(s, xa, dist, 0.014)} b={f.pt(s, xb, dist, 0.014)} r={0.016} material={mSplit} seed={b} />);
        }
      }
  }
  return <group>{out}</group>;
};

// ------------------------------------------------------------------ the little house (woven bamboo walls)
const wallShape = (w: number, h: number, holes: [number, number, number, number][]) => {
  const s = new Shape();
  s.moveTo(-w / 2, 0); s.lineTo(w / 2, 0); s.lineTo(w / 2, h); s.lineTo(-w / 2, h); s.closePath();
  for (const [x0, y0, x1, y1] of holes) {
    const hsh = new Shape();
    hsh.moveTo(x0, y0); hsh.lineTo(x0, y1); hsh.lineTo(x1, y1); hsh.lineTo(x1, y0); hsh.closePath();
    s.holes.push(hsh);
  }
  return new ShapeGeometry(s);
};

/**

  const sides = front ? [1, -1] : [-1];
  return (

// ------------------------------------------------------------------ assembly helper
/**
 * A part that arrives: it flies in from `from` (offset in metres) while growing from nothing about its own centre
 * `at`, landing at e = 1. Nothing pops: at e → 0 the part has zero size.
 */
export const Arrive: React.FC<{ e: number; at: V3; from?: V3; children: React.ReactNode }> = ({ e, at, from = [0, 1.2, 0], children }) => {
  if (e <= 0) return null;
  const k = 1 - outC(e), sc = smooth(e / 0.3);
  return (
    <group position={[at[0] + from[0] * k, at[1] + from[1] * k, at[2] + from[2] * k]} scale={[sc, sc, sc]}>
      <group position={[-at[0], -at[1], -at[2]]}>{children}</group>
    </group>
  );
};
const mid = (a: V3, b: V3): V3 => [(a[0] + b[0]) / 2, (a[1] + b[1]) / 2, (a[2] + b[2]) / 2];
/** A timber member that arrives. */
export const Member: React.FC<{ e: number; a: V3; b: V3; w: number; h: number; material: Material; from?: V3; seed?: number }> = ({ e, a, b, w, h, material, from, seed }) => (
  <Arrive e={e} at={mid(a, b)} from={from}>
    <Beam a={a} b={b} w={w} h={h} material={material} seed={seed} />
  </Arrive>
);

// ------------------------------------------------------------------ timber truss roof
export type TrussStage = {
  /** 0…1 per member group: bottom chord, principal rafters, king post, struts */
  chord: number; rafters: number; king: number; struts: number;
  /** purlins (gording), common rafters (usuk), battens (reng) */
  purlins: number; usuk: number; reng: number;
};
export const TRUSS_DONE: TrussStage = { chord: 1, rafters: 1, king: 1, struts: 1, purlins: 1, usuk: 1, reng: 1 };

/** Kuda-kuda: bottom chord, two principal rafters, king post, two struts — then purlins, usuk and reng over them. */
export const TrussRoof: React.FC<{ dims: RoofDims; xs: number[]; plate: number; stage: TrussStage; stagger?: number; slopes?: number[] }> = ({ dims, xs, plate, stage, stagger = 0.22, slopes = [1, -1] }) => {
  const f = useMemo(() => roofFrame(dims), [dims.hw, dims.ridgeY, dims.run, dims.drop]); // eslint-disable-line react-hooks/exhaustive-deps
  const J = trussJoints(dims, plate);
  const mTim = mat("timber", wood({ base: "#b07a44", dark: "#6e4526", seed: 19 }), { repeat: [1 / 1.6, 1 / 0.35], roughness: 0.78 });
  const mTim2 = mat("timber2", wood({ base: "#c08c52", dark: "#7c5230", seed: 23 }), { repeat: [1 / 1.2, 1 / 0.3], roughness: 0.8 });
  const c01 = (x: number) => Math.min(1, Math.max(0, x));
  const st = (e: number, i: number) => c01(e * (1 + stagger * (xs.length - 1)) - i * stagger);
  const out: React.ReactNode[] = [];
  xs.forEach((x, i) => {
    out.push(<Member key={`c${i}`} e={st(stage.chord, i)} from={[0, 1.6, 0]} a={[x, J.seatY - 0.075, -J.footZ - 0.25]} b={[x, J.seatY - 0.075, J.footZ + 0.25]} w={0.08} h={0.15} material={mTim} seed={i} />);
    for (const s of [1, -1])
      out.push(<Member key={`r${i}_${s}`} e={st(stage.rafters, i)} from={[0, 1.0, s * 1.0]} a={[x, J.seatY + 0.05, s * J.footZ]} b={[x, J.apexY - 0.01, s * 0.05]} w={0.08} h={0.14} material={mTim} seed={i + 3 + s} />);
    out.push(<Member key={`k${i}`} e={st(stage.king, i)} from={[0, 1.5, 0]} a={[x + 0.0, J.seatY, 0]} b={[x, J.apexY + 0.1, 0]} w={0.082} h={0.1} material={mTim} seed={i + 7} />);
    for (const s of [1, -1]) {
      const t = 0.5;
      const top: V3 = [x, J.seatY + (J.apexY - J.seatY) * (1 - t) - 0.1, s * J.footZ * t];
      out.push(<Member key={`s${i}_${s}`} e={st(stage.struts, i)} from={[0, 0.9, s * 0.6]} a={[x, J.seatY + 0.07, s * 0.09]} b={top} w={0.07} h={0.09} material={mTim} seed={i + 11 + s} />);
    }
  });
  // purlins: ridge board, two per slope
  [0.05, f.sl * 0.34, f.sl * 0.63].forEach((dist, k) => {
    for (const s of k === 0 ? [1] : slopes)
      out.push(<Member key={`p${k}_${s}`} e={c01(stage.purlins * 1.6 - k * 0.3)} from={[0, 1.5, 0]} a={f.pt(s, -dims.hw + 0.12, dist, -0.145)} b={f.pt(s, dims.hw - 0.12, dist, -0.145)} w={0.08} h={0.12} material={mTim2} seed={k * 2 + s} />);
  });
  // usuk every ~0.5 m
  const nu = Math.round((2 * dims.hw - 0.3) / 0.5) + 1;
  for (let u = 0; u < nu; u++) {
    const x = -dims.hw + 0.15 + ((2 * dims.hw - 0.3) * u) / (nu - 1);
    const e = c01(stage.usuk * (1 + 0.06 * nu) - u * 0.06);
    for (const s of slopes)

// Every generator fills an albedo + height field per pixel; the normal map is derived from the height field.
// Texture row 0 is v = 0 (DataTexture is not flipped), so "y up" in the generators is "v up" on the mesh.

import { ClampToEdgeWrapping, DataTexture, LinearFilter, LinearMipmapLinearFilter, NoColorSpace, RepeatWrapping, RGBAFormat, SRGBColorSpace, type Texture } from "three";

export type Maps = { map: Texture; normalMap: Texture };
type Rgb = [number, number, number];

export const hex = (h: string): Rgb => [parseInt(h.slice(1, 3), 16), parseInt(h.slice(3, 5), 16), parseInt(h.slice(5, 7), 16)];
export const prng = (seed: number) => {
  let s = seed % 2147483647;
  if (s <= 0) s += 2147483646;
  return () => (s = (s * 16807) % 2147483647) / 2147483647;
};

// ------------------------------------------------------------------ noise (tileable)
export const hash2 = (x: number, y: number, seed = 0) => {
  let h = Math.imul(x | 0, 374761393) ^ Math.imul(y | 0, 668265263) ^ Math.imul(seed | 0, 1442695041);
  h = Math.imul(h ^ (h >>> 13), 1274126177);
  h ^= h >>> 16;
  return (h >>> 0) / 4294967295;
};
const wrap = (i: number, p: number) => ((i % p) + p) % p;
/** Value noise, periodic with integer periods (px, py) in lattice units. */
export const vnoise = (x: number, y: number, px: number, py: number, seed = 0) => {
  const xi = Math.floor(x), yi = Math.floor(y);
  const fx = x - xi, fy = y - yi;
  const ux = fx * fx * (3 - 2 * fx), uy = fy * fy * (3 - 2 * fy);
  const x0 = wrap(xi, px), x1 = wrap(xi + 1, px), y0 = wrap(yi, py), y1 = wrap(yi + 1, py);
  const a = hash2(x0, y0, seed), b = hash2(x1, y0, seed), c = hash2(x0, y1, seed), d = hash2(x1, y1, seed);
  return a + (b - a) * ux + (c - a) * uy + (a - b - c + d) * ux * uy;
};
/** Fractal noise 0..1; (u, v) in 0..1, `f` = base lattice cells across (integer → tileable). */
export const fbm = (u: number, v: number, f: number, oct: number, seed = 0, fy = f) => {
  let a = 0.5, s = 0, n = 0, px = f, py = fy;
  for (let i = 0; i < oct; i++) {
    s += a * vnoise(u * px, v * py, px, py, seed + i * 31);
    n += a;
    a *= 0.5;
    px *= 2;
    py *= 2;
  }
  return s / n;
};

const c01 = (x: number) => (x < 0 ? 0 : x > 1 ? 1 : x);
const ss = (a: number, b: number, x: number) => {
  const t = c01((x - a) / (b - a));
  return t * t * (3 - 2 * t);
};
const lerp3 = (out: Float32Array, a: Rgb, b: Rgb, t: number) => {
  out[0] = a[0] + (b[0] - a[0]) * t;
  out[1] = a[1] + (b[1] - a[1]) * t;
  out[2] = a[2] + (b[2] - a[2]) * t;
};

const finish = (t: DataTexture, srgb: boolean, repeat: boolean) => {

/**
 * Build albedo (+alpha) and a normal map. `bump` scales the height gradient (height units × pixels);
 * `tileV` false keeps the texture clamped vertically (a section that is drawn once).
 */
export const build = (w: number, h: number, fill: Fill, bump = 1, tileV = true): Maps => {
  const col = new Uint8Array(w * h * 4);
  const H = new Float32Array(w * h);
  const px = new Float32Array(5);
  for (let y = 0; y < h; y++)
    for (let x = 0; x < w; x++) {
      px[3] = 0;
      px[4] = 1;
      fill((x + 0.5) / w, (y + 0.5) / h, px);
      const k = (y * w + x) * 4;
      col[k] = px[0] < 0 ? 0 : px[0] > 255 ? 255 : px[0];
      col[k + 1] = px[1] < 0 ? 0 : px[1] > 255 ? 255 : px[1];
      col[k + 2] = px[2] < 0 ? 0 : px[2] > 255 ? 255 : px[2];
      col[k + 3] = c01(px[4]) * 255;
      H[y * w + x] = px[3];
    }
  const nrm = new Uint8Array(w * h * 4);
  for (let y = 0; y < h; y++) {
    const yu = tileV ? (y + 1) % h : Math.min(h - 1, y + 1), yd = tileV ? (y + h - 1) % h : Math.max(0, y - 1);
    for (let x = 0; x < w; x++) {
      const xr = (x + 1) % w, xl = (x + w - 1) % w;
      const dx = (H[y * w + xr] - H[y * w + xl]) * bump, dy = (H[yu * w + x] - H[yd * w + x]) * bump;
      const il = 1 / Math.hypot(dx, dy, 1);
      const k = (y * w + x) * 4;
      nrm[k] = (-dx * il * 0.5 + 0.5) * 255;
      nrm[k + 1] = (-dy * il * 0.5 + 0.5) * 255;
      nrm[k + 2] = (il * 0.5 + 0.5) * 255;
      nrm[k + 3] = 255;
    }
  }
  return { map: finish(new DataTexture(col, w, h, RGBAFormat), true, tileV), normalMap: finish(new DataTexture(nrm, w, h, RGBAFormat), false, tileV) };
};

const cache = new Map<string, Maps>();
const memo = (key: string, make: () => Maps) => {
  let m = cache.get(key);
  if (!m) cache.set(key, (m = make()));
  return m;
};

// ------------------------------------------------------------------ brick (running bond)
/** One repeat = `across` bricks × `courses` courses. Height: brick faces proud of recessed mortar. */
export const brick = (o: { across?: number; courses?: number; seed?: number; base?: string; dark?: string; light?: string; mortar?: string } = {}) =>
  memo("brick" + JSON.stringify(o), () => {
    const { across = 6, courses = 24, seed = 3 } = o;
    const B = hex(o.base ?? "#b4522e"), D = hex(o.dark ?? "#6b2d20"), L = hex(o.light ?? "#cc7648"), M = hex(o.mortar ?? "#b3aa9b");
    const jx = 0.045, jy = 0.17; // joint width as a fraction of a brick cell (≈ 10 mm)
    return build(1024, 1024, (u, v, px) => {
      const row = Math.floor(v * courses);
      const fy = v * courses - row;
      const xo = u * across + (row % 2) * 0.5;
      const bi = wrap(Math.floor(xo), across);
      const fx = xo - Math.floor(xo);
      const r = hash2(bi, row, seed), r2 = hash2(bi, row, seed + 7), r3 = hash2(bi, row, seed + 13);
      // rounded-rectangle mask with slightly ragged arrises
      const rag = (fbm(u, v, 96, 2, seed + 40) - 0.5) * 0.03;
      const ex = Math.min(fx, 1 - fx) - jx / 2 + rag * 0.6, ey = Math.min(fy, 1 - fy) - jy / 2 + rag * 2;
      const face = ss(0, 0.012, ex) * ss(0, 0.045, ey);
      const grain = fbm(u, v, 128, 4, seed + 3) - 0.5, blot = fbm(u, v, 24, 3, seed + 5) - 0.5;
      // most bricks orange-red, a fifth burnt dark, a few pale
      const t = r < 0.2 ? 0.75 + r2 * 0.25 : r2 * 0.28;
      lerp3(px, B, D, t);
      if (r > 0.9) lerp3(px, [px[0], px[1], px[2]], L, 0.6);
      const sh = 1 + grain * 0.22 + blot * 0.18 + (r3 - 0.5) * 0.14;
      px[0] *= sh; px[1] *= sh; px[2] *= sh;
      const mo = 1 + (fbm(u, v, 160, 3, seed + 9) - 0.5) * 0.3;
      px[0] = M[0] * mo + (px[0] - M[0] * mo) * face;
      px[1] = M[1] * mo + (px[1] - M[1] * mo) * face;
      px[2] = M[2] * mo + (px[2] - M[2] * mo) * face;
      px[3] = face * (1 + grain * 0.35) + (1 - face) * 0.25 * mo;
    }, 5);
  });

// ------------------------------------------------------------------ woven bamboo (gedek, 2/2 twill herringbone)
export const weave = (o: { strips?: number; seed?: number; a?: string; b?: string } = {}) =>
  memo("weave" + JSON.stringify(o), () => {
    const { strips = 32, seed = 5 } = o;
    const A = hex(o.a ?? "#cdb47c"), Bc = hex(o.b ?? "#b99d64");
    const band = 8; // twill direction flips every `band` strips → zigzag
    return build(1024, 1024, (u, v, px) => {
      const x = u * strips, y = v * strips;
      const i = Math.floor(x), j = Math.floor(y);
      const fx = x - i, fy = y - j;
      const dir = Math.floor(i / band) % 2 ? 1 : -1;
      const over = wrap(i + dir * j, 4) < 2; // true → the vertical strip is on top
      const along = over ? fy : fx, across = over ? fx : fy;
      const id = over ? i : j;
      const tone = hash2(id, over ? 1 : 2, seed), fib = fbm(over ? u : v, over ? v : u, 256, 2, seed + (over ? 3 : 4), 8);
      lerp3(px, over ? A : Bc, over ? Bc : A, tone * 0.3);
      // strip arches across its width and dips where it passes under its neighbours
      const arch = Math.sin(Math.PI * across), dip = 0.55 + 0.45 * Math.sin(Math.PI * along);
      const edge = ss(0, 0.1, across) * ss(0, 0.1, 1 - across);
      const sh = (0.85 + 0.15 * arch * dip) * (0.94 + fib * 0.12) * (0.9 + 0.1 * edge);
      px[0] *= sh; px[1] *= sh; px[2] *= sh;
      px[3] = arch * dip * 0.9 + edge * 0.2;
    }, 5);
  });

/** One strip of folded palm leaves: blades hang from the top edge (v = 1) and end raggedly near v = 0. */
export const thatch = (o: { blades?: number; seed?: number } = {}) =>
  memo("thatch" + JSON.stringify(o), () => {
    const { blades = 90, seed = 7 } = o;
    const C1 = hex("#c2a465"), C2 = hex("#8f7440"), C3 = hex("#6f6248"), C4 = hex("#d8c088");
    return build(1024, 512, (u, v, px) => {
      // two interleaved blade layers; the nearer one wins where it exists
      let best = -1, bh = 0;
      for (let layer = 0; layer < 2; layer++) {
        const x = u * blades * (layer ? 0.83 : 1) + layer * 0.37 + (fbm(u, v, 6, 2, seed + layer) - 0.5) * 0.9 * (1 - v);
        const i = Math.floor(x), f = x - i;
        const n = Math.round(blades * (layer ? 0.83 : 1));
        const id = wrap(i, n);
        const len = 0.72 + hash2(id, layer, seed) * 0.28; // blade length (fraction of strip height)
        const wdt = 0.55 + hash2(id, layer + 5, seed) * 0.4;
        const tip = c01((v - (1 - len)) / 0.22); // taper toward the tip
        const half = (wdt * (0.25 + 0.75 * Math.sqrt(tip))) / 2;
        const d = Math.abs(f - 0.5);
        if (v > 1 - len && d < half) {
          const hgt = (1 - d / half) * 0.6 + layer * 0.5 + 0.2;
          if (hgt > bh) { bh = hgt; best = id * 2 + layer; }
        }
      }
      if (best < 0) { px[0] = 60; px[1] = 48; px[2] = 30; px[4] = 0; return; }
      const r = hash2(best, 11, seed), r2 = hash2(best, 12, seed);
      lerp3(px, C1, C2, r);
      if (r2 > 0.82) lerp3(px, [px[0], px[1], px[2]], C3, 0.7);
      else if (r2 < 0.12) lerp3(px, [px[0], px[1], px[2]], C4, 0.7);
      const streak = fbm(u, v, 512, 2, seed + 20, 6);
      const sh = (0.72 + 0.28 * c01(bh)) * (0.86 + streak * 0.28) * (0.8 + 0.2 * v);
      px[0] *= sh; px[1] *= sh; px[2] *= sh;
      px[3] = bh;
      px[4] = 1;
    }, 3, false);
  });

/**
 * Drawn once top-to-bottom (v = 1 is the ground surface), tiling sideways. Bands, as fractions of the face height:
 * a thin fill/yard layer, dark topsoil with roots and stones, subsoil, then cracked rock.
 */
export const soilSection = (o: { seed?: number; top?: string } = {}) =>
  memo("soil" + JSON.stringify(o), () => {
    const { seed = 31 } = o;
    const FILL = hex(o.top ?? "#c58c5c"), TOP = hex("#4a3320"), SUB = hex("#7a4f2e"), ROCK = hex("#77726a"), ROCKD = hex("#4f4b45"), STONE = hex("#8f867a");
    return build(2048, 1024, (u, v, px) => {
      const d = 1 - v; // depth 0 (surface) … 1 (bottom)
      const wob = (fbm(u, 0.5, 6, 3, seed, 1) - 0.5) * 0.05;
      const big = fbm(u, v, 8, 5, seed + 2, 4), mid = fbm(u, v, 48, 4, seed + 4, 24), fine = fbm(u, v, 320, 2, seed + 6, 160);
      const t1 = ss(0.035 + wob * 0.2, 0.05 + wob * 0.2, d); // fill → topsoil
      const t2 = ss(0.3 + wob, 0.42 + wob, d + (big - 0.5) * 0.12); // topsoil → subsoil
      const t3 = ss(0.58 + wob, 0.66 + wob, d + (big - 0.5) * 0.1); // subsoil → rock
      lerp3(px, FILL, TOP, t1);
      lerp3(px, [px[0], px[1], px[2]], SUB, t2);
      lerp3(px, [px[0], px[1], px[2]], ROCK, t3);
      // mottling: dark clods in the soil, pale stones, cracks in the rock
      const clod = ss(0.5, 0.72, mid) * (1 - t3) * t1;
      lerp3(px, [px[0], px[1], px[2]], TOP, clod * 0.55 * (0.3 + t2));
      const st = ss(0.72, 0.75, fbm(u, v, 110, 2, seed + 9, 55)) * t1 * (1 - t3 * 0.5);
      lerp3(px, [px[0], px[1], px[2]], STONE, st * 0.55);
      const crack = 1 - ss(0.0, 0.035, Math.abs(fbm(u, v, 10, 4, seed + 12, 5) - 0.5));
      lerp3(px, [px[0], px[1], px[2]], ROCKD, crack * t3 * 0.85);
      const sh = (0.8 + mid * 0.3 + fine * 0.1) * (1 - 0.1 * d);
      px[0] *= sh; px[1] *= sh; px[2] *= sh;
      px[3] = mid * 0.7 + fine * 0.5 + st * 1.2 - crack * t3 * 1.5 + big * 0.4;
    }, 3, false);
  });

/** River stones packed in mortar: jittered cells, each a stone with its own tone, pale recessed joints. */
export const rubble = (o: { seed?: number; cells?: number } = {}) =>
  memo("rubble" + JSON.stringify(o), () => {
    const { seed = 43, cells = 7 } = o;
    const S1 = hex("#8f8b84"), S2 = hex("#5f5c58"), S3 = hex("#a79f90"), M = hex("#b8b1a4");
    return build(512, 512, (u, v, px) => {
      const x = u * cells, y = v * cells;
      const xi = Math.floor(x), yi = Math.floor(y);
      let d1 = 9, d2 = 9, id = 0;
      for (let dy = -1; dy <= 1; dy++)
        for (let dx = -1; dx <= 1; dx++) {
          const cx = wrap(xi + dx, cells), cy = wrap(yi + dy, cells);
          const fx = xi + dx + 0.5 + (hash2(cx, cy, seed) - 0.5) * 0.8, fy = yi + dy + 0.5 + (hash2(cx, cy, seed + 1) - 0.5) * 0.8;
          const dd = Math.hypot(x - fx, y - fy);
          if (dd < d1) { d2 = d1; d1 = dd; id = cx * 31 + cy; } else if (dd < d2) d2 = dd;
        }
      const edge = d2 - d1; // 0 on the joint between two stones
      const stone = ss(0.05, 0.14, edge);
      const r = hash2(id, 3, seed), r2 = hash2(id, 5, seed);
      lerp3(px, S1, S2, r * 0.8);
      if (r2 > 0.75) lerp3(px, [px[0], px[1], px[2]], S3, 0.6);
      const grain = fbm(u, v, 90, 3, seed + 7);
      const sh = (0.82 + grain * 0.3) * (0.78 + 0.22 * ss(0.05, 0.4, edge));
      px[0] = M[0] + (px[0] * sh - M[0]) * stone;
      px[1] = M[1] + (px[1] * sh - M[1]) * stone;
      px[2] = M[2] + (px[2] * sh - M[2]) * stone;
      px[3] = stone * (0.7 + 0.5 * ss(0.05, 0.5, edge)) + grain * 0.2;
    }, 4);
  });

// ------------------------------------------------------------------ roof-tile sheet (distant houses: rows of pan tiles in one texture)
export const tileSheet = (o: { seed?: number; cols?: number; rows?: number } = {}) =>
  memo("tileSheet" + JSON.stringify(o), () => {
    const { seed = 37, cols = 12, rows = 10 } = o;

      lerp3(px, A, B, t * 0.42);

// ------------------------------------------------------------------ leaf card (palm / banana / tree foliage) with alpha
export const leafCard = (kind: "frond" | "banana" | "canopy", seed = 41) =>
  memo("leaf" + kind + seed, () => {
    const G1 = hex("#4f8a2c"), G2 = hex("#2f5f1e"), G3 = hex("#86b244");
    const rnd = prng(seed * 31 + 7);
    const leaves = Array.from({ length: 64 }, (_, i) => {
      const a = rnd() * Math.PI * 2, r = Math.sqrt(rnd()) * 0.4;
      return { x: 0.5 + Math.cos(a) * r, y: 0.5 + Math.sin(a) * r, c: Math.cos(i * 2.4), s: Math.sin(i * 2.4), l: 0.075 + rnd() * 0.05, w: 0.032 + rnd() * 0.018, tone: rnd() };
    });
    return build(512, 512, (u, v, px) => {
      const x = u - 0.5;
      let a = 0, hgt = 0, tone = 0.4;
      if (kind === "frond") {
        // one solid blade with a saw-toothed edge; the leaflets are drawn as tone, not as see-through gaps
        // (a comb of alpha stripes shimmered against the sky whenever the camera moved)
        const n = 20, s = v * n + Math.abs(x) * 6;
        const f = s - Math.floor(s);
        const reach = 0.47 * Math.sin(Math.PI * Math.pow(v, 0.8)) + 0.02;
        a = Math.abs(x) < reach * (0.8 + 0.2 * (1 - Math.abs(f - 0.5) * 2)) ? 1 : 0;
        const rachis = Math.abs(x) < 0.012;
        hgt = 1 - Math.abs(f - 0.5) * 2;
        tone = 0.25 + (1 - hgt) * 0.5 + hash2(Math.floor(s), x > 0 ? 1 : 2, seed) * 0.25 + (rachis ? 0.3 : 0);
      } else if (kind === "banana") {
        const half = 0.4 * Math.pow(Math.sin(Math.PI * Math.pow(c01(v), 0.75)), 0.7);
        const tear = fbm(u, v, 40, 2, seed, 90);
        a = Math.abs(x) < half * (tear > 0.74 && Math.abs(x) > half * 0.35 ? 0 : 1) ? 1 : 0;
        const vein = Math.abs(((v * 46 + Math.abs(x) * 30) % 1) - 0.5);
        hgt = vein + (Math.abs(x) < 0.02 ? 1.5 : 0);
        tone = 0.25 + vein * 0.5 + (Math.abs(x) < 0.018 ? -0.6 : 0);
      } else {
        // a spray of small leaves (later leaves overlap earlier ones)
        for (const L of leaves) {
          const dx = u - L.x, dy = v - L.y;
          const lx = (dx * L.c + dy * L.s) / L.l, ly = (-dx * L.s + dy * L.c) / L.w;
          const q = lx * lx + ly * ly;
          if (q < 1) { a = 1; hgt = 1.6 - Math.abs(ly) * 1.2; tone = L.tone * 0.9 + Math.abs(ly) * 0.15; }
        }
      }
      lerp3(px, G1, G2, c01(tone));
      lerp3(px, [px[0], px[1], px[2]], G3, c01(hgt * 0.35 - 0.1));
      px[3] = hgt;
      px[4] = a;
    }, 1.5, false);
  });

let _noise: Texture | null = null;
export const noiseTex = () => {
  if (_noise) return _noise;
  const N = 512;
  const data = new Uint8Array(N * N * 4);
  for (let y = 0; y < N; y++)
    for (let x = 0; x < N; x++) {
      const u = (x + 0.5) / N, v = (y + 0.5) / N;
      const k = (y * N + x) * 4;
      data[k] = fbm(u, v, 6, 5, 51) * 255;
      data[k + 1] = fbm(u, v, 24, 4, 53) * 255;
      data[k + 2] = fbm(u, v, 96, 3, 55) * 255;
      data[k + 3] = 255;
    }
  const t = new DataTexture(data, N, N, RGBAFormat);
  t.colorSpace = NoColorSpace;
  t.wrapS = t.wrapT = RepeatWrapping;
  t.minFilter = LinearMipmapLinearFilter;
  t.magFilter = LinearFilter;
  t.generateMipmaps = true;
  t.needsUpdate = true;
  return (_noise = t);
};

import { ConeGeometry, CylinderGeometry, Matrix4, Quaternion, SphereGeometry, Vector3, type BufferGeometry, type Material } from "three";
import { flat, mat } from "./mats.ts";
import { weave } from "./tex.ts";
import { DIM, solve, type Pose, type V3 } from "./rig.ts";

const UNIT = { cyl: new Map<string, BufferGeometry>(), sph: new SphereGeometry(1, 36, 24) };
/** Tapered limb along +Y from y=0 to y=1 (scaled to length by the matrix). */
const limb = (r0: number, r1: number) => {
  const k = `${r0}|${r1}`;
  let g = UNIT.cyl.get(k);
  if (!g) {
    g = new CylinderGeometry(r1, r0, 1, 32, 1);
    g.translate(0, 0.5, 0);
    UNIT.cyl.set(k, g);
  }
  return g;
};

const Yv = new Vector3(0, 1, 0);
/** Matrix: unit +Y limb from a toward b, covering fractions [f0, f1] of the span. */
const seg = (a: V3, b: V3, f0 = 0, f1 = 1) => {
  const A = new Vector3(...a), B = new Vector3(...b);
  const d = B.clone().sub(A);
  const L = d.length();
  const q = new Quaternion().setFromUnitVectors(Yv, d.clone().normalize());
  return new Matrix4().compose(A.clone().add(d.clone().multiplyScalar(f0)), q, new Vector3(1, L * (f1 - f0), 1));
};
/** Matrix from a basis (x, y, z) at p with scale s. */
const basis = (p: V3, x: V3, y: V3, z: V3, s: V3) => {
  const m = new Matrix4().makeBasis(new Vector3(...x), new Vector3(...y), new Vector3(...z));
  m.scale(new Vector3(...s));
  m.setPosition(...p);
  return m;
};
const cross = (a: V3, b: V3): V3 => [a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0]];
const nrm = (a: V3): V3 => {
  const l = Math.hypot(a[0], a[1], a[2]) || 1;
  return [a[0] / l, a[1] / l, a[2] / l];
};
const add = (a: V3, b: V3, s = 1): V3 => [a[0] + b[0] * s, a[1] + b[1] * s, a[2] + b[2] * s];

const Part: React.FC<{ g: BufferGeometry; m: Matrix4; material: Material }> = ({ g, m, material }) => (
  <mesh geometry={g} material={material} matrix={m} matrixAutoUpdate={false} castShadow receiveShadow />
);
const Ball: React.FC<{ p: V3; r: number; material: Material; s?: V3 }> = ({ p, r, material, s = [1, 1, 1] }) => (
  <mesh geometry={UNIT.sph} material={material} position={p} scale={[r * s[0], r * s[1], r * s[2]]} castShadow receiveShadow />
);

export type WorkerLook = { skin?: string; shirt?: string; shorts?: string; hat?: "caping" | "none" };

export const Worker: React.FC<{ pos?: V3; rotY?: number; pose: Pose; look?: WorkerLook }> = ({ pos = [0, 0, 0], rotY = 0, pose, look = {} }) => {
  const j = solve(pose);
  const mSkin = flat(look.skin ?? "#c9a47e", 0.62);
  const mJoint = flat("#ad8660", 0.6);
  const mShirt = flat(look.shirt ?? "#2f8f8c", 0.86);
  const mShorts = flat(look.shorts ?? "#4a4d55", 0.9);
  const mSole = flat("#2b2622", 0.9);
  const mHat = mat("caping", weave({ strips: 40, a: "#dcc48e", b: "#b99a5c" }), { repeat: [3, 1], roughness: 0.75, double: true });
  const g = useMemo(() => {
    const hat = new ConeGeometry(0.27, 0.15, 28, 1, true);
    hat.translate(0, 0.075, 0);
    const foot = new SphereGeometry(1, 16, 10);
    return { hat, foot };
  }, []);

  const D = DIM;
  // torso: an elliptical trunk from pelvis to neck, in the torso frame
  const trunk = basis(add(j.pelvis, j.up, -0.05), j.right, j.up, j.fwd, [0.158, D.spine + 0.05, 0.108]);
  const chestM = basis(add(j.chest, j.up, -0.07), j.right, j.up, j.fwd, [0.2, 0.15, 0.125]);
  const hipM = basis(add(j.pelvis, [0, -0.02, 0]), [1, 0, 0], [0, 1, 0], [0, 0, 1], [0.165, 0.13, 0.125]);
  const headM = basis(j.head, cross(j.headUp, j.headFwd), j.headUp, j.headFwd, [0.098, 0.118, 0.108]);
  const hatM = basis(add(j.head, j.headUp, 0.045), cross(j.headUp, j.headFwd), j.headUp, j.headFwd, [1, 1, 1]);
  const foot = (ankle: V3, side: 1 | -1, toe = 0) => {
    const yaw = (pose.splay ?? 0.18) * side;
    const f: V3 = [Math.sin(yaw), 0, Math.cos(yaw)];
    const fw: V3 = nrm([f[0] * Math.cos(toe), -Math.sin(toe), f[2] * Math.cos(toe)]);
    const up: V3 = nrm(cross(fw, cross([0, 1, 0], fw)));
    const rt = cross(up, fw);
    const c = add(add(ankle, fw, 0.07), up, -0.04);
    return { top: basis(c, rt, up, fw, [0.048, 0.034, 0.118]), sole: basis(add(c, up, -0.026), rt, up, fw, [0.052, 0.014, 0.128]) };
  };
  const fL = foot(j.ankleL, 1, pose.toeL ?? 0), fR = foot(j.ankleR, -1, pose.toeR ?? 0);

  const arm = (sh: V3, el: V3, wr: V3, key: string) => {
    const dir = nrm([wr[0] - el[0], wr[1] - el[1], wr[2] - el[2]]);
    return (
      <group key={key}>
        <Ball p={sh} r={0.066} material={mShirt} />
        <Part g={limb(0.046, 0.04)} m={seg(sh, el)} material={mSkin} />
        <Part g={limb(0.058, 0.053)} m={seg(sh, el, 0, 0.62)} material={mShirt} />
        <Ball p={el} r={0.041} material={mJoint} />
        <Part g={limb(0.039, 0.03)} m={seg(el, wr)} material={mSkin} />
        <Ball p={wr} r={0.031} material={mJoint} />
        <Ball p={add(wr, dir, 0.05)} r={0.043} s={[0.85, 1.15, 0.75]} material={mSkin} />
      </group>
    );
  };
  const leg = (hip: V3, kn: V3, an: V3, f: { top: Matrix4; sole: Matrix4 }, key: string) => (
    <group key={key}>
      <Part g={limb(0.066, 0.052)} m={seg(hip, kn)} material={mSkin} />
      <Part g={limb(0.084, 0.074)} m={seg(hip, kn, 0, 0.6)} material={mShorts} />
      <Ball p={kn} r={0.055} material={mJoint} />
      <Part g={limb(0.05, 0.037)} m={seg(kn, an)} material={mSkin} />
      <Ball p={an} r={0.04} material={mJoint} />
      <Part g={g.foot} m={f.top} material={mSkin} />
      <Part g={g.foot} m={f.sole} material={mSole} />
    </group>
  );

  return (
    <group position={pos} rotation={[0, rotY, 0]}>
      <Part g={limb(1, 0.92)} m={trunk} material={mShirt} />
      <Part g={UNIT.sph} m={chestM} material={mShirt} />
      <Part g={UNIT.sph} m={hipM} material={mShorts} />
      <Part g={limb(0.042, 0.04)} m={seg(j.neck, j.head)} material={mSkin} />
      <Part g={UNIT.sph} m={headM} material={mSkin} />
      {look.hat === "none" ? null : <Part g={g.hat} m={hatM} material={mHat} />}
      {arm(j.shL, j.elbowL, j.wristL, "aL")}
      {arm(j.shR, j.elbowR, j.wristR, "aR")}
      {leg(j.hipL, j.kneeL, j.ankleL, fL, "lL")}
      {leg(j.hipR, j.kneeR, j.ankleR, fR, "lR")}
    </group>
  );
};
