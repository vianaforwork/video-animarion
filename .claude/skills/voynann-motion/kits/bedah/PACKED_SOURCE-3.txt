==== 1469 project/scripts/gl.mjs
// WebGL renderer choice for headless Chrome, shared by every render script.
// Order: GL env var > the project's .gl-mode file > "angle" (GPU). When the GPU path cannot create a WebGL
// context (no GPU, some VMs/drivers), the scripts switch to "swangle" (software, much slower) automatically
// and remember it in .gl-mode, so later renders and `npx remotion render` (remotion.config.ts) use it too.
import fs from "node:fs";

const FILE = ".gl-mode";

// When the GPU path crashes, Remotion may reject an internal "Target closed" promise that nothing awaits.
// That rejection is noise from the dead tab, not a render failure; real errors still reach the caller.
export const tolerateClosedTargets = () =>
  process.on("unhandledRejection", (e) => {
    if (/Target closed/i.test(String(e?.message ?? e))) return;
    console.error(e);
    process.exit(1);
  });

export const glMode = () =>
  process.env.GL || (fs.existsSync(FILE) ? fs.readFileSync(FILE, "utf8").trim() : "") || "angle";

export const isWebGLError = (e) =>
  /webgl|WebGLRenderer|GL context|BindToCurrentSequence|GL_VENDOR = Disabled/i.test(String(e?.stack ?? e?.message ?? e));

export const switchToSoftware = () => {
  fs.writeFileSync(FILE, "swangle\n");
  console.log(
    "\n! GPU 3D rendering is not available on this computer: switched to software rendering (GL=swangle, much slower)." +
      "\n  Saved in .gl-mode; delete that file to try the GPU again.\n",
  );
  return "swangle";
};

==== 1838 project/scripts/qa-blink.mjs
// Blink scan: mean absolute luma difference between consecutive frames; reports spikes that are not at a cut.
//   node scripts/qa-blink.mjs <video> <cutSeconds,comma,separated> [threshold=6]
import { execFileSync } from "node:child_process";
const [file, cutsArg = "", thrArg = "6"] = process.argv.slice(2);
const cuts = cutsArg.split(",").filter(Boolean).map(Number);
const probe = execFileSync("ffprobe", ["-v", "error", "-select_streams", "v:0", "-show_entries", "stream=r_frame_rate", "-of", "csv=p=0", file], { encoding: "utf8" }).trim();
const [n, d] = probe.split("/").map(Number);
const fps = n / (d || 1);
const W = 90, H = 160;
const raw = execFileSync("ffmpeg", ["-v", "error", "-i", file, "-vf", `scale=${W}:${H},format=gray`, "-f", "rawvideo", "-"], { maxBuffer: 1 << 30 });
const frames = raw.length / (W * H);
const diffs = [];
for (let f = 1; f < frames; f++) {
  let s = 0;
  const a = (f - 1) * W * H, b = f * W * H;
  for (let i = 0; i < W * H; i++) s += Math.abs(raw[b + i] - raw[a + i]);
  diffs.push(s / (W * H));
}
const sorted = [...diffs].sort((x, y) => x - y);
const med = sorted[Math.floor(sorted.length / 2)];
const thr = Number(thrArg);
const hits = [];
diffs.forEach((v, i) => {
  const t = (i + 1) / fps;
  const atCut = cuts.some((c) => Math.abs(c - t) <= 1.5 / fps);
  // a pop = a one-frame jump well above its neighbours
  const nb = Math.max(diffs[i - 1] ?? 0, diffs[i + 1] ?? 0);
  if (!atCut && v > thr && v > nb * 2.2) hits.push(`${t.toFixed(2)}s (frame ${i + 1}) diff ${v.toFixed(1)} vs neighbours ${nb.toFixed(1)}`);
});
console.log(`${frames} frames @ ${fps.toFixed(2)} fps, median diff ${med.toFixed(2)}, max ${Math.max(...diffs).toFixed(1)}`);
console.log(hits.length ? `pops outside cuts: ${hits.length}\n  ` + hits.join("\n  ") : "no pops outside cuts");
process.exit(hits.length ? 1 : 0);

==== 3511 project/scripts/qa-flicker.mjs
// Local flicker map: where does brightness oscillate frame-to-frame (up-down-up), as opposed to moving smoothly?
//   node scripts/qa-flicker.mjs <video> <cutSeconds,…> [outDir]
// Per shot: per-pixel count of sign flips of the temporal derivative with amplitude above a threshold, block-averaged.
// Writes <outDir>/flicker_shot<N>.png (heat over the shot's middle frame) and prints the worst blocks.
import { execFileSync } from "node:child_process";
import fs from "node:fs";
const [file, cutsArg = "", outDir = "out/flicker"] = process.argv.slice(2);
fs.mkdirSync(outDir, { recursive: true });
const probe = execFileSync("ffprobe", ["-v", "error", "-select_streams", "v:0", "-show_entries", "stream=r_frame_rate,width,height", "-of", "csv=p=0", file], { encoding: "utf8" }).trim().split(",");
const [n, d] = probe[2].split("/").map(Number);
const fps = n / (d || 1);
const W = 270, H = 480;
const raw = execFileSync("ffmpeg", ["-v", "error", "-i", file, "-vf", `scale=${W}:${H}:flags=area,format=gray`, "-f", "rawvideo", "-"], { maxBuffer: 1024 * 1024 * 1024 });
const frames = raw.length / (W * H);
const cuts = [0, ...cutsArg.split(",").filter(Boolean).map(Number), frames / fps];
const B = 15; // block size
const bw = W / B, bh = H / B;
let report = [];
for (let s = 0; s < cuts.length - 1; s++) {
  const f0 = Math.round(cuts[s] * fps) + 1, f1 = Math.round(cuts[s + 1] * fps) - 1;
  const score = new Float32Array(W * H);
  for (let p = 0; p < W * H; p++) {
    let flips = 0, prev = 0;
    for (let f = f0 + 1; f < f1; f++) {
      const dv = raw[f * W * H + p] - raw[(f - 1) * W * H + p];
      if (Math.abs(dv) >= 7) {
        const sg = Math.sign(dv);
        if (prev && sg !== prev) flips++;
        prev = sg;
      }
    }
    score[p] = flips / Math.max(1, f1 - f0) * fps; // flips per second
  }
  const blocks = [];
  for (let by = 0; by < bh; by++)
    for (let bx = 0; bx < bw; bx++) {
      let a = 0;
      for (let y = 0; y < B; y++) for (let x = 0; x < B; x++) a += score[(by * B + y) * W + bx * B + x];
      blocks.push({ bx, by, v: a / (B * B) });
    }
  blocks.sort((a, b) => b.v - a.v);
  const mean = blocks.reduce((a, b) => a + b.v, 0) / blocks.length;
  report.push(`shot ${s + 1} [${cuts[s].toFixed(2)}–${cuts[s + 1].toFixed(2)}s] mean ${mean.toFixed(2)} flips/s/px, worst: ` + blocks.slice(0, 5).map((b) => `(${(b.bx + 0.5) / bw * 100 | 0}%,${(b.by + 0.5) / bh * 100 | 0}%)=${b.v.toFixed(1)}`).join(" "));
  // heat image: red channel = score, over the grey mid frame
  const mid = Math.floor((f0 + f1) / 2);
  const rgb = Buffer.alloc(W * H * 3);
  for (let p = 0; p < W * H; p++) {
    const g = raw[mid * W * H + p] * 0.55, h = Math.min(1, score[p] / 6);
    rgb[p * 3] = Math.min(255, g + h * 255); rgb[p * 3 + 1] = g * (1 - h); rgb[p * 3 + 2] = g * (1 - h);
  }
  fs.writeFileSync(`${outDir}/s${s + 1}.rgb`, rgb);
  execFileSync("ffmpeg", ["-v", "error", "-y", "-f", "rawvideo", "-pix_fmt", "rgb24", "-s", `${W}x${H}`, "-i", `${outDir}/s${s + 1}.rgb`, `${outDir}/flicker_shot${s + 1}.png`]);
  fs.rmSync(`${outDir}/s${s + 1}.rgb`);
}
console.log(report.join("\n"));
const list = Array.from({ length: cuts.length - 1 }, (_, i) => ["-i", `${outDir}/flicker_shot${i + 1}.png`]).flat();
if (cuts.length - 1 > 1) execFileSync("ffmpeg", ["-v", "error", "-y", ...list, "-filter_complex", `hstack=inputs=${cuts.length - 1}`, `${outDir}/_all.jpg`]);
else execFileSync("ffmpeg", ["-v", "error", "-y", ...list, `${outDir}/_all.jpg`]);
console.log(`map: ${outDir}/_all.jpg`);

// Pose QA: every library pose must keep all joints above the ground, limbs un-stretched, and hands on target.
//   node scripts/qa-pose.mjs
import { DIM, POSES, solve } from "../src/kits/bedah/rig.ts";

const d = (a, b) => Math.hypot(a[0] - b[0], a[1] - b[1], a[2] - b[2]);
let bad = 0;
for (const [name, make] of Object.entries(POSES)) {
  const p = make();
  const j = solve(p);
  const issues = [];
  const R = { kneeL: 0.05, kneeR: 0.05, elbowL: 0.04, elbowR: 0.04, wristL: 0.03, wristR: 0.03, ankleL: 0.06, ankleR: 0.06 };
  for (const [k, r] of Object.entries(R)) if (j[k][1] < r - 0.005) issues.push(`${k} below ground (y=${j[k][1].toFixed(3)})`);
  for (const s of ["L", "R"]) {
    const miss = d(j["wrist" + s], p["hand" + s]);
    if (miss > 0.01) issues.push(`hand${s} misses target by ${(miss * 100).toFixed(1)} cm`);
    const fm = d(j["ankle" + s], p["foot" + s]);
    if (fm > 0.01) issues.push(`foot${s} misses target by ${(fm * 100).toFixed(1)} cm`);
    const legStretch = d(j["hip" + s], j["ankle" + s]) / (DIM.thigh + DIM.shin);
    if (legStretch > 0.985) issues.push(`leg${s} locked (${legStretch.toFixed(3)})`);
  }
  if (j.head[1] < 0.3) issues.push("head too low");
  const fmt = (v) => v.map((x) => x.toFixed(2)).join(",");
  console.log(`${issues.length ? "✗" : "✓"} ${name.padEnd(8)} knee ${fmt(j.kneeL)} | elbow ${fmt(j.elbowL)} | head ${fmt(j.head)}${issues.length ? "\n    " + issues.join("\n    ") : ""}`);
  bad += issues.length;
}
console.log(bad ? `\n${bad} issue(s)` : "\nall poses clean");
process.exit(bad ? 1 : 0);

==== 3754 project/scripts/stills.mjs
// Visual QA: bundle once, render chosen frames of a composition, tile them into a contact sheet.
//   node scripts/stills.mjs <outDir> <frame,frame,...> <compositionId> [scale]
// Frames may be given in seconds with an "s" suffix (e.g. 4.5s), using the composition's fps.

import path from "node:path";
import fs from "node:fs";
import { execFileSync } from "node:child_process";
import { bundle } from "@remotion/bundler";
import { openBrowser, renderStill, selectComposition } from "@remotion/renderer";
import { glMode, isWebGLError, switchToSoftware, tolerateClosedTargets } from "./gl.mjs";

tolerateClosedTargets();

const [outDir, list, id, scaleArg = "1", propsFile] = process.argv.slice(2);
const inputProps = propsFile ? JSON.parse(fs.readFileSync(propsFile, "utf8")) : {};
if (!outDir || !list || !id) {
  console.error("usage: node scripts/stills.mjs <outDir> <frame,frame,...|4.5s,...> <compositionId> [scale]");
  process.exit(1);
}
fs.mkdirSync(outDir, { recursive: true });

const serveUrl = await bundle({ entryPoint: path.resolve("src/index.ts") });
// bundles are ~50 MB each; never leave them in the temp dir, even when a render throws
const cleanup = () => fs.rmSync(serveUrl, { recursive: true, force: true });
process.on("exit", cleanup);
const composition = await selectComposition({ serveUrl, id, inputProps: propsFile ? { plan: inputProps } : {} });
const frames = list.split(",").map((t) => (t.endsWith("s") ? Math.round(parseFloat(t) * composition.fps) : parseInt(t, 10)));
const files = [];
const renderAll = async (gl) => {
  // one browser for all stills: WebGL contexts are expensive to create and tear down
  const puppeteerInstance = await openBrowser("chrome", { chromiumOptions: { gl } });
  try {
    files.length = 0;
    for (const frame of frames) {
      const output = path.join(outDir, `f${String(frame).padStart(4, "0")}.png`);
      await renderStill({ composition, serveUrl, output, frame, inputProps: propsFile ? { plan: inputProps } : {}, scale: Number(scaleArg), chromiumOptions: { gl }, timeoutInMilliseconds: 180000, puppeteerInstance });
      files.push(output);
      process.stdout.write(`${frame} `);
    }
  } finally {
    // after a WebGL crash the tab may already be gone; closing must not hide the original error
    await puppeteerInstance.close({ silent: true }).catch(() => {});
  }
};
const gl = glMode();
try {
  await renderAll(gl);
} catch (e) {
  if (gl === "swangle" || !isWebGLError(e)) throw e;
  await renderAll(switchToSoftware());
}
console.log();
cleanup();

// contact sheet (needs ffmpeg; skipped if missing)
try {
  const w = Math.round(composition.width * Number(scaleArg)), h = Math.round(composition.height * Number(scaleArg));
  const cols = Math.min(4, files.length), rows = Math.ceil(files.length / cols), n = cols * rows;
  if (n > 1) {
    const inputs = files.flatMap((f) => ["-i", f]);
    const pads = Array.from({ length: n - files.length }, () => ["-f", "lavfi", "-i", `color=c=black:s=${w}x${h}`]).flat();
    const filter =
      Array.from({ length: n }, (_, i) => `[${i}:v]scale=${w}:${h},setsar=1[v${i}]`).join(";") + ";" +
      Array.from({ length: n }, (_, i) => `[v${i}]`).join("") +
      `xstack=inputs=${n}:layout=` + Array.from({ length: n }, (_, i) => `${(i % cols) * w}_${Math.floor(i / cols) * h}`).join("|") +
      `,scale='min(2160,iw)':-2`;
    execFileSync("ffmpeg", ["-v", "error", "-y", ...inputs, ...pads, "-filter_complex", filter, "-frames:v", "1", path.join(outDir, "_sheet.jpg")]);
    console.log(`sheet: ${path.join(outDir, "_sheet.jpg")}`);
  }
} catch {
  console.log("(contact sheet skipped: ffmpeg not available)");
}
// a render that failed on the GPU path leaves its timeout timer behind; do not wait for it
process.exit(0);

==== 2937 project/src/core/motion/easing.ts
// PRECISION easing vocabulary. Every curve is a pure function of normalised time, so the
// whole film stays deterministic: same frame in, same pose out.

export const clamp = (x: number, lo = 0, hi = 1) => Math.min(hi, Math.max(lo, x));
export const mix = (a: number, b: number, t: number) => a + (b - a) * t;
export const remap = (x: number, a: number, b: number) => clamp((x - a) / (b - a));

export const smoothstep = (t: number) => {
  const x = clamp(t);
  return x * x * (3 - 2 * x);
};

/** C2-continuous: zero velocity and acceleration at both ends. The house curve. */
export const smootherstep = (t: number) => {
  const x = clamp(t);
  return x * x * x * (x * (x * 6 - 15) + 10);
};

/** CSS-style cubic bezier easing (x1,y1,x2,y2), solved by Newton + bisection. */
export const bezier = (x1: number, y1: number, x2: number, y2: number) => {
  const cx = 3 * x1, bx = 3 * (x2 - x1) - cx, ax = 1 - cx - bx;
  const cy = 3 * y1, by = 3 * (y2 - y1) - cy, ay = 1 - cy - by;
  const sx = (t: number) => ((ax * t + bx) * t + cx) * t;
  const sy = (t: number) => ((ay * t + by) * t + cy) * t;
  const dx = (t: number) => (3 * ax * t + 2 * bx) * t + cx;
  return (x: number) => {
    const X = clamp(x);
    let t = X;
    for (let i = 0; i < 8; i++) {
      const e = sx(t) - X;
      const d = dx(t);
      if (Math.abs(e) < 1e-6) return sy(t);
      if (Math.abs(d) < 1e-6) break;
      t -= e / d;
    }
    let lo = 0, hi = 1;
    t = X;
    for (let i = 0; i < 30; i++) {
      const v = sx(t);
      if (Math.abs(v - X) < 1e-6) break;
      if (v < X) lo = t;
      else hi = t;
      t = (lo + hi) / 2;
    }
    return sy(t);
  };
};

// Named curves, tuned for product cinematography (long, confident tails; no bounce).
export const ease = {
  /** Default move: decisive start, very long settle. */
  glide: bezier(0.45, 0, 0.12, 1),
  /** Symmetric, weighty. Camera and large objects. */
  inOut: bezier(0.65, 0, 0.35, 1),
  /** Soft symmetric for subtle drifts. */
  sine: (t: number) => 0.5 - 0.5 * Math.cos(Math.PI * clamp(t)),
  /** Starts with energy (a magnet letting go), long deceleration. */
  release: bezier(0.2, 0.7, 0.1, 1),
  /** Gathers speed and arrives fast (magnetic pull into a dock). */
  attract: bezier(0.55, 0, 0.95, 0.55),
  /** Typographic reveal. */
  type: bezier(0.2, 0, 0, 1),
  out: bezier(0, 0, 0.2, 1),
  in: bezier(0.6, 0, 1, 1),
  smoother: smootherstep,
};

/** Eased progress of `frame` through [start, end]. */
export const seg = (frame: number, start: number, end: number, curve: (t: number) => number = ease.inOut) =>
  curve(remap(frame, start, end));

/**
 * Critically damped settle used after a mechanical stop: decays to zero with at most one
 * tiny overshoot. `t` in frames since the stop, `amp` the initial offset.
 */
export const settle = (t: number, amp: number, freq = 0.11, damping = 0.16) =>
  t <= 0 ? 0 : amp * Math.exp(-damping * t) * Math.cos(freq * t * Math.PI);

// Keyframe tracks with C1 continuity. Interior keys pass through with Catmull-Rom tangents
// (so a camera never "kinks" when one move hands over to the next); `hold` keys and the
// ends get zero tangents (a soft ease in/out). Tangents are clamped Fritsch–Carlson style
// so a curve never overshoots its keys — no accidental bounces.

/**
 * `m` overrides the tangent (units per frame), e.g. to arrive at speed for a magnetic snap;
 * `stop` makes the key an impact: it is reached at `m` and left at rest.
 */
export type Key = { f: number; v: number[]; hold?: boolean; m?: number[]; stop?: boolean };

const hermite = (p0: number, p1: number, m0: number, m1: number, t: number, h: number) => {
  const t2 = t * t, t3 = t2 * t;
  return (2 * t3 - 3 * t2 + 1) * p0 + (t3 - 2 * t2 + t) * h * m0 + (-2 * t3 + 3 * t2) * p1 + (t3 - t2) * h * m1;
};

const tangent = (keys: Key[], i: number, c: number) => {
  const k = keys[i];
  if (k.m) return k.m[c];
  if (k.hold || i === 0 || i === keys.length - 1) return 0;
  const a = keys[i - 1], b = keys[i + 1];
  const m = (b.v[c] - a.v[c]) / (b.f - a.f);
  // Monotone clamp: no overshoot past either neighbour.
  const d0 = (k.v[c] - a.v[c]) / (k.f - a.f);
  const d1 = (b.v[c] - k.v[c]) / (b.f - k.f);
  if (d0 * d1 <= 0) return 0;
  const lim = 3 * Math.min(Math.abs(d0), Math.abs(d1));
  return Math.sign(m) * Math.min(Math.abs(m), lim);
};

/** Sample a multi-channel track at `frame`. */
export const sample = (keys: Key[], frame: number): number[] => {
  if (frame <= keys[0].f) return keys[0].v.slice();
  const last = keys[keys.length - 1];
  if (frame >= last.f) return last.v.slice();
  let i = 0;
  while (i < keys.length - 2 && frame >= keys[i + 1].f) i++;
  const a = keys[i], b = keys[i + 1];
  const h = b.f - a.f;
  const t = (frame - a.f) / h;
  return a.v.map((_, c) =>
    hermite(a.v[c], b.v[c], a.stop ? 0 : tangent(keys, i, c), tangent(keys, i + 1, c), t, h),
  );
};

/** Convenience: build keys from [frame, ...values] tuples; append "hold" to pin a key. */
export const keys = (...rows: (number | "hold")[][]): Key[] =>
  rows.map((r) => {
    const hold = r[r.length - 1] === "hold";
    const nums = r.filter((x): x is number => typeof x === "number");
    return { f: nums[0], v: nums.slice(1), hold };
  });

// Camera grammar: a shot kind + what to frame → a camera that fits it in 9:16, with a slow steady drift.
// Angles are in degrees: azimuth 0 looks from the front (+z) at the subject, positive swings toward +x (the right
// gable); elevation is the camera's height angle above the subject's centre.

import type { Box, Cam, ShotKind, V3 } from "./types.ts";

type Preset = { az: number; el: number; fov: number; fit: number; rise: number; minY: number };
// fit = how much of the frame width the subject's bounding sphere takes (smaller = closer);
// rise = where the subject sits: the camera aims this fraction of the subject's height above its centre
const PRESETS: Record<ShotKind, Preset> = {
  wide: { az: 30, el: 4, fov: 42, fit: 0.98, rise: -0.2, minY: 1.6 },
  front: { az: 6, el: 5, fov: 32, fit: 1.05, rise: -0.24, minY: 1.6 },
  gable: { az: 80, el: 2, fov: 31, fit: 1.0, rise: -0.16, minY: 1.6 },
  close: { az: 38, el: 9, fov: 35, fit: 0.82, rise: 0.0, minY: 1.2 },
  high: { az: 42, el: 27, fov: 34, fit: 0.9, rise: 0.02, minY: 1.2 },
  top: { az: 56, el: 33, fov: 30, fit: 0.84, rise: -0.08, minY: 1.2 },
  low: { az: 40, el: -9, fov: 34, fit: 0.66, rise: 0.12, minY: 0.9 },
  section: { az: 4, el: -3, fov: 34, fit: 1.0, rise: 0.02, minY: 1.1 },
  detail: { az: 58, el: 10, fov: 30, fit: 1.0, rise: 0, minY: 0.6 },
  ground: { az: 34, el: 10, fov: 34, fit: 0.85, rise: -0.05, minY: 1.0 },
};

const rad = (d: number) => (d * Math.PI) / 180;
export const unionBox = (bs: Box[]): Box => ({
  min: [Math.min(...bs.map((b) => b.min[0])), Math.min(...bs.map((b) => b.min[1])), Math.min(...bs.map((b) => b.min[2]))],
  max: [Math.max(...bs.map((b) => b.max[0])), Math.max(...bs.map((b) => b.max[1])), Math.max(...bs.map((b) => b.max[2]))],
});
export const boxAround = (p: V3, r: number): Box => ({ min: [p[0] - r, Math.max(0, p[1] - r), p[2] - r], max: [p[0] + r, p[1] + r, p[2] + r] });

/** Camera for `kind` framing `box`. side = +1 swings toward +x, −1 toward −x; seed nudges the angle shot to shot. */
export const frame = (kind: ShotKind, box: Box, side: 1 | -1, move: "push" | "pull" | "still", cutZ: number, seed = 0, aspect = 9 / 16): { a: Cam; b: Cam } => {
  const P = PRESETS[kind];
  const c: V3 = [(box.min[0] + box.max[0]) / 2, (box.min[1] + box.max[1]) / 2, (box.min[2] + box.max[2]) / 2];
  const sx = box.max[0] - box.min[0], sy = box.max[1] - box.min[1], sz = box.max[2] - box.min[2];
  // landscape: a narrower vertical angle keeps the same lens feel across the wider frame
  const fov = aspect > 1 ? P.fov * 0.74 : P.fov;
  const hv = rad(fov) / 2, hh = Math.atan(Math.tan(hv) * aspect);
  const at = (azDeg: number, scale: number): Cam => {
    const az = rad(azDeg) * side, el = rad(P.el);
    // the subject's silhouette as seen from this azimuth: width across the view, height
    const w = Math.abs(sx * Math.cos(az)) + Math.abs(sz * Math.sin(az)), dpt = Math.abs(sx * Math.sin(az)) + Math.abs(sz * Math.cos(az));
    // landscape: more room above and below (the frame is short), aim a little lower so the cut ground shows
    const vk = aspect > 1 ? 1.75 : 1.25, rise = P.rise - (aspect > 1 ? 0.14 : 0);
    const dist = Math.max((w / 2 / Math.tan(hh)) * P.fit, (sy / 2 / Math.tan(hv)) * P.fit * vk) * scale + dpt / 2;
    const target: V3 = [c[0], c[1] + sy * rise, c[2]];
    let pos: V3 = [target[0] + Math.sin(az) * Math.cos(el) * dist, target[1] + Math.sin(el) * dist, target[2] + Math.cos(az) * Math.cos(el) * dist];
    if (pos[1] < P.minY) pos = [pos[0], P.minY, pos[2]];
    // "wide" shows the ground in section: stand beyond the cut, low
    if (kind === "wide" && pos[2] < cutZ + 4.5) pos = [pos[0], pos[1], cutZ + 4.5];
    return { pos, target, fov, shadowAt: [c[0], 0, c[2]], shadowBox: Math.min(22, Math.max(8, Math.hypot(sx, sz) * 1.15 + 3)) };
  };
  const wob = ((seed * 37) % 7) - 3; // ±3° so two shots of one kind never match exactly
  const far = at(P.az + wob, 1), near = at(P.az + wob - 3, 0.92);
  if (move === "still") return { a: far, b: far };
  if (move === "pull") {
    const out = at(P.az + wob + 4, kind === "wide" ? 1.35 : 1.18);
    return { a: near, b: { ...out, pos: [out.pos[0], out.pos[1] + (kind !== "wide" ? 0.5 : out.pos[1] < 3 ? 2.2 : 0.8), out.pos[2]] } };
  }
  return { a: far, b: near };
};

const prism = (x: number, y0: number, y1: number, w0: number, w1: number, z0: number, z1: number) => {
  const a = [x - w0 / 2, y0], b = [x + w0 / 2, y0], c = [x + w1 / 2, y1], d = [x - w1 / 2, y1];
  const pos: number[] = [], nor: number[] = [], uv: number[] = [], idx: number[] = [];

    idx.push(base, base + 1, base + 2, base, base + 2, base + 3);

// Camera moves for BEDAH shots: keyed in seconds, C1-smooth through the keys (Catmull-Rom tangents,
// zero speed at the first and last key), plus a deterministic handheld drift and impact shake.
// The same function drives the 3D camera and the 2D overlay projection (map labels, pins, lines).

import { PerspectiveCamera, Vector3 } from "three";
import { sample, type Key } from "../../core/motion/track.ts";

export type V3 = [number, number, number];
export type Shot = { pos: V3; target: V3; fov: number; roll?: number; focus?: number; blur?: number; shadowAt?: V3; shadowBox?: number };
export type CamKey = { t: number; pos: V3; target: V3; fov?: number; roll?: number; hold?: boolean };

export const clamp01 = (x: number) => (x < 0 ? 0 : x > 1 ? 1 : x);
export const lin = (s: number, a: number, b: number) => clamp01((s - a) / (b - a));
export const sm = (t: number) => {
  const x = clamp01(t);
  return x * x * (3 - 2 * x);
};
export const smoother = (t: number) => {
  const x = clamp01(t);
  return x * x * x * (x * (x * 6 - 15) + 10);
};
export const mix = (a: number, b: number, t: number) => a + (b - a) * t;
export const mix3 = (a: V3, b: V3, t: number): V3 => [mix(a[0], b[0], t), mix(a[1], b[1], t), mix(a[2], b[2], t)];

/** Smooth hash noise in 1D (deterministic). */
const h1 = (n: number) => {
  const x = Math.sin(n * 127.1 + 311.7) * 43758.5453;
  return x - Math.floor(x);
};
export const noise1 = (x: number, seed = 0) => {
  const i = Math.floor(x), f = x - i;
  const u = f * f * (3 - 2 * f);
  return mix(h1(i + seed * 57.3), h1(i + 1 + seed * 57.3), u) * 2 - 1;
};

/** Sample a list of camera keys at time s (seconds, local to the shot). */
export const camAt = (keys: CamKey[], s: number): Shot => {
  const k: Key[] = keys.map((c) => ({ f: c.t, v: [...c.pos, ...c.target, c.fov ?? 32, c.roll ?? 0], hold: c.hold }));
  const v = sample(k, s);
  return { pos: [v[0], v[1], v[2]], target: [v[3], v[4], v[5]], fov: v[6], roll: v[7] };
};

/**
 * Handheld drift (always on, tiny) + impact shake (decaying bursts at `hits` seconds).
 * Amplitudes are in world units at the target distance, so they scale with the framing.
 */
export const handheld = (shot: Shot, s: number, o: { drift?: number; hits?: number[]; hit?: number; seed?: number } = {}): Shot => {
  const seed = o.seed ?? 1;
  const d = new Vector3(...shot.pos).distanceTo(new Vector3(...shot.target));
  let ax = noise1(s * 0.35, seed) * (o.drift ?? 0.004) * d;
  let ay = noise1(s * 0.31, seed + 3) * (o.drift ?? 0.004) * d * 0.7;
  let roll = shot.roll ?? 0;
  for (const t of o.hits ?? []) {
    const a = s - t;
    if (a < 0 || a > 1.6) continue;
    const env = Math.exp(-a * 3.2) * (o.hit ?? 0.012) * d;
    ax += noise1(a * 22, seed + t) * env;
    ay += noise1(a * 25, seed + t + 9) * env;
    roll += noise1(a * 18, seed + t + 4) * env * 0.02 / Math.max(d, 1e-3);
  }
  const target: V3 = [shot.target[0] + ax, shot.target[1] + ay, shot.target[2]];
  return { ...shot, target, roll };
};

const tmpCam = new PerspectiveCamera();
/** Project a world point to overlay space (0..1, y down). `vis` is false behind the camera. */
export const project = (shot: Shot, aspect: number, p: V3, near = 0.1, far = 5000) => {
  tmpCam.fov = shot.fov;
  tmpCam.aspect = aspect;
  tmpCam.near = near;
  tmpCam.far = far;
  tmpCam.position.set(...shot.pos);
  tmpCam.up.set(Math.sin(shot.roll ?? 0), Math.cos(shot.roll ?? 0), 0);
  tmpCam.lookAt(...shot.target);
  tmpCam.updateProjectionMatrix();
  tmpCam.updateMatrixWorld();
  const v = new Vector3(...p).project(tmpCam);
  return { x: (v.x + 1) / 2, y: (1 - v.y) / 2, vis: v.z < 1 && v.z > -1 };
};

      await f.load();
      (document.fonts as unknown as { add(f: FontFace): void }).add(f);
    }),

      throw e;
    },
  );
  return ready;

// Geometry helpers. All UVs are in metres (1 UV unit = 1 m) so one material repeat value means the same real size
// on every piece; offsets decorrelate neighbouring pieces.

import React, { useEffect, useMemo } from "react";
import { BoxGeometry, BufferAttribute, BufferGeometry, CylinderGeometry, Float32BufferAttribute, Matrix4, PlaneGeometry, Quaternion, Vector3, type Material } from "three";
import type { V3 } from "./cam.ts";

/** Box with metre UVs. Faces: ±x use (d, h), ±y use (w, d), ±z use (w, h). `along` = axis the texture's u runs on for long members. */
export const boxUV = (w: number, h: number, d: number, ou = 0, ov = 0) => {
  const g = new BoxGeometry(w, h, d);
  const uv = g.getAttribute("uv") as BufferAttribute;
  const dims: [number, number][] = [[d, h], [d, h], [w, d], [w, d], [w, h], [w, h]];
  for (let f = 0; f < 6; f++)
    for (let i = 0; i < 4; i++) {
      const k = f * 4 + i;
      uv.setXY(k, uv.getX(k) * dims[f][0] + ou + f * 0.37, uv.getY(k) * dims[f][1] + ov + f * 0.21);
    }
  return g;
};

/** Plane in XY with metre UVs. */
export const planeUV = (w: number, h: number, ou = 0, ov = 0, sx = 1, sy = 1) => {
  const g = new PlaneGeometry(w, h, sx, sy);
  const uv = g.getAttribute("uv") as BufferAttribute;
  for (let i = 0; i < uv.count; i++) uv.setXY(i, uv.getX(i) * w + ou, uv.getY(i) * h + ov);
  return g;
};

const X = new Vector3(1, 0, 0), Y = new Vector3(0, 1, 0);
/** Matrix placing a unit-length +X member from a to b (returns length too). `roll` spins it about its own axis. */
export const spanMatrix = (a: V3, b: V3, roll = 0) => {
  const A = new Vector3(...a), B = new Vector3(...b);
  const dir = B.clone().sub(A);
  const len = dir.length();
  dir.normalize();
  const q = new Quaternion().setFromUnitVectors(X, dir);
  // keep the member's local Y as close to world up as possible (no random twist), then apply the roll
  const up = Y.clone().applyQuaternion(q);
  const want = Y.clone().sub(dir.clone().multiplyScalar(Y.dot(dir)));
  if (want.lengthSq() > 1e-6) {
    want.normalize();
    const ang = Math.atan2(up.clone().cross(want).dot(dir), up.dot(want));
    q.premultiply(new Quaternion().setFromAxisAngle(dir, ang));
  }
  if (roll) q.premultiply(new Quaternion().setFromAxisAngle(dir, roll));

/** Bamboo culm geometry along +X, length L: slightly swollen node rings every ~`node` metres. u around, v along (metres). */
export const culmGeometry = (L: number, r: number, node = 0.38, seed = 0) => {
  const rings: number[] = [];
  const n = Math.max(1, Math.round(L / node));
  for (let i = 0; i <= n; i++) {
    const x = (i / n) * L;
    if (i > 0) rings.push(x - 0.012);
    rings.push(x);
    if (i < n) rings.push(x + 0.012);
  }
  const seg = 10;
  const pos: number[] = [], nor: number[] = [], uv: number[] = [], idx: number[] = [];
  rings.forEach((x, ri) => {
    const isNode = ri % 3 === 1 || rings.length < 3;
    const rr = r * (isNode ? 1.09 : 1) * (1 - 0.1 * (x / L));
    for (let s = 0; s <= seg; s++) {
      const a = (s / seg) * Math.PI * 2;
      pos.push(x - L / 2, Math.cos(a) * rr, Math.sin(a) * rr);
      nor.push(0, Math.cos(a), Math.sin(a));
      uv.push(s / seg, x / (node * 4) + seed * 0.31);
    }
  });
  for (let ri = 0; ri < rings.length - 1; ri++)
    for (let s = 0; s < seg; s++) {
      const a = ri * (seg + 1) + s, b = a + seg + 1;
      idx.push(a, a + 1, b, a + 1, b + 1, b);
    }
  const g = new BufferGeometry();
  g.setAttribute("position", new Float32BufferAttribute(pos, 3));
  g.setAttribute("normal", new Float32BufferAttribute(nor, 3));
  g.setAttribute("uv", new Float32BufferAttribute(uv, 2));
  g.setIndex(idx);
  return g;
};

export type Yard = { x0: number; x1: number; z0: number; z1: number; seed?: number };

/** Rounded rectangle with a wandering edge (worn ground never ends on a ruler line). Flat in XZ at y. */
const patch = (y: Yard, grow: number, wobble: number, cutZ: number) => {
  const x0 = y.x0 - grow, x1 = y.x1 + grow, z0 = y.z0 - grow, z1 = Math.min(y.z1 + grow, cutZ);
  const s = new Shape();
  const pts: [number, number][] = [];
  const N = 160;
  const cx = (x0 + x1) / 2, cz = (z0 + z1) / 2, hx = (x1 - x0) / 2, hz = (z1 - z0) / 2;
  for (let i = 0; i < N; i++) {
    const a = (i / N) * Math.PI * 2;
    // superellipse → soft-cornered rectangle
    const c = Math.cos(a), sn = Math.sin(a);
    const e = 0.28;
    let px = cx + hx * Math.sign(c) * Math.pow(Math.abs(c), e), pz = cz + hz * Math.sign(sn) * Math.pow(Math.abs(sn), e);
    const n = (fbm(i / N, 0.5, 14, 3, (y.seed ?? 1) * 7 + 3, 1) - 0.5) * wobble;
    px += c * n;
    pz += sn * n;
    // the slab is cut straight at cutZ
    if (y.z1 + grow >= cutZ && pz > cutZ - 0.02) pz = cutZ;
    pts.push([px, Math.min(pz, cutZ)]);
  }
  pts.forEach(([px, pz], i) => (i ? s.lineTo(px, -pz) : s.moveTo(px, -pz)));
  const g = new ShapeGeometry(s);
  g.rotateX(-Math.PI / 2);
  // metre UVs from world xz
  const pos = g.getAttribute("position") as BufferAttribute, uv = g.getAttribute("uv") as BufferAttribute;
  for (let i = 0; i < pos.count; i++) uv.setXY(i, pos.getX(i), -pos.getZ(i));
  return g;
};

const rep = (t: Texture, rx: number, ry: number, ox = 0, oy = 0) => {
  const c = t.clone();

  c.offset.set(ox, oy);

export type MatOpts = {
  /** texture repeats per UV unit (geometry UVs are in metres when built with the geo helpers) */

  offset?: [number, number];
  color?: string;
  roughness?: number;
  metalness?: number;
  bump?: number;
  alphaTest?: number;
  side?: Side;
  double?: boolean;

  else if (o.side !== undefined) mm.side = o.side;

// The body is posed from a few targets: where the pelvis is, how the torso leans, where the hands and ankles go.
// Elbows and knees come from two-bone IK, so hands land exactly on props (no "tembus") and feet stay on the ground.
// Local frame: the worker faces +z, up is +y, the LEFT side is +x.

export type V3 = [number, number, number];

export type Pose = {
  pelvis: V3;
  /** torso lean forward (+) in radians, side lean toward the left (+), twist of the shoulders to the left (+) */
  pitch: number;
  roll?: number;
  twist?: number;
  /** head: nod down (+), turn left (+) — relative to the torso */
  nod?: number;
  turn?: number;
  handL: V3;
  handR: V3;
  /** ankle targets; `toe` pitches the foot (+ = heel up, as on a kneeling leg) */
  footL: V3;
  footR: V3;
  toeL?: number;
  toeR?: number;
  /** foot yaw (toes out +) */
  splay?: number;
  /** elbow pole bias: 0 = down/back, 1 = out to the side */
  elbowOut?: number;
  /** knee pole bias sideways (squat: knees apart) */
  kneeOut?: number;
};

export const DIM = {
  hipW: 0.09, hipDrop: 0.04, thigh: 0.42, shin: 0.41, ankle: 0.075,
  spine: 0.47, shoulderW: 0.185, shoulderUp: 0.43, upper: 0.29, fore: 0.26, hand: 0.075,
  neck: 0.07, head: 0.115,
};

const add = (a: V3, b: V3): V3 => [a[0] + b[0], a[1] + b[1], a[2] + b[2]];
const sub = (a: V3, b: V3): V3 => [a[0] - b[0], a[1] - b[1], a[2] - b[2]];
const mul = (a: V3, s: number): V3 => [a[0] * s, a[1] * s, a[2] * s];
const dot = (a: V3, b: V3) => a[0] * b[0] + a[1] * b[1] + a[2] * b[2];
const cross = (a: V3, b: V3): V3 => [a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0]];
const len = (a: V3) => Math.hypot(a[0], a[1], a[2]);
const norm = (a: V3): V3 => {
  const l = len(a) || 1;
  return [a[0] / l, a[1] / l, a[2] / l];
};

/** Two-bone IK: joint between root and target (lengths a, b), bent toward `pole`. Returns [mid, end]. */
export const ik2 = (root: V3, target: V3, a: number, b: number, pole: V3): [V3, V3] => {
  let d = sub(target, root);
  let L = len(d);
  const max = (a + b) * 0.999, min = Math.abs(a - b) + 1e-3;
  if (L < 1e-6) { d = [0, -1, 0]; L = 1e-6; }
  const dir = mul(d, 1 / L);
  const Lc = Math.min(max, Math.max(min, L));
  const x = (a * a - b * b + Lc * Lc) / (2 * Lc);
  const h = Math.sqrt(Math.max(0, a * a - x * x));
  let bend = sub(pole, mul(dir, dot(pole, dir)));
  if (len(bend) < 1e-5) bend = cross(dir, [1, 0, 0]);
  bend = norm(bend);
  const mid = add(add(root, mul(dir, x)), mul(bend, h));
  return [mid, add(root, mul(dir, Lc))];
};

export type Joints = {
  pelvis: V3; chest: V3; neck: V3; head: V3; up: V3; right: V3; fwd: V3; headUp: V3; headFwd: V3;
  hipL: V3; hipR: V3; kneeL: V3; kneeR: V3; ankleL: V3; ankleR: V3;
  shL: V3; shR: V3; elbowL: V3; elbowR: V3; wristL: V3; wristR: V3;
};

const rotX = (v: V3, a: number): V3 => [v[0], v[1] * Math.cos(a) - v[2] * Math.sin(a), v[1] * Math.sin(a) + v[2] * Math.cos(a)];
const rotY = (v: V3, a: number): V3 => [v[0] * Math.cos(a) + v[2] * Math.sin(a), v[1], -v[0] * Math.sin(a) + v[2] * Math.cos(a)];
const rotZ = (v: V3, a: number): V3 => [v[0] * Math.cos(a) - v[1] * Math.sin(a), v[0] * Math.sin(a) + v[1] * Math.cos(a), v[2]];

/** Solve a pose into joint positions (worker-local metres). */
export const solve = (p: Pose): Joints => {
  const D = DIM;
  const tw = p.twist ?? 0, rl = p.roll ?? 0;
  // torso frame: lean forward about x, lean sideways about z, twist about its own up axis
  const tf = (v: V3) => rotZ(rotX(rotY(v, tw), p.pitch), -rl);
  const up = tf([0, 1, 0]), right = tf([1, 0, 0]), fwd = tf([0, 0, 1]);
  const pelvis = p.pelvis;
  const chest = add(pelvis, mul(up, D.shoulderUp));
  const neck = add(pelvis, mul(up, D.spine));
  const hf = (v: V3) => tf(rotX(rotY(v, p.turn ?? 0), p.nod ?? 0));
  const headUp = hf([0, 1, 0]), headFwd = hf([0, 0, 1]);
  const head = add(neck, mul(headUp, D.neck + D.head * 0.75));
  // hips follow the pelvis without the shoulder twist
  const pr: V3 = rotZ([1, 0, 0], -rl * 0.5);
  const hipL = add(pelvis, add(mul(pr, D.hipW), [0, -D.hipDrop, 0]));
  const hipR = add(pelvis, add(mul(pr, -D.hipW), [0, -D.hipDrop, 0]));
  const ko = p.kneeOut ?? 0.15;
  const [kneeL, ankleL] = ik2(hipL, p.footL, D.thigh, D.shin, [ko, 0.25, 1]);
  const [kneeR, ankleR] = ik2(hipR, p.footR, D.thigh, D.shin, [-ko, 0.25, 1]);
  const shL = add(chest, mul(right, D.shoulderW)), shR = add(chest, mul(right, -D.shoulderW));
  const eo = p.elbowOut ?? 0.35;
  const poleL = add(add(mul(right, eo * 1.6), mul(up, -1)), mul(fwd, -0.55)), poleR = add(add(mul(right, -eo * 1.6), mul(up, -1)), mul(fwd, -0.55));
  const [elbowL, wristL] = ik2(shL, p.handL, D.upper, D.fore, poleL);
  const [elbowR, wristR] = ik2(shR, p.handR, D.upper, D.fore, poleR);
  return { pelvis, chest, neck, head, up, right, fwd, headUp, headFwd, hipL, hipR, kneeL, kneeR, ankleL, ankleR, shL, shR, elbowL, elbowR, wristL, wristR };
};

// Roof maths — pure (no React / three), shared by the kit, the director and the node scripts.
// A gable roof is described by RoofDims; `roofFrame` turns (side, x along the ridge, distance down the slope, lift
// above the plane) into world points, including the sag of an under-built frame.

export type V3 = [number, number, number];

export type RoofDims = {
  /** half length along the ridge (x), including the gable overhang */
  hw: number;
  ridgeY: number;
  /** horizontal run and vertical drop from the ridge to the eave edge */
  run: number;
  drop: number;
  /** sag of the ridge at mid-span in metres (a frame too weak for its load) */
  sag?: number;
  /** slope distance of the wall line that holds the rafters up (no sag there); default 75 % of the slope */
  support?: number;
};

export const roofFrame = (d: RoofDims) => {
  const sl = Math.hypot(d.run, d.drop);
  const cy = d.drop / sl, cz = d.run / sl;
  const sup = d.support ?? sl * 0.75;
  // the ridge hangs between the gables, the rafters bow between ridge and wall plate, the wall line stays put
  const sagAt = (x: number, dist: number) => {
    if (!d.sag) return 0;
    const ax = Math.pow(Math.max(0, Math.sin((Math.PI * (x + d.hw)) / (2 * d.hw))), 0.8);
    const u = Math.min(1, Math.max(0, dist / sup));
    const tail = dist > sup ? -0.25 * ((dist - sup) / Math.max(1e-3, sl - sup)) : 0; // the overhang tips up a little
    return d.sag * ax * (1 - u + 0.45 * Math.sin(Math.PI * u) + tail * 0.3);
  };
  return {
    sl,
    /** point on side s (+1 front, −1 back) at ridge-position x, `dist` metres down the slope, `lift` above the plane */
    pt: (s: number, x: number, dist: number, lift = 0): V3 => [x, d.ridgeY - dist * cy + lift * cz - sagAt(x, dist), s * (dist * cz + lift * cy)],
    normal: (s: number): V3 => [0, cz, s * cy],
    down: (s: number): V3 => [0, -cy, s * cz],
    sagAt,
  };
};

export const TILE = { W: 0.24, cover: 0.2, L: 0.34, gauge: 0.25, A: 0.013, T: 0.012 };


/** plate = where the roof plane crosses the wall line; bamboo members stop below it so nothing pokes through the thatch */
export const HUT = { hx: 2.1, hz: 1.5, plate: 2.25 };
/** tan of the roof pitch used by the film (≈ 32°, the usual "about 30°" for clay tiles) */
export const PITCH = 0.625;
/** Roof whose plane passes through the wall line (z = ±hz) at plate height, for a given slope (rise / run). */
export const roofForSlope = (slope: number, run = 2.0, hw = 2.5): RoofDims => ({ hw, ridgeY: HUT.plate + HUT.hz * slope, run, drop: run * slope });
export const HUT_ROOF: RoofDims = roofForSlope(PITCH);


/** Rows a ThatchRoof needs before its ridge cap (add 1.3 for cap + pole). */
export const thatchRows = (dims: RoofDims, gauge = 0.2) => Math.ceil((roofFrame(dims).sl + 0.06 - 0.3) / gauge) + 1;


// Stack under the roof plane (perpendicular metres): reng 0…0.025 above, usuk −0.08…−0.01, purlins −0.205…−0.085,
// principal rafter −0.345…−0.205. Nothing interpenetrates.
const UNDER = 0.275;
/** Joints of one truss (in an x = const plane): chord top `seatY`, apex height, and where the rafter meets the chord. */
export const trussJoints = (dims: RoofDims, plate: number) => {
  const f = roofFrame(dims);
  const cy = dims.drop / f.sl, cz = dims.run / f.sl;
  const seatY = plate + 0.27;
  const apexY = dims.ridgeY - UNDER / cz;
  const footZ = ((apexY - seatY) / cy) * cz;
  return { seatY, apexY, footZ };
};
/** Ridge height that makes a truss with these slopes sit exactly on walls at z = ±hz with top plate `plate`. */
export const ridgeForWalls = (hz: number, plate: number, run: number, drop: number) => {
  const sl = Math.hypot(run, drop), cy = drop / sl, cz = run / sl;
  return plate + 0.27 + (hz / cz) * cy + UNDER / cz;
};
