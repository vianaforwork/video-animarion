/** Line times from whisper words (in order, tolerant to split names and spelled numbers). */
const alignWhisper = (wav, W) => {
  sh(W.bin, ["-m", W.model, "-f", wav, "-l", "id", "-ml", "1", "-sow", "-ojf", "-of", path.join(out, "vo_words")]);
  const j = JSON.parse(fs.readFileSync(path.join(out, "vo_words.json"), "utf8"));
  const heard = j.transcription.map((x) => ({ w: norm(x.text), a: x.offsets.from / 1000, b: x.offsets.to / 1000 })).filter((x) => x.w);
  let h = 0;
  return lines.map((ln) => {
    const ws = ln.split(/\s+/).map(norm).filter(Boolean);
    const got = [];
    for (const n of ws) {
      for (let k = h; k < Math.min(heard.length, h + 12); k++) {
        const two = k + 1 < heard.length ? heard[k].w + heard[k + 1].w : "";
        if (heard[k].w === n || (n.length > 3 && heard[k].w.startsWith(n.slice(0, 4)))) { got.push(heard[k]); h = k + 1; break; }
        if (two && (two === n || (n.length > 5 && two.startsWith(n.slice(0, 6))))) { got.push({ a: heard[k].a, b: heard[k + 1].b }); h = k + 2; break; }
      }
    }
    if (!got.length) return null;
    return { a: got[0].a, b: got[got.length - 1].b };
  });
};

/** Without whisper: cut the VO at its pauses, choosing the pauses nearest to each line's expected start. */
const alignPauses = (wav) => {
  const log = spawnSync("ffmpeg", ["-i", wav, "-af", "silencedetect=noise=-34dB:d=0.22", "-f", "null", "-"], { encoding: "utf8" }).stderr;
  const sil = [];
  let s0 = null;
  for (const m of log.matchAll(/silence_(start|end): ([\d.]+)/g)) {
    if (m[1] === "start") s0 = parseFloat(m[2]);
    else if (s0 !== null) { sil.push([s0, parseFloat(m[2])]); s0 = null; }
  }
  const dur = parseFloat(sh("ffprobe", ["-v", "error", "-show_entries", "format=duration", "-of", "csv=p=0", wav]));
  const start = sil.length && sil[0][0] < 0.05 ? sil[0][1] : 0;
  const end = sil.length && sil[sil.length - 1][1] > dur - 0.05 ? sil[sil.length - 1][0] : dur;
  const chars = lines.map((l) => l.replace(/\s+/g, "").length);
  const tot = chars.reduce((a, b) => a + b, 0);
  const cuts = [];
  let acc = 0;
  const free = sil.filter(([a, b]) => a > start + 0.3 && b < end - 0.3);
  for (let i = 0; i < lines.length - 1; i++) {
    acc += chars[i];
    const want = start + ((end - start) * acc) / tot;
    let best = null;
    for (const p of free) if (!cuts.includes(p) && (!best || Math.abs((p[0] + p[1]) / 2 - want) < Math.abs((best[0] + best[1]) / 2 - want))) best = p;
    cuts.push(best ?? [want, want]);
  }
  cuts.sort((x, y) => x[0] - y[0]);
  return lines.map((_, i) => ({ a: i === 0 ? start : cuts[i - 1][1], b: i === lines.length - 1 ? end : cuts[i][0] }));
};

/** whisper.cpp stamps run a little early: move each line's start to the speech onset and its end to where the
 *  speech run really stops, so no syllable is cut when pages are moved apart. */
const snapToSpeech = (t, wav) => {
  const log = spawnSync("ffmpeg", ["-i", wav, "-af", "silencedetect=noise=-34dB:d=0.18", "-f", "null", "-"], { encoding: "utf8" }).stderr;
  const sil = [];
  let s0 = null;
  for (const m of log.matchAll(/silence_(start|end): ([\d.]+)/g)) {
    if (m[1] === "start") s0 = parseFloat(m[2]);
    else if (s0 !== null) { sil.push([s0, parseFloat(m[2])]); s0 = null; }
  }
  return t.map((x, i) => {
    let a = x.a, b = x.b;
    const inA = sil.find(([p, q]) => a >= p - 0.02 && a < q);
    if (inA && inA[1] - a < 0.8) a = inA[1];
    const next = sil.find(([p]) => p >= b - 0.05);
    if (next && next[0] - b < 0.8) b = next[0];
    if (i + 1 < t.length) b = Math.min(b, t[i + 1].a - 0.02);
    return { a, b: Math.max(b, a + 0.3) };
  });
};

==== 13408 project/scripts/book-sound.mjs
#!/usr/bin/env node
// STORYBOOK sound for a planned book: narration (the buyer's VO re-spaced per page, or TTS line clips) + original
// synthesised sound (no samples): music-box lullaby ducked under the voice, cover, page turns, footsteps, sparkles,
// wind, rain, lamp click. Instruments are the approved prototype's (scripts/make-audio.mjs).
//   node scripts/book-sound.mjs <plan.json> <out.wav> [--vo <file>] [--lines <dir with l0.aiff…>]
import fs from "node:fs";
import { execFileSync } from "node:child_process";

const args = process.argv.slice(2);
const [planFile, outFile] = args.filter((a, i) => !a.startsWith("--") && !(args[i - 1] ?? "").startsWith("--"));
const opt = (k) => { const i = args.indexOf(k); return i >= 0 ? args[i + 1] : undefined; };
const plan = JSON.parse(fs.readFileSync(planFile, "utf8"));
const SR = 48000;
const LEN = plan.duration + 0.3;
const N = Math.round(LEN * SR);
const L = new Float32Array(N), R = new Float32Array(N);
let seed = 20261110;
const rnd = () => {
  seed |= 0;
  seed = (seed + 0x6d2b79f5) | 0;
  let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
  t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
  return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
};
const noise = () => rnd() * 2 - 1;
const db = (d) => Math.pow(10, d / 20);
const buf = (sec) => new Float32Array(Math.max(1, Math.round(sec * SR)));

/** Band-pass biquad (RBJ) with an optionally sweeping centre (f0 may be a function of progress 0…1). */
const bandpass = (x, f0, q = 1) => {
  const y = new Float32Array(x.length);
  let x1 = 0, x2 = 0, y1 = 0, y2 = 0;
  for (let i = 0; i < x.length; i++) {
    const f = typeof f0 === "function" ? f0(i / x.length) : f0;
    const w = (2 * Math.PI * f) / SR, al = Math.sin(w) / (2 * q), a0 = 1 + al;
    const b0 = al / a0, b2 = -al / a0, a1 = (-2 * Math.cos(w)) / a0, a2 = (1 - al) / a0;
    const v = b0 * x[i] + b2 * x2 - a1 * y1 - a2 * y2;
    x2 = x1; x1 = x[i]; y2 = y1; y1 = v; y[i] = v;
  }
  return y;
};
const lowpass = (x, fc) => {
  const y = new Float32Array(x.length);
  const a = 1 - Math.exp((-2 * Math.PI * fc) / SR);
  let v = 0;
  for (let i = 0; i < x.length; i++) y[i] = v += a * (x[i] - v);
  return y;
};
const env = (x, fn) => x.map((v, i) => v * fn(i / SR, i / x.length));
const peakNorm = (x, peak = 1) => {
  let m = 0;
  for (const v of x) m = Math.max(m, Math.abs(v));
  return x.map((v) => (v / (m || 1)) * peak);
};
/** Place a mono sound at film time t (s), gain dB, pan −1…1. */
const place = (x, t, gainDb = 0, pan = 0) => {
  const g = db(gainDb), gl = g * Math.cos(((pan + 1) * Math.PI) / 4) * Math.SQRT2, gr = g * Math.sin(((pan + 1) * Math.PI) / 4) * Math.SQRT2;
  const at = Math.round(t * SR);
  for (let i = 0; i < x.length; i++) {
    const j = at + i;
    if (j < 0 || j >= N) continue;
    L[j] += x[i] * gl;
    R[j] += x[i] * gr;
  }
};

// ── Instruments ──────────────────────────────────────────────────────────────

const hz = (midi) => 440 * Math.pow(2, (midi - 69) / 12);
/** Music-box tine: fundamental + octave + a bright inharmonic partial that dies fast. */
const tine = (midi, dur = 1.8, bright = 1) => {
  const f = hz(midi);
  return buf(dur).map((_, i) => {
    const t = i / SR;
    const a = Math.min(1, t / 0.002);
    return a * (Math.sin(2 * Math.PI * f * t) * Math.exp(-t / 0.9) +
      0.28 * Math.sin(2 * Math.PI * f * 2.001 * t) * Math.exp(-t / 0.45) +
      0.12 * bright * Math.sin(2 * Math.PI * f * 5.43 * t) * Math.exp(-t / 0.07));
  });
};
/** Soft bell for sparkles (star sounds): two inharmonic partials. */
const bell = (midi, dur = 1.4) => {
  const f = hz(midi);
  return buf(dur).map((_, i) => {
    const t = i / SR;
    return Math.min(1, t / 0.003) * (Math.sin(2 * Math.PI * f * t) * Math.exp(-t / 0.5) + 0.35 * Math.sin(2 * Math.PI * f * 2.76 * t) * Math.exp(-t / 0.22) + 0.15 * Math.sin(2 * Math.PI * f * 5.4 * t) * Math.exp(-t / 0.08));
  });
};
/** Warm pad: detuned soft saw-ish (few harmonics) with slow attack. */
const pad = (midis, dur) =>
  buf(dur).map((_, i) => {
    const t = i / SR;
    const a = Math.min(1, t / 1.2) * Math.min(1, (dur - t) / 1.4);
    let v = 0;
    for (const m of midis) {
      const f = hz(m);
      for (const d of [-0.12, 0.11]) {
        const ff = f * Math.pow(2, d / 12);
        v += Math.sin(2 * Math.PI * ff * t) + 0.25 * Math.sin(4 * Math.PI * ff * t) + 0.08 * Math.sin(6 * Math.PI * ff * t);
      }
    }
    return (v * a) / (midis.length * 2);
  });

// ── Paper and room sounds ────────────────────────────────────────────────────

/** A page turning: corner crinkle → air swish that peaks as the page stands → soft landing flap. */
const pageTurn = (dur) => {
  const n = buf(dur + 0.4).map(noise);
  const swish = env(bandpass(n, (p) => 900 + 2600 * Math.sin(Math.PI * Math.min(1, p * 1.15)), 0.7), (t) => {
    const p = t / dur;
    return p > 1 ? Math.exp(-(t - dur) / 0.05) * 0.4 : Math.pow(Math.sin(Math.PI * p), 1.6) * 0.8 + 0.2 * Math.exp(-t / 0.15);
  });
  const land = buf(dur + 0.4);
  const la = Math.round(SR * (dur - 0.06));
  for (let i = 0; la + i < land.length; i++) {
    const t = i / SR;
    land[la + i] = (noise() * 0.6 * Math.exp(-t / 0.035) + Math.sin(2 * Math.PI * 110 * t) * Math.exp(-t / 0.06) * 0.5);
  }
  return peakNorm(swish.map((v, i) => v + land[i] * 0.55), 1);
};
const pageTurnFull = (dur) => {
  const base = pageTurn(dur);
  const cr = buf(dur + 0.4);
  for (let k = 0; k < 16; k++) {
    const at = Math.round(SR * (0.02 + 0.35 * rnd()));
    const amp = 0.4 + 0.6 * rnd();
    for (let i = 0; i < 260 && at + i < cr.length; i++) cr[at + i] += noise() * Math.exp(-i / 35) * amp;
  }
  const crb = peakNorm(bandpass(cr, 4200, 1.2), 0.35);
  return peakNorm(base.map((v, i) => v + crb[i]), 1);
};
/** Hardcover landing on the table + cloth creak while it opens. */
const coverOpen = (dur) => {
  const creak = env(bandpass(buf(dur).map(noise), (p) => 380 + 260 * p, 6), (t) => Math.sin(Math.PI * Math.min(1, t / dur)) * 0.35);
  const thump = buf(dur + 0.5);
  const at = Math.round(SR * dur);
  for (let i = 0; at + i < thump.length; i++) {
    const t = i / SR;
    thump[at + i] = Math.sin(2 * Math.PI * (85 - t * 60) * t) * Math.exp(-t / 0.09) + noise() * 0.25 * Math.exp(-t / 0.025);
  }
  const out = buf(dur + 0.5);
  for (let i = 0; i < out.length; i++) out[i] = (creak[i] ?? 0) + thump[i];
  return peakNorm(out, 1);
};
const footstep = () => peakNorm(env(bandpass(buf(0.16).map(noise), 2200, 0.8), (t) => Math.min(1, t / 0.01) * Math.exp(-t / 0.05)), 1);
const wind = (dur) => peakNorm(env(lowpass(bandpass(buf(dur).map(noise), (p) => 500 + 500 * Math.sin(p * 9), 0.6), 2400), (t) => Math.sin(Math.PI * Math.min(1, t / dur)) * (0.7 + 0.3 * Math.sin(t * 2.3))), 1);
const click = () => peakNorm(env(bandpass(buf(0.08).map(noise), 2600, 2), (t) => Math.exp(-t / 0.006)).map((v, i) => v + Math.sin(2 * Math.PI * 900 * (i / SR)) * Math.exp(-(i / SR) / 0.01) * 0.4), 1);
const poff = () => peakNorm(env(lowpass(buf(0.35).map(noise), 900), (t) => Math.min(1, t / 0.01) * Math.exp(-t / 0.08)), 1);

// ── Score: an original music-box lullaby in F, 3/4, 66 bpm ─────────────────

const BEAT = 60 / 66;
// melody (midi, beats) — two phrases of 8 bars

// Procedural, deterministic textures (no image assets): perforated grilles and boards.

import { DataTexture, LinearFilter, LinearMipmapLinearFilter, RepeatWrapping, RGBAFormat, SRGBColorSpace, type ColorSpace } from "three";

const toTexture = (data: Uint8Array, w: number, h: number, colorSpace: ColorSpace = SRGBColorSpace) => {
  const t = new DataTexture(data, w, h, RGBAFormat);
  t.colorSpace = colorSpace;
  t.generateMipmaps = true;
  t.minFilter = LinearMipmapLinearFilter;
  t.magFilter = LinearFilter;
  t.wrapS = t.wrapT = RepeatWrapping;
  t.anisotropy = 8;
  t.needsUpdate = true;
  return t;
};

type Rgb = [number, number, number];
const hex = (h: string): Rgb => [parseInt(h.slice(1, 3), 16), parseInt(h.slice(3, 5), 16), parseInt(h.slice(5, 7), 16)];

/**
 * Hex-packed micro perforation. Alpha carries the outline mask (stadium / circle) for
 * alpha-to-coverage, so edges stay anti-aliased under MSAA.
 */
export const perforation = ({
  w = 1024, h = 1024, across = 18, hole = 0.36, base = "#3a3a3d", rim = "#55565a", deep = "#0d0d0f",
  mask = "none" as "none" | "stadium" | "circle",
}) => {
  const data = new Uint8Array(w * h * 4);
  const pitch = w / across;
  const rowH = pitch * Math.sqrt(3) / 2;
  const B = hex(base), R = hex(rim), D = hex(deep);
  const r = pitch * hole;
  for (let y = 0; y < h; y++)
    for (let x = 0; x < w; x++) {
      const row = Math.round(y / rowH);
      const cy = row * rowH;
      const off = row % 2 ? pitch / 2 : 0;
      const cx = Math.round((x - off) / pitch) * pitch + off;
      // nearest of the two candidate rows keeps the hex lattice exact
      let d = Math.hypot(x - cx, y - cy);
      const row2 = row + (y > cy ? 1 : -1);
      const off2 = row2 % 2 ? pitch / 2 : 0;
      const cx2 = Math.round((x - off2) / pitch) * pitch + off2;
      d = Math.min(d, Math.hypot(x - cx2, y - row2 * rowH));
      const inHole = Math.min(1, Math.max(0, (r - d) / 1.2));
      const onRim = Math.min(1, Math.max(0, 1 - Math.abs(d - r - 1.2) / 1.6));
      const c = B.map((b, i) => b + (R[i] - b) * onRim * 0.8 + (D[i] - b) * inHole);
      let a = 255;
      if (mask !== "none") {
        const u = (x + 0.5) / w - 0.5, v = (y + 0.5) / h - 0.5;
        let dist: number;
        if (mask === "circle") dist = 0.5 - Math.hypot(u, v);
        else {
          // stadium in a w×h box (aspect-correct)
          const aspect = w / h;
          const hx = 0.5 * aspect - 0.5;
          const px = Math.max(Math.abs(u * aspect) - hx, 0);
          dist = 0.5 - Math.hypot(px, v);
        }
        a = Math.round(Math.min(1, Math.max(0, dist * h * 0.6 + 0.5)) * 255);
      }
      const k = (y * w + x) * 4;
      data[k] = c[0];
      data[k + 1] = c[1];
      data[k + 2] = c[2];
      data[k + 3] = a;
    }
  return toTexture(data, w, h);
};

/** Seeded PRNG (Park–Miller). */
export const prng = (seed: number) => () => (seed = (seed * 16807) % 2147483647) / 2147483647;

/** Circuit board: graphite substrate, routed traces, vias and champagne pads. */
export const circuitBoard = (size = 1024, seed = 11) => {
  const rnd = prng(seed);
  const S = size;
  const data = new Uint8Array(S * S * 4);
  const base = hex("#2c2d30"), trace = hex("#45474b"), pad = hex("#c8b48c"), via = hex("#8d8a84");
  const set = (x: number, y: number, c: Rgb, a = 1) => {
    if (x < 0 || y < 0 || x >= S || y >= S) return;
    const k = (y * S + x) * 4;
    for (let i = 0; i < 3; i++) data[k + i] = data[k + i] + (c[i] - data[k + i]) * a;
  };
  for (let i = 0; i < S * S; i++) {
    const n = (rnd() - 0.5) * 6;
    data[i * 4] = base[0] + n;
    data[i * 4 + 1] = base[1] + n;
    data[i * 4 + 2] = base[2] + n;
    data[i * 4 + 3] = 255;
  }
  const disc = (cx: number, cy: number, r: number, c: Rgb, hollow = 0) => {
    for (let y = Math.floor(cy - r - 1); y <= cy + r + 1; y++)
      for (let x = Math.floor(cx - r - 1); x <= cx + r + 1; x++) {
        const d = Math.hypot(x - cx, y - cy);
        const a = Math.min(1, Math.max(0, r - d + 0.5)) * (hollow ? Math.min(1, Math.max(0, d - hollow + 0.5)) : 1);
        if (a > 0) set(x, y, c, a);
      }
  };
  const line = (x0: number, y0: number, x1: number, y1: number, wdt: number) => {
    const L = Math.hypot(x1 - x0, y1 - y0);
    for (let s = 0; s <= L; s += 0.5) {
      const x = x0 + ((x1 - x0) * s) / L, y = y0 + ((y1 - y0) * s) / L;
      disc(x, y, wdt / 2, trace);
    }
  };
  // Manhattan + 45° routing from random pins
  for (let i = 0; i < 70; i++) {
    let x = rnd() * S, y = rnd() * S;
    const steps = 2 + Math.floor(rnd() * 3);
    const w = 3 + rnd() * 4;
    for (let s = 0; s < steps; s++) {
      const dir = Math.floor(rnd() * 8) * (Math.PI / 4);
      const len = 40 + rnd() * 180;
      const nx = x + Math.cos(dir) * len, ny = y + Math.sin(dir) * len;
      line(x, y, nx, ny, w);
      x = nx;
      y = ny;
    }
    disc(x, y, w * 1.6, via, w * 0.7);
  }
  for (let i = 0; i < 40; i++) disc(rnd() * S, rnd() * S, 5 + rnd() * 5, pad);
  return toTexture(data, S, S);
};

  [/menatap|melihat|memandang|mendongak/, "look"], [/memegang|membawa|menggenggam/, "hold"], [/berdiri/, "stand"],

export const parseSpec = (spec: string, cast: CastDef[]): PageSpec => {
  const t = spec.toLowerCase();
  const time = TIMES.find(([re]) => re.test(t))?.[1] ?? "siang";
  const place = PLACES.find(([re]) => re.test(t))?.[1] ?? "kebun";
  const props = PROPS.filter((p) => t.includes(p));
  const actors: Actor[] = [];
  const clauses = spec.split(/[.;]|,\s*(?=[A-Z])/).map((c) => c.trim()).filter(Boolean);
  for (const c of cast) {
    const n = c.name.toLowerCase();
    const re = new RegExp(`\\b${n.replace(/[.*+?^${}()|[\]\\]/g, "\\$&")}\\b`);
    if (!re.test(t)) continue;
    const clause = clauses.find((cl) => re.test(cl.toLowerCase())) ?? spec;
    const cl = clause.toLowerCase();
    // the verb that follows the name in its clause wins over verbs before it ("Nara memeluk Bintang")
    const after = cl.slice(cl.search(re));
    let act: Act = "stand";
    for (const [r, a] of ACTS) if (r.test(after)) { act = a; break; }
    let target: string | undefined;
    if (act === "hug") target = cast.find((o) => o !== c && new RegExp(`\\b${o.name.toLowerCase()}\\b`).test(after))?.name;
    actors.push({ who: c.name, act, target });
  }
  // the hugged one is drawn hugging back
  for (const a of actors) if (a.act === "hug" && a.target) { const b = actors.find((x) => x.who === a.target); if (b && b.act === "stand") b.act = "hug", b.target = a.who; }
  return { time, place, props, actors };
};

type RawPage = { spec: string; lines: string[] };
export const parseBook = (src: string) => {
  const out = { title: "", by: undefined as string | undefined, dedication: undefined as string | undefined, closing: undefined as string | undefined, cast: [] as CastDef[], cover: { spec: "", lines: [] as string[] }, pages: [] as RawPage[], warnings: [] as string[] };
  let cur: RawPage | null = null;
  let inCover = false;
  for (const raw of src.replace(/\r/g, "").split("\n")) {
    const line = raw.trim();
    if (!line || line.startsWith("#") || line.startsWith("//")) continue;
    const head = /^(JUDUL|OLEH|PENULIS|PERSEMBAHAN|SUARA|TOKOH)\s*:\s*(.*)$/i.exec(line);
    if (head) {
      const k = head[1].toUpperCase(), v = head[2].trim();
      if (k === "JUDUL") out.title = v;
      else if (k === "OLEH" || k === "PENULIS") out.by = v;
      else if (k === "PERSEMBAHAN") out.dedication = v;
      else if (k === "TOKOH") {
        const m = /^([^=:]+?)\s*[=:]\s*(.+)$/.exec(v);
        if (!m) { out.warnings.push(`TOKOH tanpa "=": "${v}"`); continue; }
        const look = parseLook(m[2]);
        const firstColor = Object.keys(COLORS).sort((a, b) => b.length - a.length).find((k) => new RegExp(`\\b${k}\\b`).test(m[2].toLowerCase()));
        out.cast.push(typeof look === "string" ? { name: m[1].trim(), kind: look, color: firstColor ? COLORS[firstColor] : undefined } : { name: m[1].trim(), kind: "person", look });
      }
      continue;
    }
    const tag = /^\[(SAMPUL|HALAMAN|PENUTUP)\]\s*(.*)$/i.exec(line);
    if (tag) {
      const k = tag[1].toUpperCase();
      if (k === "SAMPUL") { inCover = true; cur = null; out.cover.spec = tag[2]; }
      else if (k === "HALAMAN") { inCover = false; cur = { spec: tag[2], lines: [] }; out.pages.push(cur); }
      else { inCover = false; cur = null; out.closing = tag[2] || "Tamat"; }
      continue;
    }
    const sentences = line.split(/(?<=[.!?…]["”']?)\s+(?=[A-Z"“*])/).map((x) => x.trim()).filter(Boolean);
    if (inCover) out.cover.lines.push(...sentences);
    else if (cur) cur.lines.push(...sentences);
    else { cur = { spec: "", lines: [...sentences] }; out.pages.push(cur); }
  }
  if (!out.pages.length) out.warnings.push("Naskah belum punya [HALAMAN].");
  if (out.pages.length > 12) out.warnings.push(`${out.pages.length} halaman — lebih dari 12 membuat video panjang; pertimbangkan menggabung halaman.`);
  return out;
};

export type LineTime = { a: number; b: number };
const WPS = 2.3; // a calm read-aloud pace
const TURN = 1.8;

/**
 * Plans the film. `vo` (optional) = the voice-over's own line times, in reading order (cover lines first): the film
 * keeps every line's audio and only moves whole pages apart so each page turn falls into a pause (segments map
 * source audio → film time). Without a VO the lines are paced at a read-aloud speed.
 */
export const planBook = (src: string, opts: { id: string; vo?: LineTime[]; audio?: string } = { id: "book" }): BookPlan => {
  const raw = parseBook(src);
  const warnings = [...raw.warnings];
  const allLines = [...raw.cover.lines, ...raw.pages.flatMap((p) => p.lines)];
  const dur = (s: string) => Math.max(1.2, s.split(/\s+/).length / WPS + 0.2);
  const vo = opts.vo && opts.vo.length === allLines.length ? opts.vo : undefined;
  if (opts.vo && !vo) warnings.push(`VO punya ${opts.vo.length} kalimat, naskah ${allLines.length} — VO diabaikan, tempo dibaca otomatis.`);
  const segments: BookPlan["segments"] = [];
  let li = 0;
  let t = 0.6;
  // cover: its lines are read over the closed book; then the cover opens
  const coverLines = raw.cover.lines.map((text) => {
    const d = vo ? vo[li].b - vo[li].a : dur(text);
    const r = { text, a: t, b: t + d };
    if (vo) segments.push({ src: vo[li].a, dst: t, len: d });
    li++;
    t = r.b + 0.35;
    return r;
  });
  if (!coverLines.length) t = 1.4;
  const open: [number, number] = [t + 0.2, t + 2.0];
  t = open[1] + 0.6;
  const pages: PagePlan[] = raw.pages.map((rp, i) => {
    const spec = parseSpec(rp.spec, raw.cast);
    // every story page is a right-hand page (picture + its text); the left page facing it is printed paper.
    // Each new page = one page turn (the 3D cut-outs only ever face up on a front).
    const side: "R" | "L" = "R";
    const leaf = i;
    let turn: PagePlan["turn"];
    const pan: PagePlan["pan"] = undefined;
    if (i > 0) { turn = { leaf: i - 1, a: t, b: t + TURN }; t += TURN + 0.25; }
    const a = t;
    // a page's lines keep their natural spacing from the VO (pauses inside a page are the narrator's)
    const lines = rp.lines.map((text, k) => {
      let d: number, gap: number;
      if (vo) {
        d = vo[li].b - vo[li].a;
        gap = k + 1 < rp.lines.length ? Math.min(1.2, Math.max(0.15, vo[li + 1].a - vo[li].b)) : 0;
        segments.push({ src: vo[li].a, dst: t, len: d });
      } else { d = dur(text); gap = 0.45; }
      li++;
      const r = { text, a: t, b: t + d };
      t = r.b + gap;
      return r;
    });
    const minPage = 3.2;
    if (t - a < minPage) t = a + minPage;
    t += 0.7; // a beat to look at the picture before moving on
    return { ...spec, idx: i, side, leaf, a, b: t, lines, turn, pan, seed: i * 7 + 3 };
  });
  if (!raw.cast.length) warnings.push("Belum ada TOKOH: halaman hanya berisi latar.");
  for (const p of pages) for (const ac of p.actors) if (!raw.cast.some((c) => c.name === ac.who)) warnings.push(`Tokoh "${ac.who}" belum dideklarasikan.`);
  const dim: [number, number] = [t + 0.4, t + 2.2];
  const duration = Math.ceil((dim[1] + 0.8) * 10) / 10;
  const coverSpec = raw.cover.spec ? parseSpec(raw.cover.spec, raw.cast) : { ...(pages[0] ?? parseSpec("", raw.cast)), actors: raw.cast.slice(0, 2).map((c) => ({ who: c.name, act: "wave" as Act })) };
  return {
    id: opts.id, title: raw.title || "Tanpa Judul", by: raw.by, dedication: raw.dedication, closing: raw.closing,
    cast: raw.cast, cover: coverSpec, coverLines, open, pages, dim, duration, audio: opts.audio, segments, warnings,
  };
};

/** Every narrated sentence in reading order (for TTS / VO alignment). */
export const bookLines = (src: string) => {
  const raw = parseBook(src);
  return [...raw.cover.lines, ...raw.pages.flatMap((p) => p.lines)].map((s) => s.replace(/\*/g, ""));
};

const sm = (x: number) => (x <= 0 ? 0 : x >= 1 ? 1 : x * x * (3 - 2 * x));
const lerp = (a: number, b: number, t: number) => a + (b - a) * t;

const SLOTS: Record<number, number[]> = { 1: [0.6], 2: [-3.0, 3.4], 3: [-5.0, 0.2, 5.0], 4: [-6.2, -2.2, 1.8, 5.8], 5: [-6.6, -3.3, 0, 3.3, 6.6] };

/** Where a person's front hand is (page space), for kites, balloons and held things. */
const handOf = (x: number, y: number, scale: number, flip: boolean, armF: number, adult: boolean) => {
  const legK = adult ? 1.45 : 1;
  const shoulderY = 3.0 * legK + (adult ? 3.3 : 2.65);
  const L = 2.6 * (adult ? 1.25 : 1);
  const hx = 0.95 + Math.sin(armF) * L, hy = shoulderY - Math.cos(armF) * L;
  return [x + (flip ? -hx : hx) * scale, y + hy * scale] as [number, number];
};

/** True when every actor on the page is an animal / star (a small cast: the close-up comes nearer and lower). */
export const smallCast = (spec: PageSpec, cast: CastDef[]) => spec.actors.length > 0 && spec.actors.every((a) => cast.find((c) => c.name === a.who)?.kind !== "person");

/** Where the eye goes on a page (u, cm): the close-up centres here. */
export const pageFocus = (spec: PageSpec, cast: CastDef[]) => pageFrame(spec, cast).u;

/** The close-up's centre u and the width (cm) the cast spans on the page. */
export const pageFrame = (spec: PageSpec, cast: CastDef[]) => {
  const g = spec.actors.filter(ground);
  const slotK = smallCast(spec, cast) ? 0.55 : 1;
  const slots = (SLOTS[Math.min(5, Math.max(1, g.length))] ?? SLOTS[1]).map((u) => u * slotK);
  const us: number[] = [];
  spec.actors.forEach((ac) => {
    if (!cast.some((c) => c.name === ac.who)) return;
    if (ac.act === "peek") us.push(houseWindow(-3.6).u + 0.4);
    else if (ac.act === "fall") us.push(slots[0] + 2.6);
    else if (ac.act === "fly" || ac.act === "swim") us.push(0);
    else us.push(slots[g.indexOf(ac)] ?? 0);
  });
  if (!us.length) return { u: 0, spread: 0 };
  return { u: Math.max(-3.8, Math.min(3.8, (Math.min(...us) + Math.max(...us)) / 2)), spread: Math.max(...us) - Math.min(...us) };
};

/**
 * The band of the picture (v, cm) that holds the action, feet to the highest head, kite or flier. A 16:9 close-up is
 * much lower than a 9:16 one: it is only taken when this band fits its frame.
 */
export const pageBand = (spec: PageSpec, cast: CastDef[]) => {
  let lo = Infinity, hi = -Infinity;
  const add = (a: number, b: number) => { lo = Math.min(lo, a); hi = Math.max(hi, b); };
  const has = (p: string) => spec.props.includes(p);
  spec.actors.forEach((ac, i) => {
    const def = cast.find((c) => c.name === ac.who);
    if (!def) return;
    const person = def.kind === "person";
    const tall = person ? (def.look!.build !== "child" ? 7.4 : 5.4) : 3.2;
    if (ac.act === "peek") add(houseWindow(-3.6).v - 2.0, houseWindow(-3.6).v + 2.0);
    else if (ac.act === "fly") add(4.8, 8.4);
    else if (ac.act === "swim") add(-4.2, 1.6);
    else if (ac.act === "fall") add(FEET - 0.3, FEET + 4.6);
    else if (ac.act === "sleep") add(FEET - 0.9, spec.place === "kamar" ? 3.6 : FEET + 4.4);
    else add(FEET - 0.3, FEET + tall + (ac.act === "jump" ? 1.5 : ac.act === "happy" ? 0.6 : 0));
    if (!person || ac.act === "sleep") return;
    if (ac.act === "kite" || (has("layang") && i === 0)) add(FEET, 10.6);
    if (ac.act === "hold" && has("balon")) add(FEET, FEET + tall + 3.6);
    if (has("payung") && i === 0) add(FEET, FEET + tall + 1.6);
  });
  return hi > lo ? { lo, hi } : null;
};

type PagePrint = (g: G, r: () => number) => void;

/** A page surface: warm paper, a soft gutter shade at the spine side, optional print (page numbers, text). */
const pageCanvas = (ppc: number, spineAt: "left" | "right", print?: PagePrint, seed = 1) => {
  const k = Math.min(ppc, 2048 / PAGE.H);
  const c = document.createElement("canvas");
  c.width = Math.round(PAGE.W * k);
  c.height = Math.round(PAGE.H * k);
  const g = c.getContext("2d")!;
  g.setTransform(k, 0, 0, -k, (PAGE.W / 2) * k, (PAGE.H / 2) * k);
  g.fillStyle = PAPER_COLOR;
  g.fillRect(-PAGE.W / 2, -PAGE.H / 2, PAGE.W, PAGE.H);
  const r = rng(seed);
  // faint pulp mottling
  for (let i = 0; i < 90; i++) {
    g.fillStyle = `rgba(${r() < 0.5 ? "120,90,50" : "255,255,255"},${0.012 + r() * 0.015})`;
    g.beginPath();
    g.ellipse(-PAGE.W / 2 + r() * PAGE.W, -PAGE.H / 2 + r() * PAGE.H, 0.5 + r() * 3, 0.5 + r() * 3, r() * 3, 0, Math.PI * 2);
    g.fill();
  }
  // gutter: the page curves down into the binding
  const sx = spineAt === "left" ? -PAGE.W / 2 : PAGE.W / 2;
  const dir = spineAt === "left" ? 1 : -1;
  const grad = g.createLinearGradient(sx, 0, sx + dir * 2.2, 0);
  grad.addColorStop(0, "rgba(90,60,30,0.22)");
  grad.addColorStop(0.35, "rgba(90,60,30,0.07)");
  grad.addColorStop(1, "rgba(90,60,30,0)");
  g.fillStyle = grad;
  g.fillRect(-PAGE.W / 2, -PAGE.H / 2, PAGE.W, PAGE.H);
  print?.(g, r);
  const t = canvasTexture(c);
  return t;
};

const clothCanvas = (color: string, w: number, h: number, ppc: number) => {
  const k = Math.min(ppc * 0.6, 1400 / Math.max(w, h));
  const c = document.createElement("canvas");
  c.width = Math.round(w * k);
  c.height = Math.round(h * k);
  const g = c.getContext("2d")!;
  g.fillStyle = color;
  g.fillRect(0, 0, c.width, c.height);
  const r = rng(4242);
  // woven book cloth: fine warp/weft threads
  const step = Math.max(1.5, k * 0.05);
  for (let x = 0; x < c.width; x += step) {
    g.fillStyle = `rgba(255,255,255,${0.025 + r() * 0.03})`;
    g.fillRect(x, 0, step * 0.45, c.height);
  }
  for (let y = 0; y < c.height; y += step) {
    g.fillStyle = `rgba(0,0,0,${0.03 + r() * 0.03})`;
    g.fillRect(0, y, c.width, step * 0.4);
  }
  // wear at the edges
  g.strokeStyle = "rgba(255,240,220,0.12)";
  g.lineWidth = k * 0.25;
  g.strokeRect(0, 0, c.width, c.height);
  return canvasTexture(c);
};

const woodCanvas = (ppc: number) => {
  const W = 1024, H = 1024;
  const c = document.createElement("canvas");
  c.width = W;
  c.height = H;
  const g = c.getContext("2d")!;
  const r = rng(9001);
  const planks = 5;
  for (let p = 0; p < planks; p++) {
    const y0 = (p * H) / planks;
    const base = [118 + r() * 18, 78 + r() * 12, 48 + r() * 8];
    g.fillStyle = `rgb(${base[0]},${base[1]},${base[2]})`;
    g.fillRect(0, y0, W, H / planks);
    // grain lines
    for (let i = 0; i < 70; i++) {
      const yy = y0 + r() * (H / planks);
      const amp = 2 + r() * 6, f = 0.004 + r() * 0.01, ph = r() * 6;
      g.strokeStyle = `rgba(${r() < 0.5 ? "60,35,18" : "170,120,80"},${0.06 + r() * 0.12})`;
      g.lineWidth = 0.6 + r() * 2.2;
      g.beginPath();
      for (let x = 0; x <= W; x += 16) {
        const y = yy + Math.sin(x * f + ph) * amp + Math.sin(x * f * 3.1 + ph * 2) * amp * 0.3;
        if (x === 0) g.moveTo(x, y);
        else g.lineTo(x, y);
      }
      g.stroke();
    }
    g.fillStyle = "rgba(30,15,6,0.55)";
    g.fillRect(0, y0, W, 2);
  }
  const t = canvasTexture(c);
  t.wrapS = t.wrapT = 1000; // RepeatWrapping

type LeafState = { theta: number; k0: number; kz: number; y: number };
const NONE = NO_CEIL;
const ceil = (st: LeafState) => new Vector4(st.theta, st.k0, st.kz, st.y);

/** Turn kinematics: lift the corner (edge leads), swing over, the edge trails, land on the left stack. */
const leafState = (p: number, yR: number, yL: number): LeafState => {
  const e = smooth(smooth(clamp01(p)));
  const sw = Math.sin(2 * Math.PI * clamp01(p));
  return { theta: Math.PI * e, k0: (0.95 * sw) / PAGE.W, kz: (0.55 * sw) / (PAGE.W * (PAGE.H / 2)), y: mix(yR, yL, e) };
};

/** Write a leaf state + its clearance surfaces into a bend's uniforms. */
const applyState = (U: BendUniforms, st: LeafState, ceilR: Vector4, ceilL: Vector4, floorR: number, floorL: number) => {
  U.uTheta.value = st.theta;
  U.uK0.value = st.k0;
  U.uKz.value = st.kz;
  U.uLift.value = 1;
  U.uPage.value = new Matrix4().makeTranslation(0, st.y, 0);
  U.uPageInv.value = new Matrix4().makeTranslation(0, -st.y, 0);
  U.uCeilR.value.copy(ceilR);
  U.uCeilL.value.copy(ceilL);
  U.uFloor.value.set(floorR, floorL);
};

  const yL = (i: number) => hingeY + (i + 1) * DT; // leaf i landed on the left stack

      </TitleContent>

          spec={spec}

          ceilR={i === 0 ? ceil(coverSt) : ceil(st[i - 1])}
          ceilL={i < n - 1 ? ceil(st[i + 1]) : NONE}

// Painted paper — the picture-book surface every cut-out is made of.
//
// A picture-book illustration in this style is paper that was painted (watercolour wash, crayon, gouache),
// cut out with scissors and glued down in layers. Here each piece is a polygon in centimetres (y up),
// jittered like a scissor cut, painted procedurally into a canvas and used as an alpha-cut texture.
// Everything is seeded, so every frame and every render is identical.

import { CanvasTexture, LinearMipmapLinearFilter, SRGBColorSpace } from "three";

export type Pt = [number, number];
export type Poly = Pt[];

// ── Random ─────────────────────────────────────────────────────────────────────

export const rng = (seed: number) => () => {
  seed |= 0;
  seed = (seed + 0x6d2b79f5) | 0;
  let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
  t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
  return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
};
export const hashStr = (s: string) => {
  let h = 2166136261;
  for (let i = 0; i < s.length; i++) h = Math.imul(h ^ s.charCodeAt(i), 16777619);
  return h >>> 0;
};

// ── Shapes (centimetres, y up, closed) ─────────────────────────────────────────

export const ellipse = (cx: number, cy: number, rx: number, ry = rx, n = 64): Poly =>
  Array.from({ length: n }, (_, i) => {
    const a = (i / n) * Math.PI * 2;
    return [cx + Math.cos(a) * rx, cy + Math.sin(a) * ry];
  });

export const rect = (x0: number, y0: number, x1: number, y1: number, r = 0): Poly => {
  if (r <= 0) return [[x0, y0], [x1, y0], [x1, y1], [x0, y1]];
  const out: Poly = [];
  const corner = (cx: number, cy: number, a0: number) => {
    for (let i = 0; i <= 6; i++) {
      const a = a0 + (i / 6) * (Math.PI / 2);
      out.push([cx + Math.cos(a) * r, cy + Math.sin(a) * r]);
    }
  };
  corner(x1 - r, y0 + r, -Math.PI / 2);
  corner(x1 - r, y1 - r, 0);
  corner(x0 + r, y1 - r, Math.PI / 2);
  corner(x0 + r, y0 + r, Math.PI);
  return out;
};

export const star = (cx: number, cy: number, R: number, r: number, n = 5, rot = 0): Poly =>
  Array.from({ length: n * 2 }, (_, i) => {
    const a = rot + Math.PI / 2 + (i / (n * 2)) * Math.PI * 2;
    const q = i % 2 ? r : R;
    return [cx + Math.cos(a) * q, cy + Math.sin(a) * q];
  });

/** Closed Catmull-Rom through the points (soft blobs: bushes, clouds, hair, hills). */
export const blob = (pts: Poly, per = 10): Poly => {
  const out: Poly = [];
  const n = pts.length;
  for (let i = 0; i < n; i++) {
    const p0 = pts[(i - 1 + n) % n], p1 = pts[i], p2 = pts[(i + 1) % n], p3 = pts[(i + 2) % n];
    for (let j = 0; j < per; j++) {
      const t = j / per, t2 = t * t, t3 = t2 * t;
      const f = (a: number, b: number, c: number, d: number) =>
        0.5 * (2 * b + (-a + c) * t + (2 * a - 5 * b + 4 * c - d) * t2 + (-a + 3 * b - 3 * c + d) * t3);
      out.push([f(p0[0], p1[0], p2[0], p3[0]), f(p0[1], p1[1], p2[1], p3[1])]);
    }
  }
  return out;
};

/** Open Catmull-Rom (for hill tops, paths, strings). */
export const spline = (pts: Poly, per = 10): Poly => {
  const out: Poly = [];
  const n = pts.length;
  for (let i = 0; i < n - 1; i++) {
    const p0 = pts[Math.max(0, i - 1)], p1 = pts[i], p2 = pts[i + 1], p3 = pts[Math.min(n - 1, i + 2)];
    for (let j = 0; j < per; j++) {
      const t = j / per, t2 = t * t, t3 = t2 * t;
      const f = (a: number, b: number, c: number, d: number) =>
        0.5 * (2 * b + (-a + c) * t + (2 * a - 5 * b + 4 * c - d) * t2 + (-a + 3 * b - 3 * c + d) * t3);
      out.push([f(p0[0], p1[0], p2[0], p3[0]), f(p0[1], p1[1], p2[1], p3[1])]);
    }
  }
  out.push(pts[n - 1]);
  return out;
};

export const hill = (tops: Poly, base: number): Poly => {
  const top = spline(tops, 12);
  return [...top, [top[top.length - 1][0], base], [top[0][0], base]];
};

/** Resample a closed polygon every `step` cm. */
const resample = (p: Poly, step: number): Poly => {
  const out: Poly = [];
  for (let i = 0; i < p.length; i++) {
    const a = p[i], b = p[(i + 1) % p.length];
    const L = Math.hypot(b[0] - a[0], b[1] - a[1]);
    const n = Math.max(1, Math.ceil(L / step));
    for (let j = 0; j < n; j++) out.push([a[0] + ((b[0] - a[0]) * j) / n, a[1] + ((b[1] - a[1]) * j) / n]);
  }
  return out;
};

/** Scissor cut: the outline wanders a little (long gentle drift + small facets), never a perfect vector edge. */
export const scissor = (p: Poly, seed: number, amount = 0.035): Poly => {
  const r = rng(seed);
  const q = resample(p, 0.12);
  const n = q.length;
  const ph = [r() * 6.28, r() * 6.28, r() * 6.28];
  const out: Poly = [];
  for (let i = 0; i < n; i++) {
    const a = q[(i - 1 + n) % n], b = q[(i + 1) % n];
    let nx = -(b[1] - a[1]), ny = b[0] - a[0];
    const l = Math.hypot(nx, ny) || 1;
    nx /= l;
    ny /= l;
    const t = i / n;
    const d = amount * (0.6 * Math.sin(t * 6.28 * 3 + ph[0]) + 0.3 * Math.sin(t * 6.28 * 11 + ph[1]) + 0.25 * Math.sin(t * 6.28 * 29 + ph[2]));
    out.push([q[i][0] + nx * d, q[i][1] + ny * d]);
  }
  return out;
};

export const bbox = (polys: Poly[]) => {
  let x0 = Infinity, y0 = Infinity, x1 = -Infinity, y1 = -Infinity;
  for (const p of polys)
    for (const [x, y] of p) {
      if (x < x0) x0 = x;
      if (y < y0) y0 = y;
      if (x > x1) x1 = x;
      if (y > y1) y1 = y;
    }
  return { x0, y0, x1, y1, w: x1 - x0, h: y1 - y0 };
};

const hex = (c: string) => {
  if (c.startsWith("rgb")) return c.replace(/[^\d,.]/g, "").split(",").slice(0, 3).map(Number);
  const v = c.replace("#", "");
  const n = parseInt(v.length === 3 ? v.split("").map((x) => x + x).join("") : v, 16);
  return [(n >> 16) & 255, (n >> 8) & 255, n & 255];
};
export const shade = (c: string, k: number) => {
  const [r, g, b] = hex(c);
  const f = (x: number) => Math.round(Math.max(0, Math.min(255, k < 0 ? x * (1 + k) : x + (255 - x) * k)));
  return `rgb(${f(r)},${f(g)},${f(b)})`;
};
export const rgba = (c: string, a: number) => {
  const [r, g, b] = hex(c);
  return `rgba(${r},${g},${b},${a})`;
};

// ── Noise tiles (made once) ────────────────────────────────────────────────────

type Tiles = { mottle: HTMLCanvasElement; grain: HTMLCanvasElement; tooth: HTMLCanvasElement; streak: HTMLCanvasElement };
let tiles: Tiles | null = null;

const valueNoise = (N: number, cells: number, r: () => number) => {
  const lat = Float32Array.from({ length: cells * cells }, () => r());
  const out = new Float32Array(N * N);
  for (let y = 0; y < N; y++) {
    const fy = (y / N) * cells, y0 = Math.floor(fy), ty = fy - y0, sy = ty * ty * (3 - 2 * ty);
    for (let x = 0; x < N; x++) {
      const fx = (x / N) * cells, x0 = Math.floor(fx), tx = fx - x0, sx = tx * tx * (3 - 2 * tx);
      const g = (i: number, j: number) => lat[(j % cells) * cells + (i % cells)];
      const a = g(x0, y0) + (g(x0 + 1, y0) - g(x0, y0)) * sx;
      const b = g(x0, y0 + 1) + (g(x0 + 1, y0 + 1) - g(x0, y0 + 1)) * sx;
      out[y * N + x] = a + (b - a) * sy;
    }
  }
  return out;
};

const toCanvas = (N: number, px: (i: number) => [number, number, number, number]) => {
  const c = document.createElement("canvas");
  c.width = c.height = N;
  const g = c.getContext("2d")!;
  const img = g.createImageData(N, N);
  for (let i = 0; i < N * N; i++) {
    const [r, gg, b, a] = px(i);
    img.data[i * 4] = r;
    img.data[i * 4 + 1] = gg;
    img.data[i * 4 + 2] = b;
    img.data[i * 4 + 3] = a;
  }
  g.putImageData(img, 0, 0);
  return c;
};

const makeTiles = (): Tiles => {
  const r = rng(77031);
  const N = 512;
  const a = valueNoise(N, 5, r), b = valueNoise(N, 13, r), c = valueNoise(N, 37, r);
  // pigment pooling: multiply layer 0.72…1.0
  const mottle = toCanvas(N, (i) => {
    const v = 0.55 * a[i] + 0.3 * b[i] + 0.15 * c[i];
    const k = Math.round(255 * (0.72 + 0.28 * Math.min(1, Math.max(0, (v - 0.25) * 1.9))));
    return [k, k, k, 255];
  });
  // granulation: fine dark specks (multiply)
  const fine = valueNoise(N, 160, r);
  const grain = toCanvas(N, (i) => {
    const v = fine[i] * 0.7 + r() * 0.3;
    const k = Math.round(255 * (0.86 + 0.14 * v));
    return [k, k, k, 255];
  });
  // paper tooth for crayon: white specks where the wax skipped
  const toothN = valueNoise(N, 90, r);
  const tooth = toCanvas(N, (i) => {
    const v = toothN[i] * 0.6 + r() * 0.4;
    const al = v > 0.62 ? Math.min(255, Math.round((v - 0.62) * 900)) : 0;
    return [255, 250, 238, al];
  });
  // brush streaks (gouache): stretched noise
  const sN = valueNoise(N, 64, r);
  const streak = toCanvas(N, (i) => {
    const x = i % N, y = Math.floor(i / N);
    const v = sN[(Math.floor(y / 1) % N) * N + (Math.floor(x / 9) % N)];
    const k = Math.round(255 * (0.86 + 0.14 * v));
    return [k, k, k, 255];
  });
  return { mottle, grain, tooth, streak };
};
const getTiles = () => (tiles ??= makeTiles());

export const trace = (g: G, p: Poly) => {
  g.beginPath();
  g.moveTo(p[0][0], p[0][1]);
  for (let i = 1; i < p.length; i++) g.lineTo(p[i][0], p[i][1]);
  g.closePath();
};
const traceAll = (g: G, ps: Poly[]) => {

  for (const p of ps) {
    g.moveTo(p[0][0], p[0][1]);

const pattern = (g: G, tile: HTMLCanvasElement, cmPerTile: number, r: () => number) => {
  const pat = g.createPattern(tile, "repeat")!;
  const k = cmPerTile / tile.width;
  pat.setTransform(new DOMMatrix().translate(r() * 40, r() * 40).rotate(r() * 360).scale(k, k));
  return pat;
};

export const MARGIN = 0.3;

/** Paint polygons into a canvas laid out in cm (y up). Returns the canvas + its cm rectangle. */
export const paintCanvas = (polys: Poly[], paint: Paint, ppc: number) => {
  const T = getTiles();
  const r = rng(paint.seed ?? 1);
  const cut = paint.clean ? polys : polys.map((p, i) => scissor(p, (paint.seed ?? 1) * 31 + i * 7));
  const bb = bbox(cut);
  const m = MARGIN;
  const wcm = bb.w + 2 * m, hcm = bb.h + 2 * m;
  const scale = Math.min(ppc, 2048 / Math.max(wcm, hcm));
  const W = Math.max(4, Math.ceil(wcm * scale)), H = Math.max(4, Math.ceil(hcm * scale));
  const c = document.createElement("canvas");
  c.width = W;
  c.height = H;
  const g = c.getContext("2d")!;
  const sx = W / wcm, sy = H / hcm;
  g.setTransform(sx, 0, 0, -sy, -(bb.x0 - m) * sx, (bb.y1 + m) * sy);
  const px = sx; // pixels per cm
  const style = paint.style ?? "wash";
  const tone = paint.tone ?? shade(paint.color, -0.28);
  const e = paint.edge ?? 0.5;
  const full = () => g.fillRect(bb.x0 - m, bb.y0 - m, wcm, hcm);

  g.save();
  traceAll(g, cut);
  g.clip();
  g.fillStyle = paint.color;
  full();

  if (style === "wash") {
    g.globalCompositeOperation = "multiply";
    g.globalAlpha = 0.6 * e + 0.15;
    g.fillStyle = pattern(g, T.mottle, 9, r);
    full();
    // pooled pigment blobs
    g.globalAlpha = 1;
    const n = Math.round(2 + (bb.w * bb.h) / 40);
    for (let i = 0; i < n; i++) {
      const x = bb.x0 + r() * bb.w, y = bb.y0 + r() * bb.h, rr = 0.6 + r() * Math.min(4, Math.max(bb.w, bb.h) * 0.35);
      g.filter = `blur(${Math.max(1, rr * 0.5 * px)}px)`;
      g.fillStyle = rgba(tone, 0.12 * e + 0.04);
      g.beginPath();
      g.ellipse(x, y, rr, rr * (0.5 + r() * 0.6), r() * 3, 0, Math.PI * 2);
      g.fill();
    }
    g.filter = "none";
    g.globalCompositeOperation = "multiply";
    g.globalAlpha = 0.5;
    g.fillStyle = pattern(g, T.grain, 5, r);
    full();
  } else if (style === "crayon") {
    // waxy strokes on top of a light base, then paper tooth showing through
    g.globalCompositeOperation = "source-over";
    const area = bb.w * bb.h;
    const n = Math.round(area * 26 + 30);
    const ang = -0.9 + r() * 0.3;
    for (let i = 0; i < n; i++) {
      const x = bb.x0 + r() * bb.w, y = bb.y0 + r() * bb.h, L = 0.25 + r() * 0.7;
      const a = ang + (r() - 0.5) * 0.5;
      g.strokeStyle = r() < 0.55 ? rgba(tone, 0.16 + r() * 0.16) : rgba(shade(paint.color, 0.22), 0.14 + r() * 0.12);
      g.lineWidth = 0.05 + r() * 0.08;
      g.lineCap = "round";
      g.beginPath();
      g.moveTo(x, y);
      g.lineTo(x + Math.cos(a) * L, y + Math.sin(a) * L);
      g.stroke();
    }
    g.globalCompositeOperation = "source-atop";
    g.globalAlpha = 0.55;
    g.fillStyle = pattern(g, T.tooth, 4, r);
    full();
  } else if (style === "gouache") {
    g.globalCompositeOperation = "multiply";
    g.globalAlpha = 0.35 * e + 0.1;
    g.fillStyle = pattern(g, T.mottle, 12, r);
    full();
    g.globalAlpha = 0.55;
    g.fillStyle = pattern(g, T.streak, 6, r);
    full();
  } else {
    g.globalCompositeOperation = "multiply";
    g.globalAlpha = 0.35;
    g.fillStyle = pattern(g, T.grain, 5, r);
    full();
  }

  // edge darkening: pigment dries at the rim
  if (e > 0 && style !== "flat") {
    g.globalCompositeOperation = "multiply";
    g.globalAlpha = 0.55 * e;
    g.filter = `blur(${Math.max(1, 0.1 * px)}px)`;
    g.strokeStyle = tone;
    g.lineWidth = 0.32;
    for (const p of cut) {
      trace(g, p);
      g.stroke();
    }
    g.filter = "none";
  }

  g.globalCompositeOperation = "source-over";
  g.globalAlpha = 1;
  paint.marks?.(g, r, px);
  g.restore();

  // coloured-pencil outline, drawn twice with a small wobble like a real sketch line
  if (paint.line) {
    g.save();
    traceAll(g, cut);
    g.clip();
    g.strokeStyle = rgba(paint.line, 0.75);
    g.lineWidth = 0.05;
    g.lineJoin = "round";
    for (const p of cut) {
      trace(g, scissor(p, (paint.seed ?? 1) + 991, 0.02));
      g.stroke();
    }
    g.strokeStyle = rgba(paint.line, 0.35);
    g.lineWidth = 0.035;
    for (const p of cut) {
      trace(g, scissor(p, (paint.seed ?? 1) + 517, 0.03));
      g.stroke();
    }
    g.restore();
  }
  // the cut: a hair of lighter paper core catches the lamp
  if (paint.rim ?? !paint.clean) {
  g.save();
  traceAll(g, cut);
  g.clip();
  g.globalAlpha = 0.45;
  g.strokeStyle = "rgba(255,248,232,0.55)";
  g.lineWidth = 0.05;
  for (const p of cut) {
    trace(g, p);
    g.stroke();
  }
  g.restore();
  }
  if (paint.holes?.length) {
    const hs = paint.holes.map((h, i) => (paint.clean ? h : scissor(h, (paint.seed ?? 1) * 17 + i * 13, 0.025)));
    // the rim of a hole darkens like any cut edge, then the hole is punched out
    g.save();
    traceAll(g, cut);
    g.clip();
    g.globalCompositeOperation = "multiply";
    g.filter = `blur(${Math.max(1, 0.08 * px)}px)`;
    g.strokeStyle = rgba(tone, 0.5);
    g.lineWidth = 0.22;
    for (const h of hs) {
      trace(g, h);
      g.stroke();
    }
    g.filter = "none";
    g.globalCompositeOperation = "source-over";
    if (paint.line) {
      g.strokeStyle = rgba(paint.line, 0.7);
      g.lineWidth = 0.1;
      for (const h of hs) {
        trace(g, h);
        g.stroke();
      }
    }
    g.globalCompositeOperation = "destination-out";
    for (const h of hs) {
      trace(g, h);
      g.fill();
    }
    g.restore();
  }
  paint.over?.(g, r, px);
  return { canvas: c, x0: bb.x0 - m, y0: bb.y0 - m, w: wcm, h: hcm };
};

  const bob = st * 0.16 * Math.abs(Math.cos(ph)) - crouch * 0.55 * legK + (pose.hop ?? 0) - (L.build === "elder" ? 0.15 : 0);
  const armF = (pose.armF ?? 0.1) - st * 0.35 * Math.sin(ph);
  const armB = (pose.armB ?? -0.1) + st * 0.35 * Math.sin(ph);
  const lk = pose.look ?? [0, 0];
  const blink = (pose.blink ?? 0) > 0.5;
  const mouth = pose.mouth ?? "smile";
  const ln: Paint["line"] = L.line;
  const sx = flip ? -scale : scale;
  const long = L.outfit === "dress" || L.outfit === "long-skirt" || L.outfit === "robe";
  const legColor = L.outfit === "shirt-shorts" || L.outfit === "dress" ? L.skin : L.bottom;
  const thighColor = L.outfit === "shirt-shorts" ? L.bottom : legColor;

  const leg = (key: string, hip: number, knee: number, dx: number, zz: number) => (
    <Pivot at={[dx, hipY]} rot={hip} z={zz}>
      <Sheet id={`${id}-${key}-thigh`} shape={P(rect(dx - 0.42, hipY - 1.55 * legK, dx + 0.42, hipY + 0.2, 0.4))} paint={{ color: thighColor, style: "crayon", line: ln }} z={0} />
      <Pivot at={[dx, hipY - 1.35 * legK]} rot={-knee} z={0.008}>
        <Sheet id={`${id}-${key}-shin`} shape={P(rect(dx - 0.38, 0.45, dx + 0.38, hipY - 1.15 * legK, 0.36))} paint={{ color: legColor, style: "crayon", line: ln }} z={0} />
        <Sheet id={`${id}-${key}-shoe`} shape={P(blob([[dx - 0.55, 0.05], [dx + 0.95, 0.05], [dx + 1.0, 0.45], [dx + 0.4, 0.75], [dx - 0.5, 0.7]], 8))} paint={{ color: L.shoes, style: "gouache", line: ln }} z={0.006} />
        <Pin x={dx} y={hipY - 1.35 * legK} z={0.012} r={0.12} />
      </Pivot>
    </Pivot>
  );
  const arm = (key: string, sh: number, el: number, dx: number, zz: number) => {
    const ua = adult ? 1.25 : 1;
    return (
      <Pivot at={[dx, shoulderY]} rot={sh} z={zz}>
        <Sheet id={`${id}-${key}-upper`} shape={P(rect(dx - 0.45, shoulderY - 1.5 * ua, dx + 0.45, shoulderY + 0.35, 0.42))} paint={{ color: L.top, style: "wash", line: ln, marks: marksFor(L) }} z={0} />
        <Pivot at={[dx, shoulderY - 1.3 * ua]} rot={el} z={0.008}>
          <Sheet id={`${id}-${key}-fore`} shape={P(rect(dx - 0.33, shoulderY - 2.55 * ua, dx + 0.33, shoulderY - 1.1 * ua, 0.3))} paint={{ color: L.outfit === "robe" || adult ? L.top : L.skin, style: "crayon", line: ln }} z={0} />
          <Sheet id={`${id}-${key}-hand`} shape={P(ellipse(dx, shoulderY - 2.62 * ua, 0.42, 0.46, 24))} paint={{ color: L.skin, style: "crayon", line: ln }} z={0.004} />
          <Pin x={dx} y={shoulderY - 1.3 * ua} z={0.01} r={0.11} />
        </Pivot>
        <Pin x={dx} y={shoulderY} z={0.012} r={0.13} />
      </Pivot>
    );
  };
  // torso + skirt shapes
  const shirtTop = shoulderY + 0.3;
  const body: Poly = long
    ? blob([[-1.95 - (L.outfit === "robe" ? 0.2 : 0), adult ? 0.9 : 2.35], [-0.9, adult ? 0.75 : 2.15], [0, adult ? 0.85 : 2.3], [0.9, adult ? 0.75 : 2.15], [1.95, adult ? 0.9 : 2.35], [1.2, shoulderY - 0.45], [0.75, shirtTop], [-0.75, shirtTop], [-1.2, shoulderY - 0.45]], 10)
    : blob([[-1.25, hipY - 0.55], [0, hipY - 0.7], [1.25, hipY - 0.55], [1.3, shoulderY - 0.4], [0.85, shirtTop], [-0.85, shirtTop], [-1.3, shoulderY - 0.4]], 10);
  const hk = headK;
  const H = (pts: [number, number][]): Poly => pts.map(([a, b]) => [a * hk, neckY + (b - 6.05) * hk]);
  const hy = (v: number) => neckY + (v - 6.05) * hk;

  return (
    <group position={[x, y, z]} rotation={[0, 0, rot]} scale={[sx, scale, 1]}>
      <group position={[0, bob, 0]}>
        {leg("legB", legB, kneeB, -0.45, 0.012)}
        {leg("legF", legF, kneeF, 0.45, 0.03)}
        <Pivot at={[0, hipY - 0.4]} rot={(pose.lean ?? 0) + (L.build === "elder" ? 0.12 : 0)}>
          {arm("armB", armB, pose.elbowB ?? 0.15, -0.95, 0.0)}
          {!long && <Sheet id={`${id}-shorts`} shape={P(blob([[-1.35, hipY - 1.2], [1.35, hipY - 1.2], [1.3, hipY + 0.1], [-1.3, hipY + 0.1]], 6))} paint={{ color: L.bottom, style: "wash", line: ln }} z={0.045} />}
          <Sheet id={`${id}-top`} shape={P(body)} paint={{ color: L.top, style: "wash", line: ln, marks: marksFor(L) }} z={0.05} />
          {L.outfit === "long-skirt" && (
            <Sheet id={`${id}-skirt`} shape={P(blob([[-2.0, adult ? 0.85 : 2.3], [2.0, adult ? 0.85 : 2.3], [1.35, hipY + 0.2], [-1.35, hipY + 0.2]], 8))} paint={{ color: L.bottom, style: "wash", line: ln }} z={0.052} />
          )}
          <Sheet id={`${id}-collar`} shape={P(blob([[-0.95, shirtTop], [-0.2, shirtTop - 0.6], [0, shirtTop - 0.4], [0.2, shirtTop - 0.6], [0.95, shirtTop], [0, shirtTop + 0.2]], 8))} paint={{ color: L.accent, style: "gouache", line: ln }} z={0.058} />
          <Pivot at={[0, neckY]} rot={pose.head ?? 0} z={0.06}>
            {/* hair behind the face */}
            {L.hair === "bob" && <Sheet id={`${id}-hb`} shape={P(H([[-2.25, 6.6], [-2.35, 8.3], [-1.5, 9.85], [0, 10.25], [1.5, 9.85], [2.35, 8.3], [2.25, 6.6], [1.4, 6.3], [-1.4, 6.3]]))} paint={{ color: L.hairColor, style: "gouache", line: ln, edge: 0.7 }} z={0} />}
            {L.hair === "long" && <Sheet id={`${id}-hb`} shape={P(H([[-2.2, 4.6], [-2.4, 8.3], [-1.5, 9.85], [0, 10.25], [1.5, 9.85], [2.4, 8.3], [2.2, 4.6], [0, 4.9]]))} paint={{ color: L.hairColor, style: "gouache", line: ln, edge: 0.7 }} z={0} />}
            {L.hair === "puffs" && (
              <>
                <Sheet id={`${id}-pl`} shape={P(ellipse(-1.85 * hk, hy(9.4), 0.95 * hk, 0.9 * hk, 32))} paint={{ color: L.hairColor, style: "gouache", line: ln }} z={0} />
                <Sheet id={`${id}-pr`} shape={P(ellipse(1.85 * hk, hy(9.4), 0.95 * hk, 0.9 * hk, 32))} paint={{ color: L.hairColor, style: "gouache", line: ln }} z={0} />
              </>
            )}
            {L.hair === "bun" && <Sheet id={`${id}-bun`} shape={P(ellipse(0, hy(10.2), 0.95 * hk, 0.8 * hk, 32))} paint={{ color: L.hairColor, style: "gouache", line: ln }} z={0} />}
            {L.hair === "hijab" && <Sheet id={`${id}-hj`} shape={P(H([[-2.3, 5.4], [-2.45, 8.0], [-1.6, 9.9], [0, 10.4], [1.6, 9.9], [2.45, 8.0], [2.3, 5.4], [1.2, 4.9], [-1.2, 4.9]]))} paint={{ color: L.accent, style: "gouache", line: ln, edge: 0.6 }} z={0} />}
            <Sheet
              id={`${id}-face`}
              shape={P(H([[-1.85, 7.6], [-1.6, 6.55], [0, 6.05], [1.6, 6.55], [1.85, 7.6], [1.55, 9.0], [0, 9.55], [-1.55, 9.0]]))}
              paint={{
                color: L.skin, style: "crayon", line: ln,
                marks: (g, r) => {
                  if (L.build === "child") {
                    scribble(g, ellipse(-1.05 * hk, hy(7.2), 0.42, 0.3, 20), shade(L.skin, -0.12), r, 120, 0.45);
                    scribble(g, ellipse(1.05 * hk, hy(7.2), 0.42, 0.3, 20), shade(L.skin, -0.12), r, 120, 0.45);
                  }
                  if (L.build === "elder") { pencil(g, [[-1.2 * hk, hy(8.6)], [-0.5 * hk, hy(8.7)]], shade(L.skin, -0.4), 0.04, 0.5); pencil(g, [[0.5 * hk, hy(8.7)], [1.2 * hk, hy(8.6)]], shade(L.skin, -0.4), 0.04, 0.5); }
                  pencil(g, [[-0.05, hy(7.75)], [0.08, hy(7.55)], [-0.05, hy(7.45)]], shade(L.skin, -0.45), 0.05, 0.6);
                  if (L.beard) scribble(g, H([[-1.5, 7.2], [0, 6.1], [1.5, 7.2], [0, 6.6]]), shade(L.hairColor, 0.15), r, 160, 0.8);
                },
              }}
              z={0.01}
            />
            <group position={[lk[0] * 0.18, lk[1] * 0.12, 0]}>
              <Sheet id={`${id}-eyes`} visible={!blink} shape={P([ellipse(-0.7 * hk, hy(8.0), 0.2 * hk, 0.27 * hk, 20), ellipse(0.7 * hk, hy(8.0), 0.2 * hk, 0.27 * hk, 20)])}
                paint={{ color: "#2a1d18", style: "flat", clean: true, marks: (g) => { dot(g, -0.64 * hk, hy(8.09), 0.07, "#fff6e6"); dot(g, 0.76 * hk, hy(8.09), 0.07, "#fff6e6"); } }} z={0.02} cast={false} />
            </group>
            <Sheet id={`${id}-eyesShut`} visible={blink} shape={P([rect(-1.0 * hk, hy(7.9), -0.4 * hk, hy(8.05), 0.07), rect(0.4 * hk, hy(7.9), 1.0 * hk, hy(8.05), 0.07)])} paint={{ color: "#2a1d18", style: "flat", clean: true }} z={0.02} cast={false} />
            {L.glasses && <Sheet id={`${id}-glasses`} shape={P([ellipse(-0.7 * hk, hy(8.0), 0.45 * hk, 0.4 * hk, 24), ellipse(0.7 * hk, hy(8.0), 0.45 * hk, 0.4 * hk, 24)])} paint={{ color: "#f2ead8", style: "flat", clean: true, line: "#3b2b26", holes: [ellipse(-0.7 * hk, hy(8.0), 0.36 * hk, 0.31 * hk, 24), ellipse(0.7 * hk, hy(8.0), 0.36 * hk, 0.31 * hk, 24)] }} z={0.024} cast={false} />}
            <Sheet id={`${id}-m-smile`} visible={mouth === "smile"} shape={P(H([[-0.42, 7.05], [0, 6.75], [0.42, 7.05], [0, 6.92]]))} paint={{ color: "#8a3a33", style: "flat", clean: true }} z={0.02} cast={false} />
            <Sheet id={`${id}-m-open`} visible={mouth === "open"} shape={P(ellipse(0, hy(6.92), 0.22 * hk, 0.26 * hk, 20))} paint={{ color: "#7a2f2b", style: "flat", clean: true }} z={0.02} cast={false} />
            <Sheet id={`${id}-m-sad`} visible={mouth === "sad"} shape={P(H([[-0.38, 6.8], [0, 7.02], [0.38, 6.8], [0, 6.92]]))} paint={{ color: "#8a3a33", style: "flat", clean: true }} z={0.02} cast={false} />
            <Sheet id={`${id}-m-flat`} visible={mouth === "flat"} shape={P(rect(-0.3 * hk, hy(6.88), 0.3 * hk, hy(6.98), 0.05))} paint={{ color: "#8a3a33", style: "flat", clean: true }} z={0.02} cast={false} />
            {/* tears roll down from the eyes while `tears` > 0 */}
            {tears > 0 && [-0.75, 0.75].map((ex, i) => {
              const k = ((s * 0.9 + i * 0.45) % 1);
              return <Sheet key={i} id={`${id}-tear${i}`} y={-k * 1.4 * hk} scale={1 - k * 0.3} shape={P(ellipse(ex * hk, hy(7.7), 0.11, 0.15, 12))} paint={{ color: "#8fc6e8", style: "flat", clean: true, line: "#4f86a8" }} z={0.026} cast={false} />;
            })}
            {/* hair / cap in front */}
            {L.hair === "bob" && <Sheet id={`${id}-bangs`} shape={P(H([[-2.0, 8.3], [-1.9, 9.4], [-0.9, 10.05], [0.6, 10.1], [1.8, 9.5], [2.05, 8.4], [1.4, 8.75], [0.85, 8.55], [0.2, 8.85], [-0.5, 8.6], [-1.2, 8.85]]))} paint={{ color: L.hairColor, style: "gouache", line: ln, edge: 0.7 }} z={0.03} />}
            {(L.hair === "puffs" || L.hair === "bun" || L.hair === "long") && <Sheet id={`${id}-cap`} shape={P(H([[-1.85, 8.5], [-1.5, 9.5], [0, 9.95], [1.5, 9.5], [1.85, 8.5], [0.9, 8.95], [0, 8.75], [-0.9, 8.95]]))} paint={{ color: L.hairColor, style: "gouache", line: ln }} z={0.03} />}
            {L.hair === "short" && <Sheet id={`${id}-short`} shape={P(H([[-1.95, 8.2], [-1.8, 9.5], [-0.6, 10.15], [0.9, 10.1], [1.9, 9.4], [1.95, 8.3], [1.3, 8.95], [0.4, 8.85], [-0.6, 9.05], [-1.3, 8.7]]))} paint={{ color: L.hairColor, style: "gouache", line: ln, edge: 0.6 }} z={0.03} />}
            {L.hair === "peci" && <Sheet id={`${id}-peci`} shape={P(H([[-1.75, 9.05], [-1.6, 10.35], [1.6, 10.35], [1.75, 9.05], [0, 9.2]]))} paint={{ color: "#2b2522", style: "gouache", line: ln }} z={0.03} />}
            {L.hair === "hijab" && <Sheet id={`${id}-hjf`} shape={P(H([[-1.95, 8.2], [-1.6, 9.55], [0, 10.05], [1.6, 9.55], [1.95, 8.2], [1.5, 9.15], [0, 9.55], [-1.5, 9.15]]))} paint={{ color: L.accent, style: "gouache", line: ln }} z={0.03} />}
            {L.hair === "bald" && L.build !== "child" && <Sheet id={`${id}-fringe`} shape={P([...H([[-1.95, 7.9], [-1.7, 8.9], [-1.4, 8.3]]), ...H([[1.4, 8.3], [1.7, 8.9], [1.95, 7.9]])])} paint={{ color: L.hairColor, style: "gouache", line: ln }} z={0.03} />}
            {(L.hair === "bob" || L.hair === "puffs") && L.build === "child" && <Sheet id={`${id}-clip`} shape={P(H([[0.95, 9.55], [1.75, 9.95], [1.6, 9.35], [1.75, 8.95], [0.95, 9.3]]))} paint={{ color: L.accent, style: "gouache", line: ln }} z={0.04} />}
          </Pivot>
          {arm("armF", armF, pose.elbowF ?? -0.15, 0.95, 0.08)}
        </Pivot>
      </group>
    </group>
  );
};
